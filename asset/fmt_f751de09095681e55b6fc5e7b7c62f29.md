```c
/* 添加项 */
add()
{
    struct list_tag *ptr, *prev, *new;
    new=getdata();
    /* 同一个函数处理创建和向空列表插入 */
    if (start == NULL) {
        start = new;
        new->next = NULL;
        /* 在开头插入 */
        start = new;
        return;
    }
    /* 在间插入 */
    ptr = start;
    while(ptr->next != NULL) {
        prev = ptr;
        ptr = ptr->next;
    }
    /* 在末尾插入 */
    prev->next = new;
    new->next = NULL;
}
```
