/**
 findMultiplier 需要累加所有
 * 为每个指数计算的倍数。
 * 函数返回结果。
 */
double findMultiplier(double rate, double term) {
    double multiplier = 0;
    double factor = 1 + rate/100;
    for (int i = term; i > 0; i--) {
        multiplier += pow(factor, i);
    }
    return multiplier;
}

 * printData 打印三项。

}

 * printResult 打印结果。
