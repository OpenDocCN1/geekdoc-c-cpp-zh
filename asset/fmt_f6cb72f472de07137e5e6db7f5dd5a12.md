void stringCopy(char string1[], char string2[])
{
    int index = 0; // 计数器

    // 遍历 string1，将每个元素拷贝到
    // string2。遇到空字符时停止。
    while (string1[index] != '\0')

        index++;
    }

    // 在 string2 中放置空字符。
    string2[index] = '\0';
}

void nameSlice(char userName[]) {
    int count = 0; // 循环计数器

    // 定位第一个空格，或者如果没有空格，
    // 则定位空终止符
    while (userName[count] != ' ' && userName[count] != '\0') {

int countChars(char *userName, char letter) {
    int count = 0; // 循环计数器
    // 定位第一个空格，或者如果没有空格，
    // 则定位空终止符
    while (userName[count] != ' ' && userName[count] != '\0') {
        count++;
    }
    // 如果找到空格，将其替换为空终止符。
    if (userName[count] == ' ') {
        userName[count] = '\0';
    }
    int occurrences = 0;
    while (userName[count] != '\0') {
        if (userName[count] == letter) {
            occurrences++;
        }
        count++;
    }
    return occurrences;
}
