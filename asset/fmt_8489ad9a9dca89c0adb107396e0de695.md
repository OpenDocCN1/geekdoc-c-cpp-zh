
```c
int main(void)
{
    thrd_t t; // t 将持有线程 ID
    int arg = 3490;

    printf("Launching a thread\n");

    // 启动一个线程运行 run() 函数，传递指向 3490 的指针
    // 作为参数。并将线程 ID 存储在 t 中：
    thrd_create(&t, run, &arg);

    printf("Doing other things while the thread runs\n");

    printf("Waiting for thread to complete...\n");

    int res; // 存储线程退出时的返回值

    // 等待线程完成；将返回值存储
    // 在 res 中：
    thrd_join(t, &res);

    printf("Thread exited with return value %d\n", res);
}
```

看我们是如何使用 `thrd_create()` 调用 `run()` 函数的？接着我们在 `main()` 中做了其他事情，然后停止并使用 `thrd_join()` 等待线程完成。

示例输出（你的结果可能不同）：

```

Waiting for thread to complete...
```

你传递给函数的 `arg` 必须有足够长的寿命，以便线程在它消失之前能够拾取它。此外，在新线程使用它之前，它不能被主线程覆盖。

让我们来看一个启动 5 个线程的示例。这里需要注意的是，我们如何使用 `thread_t` 数组来跟踪所有的线程 ID。

```c
#include <stdio.h>
#include <threads.h>

int run(void *arg)
{
    int i = *(int*)arg;
    printf("THREAD %d: running!\n", i);
    return i;
}

#define THREAD_COUNT 5

int main(void)
{
    thread_t t[THREAD_COUNT];
    int i;

    printf("Launching threads...\n");
    for (int i = 0; i < THREAD_COUNT; i++)
        t[i] = thrd_create(run, &i); // 注意这里！

    printf("Doing other things while the thread runs...\n");
    printf("Waiting for thread to complete...\n");

    for (int i = 0; i < THREAD_COUNT; i++) {
        int res;
        if (thrd_join(t[i], &res) == thrd_success)
            printf("Thread %d complete!\n", i);
    }

    printf("All threads complete!\n");
    return 0;
}
```

## 39.6.1 线程本地存储 (Thread-local Storage)

首先，我们来看解决这个问题的简单方法：线程本地存储类。基本上，我们只需要给静态变量加上这个修符，一切就能正常工作了！`<threads.h>` 头文件定义 `thread_local` 为 `_ThreadLocal` 的别名，这样每个线程的代码看起来就不会那么难看。让我们采取之前的示例并将 `foo` 变成线程局部变量，这样我们就不再共享它。

```c
int run(void *arg)
{
    int n = *(int*)arg; // 用于人类区分的线程编号

    free(arg);
    static int foo = 10; // 线程共享的静态值
```

运行后我们得到：

Thread 0: x = 10, foo = 10

Thread 1: x = 11, foo = 11

Thread 2: x = 12, foo = 12

Thread 4: x = 13, foo = 13

Thread 3: x = 14, foo = 14

不再有奇怪的问题了！一点：如果 `thread_local` 变量是块作用域，它必须是 `static` 的。这是可以，因为非静态变量已经是每个线程私有的。小误导：`thread_local` 变量也可以是 `extern`。

## 39.6.2 另一种选择：线程特定存储 (Thread-Specific Storage)

线程特定存储 (TSS) 是获取线程数据的另一种方法。一个额外特性是，这些函数允许你定义一个析构函数，当 TSS 变量被删除时调用它；如果你不需要销毁任何东西，则传入 `NULL`。析构函数是一个指针，它返回 `void` 并接收一个 `void*` 作为参数（存储在变量中的数据）。换句话说，它是一个 `void (*dtor)(void*)`，用于清理工作。

查看下面的示例：

通常情况下，`thread_local` 可能是你的首选，但如果你使用析构函数，就可以利用它。这种用法有点奇怪，我们用类型 `tss_t` 初始化它，以便相对于每个线程的值为基准。我们使用 `tss_create()` 进行初始化。我们在创建之前设置好 TSS 变量，然后在线程结束后进行清理。在 `run()` 函数中，线程为字符串 `malloc()` 了一些空间，然后将该指针存储在 TSS 变量中。当线程退出时，所有线程都会调用析构函数（在本例中为 `free`）。

```c
#include <stdio.h>
#include <stdlib.h>
#include <threads.h>

tss_t str;

void some_function(void)
{
    // 获取此字符串的线程特定值
    char *tss_string = tss_get(str);

    // 打印它
    printf("TSS string: %s\n", tss_string);
}

int run(void *arg)
{
    int serial = *(int*)arg; // 获取线程特定值
    free(arg);

    // malloc 空间存储字符串
    char *s = malloc(64);
    printf("thread %d:\n", serial); // Happy string

    // 设置 TSS 指向字符串
    tss_set(str, s);

    // 调用函数获取变量
}

int main(void)
{
    thread_t t[THREAD_COUNT];

    // 创建一个新的 TSS 变量，析构函数为 free()
    if (tss_create(&str, free) == 0) {
        for (int i = 0; i < THREAD_COUNT; i++) {
            int *n = malloc(sizeof(*n)); // 存储线程序列号
            *n = i;
            thrd_create(&t[i], run, n);
        }

        for (int i = 0; i < THREAD_COUNT; i++) {
            thrd_join(t[i], NULL);
        }
    }
    tss_delete(str);
    return 0;
}
```

## 39.7 互斥锁 (Mutexes)

互斥（Mutual exclusion）的缩写，又称为对一段允许只有一个线程执行的“锁”。如果你想防止多个线程进入该代码区域并导致损坏，可以使用互斥锁。

```c
#include <stdio.h>
#include <stdlib.h>
#include <threads.h>

mtx_t mtx_serial;
int serial = 0;

int run(void *arg)
{
    int n = *(int*)arg;
    free(arg);

    mtx_lock(mtx_serial);
    printf("Thread running: %d\n", serial);
    serial++;
    mtx_unlock(mtx_serial);

    return 0;
}

int main(void)
{
    thread_t t[10];
    if (mtx_init(&mtx_serial, mtx_recursive) == 0) {
        for (int i = 0; i < 10; i++) {
            int *n = malloc(sizeof(*n));
            *n = i;
            thrd_create(&t[i], run, n);
        }

        for (int i = 0; i < 10; i++) {
            thrd_join(t[i], NULL);
        }
    }
    mtx_destroy(mtx_serial);
    return 0;
}
```
