## 计算与输出

使用二维数组进行计算和输出必须指定两个偏移量。为了说明，请考虑下一个程序，它读取一个包含10周功率输出的数据文件。数据文件的每一行代表一周的每日输出。数据存储在二维数组中，然后打印一份报告，给出该期间每周第一天的平均功率，每周第二天的平均功率，以此类推：

```cpp
/*
* 程序 chapter8_2
* 此程序计算
* 10周期间的平均功率。
 */
#include<iostream> // 需要 cin, cout, cerr
#include<fstream> // 需要 istream
#include<string> // 需要 string
using namespace std;
int main() {
    // 声明对象。
    const int NROWS = 10;
    const int NCOLS = 7;
    double power[NROWS][NCOLS], col_sum;
    string filename;
    // 打开文件并将数据读数组。
    cout << "\nEnter输入文件名: ";
    cin >> filename;
    datal.open(filename.c_str());
    if(datal.fail()) {
        cerr << "\Error打开数据文件\n";
        exit(1);
    }
    // 设置格式标志。
    cout.setf(ios::fixed);
    cout.setf(ios::showpoint);
    cout.precision(1);
    for (int i=0; i<NROWS; ++i) {
        for (int j=0; j<NCOLS; ++j) {
            datal >> power[i][j];
        }
    }
    // 计算并打印每日平均值。
    for (int j=0; j<NCOLS; ++j) {
        col_sum = 0;
        for (int i=0; i<NROWS; ++i) {
            col_sum += power[i][j];
        }
        cout << "\Day" << j+1 << " Average =" << col_sum/NROWS << endl;
    }
    // 关闭文件并退出程序。
    datal.close();
    return 0;
}
```
