/**
 * 描述一个邮件地址。
 */
public class Address {
    private String name;
    private String street;
    private String city;
    private String state;
    private String zip;

    /**
     * 构造一个邮件地址。
     * @param name 收件人姓名
     * @param street 街道
     * @param city 市
     * @param state 州代码
     * @param zip 邮政编码
     */
    public Address(String name, String street, String city, String state, String zip) {
        name = name;
        street = street;
        city = city;
        state = state;
        zip = zip;
    }

    /**
     * 格式化地址。
     * @return 格式化的地址
     */
    public String format() {
        return name + "\n" + street + "\n" + city + "\n" + state + "\n" + zip;
    }
}
