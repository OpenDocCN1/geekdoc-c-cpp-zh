using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace Loan_Qualifier
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void checkButton_Click(object sender, EventArgs e)
        {
            try
            {
                // constants
                const decimal MINIMUM_SALARY = 40000m;
                const int MINIMUM_YEARS_ON_JOB = 2;

                // variables
                decimal salary;
                int yearsOnJob;

                // 获取工资和当前工作年限
                salary = decimal.Parse(salaryTextBox.Text);
                yearsOnJob = int.Parse(yearsTextBox.Text);

                if (salary >= MINIMUM_SALARY)
                {
                    if (yearsOnJob >= MINIMUM_YEARS_ON_JOB)
                    {
                        // 用户符合条件
                        decisionLabel.Text = "You qualify for the loan";
                    }
                    else
                    {
                        // 用户不符合条件
                        decisionLabel.Text = "Minimum years at current " +
                              "job not met";
                    }
                }
                else
                {
                    // 用户不符合条件
                    decisionLabel.Text = "Minimum salary requirement " +
                          "not met";
                }
            }
            catch (Exception ex)
            {
                // 显示错误消息
                MessageBox.Show(ex.Message);
            }
        }

        private void clearButton_Click(object sender, EventArgs e)
        {
            // 清空 TextBox 和 decisionLabel
            salaryTextBox.Text = "";
            yearsTextBox.Text = "";
            decisionLabel.Text = "";

            // 重置焦点
            salaryTextBox.Focus();
        }

        private void exitButton_Click(object sender, EventArgs e)
        {
            // 关闭表单
            this.Close();
        }
    }
}
