
# MutexLock 示例

```csharp
// 此值类型封装了互斥锁锁定和解锁
internal struct MutexLock : IDisposable {
    private readonly Mutex m_mutex;

    // 此构造函数获取互斥锁的锁
    public MutexLock(Mutex m) {
        m_mutex = m;
        m_mutex.WaitOne();
    }

    // 此 Dispose 方法释放互斥锁的锁
    public void Dispose() {
        m_mutex.ReleaseMutex();
    }
}

public static class Program {
    public static void Main() {
        // 构建一个互斥锁对象
        Mutex m = new Mutex();

        // 锁定互斥锁，执行某些操作，然后解锁
        using (new MutexLock(m)) {
            // 在此处执行某些线程安全的操作...
        }
    }
}
```
