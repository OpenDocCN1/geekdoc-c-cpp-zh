```cpp
int main()
{
    Teapot t = Teapot("blue");
    t.change_tea("oolong");
    t.add_water(25.5);
    t.heat_up(95);
    cout << "Teapot status:" << endl;
    cout << "Color: " << t.get_color() << endl;
    cout << "Temperature: " << t.get_temperature() << endl;
    cout << "Tea Type: " << t.get_tea_type() << endl;
    cout << "Water Amount: " << t.get_water_amount() << endl;
    return 0;
}
```
