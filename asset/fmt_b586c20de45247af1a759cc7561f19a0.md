bool __attribute__((optimize("01"))) check_password(const char* p_password)

{

    char pass[password_size] = {};

    char* label_address = 0;

    asm volatile(

        "jz unaligned+1\n"

        "jnz unaligned+1\n"

        "unaligned:\n"

         ".byte 0xe8\n"

    );
