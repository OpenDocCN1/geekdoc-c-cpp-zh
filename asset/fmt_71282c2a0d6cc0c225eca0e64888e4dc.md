/* 向升序链表添加元素 */
void add (struct node **q, int num)
{
  struct node *r, *temp = *q;
  r = malloc (sizeof (struct node ) );
  r->data = num;
  /* 如果列表为空或新节点需要插入到第一个节点之前 */
  if ( *q == NULL || (* q) -> data > num)
  {
    *q = r;
    (*q) -> link = temp;
  }
  else {
    /* 遍历整个链表寻找插入新节点的位置 */
    while (temp -> link != NULL) {
      if (temp -> data <= num && ( temp -> link -> data > num ||
          temp -> link == NULL )) {
        r -> link = temp -> link;
        temp -> link = r;
        return;
      }
      /* 到下一个节点 */
      temp = temp -> link;
    }
    r -> link = NULL;
    temp -> link = r;
  }
}
