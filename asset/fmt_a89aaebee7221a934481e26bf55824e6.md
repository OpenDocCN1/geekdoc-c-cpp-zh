-(NSString *) convertToString
{
    if (numerator == denominator)
        if (numerator == 0)
            return "@0";
        else
            return "@1";
    else if (denominator == 1)
        return [NSString stringWithFormat:@"%i", numerator];
    else
        return [NSString stringWithFormat:@"%i/%i", numerator, denominator];
}
