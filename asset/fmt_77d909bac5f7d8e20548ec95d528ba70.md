#include <fstream>

void partition(int a [], int start, int end);

// // Range 类用于表示数组中
// // 待排序的段
class Range
{
    // // 将 qsort 设置为友元
    friend void qsort(int a [], int start, int end);
    int start;
    int end;
public:
    Range(int s, int e)
    {
        start = s;
        end = e;
    }
};

const int MAX = 100;
int main()
{
    ifstream inputFile;
    string filename = "sort.dat";
    int array[MAX];
    int size;

        exit(1);
    }

// // 读取文件以及文件中的项数
// // 注意不要超出数组范围
size = 0;

        break;

// 执行非递归的快速排序

void qSort(int arr [], int size) {
    // qStack 处理数组中尚未排序的段
    stack<Range> qStack;
    int pivot, start, end;
    qStack.push(Range(0, size-1));
    // 只要栈中还有待排序的范围
    // 就将其从栈中取出，进行分区，然后
    // 将生成的较小范围放入栈中
    while (!qStack.empty()) {
        Range currentRange = qStack.top();
        qStack.pop();
        start = currentRange.start;
        end = currentRange.end;
        if (start < end) {
            pivot = partition(arr, start, end);
            // 存储生成的较小范围以便
            // 处理
            qStack.push(Range(start, pivot-1));
            qStack.push(Range(pivot + 1, end));
        }
    }
// partition 重新排列从 start 到 end 的数组项
// 所有大于等于 pivot 的值都在 pivot 右侧
// 所有小于 pivot 的值都在 pivot 左侧。
