from random import *

from randoml import *

from modfunc import *

from graphlib import *

pi4 = 4 * pi

def func(x,y):                      # integrand

    r2 = x*x + y*y

    return r2*exp(-0.5e0*r2)/pi4

#### main

L = 8e0                             # integration domain [-L,L] x [-L,L]

L2 = L * L                          # area of sampling domain

nn = [0]*4                          # ending indexes of plots

        f = func(x,y)               # integrand
