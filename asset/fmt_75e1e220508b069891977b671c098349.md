```c
        for (j=0; j<=4;j++)
```
```c
        {
```

## 74. 编写程序读取两个 3*3 矩阵并相乘结果存储在第三矩阵中。

```c
#include<stdio.h>
#include<conio.h>
void main()
{
    int a[3][3],b[3][3],c[3][3],i,j,k;
    clrscr();
    printf("\n 输入第一个矩阵元素");
    for (i=0;i<3;i++)
    {
        for (j=0; j<3;j++)
        {
            scanf("%d",a[i][j]);
        }
    }
    printf("\n 输入第二个矩阵元素");
    for (i=0;i<3;i++)
    {
        for (j=0; j<3;j++)
        {
            scanf("%d",b[i][j]);
        }
    }
    /*矩阵乘法*/
    for (i=0; i<3;i++)
    {
        for (j=0; j<3;j++)
        {
            c[i][j]=0;
            for(k=0; k<3;k++)
            {
                c[i][j]+=a[i][k]*b[k][j];
            }
        }
    }
    printf("\n 两个矩阵相乘的结果 \n");
    for (i=0; i<3;i++)
    {
        for (j=0; j<3;j++)
        {
            printf("%d",c[i][j]);
        }
    }
    printf("\n");
    getch();
}
```
