
```c
#include <stdio.h>
#include <stdlib.h>

    // define dynamic array of pointer
    int **matrix; // two dimensional array pointer

    // a number of array
    int M = 3;
    int N = 5;

    // allocate memory
    matrix = malloc( M * sizeof(int*));

    // set values
    int i,j;
    for(i=0;i<M;i++){
        matrix[i] = malloc( N * sizeof(int));
        for(j=0;j<N;j++){
            matrix[i][j] = i + j;
        }
    }

    // display values
    for(i=0;i<M;i++){
        for(j=0;j<N;j++){
            printf("%d ",matrix[i][j]);
        }
        printf("\n");
    }

    // free memory
    free(matrix);

    return 0;
}
```

## 4.4 函数与指针

我们可以在函数中将指针作为参数传递。为了演示，我们可以创建 `swap()` 来交换我们的值。

```c
int main(int argc, const char* argv[]) {
    int *x, *y;
    int a, b;

    a = 10;
    b = 5;

    // set value
    x = &a;
    y = &b;

    printf("value pointer x: %d \n",*x);
    printf("value pointer y: %d \n",*y);

    printf("swap()\n");

    printf("value pointer y: %d \n",*y);
    return 0;

void swap(int *px, int *py){
    int temp;

    // store pointer px value to temp
    temp = *px;
    // set pointer px by py value
    *px = *py;
    // set pointer py by temp value
    *py = temp;
}
```

### 5.2.2 gets() 和 puts() 函数

`gets()` 用于读取文本。当你输入换行符（回车键）时，它会停止读取。遗憾的是，`gets()` 是一个不安全的函数。我们可以使用 `fgets()` 来按长度限制读取文本。

为了测试，我们在 `keyboard.c` 文件中添加 `get_puts()` 函数，并编写以下代码。

```c
#include <stdio.h>

void gets_puts();

int main() {

    gets_puts();

    return 0;
}

void gets_puts(){
    printf("-----unsafe input-----\n");
    char name[256];

    printf ("Your name: ");
    gets (name);
    printf ("name: %s\n", name);
    puts(name);

    printf ("-----fgets() input-----\n");
    name[0] = '\0'; // clear
    printf ("Your name: ");
    fgets(name,256,stdin);
    printf ("name: %s\n",name);
    puts(name);
}
```
