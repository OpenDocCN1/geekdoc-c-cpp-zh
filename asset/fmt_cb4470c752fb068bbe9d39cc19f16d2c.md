
## 第 12 章 二维数组

```cpp
//Lab12-1.cpp 计算总销售额
// #由 <你的姓名> 于 <当前日期> 创建/修订
#include <iostream>
using namespace std;
int main() {
    // 声明数组和变量
    int company[2][6] = {{12000, 45000, 32000,
                          67000, 24000, 55000},
                          10000, 56000, 42000,
                          23000, 12000, 34000}};
    int companySales = 0; // 累加器
    // 累加销售额
    for (int location = 0; location < 2; location += 1) {
        for (int month = 0; month < 6; month += 1) {
            companySales += company[location][month];
        }
    }
    // 显示总销售额
    cout << "Company sales: $" << companySales << endl;
    system("pause");
    return 0;
}
```
