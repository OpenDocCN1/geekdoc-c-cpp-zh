#include <iostream>
#include <cmath>
#include <iomanip>

using namespace std;

int main() {
    int principal = 1000;
    double rate = .03;
    int years = 1;
    double balance = 0.0;

    while (years < 6) {
        cout << "Year " << years << ":" << endl;
        balance = principal * pow(1 + rate, years);

        years++;
    }
    system("pause");
    return 0;
}
