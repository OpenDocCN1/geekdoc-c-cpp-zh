```cpp
Student::Student(const string& n, float g) {
    name = n;
    grd = g;
}

void Student::show() {
    cout << "N:" << name << " G:" << grd << '\n';
}

void sort(vector<Student>& v) {
    int i, j, size;
    Student tmp;

    size = v.size();
    for (i = 0; i < size-1; i++) {
        for (j = i+1; j < size; j++) {
            if (v[i].grd > v[j].grd) {
                tmp = v[i];
                v[i] = v[j];
                v[j] = tmp;
            }
        }
    }
}
```
