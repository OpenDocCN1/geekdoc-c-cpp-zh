
# 结构体示例与错误分析

```c
#include <stdio.h>
struct virus
{
    char signature[25];
    char status[20];
    int size;
};
struct virus v[] = {"Yankee Doodle", "Deadly", 183, "Killer", 195 };
int main()
{
    int i;
    for (i=0; i<=1; i++)
    printf("%s %s\n", v.signature, v.status);
    return 0;
}
```

**错误：** 打印 `signature` 和 `status` 时，应使用 `v[i].signature` 和 `v[i].status`。

```c
#include <stdio.h>
struct s
{
    char ch;
    int i;
    float a;
};
void f (struct s);
void g (struct s *);
int main()
{
    struct s var = {'C', 100, 5.5};
    f (var);
    g (&var);
    return 0;
}
void f (struct s v) {
    printf("%c %d %f\n", v->ch, v->i, v->a);
}
void g (struct s v) {
    printf("%c %d %f\n", v.ch, v.i, v.a);
}
```

**错误：** 在函数 `f()` 中应使用点运算符访问结构体元素，而在函数 `g()` 中应使用 `->` 运算符。

- (e)

```c
#include <stdio.h>
struct s {
    int i;
    struct s *p;
}
int main()
{
    struct s var1, var2;
    var1.i = 100;
    var2.i = 200;
    var1.p = &var2;
    var2.p = &var1;
    printf("%d %d\n", var1.p->i, var2.p->i);
    return 0;
}
```

**错误：** 结构体定义缺少分号。

void deposit (int ano, float amount)
{
    int i, j = 0;

void withdrawal (int ano, float amount)
