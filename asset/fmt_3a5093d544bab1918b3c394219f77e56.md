    int lineno = 0;

        ++lineno;
        smatch matches; // 匹配的字符串存储在这里
        if (regex_search(line, matches, pat)) {
            cout << lineno << ": " << matches[0] << '\n'; // 完全匹配项
            if (1<matches.size() && matches[1].matched) // 如果存在子模式
                cout << ": " << matches[1] << '\n'; // 如果它被匹配
                // 子匹配
        }
    }
}
