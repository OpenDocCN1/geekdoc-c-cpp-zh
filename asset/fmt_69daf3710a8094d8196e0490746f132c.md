## 计数器循环 (Counter-Controlled Loops)

假设你想累加这些数字并求它们的平均值。考虑以下程序：

```cpp
#include <iostream>
using namespace std;
int main() {
    int limit; // 存储项目个数
    int number; // 存储数字
    int sum; // 存储和
    int counter; // 计数
    cout << "Enter number of items: ";
    cin >> limit;
    sum = 0;
    counter = 0;
    cout << "Enter integers: ";
    while (counter < limit) {
        cin >> number;
        sum = sum + number;
        counter++;
    }
    cout << "The sum of the " << limit << " numbers = " << sum << endl;
    if (counter != 0) {
        cout << "The average = " << sum / counter << endl;
    } else {
        cout << "No input." << endl;
    }
    return 0;
}
```
