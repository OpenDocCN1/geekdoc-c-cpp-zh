class Triangle
{
    private int side1Length;
    private int side2Length;
    private int side3Length;
    // 默认构造函数 - 所有边的默认值
    public Triangle()
    {
        side1Length = 0;
        side2Length = 0;
        side3Length = 0;
    }
    // 为 side1Length 指定长度，其他边使用默认值
    public Triangle(int length1)
    {
        side1Length = length1;
        side2Length = 0;
        side3Length = 0;
    }
    // 为 side1Length 和 side2Length 指定长度
    // side3 使用默认值
    public Triangle(int length1, int length2)
    {
        side1Length = length1;
        side2Length = length2;
        side3Length = 0;
    }
    // 为所有边指定长度
    public Triangle(int length1, int length2, int length3)
    {
        side1Length = length1;
        side2Length = length2;
        side3Length = length3;
    }
}

{
    private int bits;
    // 简单构造函数
    public IntInts(int initialValue)
    {
        bits = initialValue;
    }
    // 索引器
    public bool this[int index]
    {
        get
        {
            return (bits & (1 << index)) != 0;
        }
        set
        {
            if (value)
            {
                bits |= (1 << index);
            }
            else
            {
                bits &= ~(1 << index);
            }
        }
    }
}
extension IntInts
{
    public override string ToString()

```csharp
class Queue
{
    private const int DEFAULTQUEUESIZE = 100;
    private int[] data;
    private int head = 0, tail = 0;
    private int numElements = 0;
    public Queue()
    {
        this.data = new int[DEFAULTQUEUESIZE];
    }
    public Queue(int size)
    {
        if (size > 0)
        {
            this.data = new int[size];
        }
        else
        {
            throw new ArgumentOutOfRangeException("size", "必须大于 0");
        }
    }
    public void Enqueue(int item)
    {
        if (this.numElements == this.data.Length)
        {
            throw new Exception("队列满");
        }
        this.data[head] = item;
        this.head++;
        this.head %= this.data.Length;
        this.numElements++;
    }
    public int Dequeue()
    {
        if (this.numElements == 0)
        {
            throw new Exception("队列空");
        }
        int queueItem = this.data[this.tail];
        this.tail++;
        this.tail %= this.data.Length;
        this.numElements--;
        return queueItem;
    }
}
```
