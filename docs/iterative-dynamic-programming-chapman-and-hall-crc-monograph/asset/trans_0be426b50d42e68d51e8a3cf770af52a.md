SUBROUTINE RLEQ(A,D,X,N)
IMPLICIT DOUBLE PRECISION(A-H,O-Z)
DIMENSION A(20,20),D(20),X(20)
NQ = N - 1
DO 5 K=1,NQ
DEN = 1.0D0/A(K,K)
D(K) = DEN*D(K)
KK = K + 1
DO 6 J = KK,N
A(K,J) = DEN*A(K,J)
DO 7 I=KK,N
D(I) = D(I) - D(K)*A(I,K)
DO 7 J=KK,N
7 A(I,J) = A(I,J) - A(K,J)*A(I,K)
5 CONTINUE
X(N) = D(N)/A(N,N)
DO 8 I=1,NQ
K = N - I
X(K) = D(K)
DO 9 J=1,I
L = K+J
9 X(K) = X(K) - A(K,L)*X(L)
8 CONTINUE

SUBROUTINE SIMP(A,N,M,B,C,M1)
DIMENSION A(10,20),B(10),C(20),CC(20),Z(21),ZZ(21)
COMMON CC,ZZ,Z
DO 8 I=1,N
NN = M1-N+I
CC(I) = C(NN)
8 DO 39 KK=1,100
IF(KK.EQ.1) GO TO 14
TEST = Z(1)
LC=1
DO 30 I=2,M1
IF(Z(I).GT.TEST) GO TO 30
TEST = Z(I)
LC = I
CONTINUE
30 IF(TEST.GT.-0.001) GO TO 1000
TEST = 1.0E7
DO 31 I=1,N
IF(A(I,LC).LT.0.001) GO TO 31
RAT = B(I)/A(I,LC)
IF(RAT.GT.TEST) GO TO 31
TEST = RAT
LR = I
CONTINUE
31 AA = 1.0/A(LR,LC)
B(LR) = B(LR)*AA
DO 32 I=1,M1
A(LR,I) = A(LR,I) *AA
LR1=LR-1
IF(LR1.EQ.0) GO TO 35
LR2 = LR-1
DO 33 I=1,LR1
AA = A(I,LC)
B(I) = B(I) - AA*B(LR)
DO 33 J=1,M1
A(I,J) = A(I,J) - AA*A(LR,J)
IF(LR2.GT.N) GO TO 37
DO 34 I=LR2,N
AA = A(I,LC)
B(I) = B(I) - AA*B(LR)
DO 34 J=1,M1
A(I,J) = A(I,J) - AA*A(LR,J)
GO TO 37
35 CONTINUE
DO 36 I=2,N
AA = A(I,LC)
B(I) = B(I) - AA*B(LR)
DO 36 J=1,M1
A(I,J) = A(I,J) - AA*A(LR,J)
CONTINUE
37 CC(LR) = C(LC)
CONTINUE
14 DO 15 J=1,M
15 ZZ(J) = 0.0
DO 17 J=1,M1
DO 16 I=1,N
16 ZZ(J) = ZZ(J)+CC(I)*A(I,J)
17 Z(J) = ZZ(J) - C(J)
DO 18 I=1,N
18 ZZ(M) = ZZ(M) + CC(I)*B(I)
Z(M) = ZZ(M)
39 CONTINUE
1000 CONTINUE
RETURN
END
