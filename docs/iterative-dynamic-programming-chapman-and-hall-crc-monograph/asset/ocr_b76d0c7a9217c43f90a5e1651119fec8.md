SUBROUTINE RLEQ(A,D,X,N)
IMPLICIT DOUBLE PRECISION(A-H,O-Z)
DIMENSION A(20,20),D(20),X(20)
NQ = N - 1
DO 5 K=1,NQ
DEN = 1.0D0/A(K,K)
D(K) = DEN*D(K)
KK = K + 1
DO 6 J = KK,N
A(K,J) = DEN*A(K,J)
DO 7 I=KK,N
D(I) = D(I) - D(K)*A(I,K)
DO 7 J=KK,N
7 A(I,J) = A(I,J) - A(K,J)*A(I,K)
5 CONTINUE

X(K) = D(K)
DO 9 J=1,I

9 X(K) = X(K) - A(K,L)*X(L)
8 CONTINUE
RETURN
END
