```
16. Write a program to carry out the arithmetic operations addition, subtraction, multiplication, and division between two variables.

**Solution:**

Use the switch construct to choose the operations.

```
```c
#include<stdio.h>
int main()
{
    int value1, value2;
    char operator;
    printf("Type in your expression. \n");
    scanf("%d %c %d ",&value1,&operator,&value2);
    switch(operator)
    {
        case '+':
            printf("%d \n", value1 + value2);
            break;
        case '-':
            printf("%d \n", value1 - value2);
            break;
        case '*':
            printf("%d \n", value1 * value2);
            break;
        case '/':
            if(value2 == 0)
                printf("division by zero. \n");
            else
                printf("%d \n", value1 / value2);
            break;
        default:
            printf("Unknown Operator \n");
            break;
    }
    return 0;
}
```

```c
int main(void)
{

        case 'e': case 'E':
        case 'i': case 'I':
        case 'o': case 'O':
        case 'u': case 'U':
```
