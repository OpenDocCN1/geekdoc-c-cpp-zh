template <typename InIt>
void sort(InIt begin, InIt end)
{
    while (begin != end)            // (1)
    {
        auto src = begin;           // (2)
        auto next = std::next(begin);   // (3)
        while (next != end)
        {
            if (*next < *src) {     // (4)
                std::iter_swap(src++, next++);
            }
        }
        --end;
    }
}
