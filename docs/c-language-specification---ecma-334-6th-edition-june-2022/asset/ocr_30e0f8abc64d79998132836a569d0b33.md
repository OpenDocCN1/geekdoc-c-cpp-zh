    : 'orderby' orderings

    ;

select_clause
    : 'select' expression
    ;

group_clause
    : 'group' expression 'by' expression
    ;

query_continuation

    ;

assignment
    : unary_expression assignment_operator expression
    ;

assignment_operator

    | right_shift_assignment
    ;

// Source: §11.19 Expression
expression
    : non_assignment_expression
    | assignment
