An Overview of C Programming

Dennis Ritchie created the C programming language at Bell Labs in the early 1970s, and it is still used today. It was developed as an extension to the B programming language, featuring new data types and functionalities. Over the years, C has evolved into one of the most extensively used programming languages in the world, influencing the development of numerous modern programming languages such as C++, C#, Objective-C, and, more indirectly, languages such as Java and Python.

The significance of C

C is frequently used in system programming, particularly in the creation of operating systems. Linux and Windows kernels, for example, contain C code.

C is a compiled language, thus it can run quicker than interpreted languages. As a result, it is appropriate for resource-constrained environments or when high performance is required.

One of the most major advantages of C is its ability to code once and run anywhere (with little modifications). This means that a C programme can be created on one system and then run on another with no or little changes.

C follows a procedural programming paradigm, which is useful for jobs that require a step-by-step approach.

Modern Language basis: Learning C lays a solid basis for understanding more sophisticated languages. C is the source of syntax and concepts in many current languages.

// #4-3

int factorial_recursive(int n) {

if (n == 0 || n == 1) {

return 1;

} else {

return n * factorial_recursive(n - 1);

}

}

// #4-4

bool is_prime(int n) {

if (n <= 1) return false;

for (int i = 2; i * i <= n; i++) {

if (n % i == 0) return false;

}

return true;

}

// #4-5

void to_uppercase(char *str) {

for (int i = 0; str[i]; i++) {

if (str[i] >= 'a' && str[i] <= 'z') {

str[i] -= 'a' - 'A';

}

}

}

}

}

// #6-2

bool is_palindrome(char *str) {

int len = string_length(str);

for (int i = 0; i < len / 2; i++) {

if (str[i] != str[len - 1 - i]) {

return false;

}

}

return true;

}

// #6-3

void concatenate_strings(char *dest, char *src) {

while (*dest) {

dest++;

}

while ((*dest++ = *src++));

}

// #6-4

void replace_spaces_with_hyphen(char *str) {

while (*str) {

if (*str == ' ') {

*str = '-';

}

str++;

}

}

// #6-5

void count_vowels_and_consonants(char *str, int *vowels, int *consonants) {

*vowels = 0;

*consonants = 0;

while (*str) {

if ((*str >= 'A' && *str <= 'Z') || (*str >= 'a' && *str <= 'z')) {

switch (*str) {

case 'A': case 'E': case 'I': case 'O': case 'U':

case 'a': case 'e': case 'i': case 'o': case 'u':

(*vowels)++;

break;

default:

(*consonants)++;

}

}

str++;

}

}

int fibonacci(int n) {

if (n <= 1) return n;

return fibonacci(n-1) + fibonacci(n-2);

}

// #8-2

int binary_search(int arr[], int l, int r, int x) {

if (r >= l) {

int mid = l + (r - l) / 2;

if (arr[mid] == x) return mid;

if (arr[mid] > x) return binary_search(arr, l, mid-1, x);

return binary_search(arr, mid+1, r, x);

}

return -1;

}

// #8-3

double power(double base, int exponent) {

if (exponent == 0) return 1;

else if (exponent < 0) return 1 / power(base, -exponent);

return base * power(base, exponent-1);

}

typedef struct {

float real;

float imag;

} Complex;

Complex add(Complex a, Complex b) {

Complex result;

result.real = a.real + b.real;

result.imag = a.imag + b.imag;

return result;

}

Complex subtract(Complex a, Complex b) {

Complex result;

result.real = a.real - b.real;

result.imag = a.imag - b.imag;

return result;

}

Complex multiply(Complex a, Complex b) {

Complex result;

result.real = a.real * b.real - a.imag * b.imag;

result.imag = a.real * b.imag + a.imag * b.real;

return result;

}

void countFileContents() {

void copyFile() {

FILE *srcFile, *destFile;

char ch;

srcFile = fopen("user_input.txt", "r");

if (srcFile == NULL) {

printf("Unable to open source file to read.\n");

return;

}

destFile = fopen("copy.txt", "w");

if (destFile == NULL) {

printf("Unable to open destination file to write.\n");

return;

}

ch = fgetc(srcFile);

while (ch != EOF) {

fputc(ch, destFile);

ch = fgetc(srcFile);

}

printf("File copied successfully.\n");

fclose(srcFile);

fclose(destFile);

}
