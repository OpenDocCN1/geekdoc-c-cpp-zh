void ASynth::update() {
  // --- Voice:
  m_GlobalSynthParams.voiceParams.uVoiceMode = Globals()->GetParameter(VOICE_MODE);
  m_GlobalSynthParams.voiceParams.dPortamentoTime_mSec = Globals()->GetParameter(PORTAMENTO_TIME_MSEC);

  m_GlobalSynthParams.voiceParams.doscFoPitchModRange = Globals()->GetParameter(PITCHBEND_RANGE);

  // --- intensities
  m_GlobalSynthParams.voiceParams.dFilterKeyIntensity = Globals()->GetParameter(FILTER_KEYTRACK_INTENSITY);
  m_GlobalSynthParams.voiceParams.dLF01FilterIntensity = Globals()->GetParameter(LF01_TO_FILTER_INTENSITY);
  m_GlobalSynthParams.voiceParams.dLF01ScModIntensity = Globals()->GetParameter(LF01_TO_OSC_INTENSITY);
  m_GlobalSynthParams.voiceParams.dLF0DCAmpIntensity = Globals()->GetParameter(LF01_TO_DCA_INTENSITY);

  m_GlobalSynthParams.voiceParams.dEGI1FilterIntensity = Globals()->GetParameter(EGI1_TO_FILTER_INTENSITY);
  m_GlobalSynthParams.voiceParams.dEGI DCAAmpIntensity = Globals()->GetParameter(EGI1_TO_DCA_INTENSITY);

  double dNoiseAmplitude = Globals()->GetParameter(NOISE_OSC_AMP_DB) == -96.0 ? 0.0 : pow(10.0, Globals()->GetParameter(NOISE_OSC_AMP_DB) / 20.0);
