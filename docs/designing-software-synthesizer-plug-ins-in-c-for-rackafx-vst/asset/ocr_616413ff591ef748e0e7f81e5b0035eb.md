ComponentResult ASync::ParameterValueStrings(args...)
(
  if(inScope == kaudioUnitScope_Global)
  {
    if(outStrings == NULL)
      return noErr;

    // --- decode the ID value and the string list; I do it this
    // way to match the "enum" described
    // in the book; take the strings from the GUI tables and
    // embed here
    switch(inParameterID)
    {
      case OSC_WAVEFORM:
        {
          setAUParamStringList(CFSTR("SINE,SAW1,
W2,SAW3,TRI,
QUARE,NOISE,PNOISE"),noutStrings);
          return noErr;
          break;
        }
      case LFO1_WAVEFORM:
        {
          setAUParamStringList(CFSTR("sine,usaw,dsaw,
tri,square,expo,rsh,qrsh"),noutStrings);
          return noErr;
          break;
        }
      case LFO1_MODE:
        {
          setAUParamStringList(CFSTR("sync,shot,free"),noutStrings);
          return noErr;
          break;
        }
    }
  }
  return kAudioUnitErr_InvalidParameter;
)
