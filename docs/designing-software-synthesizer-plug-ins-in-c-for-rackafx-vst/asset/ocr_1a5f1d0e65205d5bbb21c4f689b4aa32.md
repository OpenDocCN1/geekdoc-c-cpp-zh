// NOTE: we only have one EventBus(0)
// but it has 16 channels on it
if(busIndex == 0)
{
    id = -1;
    switch (midiControllerNumber)
    {
        // see Processor::process() method for handling
        case kPitchBend:
            id = MIDI_PITCHBEND;
            break;
        case kCtrlModWheel:
            id = MIDI_MODWHEEL;
            break;
        case kCtrlVolume:
            id = MIDI_VOLUME_CC7;
            break;
        case kCtrlPan:
            id = MIDI_PAN_CC10;
            break;
    }
