```cpp
inline bool inoscSplitRange(int noscIndex) {
  if (noscIndex == 0) {
    if (m_uUMIDINoteNumber >= m_Osc1NoteRegion.uMIDINoteLow &&
        m_uUMIDINoteNumber <= m_Osc1NoteRegion.uMIDINoteHigh)
      return true;
  } else if (noscIndex == 1) {
    if (m_uUMIDINoteNumber >= m_Osc2NoteRegion.uMIDINoteLow &&
        m_uUMIDINoteNumber <= m_Osc2NoteRegion.uMIDINoteHigh)
      return true;
  }
  return false;
}

// --- test to see if in Osc Velocity Range
inline bool inoscVelocityRange(int noscIndex) {
  if (noscIndex == 0) {
    if (m_uUMIDVelocity >= m_Osc1NoteRegion.uMIDIVelocityLow &&
        m_uUMIDVelocity <= m_Osc1NoteRegion.uMIDIVelocityHigh)
      return true;
  }
  return false;
}
```

```cpp
inline bool isOscSingleCycleSample(UINT uosc)
{
    // Decode Oscillator Index
    CSampleOscillator* p0sc = NULL;
    switch(uosc) {
        case 0:
            p0sc = (CSampleOscillator*)m_p0sc1; break;
        case 1:
            p0sc = (CSampleOscillator*)m_p0sc2; break;
        case 2:
            p0sc = (CSampleOscillator*)m_0sc3; break;
        case 3:
            p0sc = (CSampleOscillator*)m_p0sc4; break;
        default:
            break;
    }

    // --- return the flag
    if(p0sc) {
        return p0sc->m_bSingleCycleSample;
    }
    return false;
}
```
