```cpp
inline void update() {

if(m_pModulationMatrix || !m_bOutputEG) {
    // --- 使用调制矩阵时，当值为 0 时没有调制，
    // 所以在这里我们计算 (1.0 - dScale) 作为比较
    if(m_uSourceEGAttackScaling != DEST_NONE && m_dAttackTimeScalar == 1.0) {
        double dScale = m_pModulationMatrix->m_dDestinations;
        [m_uSourceEGAttackScaling];
    }
}
```
