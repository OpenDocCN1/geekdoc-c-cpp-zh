    // 0 <= t < dInc

        // --- calculate distance
        double t = dModulo/dInc;

        // --- calculate residual
        dPolyBLEP = dHeight*(2.0*t - t*t - 1.0);
    }

    if(!bRisingEdge)
        dPolyBLEP *= -1.0;
