void CSampleOscillator::update()
{
    // --- this calculates the modulated pitch
    COscillator::update();

    // --- Global Parameters
    if(m_pGlobal0scParams) {
        m_uLoopMode = m_pGlobal0scParams->uLoopMode;
    }

    // --- check sample
    CWaveData* pSample = m_pWaveSample;
    if(!pSample) {
        pSample = m_pWaveSamplePtrArray[m_uMIDINoteNumber];
    }

    if(!pSample) return;

    // --- pitch-less is simple
    if(m_bPitchlessSample) {
        m_dInc = 1.0;
        return;
    }
    // --- get unity frequency
    double dUnityityFrequency = m_bSingleCycleSample ?
    (m_dSampleRate/pSample->uSampleCount/pSample->m_uNumChannels) :
    midFreqTable[pSample->m_uMIDINote];

    // equivalent length
    double dLength = m_dSampleRate/dUnityFrequency;

    // calculate increment
    // inc - L(fo/fs) = L(inc)
    m_dInc *= dLength;
}
