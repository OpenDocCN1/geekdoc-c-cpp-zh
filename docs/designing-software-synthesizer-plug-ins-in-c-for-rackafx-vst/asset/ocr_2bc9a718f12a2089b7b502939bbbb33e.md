void cOscillator::reset()
{
    // -- Pitched moduli, wavetables start at 0.0
    m_dModulo = 0.0;

    // -- needed for triangle algorithm, DPW
    m_dDPWSquareModulator = -1.0;

    // -- flush DPW registers
    m_dDPW_z1 = 0.0;

    // -- for random stuff
    srand(time(NULL));
    m_uPNRegister = rand();
    m_nrSHCounter = -1; // flag for reset condition
    m_dRSHValueValue = 0.0;

    // -- modulation variables
    m_dAmpMod = 1.0; // note default to 1 to avoid silent osc
    m_dDPWMod = 0.0;
    m_dPitchBendMod = 0.0;
}
