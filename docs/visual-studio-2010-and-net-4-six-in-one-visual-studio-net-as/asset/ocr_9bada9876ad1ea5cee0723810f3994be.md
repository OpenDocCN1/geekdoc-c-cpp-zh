```
behaviorConfiguration="*TestBehavior"
<host>
  <baseAddresses>
    <add baseAddress="http://localhost:1809/TestService/"
  </baseAddresses>
</host>
<endpoint
  address=""
  binding="wsHttpBinding name="TestRoutingEndpoint"
  contract="System.ServiceModel.Routing.IRequestReplyRouter"/>
<services>
  <service>
</services>
<behaviors>
  <serviceBehaviors>
    <behavior name="TestBehavior*">
    <serviceMetadata getHttpGetEnabled="True"/>
    <serviceDebug includeDebugDetailInFaults="True"/>
    <routing routingTableName="ServiceRouterTable"/>
    <!--The Router Table Contains Entries for services-->
    </behavior>
  </serviceBehaviors>
</behaviors>
<!---Define Services Here-->
<client>
<endpoint
  name="WroxService" binding="wsHttpBinding"
  address="http://localhost:2709/Services/WroxService.svc"
  contract="*"/>
</endpoint>
</client>
<!---Routing Definition-->
<routing>
<!---Filter For Detecting Messages Headers to redirect-->
<filters>
  <filter name="TestFilter" filterType="MatchAll"/>
</filters>
<!---Define Routing Table, this will Map the service with Filter-->
<routingTables>
  <table name="ServiceRouterTable">
    <entries>
      <add filterName="TestFilter" endpointName="WroxService"/>
    </entries>
  </table>
</routingTables>
</routing>
</system.service.Model>
</configuration>
```

Note that the service shown in the previous code snippet is hosted at http://localhost:1809/TestService. It uses wsHttpBinding.
