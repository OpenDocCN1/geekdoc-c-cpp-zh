```
using System;
using System.Data.OleDb;
using System.Data;
using System.Web.UI.DataVisualization.Charting;
```

```
namespace PetShopCharts
```

```
public partial class DataBindCrossTable : System.Web.UI.Page
{
    protected void Page_Load(object sender, EventArgs e)
    {
        var connection = Helpers.Utility.GetDataConnection(this);
        var commandStr = "SELECT Period, Total, Category from SalesByYearAndPetKind";
        var command = new OleDbConnection(commandStr, connection);
        command.Open();
        var reader = command.ExecuteReader(CommandBehavior.CloseConnection);
        Chart1.DataBindCrossTable(
            "Category",
            "Period",
            "Total",
            "*"
        );
        reader.Close();
        connection.Close();
        foreach (var series in Chart1.Series)
        {
            series.ChartType = SeriesType.Spline;
            series.BorderWidth = 4;
        }
    }
}
```

The page uses the standard pattern to set up an Access database and prepares a reader object to execute a query. The GetDataConnection method of the Helpers.Utility static class is responsible for preparing this connection, and its source code is:

```
public static OleDbConnection GetDataConnection(Page page)
{
    string fileName = page.MapPath(".*") + "\_App_Data\PetShopData.mdb";
    string connectionString = "PROVIDER=Microsoft.Jet.OLEDB.4.0;Data Source="+
    + fileName;
    return new OleDbConnection(connectionString);
}
```
