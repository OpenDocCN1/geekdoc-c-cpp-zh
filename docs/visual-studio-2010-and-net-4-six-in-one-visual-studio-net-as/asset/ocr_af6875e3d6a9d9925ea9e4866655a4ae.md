let rec MatchTokenNoWS s pattern =
    match ([WHITESPACE|_]) with s
    | Some (_, rest) ->
        rest
        |> matchToken pattern
    | None ->
        s
        |> matchToken pattern

let MatchSymbol s pattern =
    pattern
    |> MatchToken s (fun (_, rest) -> rest |> Some)
let rec ([Star|_]) f acc s =
    match f s
    | Some (_, rest) ->
        ([Star|_]) f (rest :: acc)
    | None ->
        (acc |> List.rev, s) |> Some
