function onCommand(sender, e) {
  switch (e.get_commandName()) {
    case "Select":

      return;
    case "ShowFlag":
      var country = e.get_commandArgument();
      var img = $(get("ImageFlag"));
      img.src = String.format("/images/flag_{0}.gif", country);

  }
}
