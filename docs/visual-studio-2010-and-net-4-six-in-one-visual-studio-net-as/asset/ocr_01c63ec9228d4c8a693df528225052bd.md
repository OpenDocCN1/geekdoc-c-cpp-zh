protected void ExpertiseList_SelectedIndexChanged(object sender, EventArgs e)
{
    var experts = new Dictionary<int, Expert>();
    ExpertiseList.Items.Clear();
    experts.Clear();
    var da = new DataAccess(Server);
    foreach (var item in da.GetList(ExpertiseList.SelectedValue))
    {
        ExpertiseList.Items.Add(new ListItem(item.Name, item.Id.ToString()));
        experts.Add(item.Id, item);
    }
    Session("Experts") = experts;
    if (ExpertiseList.Items.Count > 0)
    {
        ExpertiseList.SelectedIndex = 0;
        ExpertiseList_SelectedIndexChanged(ExpertiseList, EventArgs.Empty);
    }
}

protected void ExpertiseList_SelectedIndexChanged(object sender, EventArgs e)
{
    var experts = Session("Experts") as Dictionary<int, Expert>;
    Expert expert;
    if (experts.TryGetValue(Int32.Parse(ExpertiseList.SelectedValue),
        out expert))
    {
        Description.Text = expert.Description;
        Photo.ImageURL = "/Photos/"" + expert.Photo + ".png";
        Photo.Visible = true;
    }
}

# Improving FindAnExpertDemo with Ajax Control Toolkit

There are a few issues with the FindAnExpertDemo solution. This is the first time you select an item from the expertise or expert drop-downs, a pageback occurs and the page flickers. The second is that you do not have a similar to "Select an expertise" or "Select an expert" in the drop-downs.

You can use the Ajax Control Toolkit to solve these issues. The Toolkit contains a control named CascadingDropDown that can be attached to ASP.NET DropDownList to get automatic population of a set of DropList controls. CascadingDropDown also enables a common scenario where the contents of one list depends on selection of another list, and does so without having to embed the entire data set in the page, or transfer it to the client.
