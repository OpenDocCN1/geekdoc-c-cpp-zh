```
using System;
using System.Windows.Forms;
using System.Windows.Forms.DataVisualization.Charting;

namespace ProcessInfoChart
{
    public partial class MainForm : Form
    {
        public MainForm()
        {
            InitializeComponent();
        }

        private void MainForm_Load(object sender, EventArgs e)
        {
            var revnSeries = chart.Series("Revenue");
            var costSeries = chart.Series("Cost");
            for (int i = 0; i < 5; i++)
            {
                revnSeries.Points.Add(new DataPoint(i, i + 1));
                costSeries.Points.Add(new DataPoint(i, 5-i));
            }
        }
    }
}
```



You can access a series through the Series property of the chart. You can then index by the series name or its position. The name is used. The data points of the series can be manipulated through the Points property. In the list, new DataPoint instances are added with the XValue and YValues properties specified in the constructor.
