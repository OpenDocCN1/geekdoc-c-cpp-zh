// R2 ()
DBL LinReg::R2(const V2DD& v)\{
    if(v[0].size() < 3) m_utils.ErrorChk("More than one time series required!");

    UINT n = v.size();
    DBL xy, xx, yy; xy = (x-xbar)*(y-ybar), xx = (x-xbar)^2, yy = (y-ybar)^2

    V1DD avg = m_stats.Mean(v); // Calculate Means using the Stats class

    // Calculate R-Squared using Eqn. (3.24)
    for(UINT j=1; j<v[0].size(); j++)
    {
        xy = 0.0, xx = 0.0, yy = 0.0;
        for(UINT i=0; i<n; i++)
        {
            xy += (v[i][j-1] - avg[j-2]) * (v[i][j] - avg[j]);
            xx += (pow(v[i][j-1] - avg[j-2], 2);
            yy += (pow(v[i][j] - avg[j-1], 2);
        }
    }

    return (pow(xy / sqrt(xx * yy), 2);
}
