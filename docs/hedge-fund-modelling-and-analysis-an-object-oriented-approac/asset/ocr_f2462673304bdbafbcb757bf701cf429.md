// TR()
V1DD PMetrics::TR(V2DD& v, const DBL& r)
{
    UINT n = v.size();
n    DBL tmp; // Declare a temporary holding variable
    V1DD tr; // Declare a 1D vector for the Treynor ratio

    V1DD mean = Mean(v, 12); // Inherit Mean from the Stats class
    V1DD slope = GetSlope(v); // Call GetSlope() member function

    // Calculate Treynor ratio using Eqn. (5.14)
    for (UINT j=1; j<v[0].size()-1; j++) {
        tmp = 0.0;
        for (UINT i=0; i<n; i++) {
            tmp = (mean[j - 1] - r) / slope[j - 1];
        }
        tr.push_back(tmp);
    }

    return tr;
}
