Matrix& Matrix::operator=(const Matrix& a)
{
rows = a.rows;
cols = a.cols;
p = new DBL*[a.rows];

for (UINT r=0; r<a.rows; r++)
{
    p[r] = new DBL[a.cols];
}

// Copy values from Matrix a
for (UINT c=0; c<a.cols; c++)
{
    p[r][c] = a.p[r][c];
}

return *this;
}

// Note: indices are one-based, not zero-based
double& Matrix::operator()(const int r, const int c)
{
if (p != NULL && r >= 0 && r <= rows && c > 0 && c <= cols)
{
    return p[r-1][c-1];
}
else
{
    cout << "\nSubscript out of range!\n";
}
