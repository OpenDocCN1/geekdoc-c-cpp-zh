V1DD Stats::Corr(const V2DD& v)
{
    if(v[0].size() < 3) m_utils.ErrorChk("More than one time series
required!");

    UINT n = v.size();
    DBL tmp; // Declare a temporary holding variable

    V1DD mean = Mean(v); // Declare a 1D vector of dou-
dbles for the Mean
    V1DD std = StdDev(v, 's'); // Declare a 1D vector of doubles
    for the St. Dev.
    V1DD corr; // Declare a 1D vector of doubles for the Correl.

    // Calculate correlation using Eqn (3.19)
    for(UINT k=0; k<mean.size(); k++)
    {
        for(UINT j=1; j<v[0].size(); j++)
        {
            tmp = 0.0;
            for(UINT i=0; i<n; i++)
            {
                tmp += ((v[i][j] - mean[j-1]) * (v[i][k] - 
mean[k])) / (std[j-1] * std[k]);
            }
            tmp /= (n-1);
        }
        corr.push_back(tmp);
    }

    return corr;
}
