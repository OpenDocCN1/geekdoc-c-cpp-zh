wT(1, i) = 1 / (DBL)n; // Equal weights

// Covariance matrix
V1DD cov = Cov(v);

// VCV matrix
int k = 0; // Covariance offset
for (UINT i=1; i<=n; i++)
{
  for (UINT j=1; j<=n; j++)
{
    VCV(i, j) = cov[j+k-1];
    if (i == j)
      VCV(i, j) *= 12; // Annualise variance
  }
k+=10;
}

return wT * VCV * w;
