// Slope()
DBL LinReg::Slope(const V2DD& v)
{
    if(v.size() < 3) m_utils.ErrorChk("More than one time series required!");

    UINT n = v.size();
    DBL xy, xx; // xy = (x-xbar)*(y-ybar), xx = (x-xbar)^2

    VIDD avg = m_stats.Mean(v); // Calculate Means using the Stats class

    // Calculate Slope using Egn. (3.223)
    for(UINT j=1; j[0].size(); j++)
    {
        xy = 0.0, xx = 0.0;
        for(UINT i=0; i<n; i++)
        {
            xy += (v[i][j-1] - avg[j-2]) * (v[i][j] - avg[j-1]);
            xx += (pow(v[i][j-1] - avg[j-2], 2));
        }
    }

    return (xy / xx);
}
