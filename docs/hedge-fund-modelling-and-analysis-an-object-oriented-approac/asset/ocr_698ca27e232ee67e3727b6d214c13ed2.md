V1DD PMetrics::Sharpe(const V2DD& v, const DBL& r)
{
    UINT n = v.size();
    DBL tmp; // Declare a temporary holding variable

    V1DD sharpe; // Declare a 1D vector of doubles for the Sharpe ratio
    V1DD mean = Mean(v, 12); // Inherit Mean from the Stats class
    V1DD std = StDev('p', 12); // Inherit St. Dev. from the Stats class

    // Calculate Sharpe ratio using Eqn. (5.4)
    for(UINT j=1; j<v[0].size(); j++) {
        tmp = 0.0;
        for (UINT i=0; i<n; i++) {
            tmp = (mean[j-1] - r) / std[j-1];
        }
        sharpe.push_back(tmp);
    }

    return sharpe;
}
