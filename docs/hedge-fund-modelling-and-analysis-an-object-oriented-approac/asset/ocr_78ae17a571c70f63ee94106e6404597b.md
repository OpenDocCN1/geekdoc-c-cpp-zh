V1DD Stats::XSKurt(const V2DD& v)
{
    UINT n = v.size();
    DBL tmp; // Declare a temporary holding variable

    V1DD xskurt;
    V1DD avg = Mean(v);
    V1DD std = StdDev(v, 's');

    // Calculate XS Kurt. using Eqn. (3.15)
    for (UINT j=1; j<v[0].size(); j++)
{
        tmp = 0.0;
        for (UINT i=0; i<v.size(); i++)
        {
            tmp += pow((v[i][j] - avg[j-1]) / std[j-1]), 4);
        }
tmp *= (DBL)n * (n+1) / ((n-2) * (n-3));ntmp = 3 * pow((DBL) (n-1), 2) / ((n-2) * (n-3));
xskurt.push_back(tmp);

return xskurt;
}
