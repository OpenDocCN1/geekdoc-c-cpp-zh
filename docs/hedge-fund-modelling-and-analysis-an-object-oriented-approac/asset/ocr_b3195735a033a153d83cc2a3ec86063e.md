V1DD Stats::Mean(const V2DD& v)
{
  UINT n = v.size();
  double tmp; // Declare a temporary holding variable

 V1DD mean; // Declare a 1D vector of doubles for the Mean

// Calculate mean using Equn. (3.10)
for(UINT j=1; j<v[0].size(); j++)
{
  tmp = 0.0;
  for(UINT i=0; i<n; i++)
  {
    tmp += v[i][j];
  }
  tmp /= n;
  //std::cout << " " << j << " << " << j << " << << " << tmp << "\n";
  mean.push_back(tmp);
}

return mean;
}
