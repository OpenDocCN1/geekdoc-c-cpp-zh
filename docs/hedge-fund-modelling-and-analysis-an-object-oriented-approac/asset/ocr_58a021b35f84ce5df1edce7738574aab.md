Matrix Optimise::PRet(const V2DD& v) {
  UINT n = v[0].size()-1;

  // Declare WT and R matrices
  Matrix WT = Matrix(1, n);
  Matrix R = Matrix(n, 1);

  // Transpose weights
  for (UINT i=1; i<=n; i++)
    WT(1, i) = 1 / (DBL)n; // Equal weights

  // Mean returns
  1DD r = Mean(v, 12);

  // R matrix
  for (UINT i=1; i<=n; i++)
    R(i, 1) = r[i-1];

  return WT * R;
}
