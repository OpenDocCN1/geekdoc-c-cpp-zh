void Utils::PrintFile(const V22D& v, const V1DS& s, const int& n)
{
    cout << "\n";
    for (UINT i=0; i<s.size(); i++) {
        cout << std::left << setw(6) << s[i];
        cout << "\n";
    }
    for (UINT i=0; i<n; i++) {
        for (UINT j=0; j<v[0].size(); j++) {
            cout << right << setw(6) << v[i][j];
