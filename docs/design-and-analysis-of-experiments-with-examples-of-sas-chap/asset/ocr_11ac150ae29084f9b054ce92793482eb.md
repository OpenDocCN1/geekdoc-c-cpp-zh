```
data conf;
do x4=-1 to 1 by 2;
   do x3 = -1 to 1 by 2;
      do x2 = -1 to 1 by 2;
         do x1 = -1 to 1 by 2;
            A=x1; B=x2; C=x3; D=x4;
            ABD=A*B*D; BCD=B*C*D;
            if ABD=-1 and BCD=-1 then Block=1;
            if ABD = 1 and BCD=-1 then Block=2;
            if ABD=-1 and BCD = 1 then Block=3;
            if ABD = 1 and BCD = 1 then Block=4;
            output;
         end;
      end;
   end;
end;
keep A B C D Block;
run;
proc sort; by Block;
proc print; run;
```
