// do the division if legal
public double DivideValues(double a, double b)
{
	if (b == 0)
	{
		MyCustomException e = 
		new MyCustomException(
			"Can't be divided by zero, result is Infinity.");
		Console.WriteLine(e);
	}

	if (a == 0)
	{
		// create a custom exception instance
		MyCustomException e = 
		new MyCustomException(
			"Can't have a zero divisor");
		Console.WriteLine(e);
	}

	return a / b;
}
