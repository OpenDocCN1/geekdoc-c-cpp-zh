public void Show()
{
    string str1 = "";
    Console.Write("Please enter a string : ");
    str1 = Console.ReadLine();
    Regex reg = new Regex(" |, |");
    StringBuilder sb = new StringBuilder();
    int i = 1;
    for(string subString as reg.Split(str1))
    {
        sb.AppendFormat("{{0}}: {{1}}\n", i++, subString);
    }
    Console.WriteLine("{{0}}", +sb);
}
}
