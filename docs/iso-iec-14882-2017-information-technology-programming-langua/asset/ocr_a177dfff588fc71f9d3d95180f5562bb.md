# ISO/IEC 14882:2017(E)

# 29.7.2 Class template varray overview

# 29.7.2.1 Class template varray overview

```
namespace std {
    template<class T> varray {
    public:
        using value_type = T;

        // 29.7.2.2, construct/destroy
        varray(size_t);
        explicit varray(const T, size_t);
        varray(const T, size_t);
        varray(const varray&);
        varray(varray&&) noexcept;
        varray(const slice_array<T>);
        varray(const glice_array<T>);
        varray(const mask_array<T>);
        varray(const indirect_array<T>);
        varray(initializer_list<T>);
        ~varray();

        // 29.7.2.3, assignment
        varray& operator=(const varray&);
        varray& operator=(varray&&) noexcept;
        varray& operator=(initializer_list<T>);
        varray& operator=(const T&);
        varray& operator=(const slice_array<T>);
        varray& operator=(const glice_array<T>);
        varray& operator=(const mask_array<T>);
        varray& operator=(const indirect_array<T>);

        // 29.7.2.4, element access
        const T& operator[](size_t) const;
        T& operator[](size_t);
        varray operator[](slice) const;
        slice_array<T> operator[](const slice) const;
        glice_array<T> operator[](const glice_array<T>) const;
        varray operator[](const varray<bool>) const;
        mask_array<T> operator[](const varray<bool>) const;
        varray operator[](const varray<size_t>) const;
        indirect_array<T> operator[](const varray<size_t>) const;

        // 29.7.2.6, unary operators
        varray& operator++() const;
        varray& operator--() const;
        varray operator-() const;
        varray operator!() const;
        varray operator bool() const;

        // 29.7.2.7, compound assignment
        varray& operator+=(const T&);
        varray& operator/=(const T&);
```
