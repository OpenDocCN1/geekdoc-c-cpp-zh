# ISO/IEC 14882:2017(E)

```
struct B : A { using A::A; };

struct C1 : B { using B::B; };
struct C2 : B { using B::B; };

struct D1 : C1, C2 {
    using C1::C1;
    using C2::C2;
};

struct V1 : virtual B { using B::B; };
struct V2 : virtual B { using B::B; };

struct D2 : V1, V2 {
    using V1::V1;
    using V2::V2;
};

d D1(0());
d D2(0());
// ill-formed: ambiguous
// OK: initializes virtual B base class, which initializes the A base class
// then initializes the V1 and V2 base classes as if by a default constructor

struct M { M(int); };
struct N : M { using M::M; };
struct O : M {};
struct P : N, O { using N::N; using O::O; };
// OK use M(0) to initialize N's base class,
// use M(O) to initialize O's base class
```

When an object is initialized by an inherited constructor, initialization of the object is complete when the initialization of all subobjects is complete.

# 15.7 Construction and destruction

For an object with a non-trivial constructor, referring to any non-static member or base class of the object before the constructor begins execution results in undefined behavior. For an object with a non-trivial destructor, referring to any non-static member or base class of the object after the destructor finishes execution results in undefined behavior. | Example:

```
struct X { int i; };
struct Y { Y() {} }; // non-trivial
struct A { int a; };
struct B : public A { int j; Y y; }; // non-trivial

extern B bobj;
B* p = &bobj; // OK
int* pi = &bobj.i; // undefined, refers to base class member
int* p2 = &bobj.y; // undefined, refers to member's member

A* pa = &bobj; // undefined, upcast to a base class type
 bobj; // definition of bobj

extern X x;
int* p3 = &x.i; // OK, X is a trivial class
X x();
```
