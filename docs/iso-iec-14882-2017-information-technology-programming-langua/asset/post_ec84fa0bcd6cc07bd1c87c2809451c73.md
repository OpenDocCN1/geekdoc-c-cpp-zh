## 26.5.6.2 unordered_set constructors [unord.set.constr]

```cpp
unordered_set() : unordered_set(size_type(see below)) {
explicit unordered_set(size_type n,
    const hasher& h = hasher(),
    const key_eql eql = key_equal(),
    const allocator_type& a = allocator_type());
```

- 1 Effects: Constructs an unordered_set using the specified hash function, key predicate, and allocator, and at least n buckets. For the default constructor, the number of buckets is implementation-defined. max_load_factor() returns 1.0.
- 2 Complexity: Constant.

```cpp
template <class InputIterator>
unordered_set(InputIterator first,
    size_type n = see below,
    const hasher& h = hasher(),
    const key_eql eql = key_equal(),
    const allocator_type& a = allocator_type());
unordered_set(initializer_list<value_type> il,
    size_type n = see below,
    const hasher& h = hasher(),
    const key_eql eql = key_equal(),
    const allocator_type& a = allocator_type());
```

- 3 Effects: Constructs an unordered_set using the specified hash function, key predicate, and at least n buckets. If n is not specified, the number of buckets is implementation-defined. Then inserts elements from the range [first, last) of the first form, or from the range [il.begin(), il.end()) of the second form, max_load_factor() returns 1.0.
- 4 Complexity: Average linear, worst-case quadratic.

```cpp
template <class Key, class Hash, class Pred, class Alloc>
void swap(unordered_set<Key, Hash, Pred, Alloc>& x, unordered_set<Key, Hash, Pred, Alloc>& y) noexcept(x.swap(y));
```

- 1 Effects: By x.swap(y).

## 26.5.7 Class template unordered_multiset [ord.multiset.overview]

### 26.5.7.1 Class unordered_multiset

unordered_multiset is an associative container that supports equivalent keys (instance of unordered_multiset may contain multiple copies of the same value) and where each element is the element itself. The unordered_multiset supports forward iterators. unordered_multiset satisfies all of the requirements of a container, an associative container, and an allocator-aware container. It provides the operations described in the preceding tables for equivalent keys: for a Key type, the iterator and const_iterator types are unspecified whether they are the same type. This section only describes operations on unordered_multiset that are not described in the tables or for which there is additional information.

```cpp
namespace std {
    template <class Key,
    class Hash = hash<Key>,
    class Pred = equal_to<Key>,
    class Alloc = allocator<Key>>
    class unordered_multiset {
    public:
        // types:
        using key_type = Key;
        using value_type = Key;
        using hasher = Hash;
        using key_equal = Pred;
        using allocator_type = Alloc;
        using pointer = typename allocator_traits<Allocator>::pointer;
        using const_pointer = typename allocator_traits<Allocator>::const_pointer;
        using reference = value_type&;
        using const_reference = value_type const&;
        using size_type = size_type;
        using implementation_type = __;

        // iterators:
        using iterator = forward_iterator;
        using const_iterator = forward_iterator;
        using local_iterator = local_iterator;
        using const_local_iterator = local_iterator;
        using node_type = unspecified;

        // 26.5.7.2 construct/destroy
        unordered_multiset(size_type n = 0);
        explicit unordered_multiset(size_type n, const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());

        template <class InputIterator>
        unordered_multiset(InputIterator first, InputIterator last, 
            size_type n = see below,
            const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());

        unordered_multiset(const unordered_multiset& s);
        explicit unordered_multiset(const Allocator& a);
        unordered_multiset(const unordered_multiset& s, const Allocator& a);
        unordered_multiset(const unordered_multiset& s, const Allocator& a);
        unordered_multiset(const initializer_list<value_type>& il,
            size_type n = see below,
            const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());

        unordered_multiset(size_type n, const allocator_type& a)
            : unordered_multiset(n, hasher(), key_equal(), a) {}
        unordered_multiset(size_type n, const hasher& h, const allocator_type& a)
            : unordered_multiset(n, h, key_equal(), a) {}

        template <class InputIterator>
        unordered_multiset(InputIterator first, InputIterator last, 
            size_type n, const allocator_type& a, const key_equal& eq)
            : unordered_multiset(first, last, n, hasher(), eq, a) {}

        unordered_multiset& operator=(const unordered_multiset& other)
            noexcept(allocator, traits<allocator>::is_always_equal<value> &&
            is_nothrow_move_assignable<Pred> &&);
        unordered_multiset& operator=(initializer_list<value_type>);
        allocator_type get_allocator() const noexcept;

        // iterators:
        iterator begin() const noexcept;
        const_iterator begin() const noexcept;
        iterator end() const noexcept;
        const_iterator cbegin() const noexcept;
        const_iterator cend() const noexcept;

        // capacity:
        bool empty() const noexcept;
        size_type size() const noexcept;
        size_type max() const noexcept;

        // modifiers:
        template <class... Args> iterator insert(Args&&... args);
        template <class... Args> iterator emplace_hint(const_iterator position, Args&&... args);
        iterator insert(const value_type& obj);
        iterator insert(const_iterator hint, const value_type& obj);
        iterator insert(const_iterator hint, value_type&& obj);
        template <class InputIterator> void insert(InputIterator first, InputIterator last);
        iterator insert(initializer_list<value_type> init);

        node_type extract(iterator pos);
        node_type extract(const key_type& x);
        iterator insert(node_type node);
        iterator insert(const_iterator hint, node_type node);

        iterator erase(iterator pos);
        iterator erase(const key_type& x);
        iterator erase(const_iterator hint);
        template <class InputIterator> void erase(InputIterator first, InputIterator last);
        iterator erase(initializer_list<value_type> init);

        void clear();

        // bucket:
        size_type bucket_count() const noexcept;
        size_type max_bucket_count(size_type n) const noexcept;
        size_type count(size_type n) const;
        local_iterator begin(size_type n);
        const_local_iterator begin(size_type n) const;
        const_local_iterator cbegin(size_type n) const;
        const_local_iterator cend(size_type n) const;

        // policy:
        float load_factor() const noexcept;
        float max_load_factor() const noexcept;
        void max_load_factor(float n);
        void rehash(size_type n);
        void reserve(size_type n);

        void swap(unordered_multiset& other) noexcept(allocator, traits<allocator>::is_always_equal<value> &&
        is_nothrow_move_assignable<Pred> &&);

        // observers:
        hasher hash_function() const;
        key_equal eq() const;

        // set operations:
        const_iterator find(const key_type& k);
        const_iterator count(const key_type& k);
        size_type count(const key_type& k);
        pair<iterator, iterator> range_find(const key_type& k);
        pair<const_iterator, const_iterator> range_equal(const key_type& k);
    };
}
```
