## 25.4.7 The message retrieval category

Class messages<charT> implements retrieval of strings from message catalogs.

### 25.4.7.1 Class template messages

```cpp
namespace std {
    class messages_base {
    public:
        using catalog = unsigned signed integer type;
    };

    template<class charT>
    class messages : public locale::facet, messages_base {
    public:
        using char_type = charT;
        using string_type = basic_string<charT>;
    };
```

std::istream& operator>>(std::istream& s, Date& d) {
    using namespace std;
    istream::sentry ceros(s);
    if (ceros) {
        ios_base::state err = goodbit;
        struct tm t;
        use_time_get<char*>(s.getloc()).get_date(s, 0, s, err, &t);

        else s.setstate(err);
    }
    return s;
}
