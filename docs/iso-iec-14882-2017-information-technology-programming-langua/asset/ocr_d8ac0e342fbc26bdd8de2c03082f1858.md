# ISO/IEC 14882:2017(E)

```
namespace std {
    template <class Key>
    class Hash = hash<Key>,
    class Pred = equal_to<Key>>
    class Allocator = allocator<Key>
    class unordered_multiset {
    public:
        // types:
        using key_type = Key;
        using value_type = Key;
        using hasher = Hash;
        using key_equal = Pred;
        using allocator_type = Allocator;
        using pointer = typename allocator_traits<Allocator>::pointer;
        using const_pointer = typename allocator_traits<Allocator>::const_pointer;
        using reference = value_type&
        using const_reference = const value_type&
        using size_type = implementation_size; // sec 36.2
        using implementation_type = implementation_defined; // sec 36.2

        using iterator = implementation_iterator; // sec 36.2
        using const_iterator = implementation_iterator; // sec 36.2
        using local_iterator = implementation_local; // sec 36.2
        using const_local_iterator = implementation_defined; // sec 36.2
        using node_type = unspecified;
        // 36.5.7. construct/destroy
        unordered_multiset(size_type n = 0);
        explicit unordered_multiset(size_type n, const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());

        template <class InputIterator>
        unordered_multiset(InputIterator f, InputIterator last, 
            size_type n = see below,
            const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());
        unordered_multiset(const unordered_multiset& s);
        explicit unordered_multiset(const Allocator& a);
        unordered_multiset(const unordered_multiset& s, const Allocator& a);
        unordered_multiset(const unordered_multiset& s, const Allocator& a);
        unordered_multiset(const initializer_list<value_type>& i,
            size_type n = see below,
            const hasher& h = hasher(),
            const key_equal& eq = key_equal(),
            const allocator_type& a = allocator_type());
        unordered_multiset(size_type n, const allocator_type& a)
            : unordered_multiset(n, hasher(), key_equal(), a) {}
        unordered_multiset(size_type n, const hasher& h, const allocator_type& a)
            : unordered_multiset(n, h, key_equal(), a) {}
        template <class InputIterator>
        unordered_multiset(InputIterator f, InputIterator last, size_type n, const allocator_type& a, const key_equal& eq)
            : unordered_multiset(f, l, n, hasher(), eq, a) {}
};
```
