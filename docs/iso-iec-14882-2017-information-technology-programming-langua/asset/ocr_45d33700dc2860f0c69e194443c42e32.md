# ISO/IEC 14882:2017(E)

```
// 23.2.3, swap
template <class T>
void swap(T&& a, T&& b) noexcept(see below);
template <class T, size_t N>
void swap(T&&[N], T&&[N]) noexcept(is_nothrow_swappable_v<T>);
```

```
// 23.2.4, exchange
template <class T, class U = T>
exchange(T obj, U& new_val);
```

```
// 23.2.5, forward/move
template <class T>
constexpr T&& forward(remove_reference_t<T>&) noexcept;
template <class T>
constexpr T&& forward(remove_reference_t<T>&&) noexcept;
template <class T>
constexpr move(remove_reference_t<T>&&) noexcept;
template <class T>
constexpr conditional_t<
    !is_nothrow_constructible_v<T> && is_copy_constructible_v<T>, const T&, T&&>
move_if_noexcept(T&&) noexcept;
```

```
// 23.2.6, as
template <class T>
constexpr add_const<T>& as(const T&&) noexcept;
template <class T>
void as_const(const T&&) = delete;
```

```
// 23.2.7, declval
_value_reference<T> declval() noexcept; // as unvaluated operand
```

```
// 23.2.8, compile-time sequences
template<class T, T...>
struct integer_sequence;
template<size_t... I...>
using index_sequence = integer_sequence<size_t, I...>;
ntemplate<class T, size_t N>
using make_integer_sequence = integer_sequence<T, see below>
template<size_t N>
using make_index_sequence = make_integer_sequence<size_t, N>;
```

```
template<class T>
using index_sequence_for = make_index_sequence<sizeof(T)>(T>;
```
