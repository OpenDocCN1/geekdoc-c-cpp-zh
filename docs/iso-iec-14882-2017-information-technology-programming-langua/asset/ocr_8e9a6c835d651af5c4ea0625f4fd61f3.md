std::istream& operator>>(std::istream& s, Date& d) {
    using namespace std;
    istream::sentry ceros(s);
    if (ceros) {
        ios_base::state err = goodbit;
        struct tm t;
        use_time_get = get<char*>(s.getloc()).get_date(s, 0, s, err, &t);
        if (!err) d = Date(t.tm_mday, t.tm_mon + 1, t.tm_year + 1900);
        else s.setstate(err);
    }
    return s;
}
