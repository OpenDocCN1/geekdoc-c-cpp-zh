namespace std {
    template <class T, class Container = deque<T>> class queue;

        class Compare = less<typename Container::value_type>
    class priority_queue;

    template <class T, class Container>
    bool operator=(const queue<T>& x, const queue<T, Container>& y);
    template <class T, class Container>
    bool operator=(const queue<T>& x, const queue<T, Container>& y);

    bool operator=(const queue<T>& x, const queue<T, Container>& y);

    bool operator=(const queue<T>& x, const queue<T, Container>& y);
    template <class T, class Container>
    bool operator<(const queue<T>& x, const queue<T, Container>& y);

    void swap(queue<T, Container>& x, queue<T, Container>& y) noexcept(noexcept(x.swap));

# 26.6.4 Class template queue [queue.defn]

# 26.6.4.1

Any sequence container operations front(), back(), push_back() and pop_front() can be used to instantiate queue. In particular, list (26.3.10) and deque (26.3.8) can be used.
