# ISO/IEC 14882:2017(E)

# 30.7.4.6.2 basic_iostream destructor

virtual ~basic_iostream();
Effects: Destroys an object of class basic_iostream.
Remarks: Does not perform any operations on rdbuf.

# 30.7.4.6.3 basic_iostream assign and swap

basic_iostream operator=(basic_iostream&& rhs);
Effects: As if swap(rhs).
void swap(basic_iostream& lhs, traits& rhs);
Effects: Calls basic_iostream<charT, traits>::swap(rhs).
