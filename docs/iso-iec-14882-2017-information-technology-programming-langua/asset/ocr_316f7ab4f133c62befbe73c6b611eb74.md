```
InputIterator2 first, InputIterator2 last);
template<class ExecutionPolicy, class ForwardIterator1, class ForwardIterator2>
mismatch(ExecutionPolicy&& exec, ForwardIterator2 last
ForwardIterator1 first, ForwardIterator1 last,
ForwardIterator2 first, ForwardIterator2 last);
template<class Iterator1, class Iterator2, class BinaryPredicate>
pair<InputIterator1, InputIterator2>
mismatch(InputIterator1 first, InputIterator1 last,
InputIterator2 first, InputIterator2 last,
BinaryPredicate pred);
template<class ExecutionPolicy, class ForwardIterator1, class ForwardIterator2>
mismatch(ExecutionPolicy&& exec, ForwardIterator1 first, ForwardIterator1 last,
ForwardIterator2 first, ForwardIterator2 last);
```

+   1 Remarks: If last2 was not given in the argument list, it denotes first + (last1 – first) below, respectively.

+   2 Returns: A pair of first + n and first2 + n, where n is the smallest integer such that,

```
!(*(first + n) == *(first2 + n)) or
pred(*(first + n), *(first2 + n)) == false;
```

+   or min(last1–first, last2–first) if no such integer exists.

+   3 Complexity: At most min(last1–first, last2–first) applications of the corresponding predicate.
