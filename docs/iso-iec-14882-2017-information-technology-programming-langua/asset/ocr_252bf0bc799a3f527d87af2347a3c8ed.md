# ISO/IEC 14882:2017(E)

# 25.4.1.3 ctype specializations [facet.ctype.special]

```
namespace std {
template <>
    class ctype<char> : public locale::facet, public ctype_base {
public:
    using char* type = char;

    explicit ctype(const mask* tab = nullptr, bool del = false, size_t refs = 0);
    bool is(mask m, const char c) const;
    const* is(const char* low, const char* high, mask* vec) const;
    const* scan(mask m, const char* low, const char* high) const;
    const* scan(const char* not, mask m, const char* low, const char* high) const;

    char toupper(const char c) const;
    const* toupper(const char* low, const char* high) const;
    char tolower(const char c) const;
    const* tolower(const char* low, const char* high) const;

    char widen(const char c) const;
    const* widen(const char* low, const char* high, char* to) const;
    char narrow(const char c, char default) const;
    const* narrow(const char* low, const char* high, char dfault, char* to) const;

    static locale::id id;
    static size_t table_size = implementation-defined;

    const mask table() const noexcept;
    const mask classic_table() const noexcept;

protected:
    ~ctype();
    virtual char toupper(const char c) const;
    virtual const* toupper(const char* low, const char* high) const;
    virtual char tolower(const char c) const;
    virtual const* tolower(const char* low, const char* high) const;

    virtual char widen(const char c) const;
    virtual const* widen(const char* low, const char* high, char* to) const;
    virtual char narrow(const char c, char default) const;
    virtual const* narrow(const char* low, const char* high, char dfault, char* to) const;
};
```

A specialization ctype<char> is provided so the member functions on type char can be implemented inline. The implementation-value of member table_size is at least 26.

# 25.4.1.3.1 ctype<char> destructor [facet.ctype.dtor]

```
~ctype();
```

Effects: If the constructor's first argument was nonzero, and its second argument was true, delete table().

235) Only the (unsigned char and signed char) form is provided. The specialization is in the standard, and not as an implementation detail, as it affects the derivation interface for ctype<char>.
