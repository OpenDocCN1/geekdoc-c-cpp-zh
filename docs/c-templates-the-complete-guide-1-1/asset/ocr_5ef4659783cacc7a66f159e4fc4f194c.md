Note again that hasSizeType uses std::decay to remove the references from the passed types because you can't access a type member from a reference. If you skip that, the traits will always yield false because the second overload and overloaded std::declval() is used.

#include <iostream>

#include <string>

#include <utility>

constexpr auto hasFirst
    = std::is_valid([](auto&& x) -> decltype((void)x.first)
    {
    });

template<typename T>
using HasFirst = decltype(hasFirst(std::declval<T>()));

constexpr auto hasSizeType
    = std::is_valid([](auto&& x)
        -> typename std::decay<decltype(x)>::type::size_type
    {
    });

template<typename T>
using HasSizeType = decltype(hasSizeType(std::declval<T>()));

constexpr auto hasSize
    = std::is_valid([](auto&& x) -> decltype(x.size())
    {
    });

template<typename T>
using HasSize = decltype(hasSize(std::declval<T>()));

int main()
{
    using namespace std;

    cout << "<first: " << HasFirst<int>::value << '\n';

    struct CX {
        using size_type = std::size_t;
    };

    cout << "Has_size: " << HasSizeType<CX>::value << '\n';
    cout << "HasSizeType<int>: " << HasSizeType<int>::value << '\n';
    cout << "HasSize<int>: " << HasSize<int>::value << '\n';
    cout << "HasLastType<int>: " << HasLastType<int>::value << '\n';
    cout << "HasLastType<string, int>: " << HasLastType<string, int>::value << '\n';
}
