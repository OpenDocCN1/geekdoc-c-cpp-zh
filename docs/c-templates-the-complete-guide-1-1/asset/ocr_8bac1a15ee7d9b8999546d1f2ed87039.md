Given `BaseMemberPair`, it is clear that its use makes the implementation of `Optional` more verbose. However, various template library implementations have reported that the performance gains (for the clients of their libraries) do justify the added complexity. #define BASE_MEMBER_PAIR_HPP

class BaseMemberPair : private Base
{
private:
    Member member;

public:

    // constructor
    BaseMemberPair(Base const & b, Member const & m)
      : Base(b), member(m)
    {
    }

    // access base class data via first()
    Base const & first() const
    {
        return static_cast<Base const&>(*this);
    }

    Base& first()
    {
        return static_cast<Base&>(*this);
    }

    // access member data via second()
    Member const& second() const
    {
        return this->member;
    }

    Member& second()
    {
        return this->member;
