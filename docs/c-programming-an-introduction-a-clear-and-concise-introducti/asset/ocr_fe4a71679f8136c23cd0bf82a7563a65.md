A more realistic program will allow the user to enter data into the file using a loop. Here is an example:

```cpp
// CreateFile.cpp : main project file.

#include "stdafx.h"
using namespace System;

#include <iostream>
#include <fstream>
using namespace std;

int main(array<System::String ^> ^args)
{
    ofstream outFile("C:\\data\\grades.txt");
    int grade;
    cout << "Enter a grade (-1 to quit): ";
    cin >> grade;
    while (grade > -1)
    {
        outFile << grade << endl;
        cout << "Enter a grade (-1 to quit): ";
        cin >> grade;
    }
    outFile.close();
    Console::ReadKey();
}
