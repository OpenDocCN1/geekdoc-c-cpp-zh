Now let’s look at an example where we let some of the case clauses “fall through.” You do this if you want the same set of statements to be executed when two or more case clauses are true. Sticking with letter grades, here's a program that determines if a student passes or fails a test, based on the letter grade (A, B, C, and D grades are all passing; only F is a failing grade):

```
```cpp
// Switch.cpp : main project file.

#include "stdafx.h"
using namespace System;

#include <iostream>
using namespace std;

int main(array<System::String ^> ^args)
{
        char letterGrade;
        cout << "Enter your letter grade: ";
        cin >> letterGrade;
        switch (letterGrade)
        {
        case 'A':
        case 'B':
        case 'C':
        case 'D':
                cout << "You passed the test!" << endl;
                break;
        case 'F':
                cout << "Sorry, you failed the test."
                     << endl;
                break;
        default:
                cout << "Enter a valid letter grade."
                     << endl;
        }
        Console::ReadKey();
        return 0;
}
