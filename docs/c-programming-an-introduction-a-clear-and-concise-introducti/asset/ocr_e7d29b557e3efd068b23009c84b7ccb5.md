string empName;
double pay;

public:
    Employee(string name, double payRate)
    {
        empName = name;
        pay = payRate;
    }

    string getName()
    {
        return empName;
    }

    double getPay()
    {
        return pay;
    }

    double grossPay(int hours)
    {
        return pay * hours;
    }
};

class Manager : public Employee
{
protected:
    bool salaried;

public:
    Manager(string name, double payRate,
            bool isSalaried)
        : Employee(name, payRate)
    {
        salaried = isSalaried;
    }

    bool getSalaried()
    {
        return salaried;
    }
