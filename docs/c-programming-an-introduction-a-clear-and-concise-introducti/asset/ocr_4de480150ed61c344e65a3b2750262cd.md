using namespace System;
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main(array<System::String ^> ^args)
{
    ifstream myFile;
    string line;
    myFile.open("example.txt");
    while (!myFile.eof())
    {
        getline(myFile, line);
        cout << line << endl;
    }
    myFile.close();
    Console::ReadKey();
    return 0;
