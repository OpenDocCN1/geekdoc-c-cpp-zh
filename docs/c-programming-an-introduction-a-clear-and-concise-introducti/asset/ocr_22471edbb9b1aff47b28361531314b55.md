int SumVector(vector<int> n)
{
        int sum = 0;
        for(int i = 0; i < n.size(); ++i)
        {
                sum += n[i];
        }
        return sum;
}

void CurveGrades(vector<int> &n)
{
        for(int i = 0; i < n.size(); ++i)
        {
                n[i] += 5; // add 5 to each vector element
        }
}

int main(array<System::String ^> ^args)
{
        vector<int> grades;
        srand(time(NULL));
        for(int i = 1; i <= 10; ++i)
        {
                grades.push_back(rand() % 100 + 1);
        }
        DisplayVector(grades);
        CurveGrades(grades);
        DisplayVector(grades);
        Console::ReadKey();
        return 0;
}
