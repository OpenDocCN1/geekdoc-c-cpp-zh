This program displays the value 8, as expected. Now let's try it with multiples powers of 2. Here is the complete program:

```
```cpp
// Power.cpp : main project file.

#include "stdafx.h"
using namespace System;
#include <iostream>
using namespace std;

int power(int base, int exp)
{
        int number = 1;
        for(int i = 1; i <= exp; ++i) {
                number *= base;
        }
        return number;
}

int main(array<System::String ^> ^args)
{
        for(int i = 1; i <= 10; ++i) {
                cout << i << "\t" << power(2,i) << endl;
        }
        Console::ReadKey();
        return 0;
}
```
