#include "stdafx.h"
using namespace System;

#include <iostream>
#include <string>
using namespace std;

int main(array<System::String ^> ^args)
{
    int product;
    for(int lOp = 1; lOp <= 5; ++leftOp)
    {
        for(int rOp = 1; rOp <= 5; ++rightOp)
        {
            product = lOp * rOp;
            cout << lOp << " x " << rOp << " = "
                 << product << "\t";
        }
        cout << endl;
    }

    return 0;
