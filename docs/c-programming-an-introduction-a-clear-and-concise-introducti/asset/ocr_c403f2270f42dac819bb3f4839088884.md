```cpp
#include "stdafx.h"
using namespace System;

#include <iostream>
#include <vector>
using namespace std;

void DisplayVector(vector<int> n)
{
    for(int i = 0; i < n.size(); ++i)
    {
        cout << n[i] << " ";
    }
    cout << endl;
}

int main(array<System::String ^> ^args)
{
    vector<int> numbers;
    for(int i = 1; i <= 10; ++i)
    {
        numbers.push_back(i);
    }
    DisplayVector(numbers);
    Console::ReadKey();
    return 0;
}
