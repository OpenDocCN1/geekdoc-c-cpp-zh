After selecting a zipped file from the dialog box, the file information is extracted and the `UnzipFiles()` method is called, which is defined in the `ZipSource` class library. The `UnzipFiles()` method takes two arguments: the file path information and the output location. In our case, the output folder is created at the same path from where the user selects the zip file. At first, it opens the zip file for read and returns a `FileStream` object.

+   2. An object of the `ZipFile` class is created that contains the collection of zipped files.
+   3. The unzip file operation creates an output directory in the output location. Using a fixed-size buffer, it uses the `Copy()` method of the `StreamUtils` class to copy the file from the zipped folder to the created output directory.
+   4. At the end of the process, the resources held by the `ZipFile` class are released.

public static void UnzipFiles(string fileName, string outputLocation)
{
    //Opening the zip files that is to be decompressed
    FileStream fs = File.OpenRead(fileName);

    //Creating the ZipFile object that contains all the ZipEntry objects
    ZipFile zipFile = new ZipFile(fs);

    foreach (ZipEntry zipEntry in zipFile)
    {
        //Ignoring the directories
        if (zipEntry.IsFile)
        {
            continue;
        }
        byte[] buffer = new byte[4096]; //Choosing the optimum storage
        Stream zipStream = zipFile.GetInputStream(zipEntry);
        //Creating the output file directory
        string outputDirectory = fileName.Substring
        (0, fileName.LastIndexOf("\\"));
        string outputFileName = outputDirectory + "\\" + zipEntry.Name;
        if (!Directory.Exists(outputDirectory))
        {
            Directory.CreateDirectory(outputDirectory);
        }
        //Copying the zipped file content to the unzipped file content using the buffer storage
        using (FileStream streamWriter = File.Create(outputFileName))
        {
            StreamUtils.Copy(zipStream, streamWriter, buffer);
        }
    }
    //Releasing the resources held by zipfile object
    zipFile.Close();
}
