    int hour;
    int minutes;
    int seconds;

public:
    Time()
    {
        hour=minutes=seconds=0;
    }
    Time(int hr, int min, int sec)
    {
        hour = hr;
        minutes = min;
        seconds = sec;
    }
    int showTime ()
    {
        cout<<"The Time is: "<<hour<<":"<<minutes<<":"<<seconds<<endl;
    }
    Time operator + (Time T2)
    {
        int hr,min,sec;
        hr=min=sec=0;
        hr = hour + T2.hour;
        sec = seconds + T2.seconds;
        if (sec >= 60) {
            min = sec / 60;
            sec = sec % 60;
        }
        min += minutes + T2.minutes;
        if (min >= 60) {
            hour += (min / 60);
            min = min % 60;
        }
        return Time(hr,min,sec);
    }
};

int main()
{
    Time obj1(12,20,23);
    Time obj2(20,45,30);
    Time obj3, obj4;

    obj1.showTime();
    obj2.showTime();
    obj3.showTime();
    obj4.showTime();

    obj3 = obj1 + obj2;

    obj1.showTime();
    obj2.showTime();
    obj3.showTime();
    obj4.showTime();
}
