found, then it compares the key value with the middle element. If the key < middle element, then it considers the left-hand side array of the middle element, as the key may exist in the left-hand side sub-array; otherwise, it searches in the right-hand side sub-array of the middle element. Again, the middle element is selected from the sub-array and that element is matched with the key value. This whole process continues until the minimum index is less than the maximum index of the array or the sub-arrays.

#include <stdio.h>

int Binary_Search(int a[],int key,int n)
{
    int mid;
    int first = 0;
    int last = n-1;
    while(first < last)
    {
        mid = (first + last)/2;
        if(a[mid] == key)
        {
            return mid;
        }
        if(n < a[mid])
        {
            last = mid - 1;
        }
        if(n > a[mid])
        {
            first = mid + 1;
        }
    }
    return -1;
}

void main()
{
    int a[50];
    int key,n,x,result;
    printf("Enter the number of array elements. . .\n");
    scanf("%d",&n);
    printf("Enter the array elements. . .\n");

    for(x=0;x<n;x++)
        scanf("%d",&a[x]);

    printf("Enter the number to be searched : ");
    scanf("%d",&key);
    result = Binary_Search(a,key,n);
    if(result == -1)
        printf("Element not found");
    else
        printf("%d is found in the list, at position %d\n",key,result);
}
