2. After selecting the directory or folder from the dialog box, the application finds out the path of the directory. It then sets the output location path as C:\ZipFiles\, where the compressed file will be created. Listing 21.15 shows the code for the Zip files button:

**Listing 21.15:** Showing the Code for the `btnZipFiles_Click()` Method

```
using ZipSource;

private void btnZipFiles_Click(object sender, EventArgs e)
{
    try
    {
        fbdFolderBrowserDialog.ShowDialog();
        string path = fbdFolderBrowserDialog.SelectedPath.ToString();
        string fileName = path.Substring
            (path.LastIndexOf("\\") + 1) + "_zip.zip";

        string outputFileLocation = @"C:\ZipFiles\" + fileName;
        zipSource.ZipFolderSource.ZipFilesAndFolders(path, outputFileLocation);
        lblMessage.Text = "Zipped successfully!!! Please finds the
            zipped folder in the path " + outputFileLocation;
    }

    catch (Exception ex)
    {
        MessageBox.Show(ex.Message, "Zip File Error", MessageBoxButtons.OK);
    }
}
```
