# CONDITION VARIABLES

```
int buffer[MAX];
int fill_ptr = 0;
int use_ptr = 0;
int count = 0;

void put(int value) {
    buffer[fill_ptr] = value;
    fill_ptr = (fill_ptr + 1) % MAX;
    count++;
}

int get() {
    int tmp = buffer[use_ptr];
    use_ptr = (use_ptr + 1) % MAX;
    count--;
    return tmp;
}
```

```
cond_t empty, fill;
mutex_t mutex;

void *producer(void *arg) {
    int i;
    for (i = 0; i < loops; i++) { // p1
        pthread_mutex_lock(&mutex);
        while (count == MAX) // p2
            pthread_cond_wait(&empty, &mutex); // p3
        put(i); // p4
        pthread_cond_signal(&fill); // p5
        pthread_mutex_unlock(&mutex); // p6
    }
}



void *consumer(void *arg) {
    int i;
    for (i = 0; i < loops; i++) { // c1
        pthread_mutex_lock(&mutex); // c2
        while (count == 0) // c3
            pthread_cond_wait(&fill, &mutex); // c4
        int tmp = get(); // c5
        pthread_cond_signal(&empty); // c6
        pthread_mutex_unlock(&mutex);
        printf("%d\n", tmp);
    }
}
```
