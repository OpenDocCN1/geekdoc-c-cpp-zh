```
void animal::simulate_week( )
{
    organism::simulate_week( );
    if (eaten_this_week < need_each_week)
        death( );
    eaten_this_week = 0;
}
```

```
double animal::still_need( ) const
{
    if (eaten_this_week >= need_each_week)
        return 0;
    else
        return need_each_week - eaten_this_week;
}
```
