using System;
public class SumValues
{
    static void Main()
    {
        GetSum(10, 20, 30);
        GetSum(10, 20, 30, 40, 50);
    }

    static void GetSum(params int[] numlist)
    {
        Console.WriteLine("There are {0} numbers", numlist.Length);

        int sum = 0;
        foreach (int cnt in numlist)
        {
            sum = sum + cnt;
        }

        Console.WriteLine("Sum is {0}", sum);
    }
}
