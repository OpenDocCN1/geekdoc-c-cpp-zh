# break;

# Example

```
// Illustration of break statement in which a loop breaks after second iteration.
using System;
namespace CSharpJump
{
class breakStatement
{
static void Main()
{
int cnt;
for (cnt = 100; cnt <= 105; cnt++)
{
Console.WriteLine(""+ cnt);
if (cnt == 102)
{
break;
}
}
Console.ReadLine();
}
}
}
```

Input nil

# Output

100
101
