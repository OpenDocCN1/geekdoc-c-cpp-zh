var MongoClient = require('mongodb').MongoClient;
exports.register = function (plugin, mongodbOptions, next) {
  MongoClient.connect(
    mongodbOptions.connection,
    mongodbOptions.options,
    function (err, client) {
      if (err) {
        throw err;
      }

      plugins.expose('mongodbClient', client);
      next();
    });
}
