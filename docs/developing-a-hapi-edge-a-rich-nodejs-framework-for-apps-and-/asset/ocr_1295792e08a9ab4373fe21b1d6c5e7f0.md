```
var server = new Hapi.Server();
server.connection();

var options = {
  opsInterval: 1000,
  reporters: {
    reporter: 'good-file',
    events: { ops: '+' },
    config: '/var/log/example/ops.json'
  }
};

server.register({
  register: Good,
  options: options,
}, function (err) {

  if (err) {
    console.error(err);
    return;
  }

  server.start();
});
```
