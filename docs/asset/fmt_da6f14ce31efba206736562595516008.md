```cs
/* CarpetCalculator.cs 作者: 
* 计算地毯的总成本，给定
* 以英尺和英寸为单位的房间尺寸以及
* 每平方码的地毯价格.*/

using System;
namespace CarpetExample
{
    class CarpetCalculator
    {
        static void Main()
        {
            const int SQ_FT_PER_SQ_YARD = 9;
            const int INCHES_PER_FOOT = 12;
            const string BEST_CARPET = "Berber";
            const string ECONOMY_CARPET = "Pile";
            int roomLengthFeet = 12,
            roomLengthInches = 2,
            roomWidthFeet = 14,
            roomWidthInches = 7;

            double roomLength,
            roomWidth,
            carpetPrice,
            numSquareFeet,
            numSquareYards,
            totalCost;

            roomLength = roomLengthFeet + (double)roomLengthInches / INCHES_PER_FOOT;
            roomWidth = roomWidthFeet + (double)roomWidthInches / INCHES_PER_FOOT;
            numSquareFeet = roomLength * roomWidth;
            numSquareYards = numSquareFeet / SQ_FT_PER_SQ_YARD;
            carpetPrice = 27.95;
            totalCost = numSquareYards * carpetPrice;
            Console.WriteLine("The cost of " + BEST_CARPET +
             is {0:C}", totalCost);
            carpetPrice = 15.95;
            totalCost = numSquareYards * carpetPrice;
            Console.WriteLine("The cost of " + ECONOMY_CARPET +
             is {0:C}", totalCost);

            Console.Read();
        }
}
```

## 编码标准

正如第一章一样，编码或编程风格是描述编写源代码规范的术语。它超越了语言规则，并提供了应遵循的指南或建议，以确保朝着共同目标工作的程序员之间的一致性。遵循这些指南可以减少引入项目中的错误或缺陷，节省时间并提高代码的可维护性。本章稍后将介绍命名规范。

### 命名规范

- 使用有意义的、描述性的单词作为标识符。
- 除非是已知的缩写，否则不要使用缩写。
- 除了作为循环迭代器的变量外，不要使用单字符标识符。
- 对以下内容使用帕斯卡命名法（所有单词的首字母大写；其他字母均为小写）：
    - 类和类型标识符
    - 方法名
    - 命名空间标识符
    - 属性名
- 对以下内容使用驼峰命名法（除第一个单词外的所有单词的首字母大写；其他字母均为小写）：
    - 局部变量标识符
    - 对象标识符
    - 私有数据成员
    - 参数
- 文件名应匹配类名。
- 使用全大写字母命名常量。
- 标识符中不要下划线。
- 使用名词或名词短语命名类。

### 间距规范

- 使用制表符代替空格进行缩进。
- 使用空白（空行）来分隔和组织代码组。
- 将大括号 `{ }` 放置与括号外代码相同的水平。
- 将大括号放在新行。
- 独立地分行声明每个变量。允许重用数据类型，但每个变量应该单独占一行。
- 避免长代码。行不应超过 80 个字符。
- 如果语句跨行，请使用缩进来提高可读性。

### 声明规范

- 使用内置的 C# 数据类型。
- 在靠近使用它们声明局部变量。
- 在声明时尝试进行初始化。
- 浮点数在小数点前应包含一个数字。
- 使用 `const` 关键字定义常量值。
