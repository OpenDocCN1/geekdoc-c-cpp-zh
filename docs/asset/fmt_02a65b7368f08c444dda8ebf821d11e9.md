- 11. 如果栈非空，语句 `stack.top()` 返回栈顶元素，`stack.pop()` 移除栈顶元素。

```
template <class>
void second(stack<elementType> stack){
    elementType temp1, temp2;
    if (stack.isEmpty())
        cout << "Stack is empty." << endl;
        exit(0); //终止程序
    }
    temp1 = stack.top();
    stack.pop();
    if (stack.isEmpty())
        cout << "Stack has only one element." << endl;
        exit(0);//终止程序
    }
    temp2 = stack.top();
    stack.push(temp1);
    return temp2;
}
```
