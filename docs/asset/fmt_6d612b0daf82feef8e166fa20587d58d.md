```c
    int n, temp =0, remainder, num;

    printf("输入一个整数: ");

    scanf("%d",&n);

    num= n;

    // 反转后的整数存储在 temp 中

    while(n!=0){

        remainder= n%10;

        temp= temp *10+ remainder;

        n/=10;

    }

    // 如果原始数字和反转后的数字相等则是回文

    if(num==temp)

        printf("%d 是回文数。",num);

    else

        printf("%d 不是回文数。",num);

    return 0;

}
```
