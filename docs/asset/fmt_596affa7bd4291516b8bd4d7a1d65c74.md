lval* builtin_ord(lenv* e, lval* a, char* op) {
    ASSERT_NUM(a, 2);
    ASSERT_TYPE(a, 0, LVAL_NUM);
    ASSERT_TYPE(a, 1, LVAL_NUM);
    int r;

    if (strcmp(op, ">") == 0) {
        r = (a->cell[0]->num > a->cell[1]->num);
    }
    if (strcmp(op, "<") == 0) {
        r = (a->cell[0]->num < a->cell[1]->num);
    }
    if (strcmp(op, ">=") == 0) {
        r = (a->cell[0]->num >= a->cell[1]->num);
    }
    if (strcmp(op, "<=") == 0) {
        r = a->cell[0]->num < a->cell[1]->num;
    }
    lval_del(a);
    return lval_num(r);
}

    int r;
    if (strcmp(op, "=") == 0) {
        r = lval_eq(a->cell[0], a->cell[1]);

    }
    lval_del(a);
    return lval_num(r);
}
