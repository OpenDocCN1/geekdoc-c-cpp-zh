```c
    printf("\n");

    maxim = max(i, m);
    printf("\n%d 个元素的最大值是 %d \n", n, maxim);
    i=0;
    m = n -1;
    minim = min(i, m);
    printf("\n%d 个元素的最小值是 %d \n", n, minim);
}

/* 递归寻找向量的最大值 */
int max(int i, int m, int a[])
{
    int ml, m2;
    if (i == m)
        return a[i];
    else
    {
        ml = max(i, m-1, a);
        m2 = max(i+1, m);
        if (ml > m2)
            return ml;
        else
            return m2;
    }
}
/* 递归寻找向量的最小值 */
int min(int i, int m, int a[])
{
    int ml, m2;
    if (i == m)
        return a[i];
    else
    {
        ml = min(i, m-1, a);
        m2 = min(i+1, m);
        if (ml < m2)
            return ml;
        else
            return m2;
    }
}
```

```c

#include<stdio.h>

#include<math.h>

main() {

    int i, n;

    float x[50],mean, variance, sdn, sum = 0.0, sumsq = 0.0;

    printf("输入列表大小: ");

    scanf("%d", &n);

    printf("\n项目:\n");

    for (i=0; i<n; i++) {

        scanf("%f", &x[i]);

        sum += x[i];

    }

    mean = sum/(float)n;

    for (i=0; i<n; i++)

        sumsq += (mean - x[i])*(mean - x[i]);

    variance = sumsq/(float)n;

    sdn = sqrt(variance);

    printf("\n%d 个项目的平均值: %10.6f\n", n,mean);

    printf("\n方差: %10.6f\n", variance);

    printf("\n标准差: %10.6f\n", sdn);

}
```
