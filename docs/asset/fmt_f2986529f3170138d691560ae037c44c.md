public PartialSolution[] extend() {

    /* 为每一列生成一个新的解 */

    PartialSolution[result] = new PartialSolution[NQUEENS];

        int size = queens.length;

        /* 新解比这个多一个 */

        result[i] = new PartialSolution(size + 1);

        /* 将此解拷贝到新解中 */

        }

        /* 在第 i 列中添加新皇后 */

        result[i].queens[size] = new Queen(size, 1);

    }

    return result;

}
