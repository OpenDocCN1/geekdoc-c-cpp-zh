    private void Merge(int[] arr, int left, int mid, int right)

        int[] R = new int[n2];

            L[i] = arr[left + i];

            R[j] = arr[mid + 1 + j];

        // 合并临时数组

        while (i < n1 && j < n2)
        {

            {
                arr[k] = L[i];
                i++;
            }

            {
                arr[k] = R[j];
                j++;
            }
            k++;
        }

        {
            arr[k] = L[i];
            i++;
            k++;
        }
        while (j < n2)

            j++;
            k++;
        }

# OOP 中的二分查找

## 二分查找算法

```
public class Searching
{
    public int BinarySearch(int[] arr, int target)
    {
        int left = 0;
        int right = arr.Length - 1;

        while (left <= right)
        {
            int mid = left + (right - left) / 2;

            if (arr[mid] == target)
            {
                return mid; // 如果找到目标，返回索引
            }
            else if (arr[mid] < target)
            {
                left = mid + 1;
            }
            else
            {
                right = mid - 1;
            }
        }
        return -1; // 如果未找到目标，返回 -1
    }
}
```

## 二分查找的功能

二分查找算法通过将排序数组对半并高效地确定目标元素是否存在上半部分或下半部分来工作。通过系统地缩小搜索范围，二分查找每次迭代都会极大地减少搜索空间，从而快速检索所需的元素。

## 优势与复杂度

二分查找由于每次迭代都将搜索空间减半，具有 `O(log n)` 的时间复杂度。然而，为了让二分查找正常工作，数组必须是排序的，且该算法在未排序数据上可能性能最优。
