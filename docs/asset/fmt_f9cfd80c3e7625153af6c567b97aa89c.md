# 97. 给定一个大小为 n 的数组，找到一个出现次数超过 n/2 次的元素。（该元素被称为多数元素。）

```cpp
#include <iostream>
int main() {
    int n;
    // 获取数组大小
    std::cout << "Enter the number of elements:";
    std::cin >> n;
    int arr[n];
    // 输入元素
    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> arr[i];
    }
    // Boyer-Moore 投票算法
    int candidate = -1;
    int count = 0;
    for (int i = 0; i < n; i++) {
      if (count == 0) {
        candidate = arr[i];
      }
      if (arr[i] == candidate) {
        count++;
      } else {
        count--;
      }
    }
    // 验证候选元素确实是多数元素
    count = 0;
    for (int i = 0; i < n; i++) {
      if (arr[i] == candidate) {
        count++;
      }
    }
    if (count > n / 2) {
        std::cout << "The majority element is: " << candidate << std::endl;
    } else {
        std::cout << "There is no majority element.\n";
    }
    return 0;
}
```

```cpp
    std::cin >> n;
    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> input;

        set.insert(input);

        int currentNum = num;
        int currentStreak = 1;
        // 检查序列中的后续元素
        while (set.find(currentNum + 1) != set.end()) {
          currentNum += 1;
          currentStreak++;
        }
        // 更新最长长度
        if (currentStreak > longestStreak) {
          longestStreak = currentStreak;
        }
      }
    }
    std::cout << "The length of the longest consecutive sequence is: " << longestStreak << std::endl;
    return 0;
}
```
