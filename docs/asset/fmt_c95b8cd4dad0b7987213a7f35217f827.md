    // 检查文件结束
    if (myStrRead == null || myStrRead == "")
    {
        mySR.Close();
        mySR.Dispose();
        return;
    }
    idxItemFound = myStrRead.IndexOf(mySeekFinal);
//
    if (myStrRead != null && myStrRead != "") // 大 IF 循环
    {
        // 判断读取的字符串中的月份是否符合要求

        string SubStrYear = myStrRead.Substring(6, 4);

        // 与所需的月份和日期字符串进行比较
        int intCompSubs1;

        // 如果结果都为 "0"，即相同，
