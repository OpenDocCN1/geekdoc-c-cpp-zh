## 项目 1A

这里是一个解决方案——当然，还有许多其他的。第一个是一个返回函数的函数，该函数使用 `eval` 评估给定的公式。这个函数可以传递给绘图器，并针对 x 的值进行多次求值：

```python
def farg(arg):

  def f(x): return eval(arg)

  return f
```

第二个仅仅是循环固定数量颜色的方法，以防用户想要绘制比我们预想的更多的图表。假设我们有一个长度为四的颜色列表：

`colors = ["black", "red", "green", "blue"]`

然后我们可以使用 `%` 运算符循环遍历它们：

```python
t.pcolor(colors[n % 4])
```

这里是完整程序：

```python
import sys
import turtle
import math
t = turtle.Turtle()
if len(sys.argv) < 2:
    print('未提供公式')
    sys.exit(0)
def plot(f):
    t.penup()
    t.goto(-300, f(-300))
    t.pendown()
    for x in range(-300, 300, 1):
        t.goto(x, f(x))
def farg(arg):
    def f(x): return eval(arg)
    return f
def key(n, formula_text, color):
    t.color(color)
    t.penup()
    t.goto(-300, -20 * n)
    t.pendown()
    t.write(formula_text, font ("Arial", 16, "normal"))
def line(x0, y0, x1, y1):
    t.penup()
    t.goto(x0, y0)
    t.pendown()
    t.goto(x1, y1)
def axes():
    t.color("black")
line(-300, -300, 0, 0)
line(0, -300, 0, 300)
for x in range(-300, 301, 50):
  if x != 0:
    t.penup()
    t.goto(x, -20)
    t.pendown()
    t.write(str(x), font ("Arial", 12, "normal"))
    line(x, -5, x, 5)
for y in range(-300, 301, 50):
  if y != 0:
    t.penup()
    t.goto(-20, y)
    t.pendown()
    t.write(str(y), font ("Arial", 12, "normal"))
    line(-5, y, 5, y)
colors = ["black", "red", "green", "blue"]
t.speed(0)
axes()
for n, arg in enumerate(sys.argv[1:]):
  t.pcolor(colors[n % 4])
  plot(farg(arg))
  key(n, arg, colors[n % 4])
turtle.mainloop()
```

import turtle
import time
def hand(length, thickness, angle):
    t.penup()
    t.home()
    t.setheading(90)
    t.pensize(thickness)
    t.pendown()
    t.fd(length)
def tickmarks():
    t.pensize(1)
    for a in range(60):
        t.penup()
        t.home()
        t.setheading(90)
        t.rt(360 / 60 * a)
        t.pendown()
        t.fd(295)
def clockface(h, m, s):
    t.penup()
    t.goto(0, -300)
    t.pensize(1)
    t.pendown()
    t.circle(300)
    tickmarks()
    hand(200, 3, 360 / 12 * h)
    hand(280, 3, 360 / 60 * m)
    hand(295, 1, 360 / 60 * s)
t = turtle.Turtle()
t.hideturtle()
turtle.Screen().tracer(0, 0)
while True:

    t.home()
    t.clear()
