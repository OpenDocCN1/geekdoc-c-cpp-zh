```cpp
#include <iostream>
```

# 练习 34.3-6 寻找最冷冷的城市

编写一个 C++ 程序，让用户输入 `10` 个城市的名称以及每天同一小时记录的温度（华氏度）。该 C++ 程序必须显示温度最低的城市以及是在哪一天。

## 解决方案

在本次练习中，你需要以下数组。

`names = t`

解决方案是简单的。每当变量 `minimum` 更新其值时，两个变量 `m_i` 和 `m_j` 分别存储 `i` 和 `j`。最后，它们将包含最小值所在的行索引和列索引。解决方案如下：

```cpp
#include <iostream>
using namespace std;
const int CITIES = 10;
const int DAYS = 31;
int main() {
    int i, j, minimum, m_i, m_j;
    string names[CITIES];
    int t[CITIES][DAYS];
    for (i = 0; i <= CITIES - 1; i++) {
        cin >> names[i];
        for (int j = 0; j <= DAYS - 1; j++) {
            cin >> t[i][j];
        }
    }
    minimum = t[0][0];
    m_i = 0;
    m_j = 0;
    for (i = 0; i <= CITIES - 1; i++) {
        for (j = 0; j <= DAYS - 1; j++) {
            if (t[i][j] < minimum) {
              minimum = t[i][j];
              m_i = i;
              m_j = j;
            }
        }
    }
    cout << "Minimum temperature: " << minimum << endl;
    cout << "City: " << names[m_i] << endl;
    cout << "Day: " << m_j + 1 << endl;

}
```
