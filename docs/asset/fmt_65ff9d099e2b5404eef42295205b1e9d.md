stringmerge.c — 给定两个排序的字符串文件，它创建一个包含它们所有元素的排序后的文件。文件名作为命令行参数传递的。

```c
#include <stdio.h>

#define MAXBUFFER 128

int getline(FILE *fd, char *buffer, int max)
{
    int c, i = 0;
    while ((c = fgetc(fd)) != EOF && c != '\n' && i < max - 1)
    {
        buffer[i++] = c;
    }
    buffer[i] = '\0';
    return (c == EOF && i == 0) ? EOF : i;
}

int mergeMerge(char *file1, char *file2, char *file3)
{
    FILE *fd1, *fd2, *fd3;
    char buffer1[MAXBUFFER], buffer2[MAXBUFFER];
    int In1, In2, n = 0;

    if ((fd1 = fopen(file1, "r")) == NULL)
    {
        perror("open");
        exit(1);
    }
    if ((fd2 = fopen(file2, "r")) == NULL)
    {
        perror("open");
        exit(1);
    }
    if ((fd3 = fopen(file3, "w")) == NULL)
    {
        perror("open");
        exit(1);
    }

    In1 = getline(fd1, buffer1, MAXBUFFER - 1);
    In2 = getline(fd2, buffer2, MAXBUFFER - 1);

    while ((In1 != EOF) && (In2 != EOF))
    {
        if (strcmp(buffer1, buffer2) < 0)
        {
            fprintf(fd3, "%s\n", buffer1);
            In1 = getline(fd1, buffer1, MAXBUFFER - 1);
        }
        else
        {
            fprintf(fd3, "%s\n", buffer2);
            In2 = getline(fd2, buffer2, MAXBUFFER - 1);
        }
        n++;
    }

    while (In1 != EOF)
    {
        fprintf(fd3, "%s\n", buffer1);
        In1 = getline(fd1, buffer1, MAXBUFFER - 1);
        n++;
    }

    while (In2 != EOF)
    {
        fprintf(fd3, "%s\n", buffer2);
        In2 = getline(fd2, buffer2, MAXBUFFER - 1);
        n++;
    }

    fclose(fd1);
    fclose(fd2);
    fclose(fd3);

    return n;
}

int main(int argc, char *argv[])
{
    if (argc != 4)
    {
        printf("Usage: %s sortedfile1 sortedfile2 mergefile\n", argv[0]);
        exit(0);
    }

    printf("We have %d merged records\n", mergeMerge(argv[1], argv[2], argv[3]));
}
```

makebinfile.c - 读取一个包含文本记录的文件并写入新的二进制文件。文件名作为命令行参数传递。

```c
#include <stdio.h>

#define SIZE 10
#define NAMESIZE 25

typedef struct {
    char name[NAMESIZE];
    int midterm;
    int final;
    int homeworks;
} student;

int writestudent(FILE *fdout, student *who)
{
    char *p;
    char *limit = ((char *)who) + sizeof(student);

    for (p = (char *)who; p < limit; p++)
    {
        fputc(*p, fdout);
    }

    return (limit - (char *)who);
}

int main(int argc, char *argv[])
{
    int n = 0;
    int m;
    student who;
    FILE *fdin;
    FILE *fdout;

    if (argc != 3)
    {
        printf("Usage: %s infile outfile\n", argv[0]);
        exit(0);
    }

    if ((fdin = fopen(argv[1], "r")) == NULL)
    {
        perror("fopen");
        exit(1);
    }

    if ((fdout = fopen(argv[2], "wb")) == NULL)
    {
        perror("fopen");
        exit(1);
    }

    while (fscanf(fdin, "%s %d %d %d", who.name, &who.midterm, &who.final, &who.homeworks) != EOF)
    {
        m = writestudent(fdout, &who);
        printf("m=%d\n", m);
        n++;
    }

    printf("n=%d\n", n);

    fclose(fdin);
    fclose(fdout);
}
```
