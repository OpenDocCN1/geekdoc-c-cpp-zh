```cpp
// 练习 22.13: Ex22_13.cpp
#include <iostream>
using namespace std;
bool mystery(unsigned);
int main() {
    unsigned x;
    cout << "输入一个整数:";
    cin >> x;
    cout << "结果是 " << mystery(x) << endl;
}
// 这个函数是做什么的？
bool mystery(unsigned bits) {
    const int SHIFT{8 * sizeof(unsigned) - 1};
    unsigned MASK{1 << SHIFT};
    unsigned total{0};
    for (int i{0}; i <= SHIFT + 1; ++i, bits <<=1) {
        if ((bits & MASK) == MASK) {
            total++;
        }
    }
    return !(total % 2);
}
```
