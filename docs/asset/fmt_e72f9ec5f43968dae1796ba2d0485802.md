## insertNodeAt()

`insertNodeAt()` 函数在指定索引处插入新节点。以下是完整定义：

```cpp
void insertNodeAt(int index, int data) {
    if (index < 0 || index > size) {
        return;
    }
    NODE* n = new NODE(data);
    if (size == 0) {
        front = n;
        back = n;
    } else if (index == 0) {
        n->next = front;
        front->previous = n;
        front = n;
    } else if (index == size) {
        back->next = n;
        n->previous = back;
        back = n;
    } else {
        NODE* temp = front;
        for (int i = 0; i < index - 1; i++) {
            temp = temp->next;
        }
        n->next = temp->next;
        n->previous = temp;
        temp->next->previous = n;
        temp->next = n;
    }
    size++;
}
```

## peek()

`peek()` 函数通过传递给它的索引获取存储链表中节点中的数据，并返回该数据。

让我们仔细查看 `peek()` 函数如何工作。它首先使用本章 `removeNodeAt()` 函数中讨论的验证程序检查索引。如果索引无效，则返回零。

如果索引有效，则声明一个名为 `temp` 的指针并赋值链表头部的节点。`peek()` 函数开始搜索链表，直到到达感兴趣的节点。此搜索过程与 `insertNodeAt()` 函数相同。

当 `peek()` 函数退出循环时，`temp` 指针指向包含数据的节点。返回语句将数据返回给调用 `peek()` 函数的语句。

以下是 `peek()` 函数的完整定义：

```cpp
int peek(int index) {
    if (index < 0 || index >= size) {
        return 0;
    }
    NODE* temp = front;
    for (int i = 0; i < index; i++) {
        temp = temp->next;
    }
    return temp->data;
}
```

## getSize()

`getSize()` 函数获取 `LinkedList` 类中 `size` 成员的值。你会发现 `getSize()` 函数包含一条语句，简单地返回 `size` 成员的值。

你可能在想为什么需要 `getSize()` 函数，因为你可以通过将 `size` 成员放在 `LinkedList` 类的公共访问器中使其对应用程序可见。

正如你记得的，类的大数据成员应该只由类内部函数成员或派生类访问。这样，你始终可以控制对数据的访问，从而保护数据不受应用程序意外更改。允许类用户从外部更改它会导致错误。

```cpp
int getSize() {
    return size;
}
```

#include <iostream>
using namespace std;
void main() {
    LinkedList* list = new LinkedList();
    list->appendNode(10);
    list->appendNode(20);
    list->appendNode(30);
    list->appendNode(40);
    list->appendNode(50);
    list->removeNodeAt(3);
    int index = list->findNode(20);
    list->deleteNode(20);
    list->insertNodeAt(1, 35);
    int data = list->peek(3);

```cpp
// LinkedList.h
typedef struct Node {
    Node(int data) {
        this->data = data;
        previous = NULL;
        next = NULL;
    }
    int data;
    struct Node* previous;
    struct Node* next;
} NODE;

class LinkedList {
protected:
    NODE* front;
    NODE* back;
    int size;
    void removeNode(NODE* node);
public:
    LinkedList();
    virtual ~LinkedList();
    void appendNode(int data);
    void removeNodeAt(int index);
    int findNode(int data);
    void deleteNode(int data);
    void insertNodeAt(int index, int data);
    int peek(int index);
    int getSize();
};
```

```cpp
// LinkedList.cpp
#include "LinkedList.h"
LinkedList::LinkedList() {
    front = NULL;
    back = NULL;
    size = 0;
}
LinkedList::~LinkedList() {
    destroyList();
}
void LinkedList::appendNode(int data) {
    NODE* n = new NODE(data);
    if (size == 0) {
        front = n;
        back = n;
    } else {
        back->next = n;
        n->previous = back;
        back = n;
    }
    size++;
}
void LinkedList::removeNodeAt(int index) {
    if (index < 0 || index >= size) {
        return;
    }
    NODE* temp = front;
    for (int i = 0; i < index; i++) {
        temp = temp->next;
    }
    if (temp == front) {
        front = front->next;
        if (front) {
            front->previous = NULL;
        } else {
            back = NULL;
        }
    } else if (temp == back) {
        back = back->previous;
        back->next = NULL;
    } else {
        temp->previous->next = temp->next;
        temp->next->previous = temp->previous;
    }
    delete temp;
    size--;
}
int LinkedList::findNode(int data) {
    int index = 0;
    NODE* temp = front;
    while (temp != NULL) {
        if (temp->data == data) {
            return index;
        }
        temp = temp->next;
        index++;
    }
    return -1;
}
void LinkedList::deleteNode(int data) {
    NODE* temp = front;
    while (temp != NULL) {
        if (temp->data == data) {
            if (temp == front) {
                front = front->next;
                if (front) {
                    front->previous = NULL;
                } else {
                    back = NULL;
                }
            } else if (temp == back) {
                back = back->previous;
                back->next = NULL;
            } else {
                temp->previous->next = temp->next;
                temp->next->previous = temp->previous;
            }
            delete temp;
            size--;
            return;
        }
        temp = temp->next;
    }
}
void LinkedList::insertNodeAt(int index, int data) {
    if (index < 0 || index > size) {
        return;
    }
    NODE* n = new NODE(data);
    if (size == 0) {
        front = n;
        back = n;
    } else if (index == 0) {
        n->next = front;
        front->previous = n;
        front = n;
    } else if (index == size) {
        back->next = n;
        n->previous = back;
        back = n;
    } else {
        NODE* temp = front;
        for (int i = 0; i < index - 1; i++) {
            temp = temp->next;
        }
        n->next = temp->next;
        n->previous = temp;
        temp->next->previous = n;
        temp->next = n;
    }
    size++;
}
int LinkedList::peek(int index) {
    if (index < 0 || index >= size) {
        return 0;
    }
    NODE* temp = front;
    for (int i = 0; i < index; i++) {
        temp = temp->next;
    }
    return temp->data;
}
int LinkedList::getSize() {
    return size;
}
void LinkedList::destroyList() {
    NODE* temp = front;
    while (temp != NULL) {
        NODE* next = temp->next;
        delete temp;
        temp = next;
    }
    front = NULL;
    back = NULL;
    size = 0;
}
```
