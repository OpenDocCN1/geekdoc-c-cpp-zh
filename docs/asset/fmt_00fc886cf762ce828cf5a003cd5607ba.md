```cpp
template <class Type>
void doublyLinkedList<Type>::deleteNode(const Type& deleteItem)
{
    nodeType<Type>* *current; //用于遍历链表的指针
    nodeType<Type>* *trailCurrent; //指向 current 之前的指针
    bool found;
    if (first == NULL)
        cout << "Cannot delete from an empty list." << endl;
    else if (first->info == deleteItem) //要删除的节点
        //删除第一个节点
    {
        current = first;
        first = first->next;
        if (first != NULL)
            first->back = NULL;
        else
            last = NULL;
        count--;
        delete current;
    }
    else{
        found = false;
        current = first;
        while (current != NULL && !found) //搜索链表
            if (current->info >= deleteItem)
                found = true;
            else
                current = current->next;
        if (current == NULL)
            cout << "The item to be deleted is not in " << "the list." << endl;
        else if (current->info == deleteItem) //检查
            //相等性
        {
            trailCurrent = current->back;
            trailCurrent->next = current->next;
            if (current->next != NULL)
                current->next->back = trailCurrent;
            if (current == last)
                last = trailCurrent;
            count--;
            delete current;
        }
    }
} else {
    cout << "The item to be deleted is not in list." << endl;
} //end else
//end deleteNode
```
