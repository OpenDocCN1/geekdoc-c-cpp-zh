# `chr()` 和 `ord()` 函数

`chr()` 函数（"character" 的缩写）接收一个单个字符字符串作为参数，并返回该字符串的整数 ASCII 码。`ord()` 函数（"ordinal" 的缩写）接收一个整数作为参数，并返回该字符串对应的 ASCII 字符。尝试在交互式 shell 中输入以下内容：

```python
# 凯撒密码 - 简单替换
MAX_KEY_SIZE = 26

def getMode():
    while True:
        print('你想加密还是解密消息？')
        mode = input().lower()
        if mode in ['encrypt', 'decrypt']:
            return mode[0]
        else:
            print('请输入加密(encrypt) 或解密(decrypt)')

def getMessage():
    print('输入消息:')
    return input()

def getKey():
    key = 0
    while True:
        print('请输入密钥数字 (1-%d):' % MAX_KEY_SIZE)
        key = int(input())
        if 1 <= key <= MAX_KEY_SIZE:
            return key
        else:
            print('密钥必须在 1-%d 之间' % MAX_KEY_SIZE)

def getTranslatedMessage(mode, message, key):
    translated = ''
    if mode == 'd':
        key = -key
    for symbol in message:
        if symbol.isalpha():
            num = ord(symbol)
            if symbol.isupper():
                if num > ord('Z'):
                    num -= 26
                elif num < ord('A'):
                    num += 26
            elif symbol.islower():
                if num > ord('z'):
                    num -= 26
                elif num < ord('a'):
                    num += 26
            translated += chr(num)
        else:
            translated += symbol
    return translated

mode = getMode()
message = getMessage()
key = getKey()
print('你的翻译结果是：')
print(getTranslatedMessage(mode, message, key))
```
