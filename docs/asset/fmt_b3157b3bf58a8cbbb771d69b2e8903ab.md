using namespace std;

bool multiple(int);

int main() {

    int y(0);

    cout << "输入一个 1 到 32000 之间的整数: ";

    cin >> y;

    if (multiple(y)) {

        cout << y << " 是 X 的倍数" << endl;

    }
    else {
        cout << y << "不是 X 的倍数" << endl;
    }
}

// 判断 num 是否为 X 的倍数
bool multiple(int num) {
    bool mult(true);
    for (int i=0, mask{1}; i < 10; ++i, mask <<=1) {
        if ((num & mask) != 0) {
            mult = false;
            break;
        }
    }
    return mult;
}

```cpp
// Exercise 22.13: Ex22.13.cpp
#include <iostream>
using namespace std;

bool mystery(unsigned);

int main() {
    unsigned x;
    cout << "输入一个整数:";
    cin >> x;
    cout << "结果是 " << mystery(x) << endl;
}

// 这个函数的作用是什么？
bool mystery(unsigned bits) {
    const unsigned SHIFT{sizeof(unsigned) - 1};
    const unsigned MASK{1 << SHIFT};
    unsigned total(0);
    for (int i=0; i < SHIFT + 1; ++i, bits <<=1) {
        if ((bits & MASK) == MASK) {
            total++;
        }
    }
    return !(total % 2);
}
```

            return false;
        }
    }
