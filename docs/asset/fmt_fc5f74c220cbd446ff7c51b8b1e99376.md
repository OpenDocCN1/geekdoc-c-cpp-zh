
## 24. 编写使用 switch Case 开发一个简易计算器的程序。

```c
/* calculator*/

#include<stdio.h>

#include<conio.h>

void main() {
    int a,b,ch;
    clrscr();
    printf("Enter two no\n");
    scanf("%d %d",&a,&b);
    printf("/1/t add\n 2/t subtract\n 3/t multiplication\n 4/t divide\n 5/t modulus\n Enter choice\n");
    scanf("%d",&ch);
    switch(ch) {
        case 1:
            printf("sum=%d",a+b);
            break;
        case 2:
            printf("subtraction=%d",a-b);
            break;
        case 3:
            printf("multiplication=%d",a*b);
            break;
        case 4:
            printf("division=%f",(float)a/b);
            break;
        case 5:
            printf("modulus=%d",a%b);
            break;
        default:
            printf("invalid input");
    }
    getch();
}
```

## 33. 编写电话费计费业务的程序。

#include<conio.h>

{
    int n;
    float bill;
    clrscr();
    printf("Enter units");
    scanf("%d",&n);
    if(n <= 100)
        bill = n * 1.5;
    else if(n <= 200)
        bill = (100 * 1.5) + (n-100) * 2.0;
    else
        bill = (100 * 1.5) + (100 * 2.0) + (n-200) * 3.0;
    printf("Total bill = %f",bill);
    getch();
}
