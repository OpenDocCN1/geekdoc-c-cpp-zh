CommitChunk(c: Chunk) {

  if (NumAllocatedChunks[c.addr] == 1) {

    Publish(c)

    return;

  }

  lock(LockTable[c.addr]);

  foreach (x : CommitTable[c.addr]) {

    if (x.thread > c.thread) {

      c.bitvector ^= x.bitvector

    }

  }

  Publish(c)

  CommitTable[c.addr] += c

  unlock(LockTable[c.addr]);

}
