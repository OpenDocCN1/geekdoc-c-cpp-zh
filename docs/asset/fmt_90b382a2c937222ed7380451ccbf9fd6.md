# 排序一维数组中的数据

图 11-41 所示程序使用冒泡排序对四个元素数组的内容进行升序排序。它在屏幕上显示排序后数组的内容。

```cpp
//BubbleSort.cpp - 使用冒泡排序对
//一维数组的内容进行
//升序排序
// 创建/修改：<your name> 于 <current date>

#include <iostream>
using namespace std;

int main()
{
int numbers[4] = {23, 46, 12, 35};
int sub = 0; // 记录下标
int temp = 0; // 用于交换的变量
int maxSub = 3; // 最大下标
int lastSwap = 0; // 最后一次交换的位置
char swap = 'Y'; // 表示是否执行了交换

// 只要发生了交换，就重复循环指令
while (swap == 'Y')
{
swap = 'N'; // 假设不需要交换

sub = 0; // 从第一个开始比较
// 数组元素

// 比较相邻的数组元素以确定
// 是否需要交换
while (sub < maxSub)
{
if (numbers[sub] > numbers[sub + 1])
{
// 需要交换
temp = numbers[sub];
numbers[sub] = numbers[sub + 1];
numbers[sub + 1] = temp;
swap = 'Y';
lastSwap = sub;
} // 结束
sub++; // 增加下标
} // 结束

maxSub = lastSwap; // 重置最大下标
} // 结束

// 显示排序后的数组
for (int x = 0; x < 4; x += 1)
cout << numbers[x] << endl;

// 结束

system("pause");
return 0;
} // main 函数结束
```
