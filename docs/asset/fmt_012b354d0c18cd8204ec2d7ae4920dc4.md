class ArrayMethods
{
    // 创建一个 Random 对象来随机挑选项。
    private static Random Rand = new Random();

    // 随机化数组中的项。
    public static void Random<T>(T[] items)
    {
        // 对于数组中的每个位置，挑选
        // 一个随机项交换到该位置。
        for (int i = 0; i < items.Length - 1; i++)
        {
            // 在 i 到最后一个元素之间选取一个随机项。
            int j = Rand.Next(i, items.Length);

            // 将项 j 交换到位置 i。
            T temp = items[i];
            items[i] = items[j];
            items[j] = temp;
        }
    }
}
