```cpp
void swap(int &v1, int &v2)
{
    // 如果值相同，不需要交换
    if (v1 == v2)
        return;
    // 在这里，交换是有效的
    int tmp = v2;
    v2 = v1;
    v1 = tmp;
    // 不需要返回
}
```

```cpp
bool str_range(const string &v1, const string &v2)
{
    // 相同的大小：返回测试
    if (v1.size() == v2.size())
        return v1; // OK：返回 bool
    // 较短字符串的大小；条件运算符，见 4.7
    auto size = (v1.size() < v2.size()) ? v1.size() : v2.size();
    // 遍历较短字符串的每个元素

        if (v1[i] != v2[i])
            return; // 错误1：没有返回值；编译器应检测到此错误
    }
}
```
