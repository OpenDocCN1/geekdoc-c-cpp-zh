
// 移动回该记录开头。

`inventory.seekp(recNum * sizeof(record), ios::beg);`

// 将新记录写入当前记录。

`inventory.write(reinterpret_cast<char*>(&record), sizeof(record));`

// 关闭文件。

`inventory.close();`

`return 0;`

}
