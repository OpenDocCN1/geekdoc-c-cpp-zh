
```c
// fig15_03.c
// 使用 exit 和 atexit 函数
#include <stdio.h>
#include <stdlib.h>

void print(void); // 原型声明

main(void) {
    atexit(print); // 注册函数 print
    puts("输入 1 使用 exit 终止程序\n");
    puts("输入 2 正常终止程序\n");
    int answer = 0; // 用户
    scanf("%d", &answer);

    // 如果 answer 为 1 则调用 exit
    if (answer == 1) {
        puts("\n使用函数 exit 终止程序\n");
        exit(EXIT_SUCCESS);
    }

    puts("\n通过到达 main 结束终止程序\n");
}

// 终止前显示消息
void print(void) {
    puts("程序终止时执行函数 print\n");
    puts("程序已终止\n");
}
```
