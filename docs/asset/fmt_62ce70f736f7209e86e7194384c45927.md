uses CRT;
  const FileName = 'F23.G3D';
  Rf = 0.5; (总半径)
  nc = 8.0; (线圈数)
  nl = 36; (每个圈的顶点数)
  p = 1/8; (轴距)
  var FT: Text; t, Y, Z: double; n: integer;
  procedure FT(Theta: double)
  BEGIN
    r=Rf*Sqr(Theta/(2*Pi));
    X=r*cos(Theta);
    Y=r*sin(Theta);
    Z=p*Theta/(2*Pi);
  END;
  XYZ();
  BEGIN
    ClrSc;
    Assign(FT, FileName); Rewrite(FT);
    for i=1 to n do
      Round(nc*i);
      t=(2*Pi/n)*(i-1);
      XYZ(X, Y);
      WriteLine(FT, 'X:',16:6, ' Y:',16:6, ' Z:',16:6);
    END;
    Close(FT);
    Write(FileName+' generated successfully. <CR>');
END.
