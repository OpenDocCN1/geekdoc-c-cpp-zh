#pragma once
#include "ecs/systems/text_renderer_systemh"
#include "ecs/systems/sprite_renderer_systemh"
#include "ecs/systems//frame_animation_systemh"

namespace fuse::e cs {
struct scene {
    FUSE_INLINE scene(SDL_Renderer* rd): _renderer(rd) {
        register_systems<ecs::frame_animation_system>();
        register_systems<ecs::sprite_renderer_system>();
        register_systems<ecs::text_renderer_system>();
    }

    FUSE_INLINE void start() {
        // load animation textures
        auto f1 = _assets.load_texture("assets/f1.png", "f1", _ren-
derer);
        auto f2 = _assets.load_texture("assets/f2.png", "f2", _ren-
derer);
        auto f3 = _assets.load_texture("assets/f3.png", "f3", _ren-
derer);

        // add animation asset
        auto animation = _assets.add<animation_asset>("dance");
        animation->instance.frames.push_back(f1->d);
        animation->instance.frames.push_back(f2->d);
        animation->instance.frames.push_back(f3->d);
        animation->instance.speed = 300;

        // add entity with animation component
        ecs::entity entity = add_entity("entity");
        auto& a = entity.add_component<ecs::animation_component>();
        a.animation = animation->id;

        // start systems
        for (auto& sys : _systems) { sys->start(); }
    }
};
}
