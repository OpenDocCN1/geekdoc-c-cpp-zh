```
// serialize info_component
FUSE_INLINE void d serialize::info(YAML::Emitter & em, entity & e) {
    if (e.has_component<ecs::info_component>()) {
        auto & i = e.get_component<ecs::info_component>();
        em << YAML::Key << "info_component";
        em << YAML::BeginMap;
        em << YAML::Key << "uuid" << YAML::Value << i.uuid;
        em << YAML::Key << "name" << YAML::Value << i.name;
        em << YAML::Key << "tag" << YAML::Value << i.tag;
        em << YAML::EndMap;
    }
}
```

```
// deserialize info_component
FUSE_INLINE void d deserialize_info(YAML::Node node, entity & e) {
    if (auto data = node["info_component"]) {
        auto & i = e.add_component<ecs::info_component>();
        i.name = data["name"].as<std::string>();
        i.tag = data["tag"].as<std::string>();
        i.uuid = data["uuid"].as<uint64>();
    }
}
```

The functions above show how the info component is de- and serialized. We always have to check if the entity has the component before attempting any serialization. The same thing is done while deserializing. We need to make sure there is a node available with that component's name.

But as you can see, there is no difference in the way a vec2f is serialized or deserialized compared to an integer, for example. We are using the same operator because we have overloaded it.
