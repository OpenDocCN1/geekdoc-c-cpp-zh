```
// ...
FUSE_INLINE void update_pipes() {
    auto it = pipes.begin();
    while (it != pipes.end()) {
        // remove dead pipe
        auto &pipe = (*it);
        if (!pipe.is_alive()) {
            it = pipes.erase(it);
            continue;
        }

        // destroy if out of the screen and set score
        auto &tr = pipe.get_component<ecs::transform_component>();
        if (tr.translate.x < -50) {
            pipe.destroy();
            auto &tx = find_entity("score").get_component<ecs::text_component>();
            tx.text = "Score: " + std::to_string(++score);
        }

        it++;
    }
}
// ...
```

In the "update_pipes" function, we are updating the text of the score by converting the current score's value to a string. This simply means that when the pair of sprites is out of sight, the score will be increased by one. There are better ways of dealing with that, but that's your homework.
