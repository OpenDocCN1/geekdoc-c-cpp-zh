# engine project (shared lib)
project(engine VERSION 0.0.1)

# get all cpp-files
file(GLOB_RESOURCE SRC_FILES src/*.cpp OMAKE_CONFIGURE_DEPENDS)
add_library(${PRJ ECT_NAME} ${SRC_FILES})

# define symbols
add_definitions(-DFUSE_EXPORT)
add_definitions(-D_DEBUG)

# windows platform
if (WNB2)
target_link_libraries(${PRJ ECT_NAME} SDL2 SDL2main SDL2_image
SDL2_ttf SDL2_mixer)
target_include_directories(${PRJ ECT_NAME} PUBLIC C
${OMAKE_SOURCE_DIR}/windows/includes)
target_link_directories(${PRJ ECT_NAME} PUBLIC C
${OMAKE_SOURCE_DIR}/windows/libs)
endif(WNB2)

# linux platform
if (UN X)
    # search libs
    find_package(SDL2 REQUIRED)
    find_package(SDL2_ttf REQUIRED)
    find_package(SDL2_mixer REQUIRED)
    find_package(SDL2_image REQUIRED)
    # include headers
    target_include_directories(${PRJ ECT_NAME} PUBLIC C ${SDL2_IN-
CLUDE_DIR})
    # link lib files
    target_link_libraries(${PRJ ECT_NAME} ${SDL2_LI BRARIES}
${SDL2_IMAGE_LI BRARIES})
    target_link_libraries(${PRJ ECT_NAME} ${SDL2_TTF_LI BRARIES}
${SDL2_M_XER_LI BRARIES})
endif(UN X)

# precompile headers
target_precompile_headers(${PRJ ECT_NAME} PUBLIC C ${OMAKE_CUR-
RENT_SOURCE_DIR}/src/pch.h)
# add include directory
target_include_directories(${PRJ ECT_NAME} PUBLIC C ${OMAKE_CUR-
RENT_SOURCE_DIR}/src)
