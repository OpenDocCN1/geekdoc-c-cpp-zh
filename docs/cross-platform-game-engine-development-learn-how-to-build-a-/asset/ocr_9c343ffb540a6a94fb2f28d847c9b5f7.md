```
#pragma once
#include "ecs/systems/sprite_renderer_system.h"
namespace fuse::ecs {
struct scene {
    FUSE_INLINE scene(SDL_Renderer* rd): _renderer(rd) {
        register_systems<ecs::sprite_renderer_system>();
    }
    // ...
    FUSE_INLINE void start() {
        // load texture asset
        auto sprite = assets::load_texture("assets/test.png",
            "test", _renderer);
        // create entity
        ecs::entity entity = add_entity("entity");
        // add sprite component
        entity.add_component<ecs::sprite_component>().sprite = sprite->id;
        // start systems
        for (auto& sys : _systems) { sys->start(); }
    }
    // ...
};
}
```

We start by registering the system in the scene's constructor. We then go on to load the texture from a file and add an entity with a sprite component in the start() function.

Put any texture file with the name "test.png" in your "resource" folder and compile your project. You will get the sprite rendered on the screen like shown below.
