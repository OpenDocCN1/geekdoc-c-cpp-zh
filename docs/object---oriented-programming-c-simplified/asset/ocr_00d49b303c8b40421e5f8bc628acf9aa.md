{
	int h,m,s;
	time temp;
	s = A.secs + B.secs;
	m = A.minutes + B.minutes + s/60;
	h = A.hours + B.hours + m/60;
	temp.secs = s%60;
	temp.minutes = m%60;
	temp.hours = h;
	return temp;
}
