matrix ::matrix(int r, int c)
{
    row = r; col=c;
    ptr=new int *[row];
    for(int i=0;i<row;i++)
        ptr[i]=new int[col];
}
void matrix ::input()
{
    int i,j;
    for(i=0;i<row;i++)
    for(j=0;j<col;j++)
    {
        cout << "\nEnter ptr[" << i << "][" << j << "] element :";
        cin >> ptr[i][j];
    }
}

void matrix :: show()
{
    int i,j;
    cout << "\n Matrix is \n";
    for(i=0;i<row;i++)
    {
        for(j=0;j<col;j++)
            cout << ptr[i][j] << "\t";
        cout << endl;
    }
}
