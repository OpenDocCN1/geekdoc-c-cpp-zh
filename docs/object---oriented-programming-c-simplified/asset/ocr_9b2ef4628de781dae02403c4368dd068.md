class string
{
    char str [20];
    int len;
public :
    void input( )
    {
        cout<<"Enter a string \n";
        cin.getline(str,20);
        len=strlen(str);
    }
    char* reverse( )
    {
        int i,j;
        char* rev = new char[len+1] // Dynamic memory
        // allocation
        for(i=len-1,j=0,i>=0,j<len;i--,j++)
            rev[j]=str[i];
        rev[j]='\0';
        return rev;
    }
};
