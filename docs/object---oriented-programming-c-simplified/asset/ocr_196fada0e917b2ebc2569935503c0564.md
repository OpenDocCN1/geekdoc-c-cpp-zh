It can also be written as :

```
demo *ptr;
{
    ptr= new demo ();
}
```

Even writing simply new demo (); in the main will work.

```
/* PROG 7.23 CONSTRUCTING OBJECTS DYNAMICALLY VER 2 */

#include <iostream.h>
#include <conio.h>

class demo
{
public :
    demo()
    {
        cout<<"DEFAULT CONSTRUCTOR"<<endl;
    }
    demo (int x)
    {
        cout<<"ONE INT ARGUMENT CONSTRUCTOR"<<endl;
        cout<<"X = "<<x<<endl;
    }
    demo (float f)
    {
        cout<<"ONE FLOAT ARGUMENT CONSTRUCTOR"<<endl;
        cout<<"f :="<<f<<endl;
    }
    demo (int x, char y)
    {
        cout<<"ONE INT AND ONE CHAR ARGUMENT CONSTRUCTOR"<<endl;
        cout<<"x :="<<x<<"\t"<<"y :="<<y<<endl;
    }
    demo(char *p)
    {
        cout<<"ONE CHAR* ARGUMENT CONSTRUCTOR"<<endl;
        cout<<"p :="<<p<<endl;
    }
};
```
