/*PROG 7.13 DEMO OF COPY CONSTRUCTOR VER 1*/

#include <iostream.h>
#include <conio.h>

class demo
{
    int data;
public :
    demo( )
    {
        data = 200;
        cout << “Default constructor is called\n”;
    }
    demo(int x)
    {
        data = x;
    }
    demo (demo &d)
    {
        data = d.data;
        cout << “copy constructor is called\n”;
    }
    void show( )
    {
        cout << “data = ” << data << endl;
    }
};
void main( )
{
    clrscr( );
    demo d1(300);
    demo d2 = d1;
    demo d3;
    d3 = d2;
    demo d4 = demo (d3);
    d1.show( );
    d2.show( );
    d3.show( );
    d4.show( );
    getch( );
}
