/*PROG 10.4 DEMO OF POINTER TO OBJECTS VER 4*/

#include <iostream.h>
#include <conio.h>
#include <string.h>
#define S 6

class demo
{
    char name [30];
public :
    demo( ) {}
    demo(char *s)
    {
        strcpy(name,s);
    }
    void show( )
    {
        cout << "Name :=" << name << endl;
    }
};
void main( )
{
    clrscr( );
    demo *ptr = new demo [S];
    demo *temp = ptr;
    int i;
    char s[30];
    for(i=0;i<S;i++)
    {
        cout << "Enter a number" << endl;
        cin.getline(s,30);
        ptr[i]=demo(s);
    }
    cout << "The names are" << endl;
    for(i=0;i<S;i++)
        temp++->show( );
    delete []ptr;
    getch( );
}
