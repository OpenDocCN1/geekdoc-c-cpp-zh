# /* PROG 14.8 EXCEPTION HANDLING WITH TWO CLASSES */

```
#include <iostream.h>
class demo1
{
};
class demo2
{
};
void main()
{
    for(int i=1;i<=2;i++)
    {
        try
        {
            if(i == 1)
                throw demo1();
            else if(i == 2)
                throw demo2();
        }
        catch(demo1 d1)
        {
            cout << "Caught exception of demo1 class \n";
        }
        catch(demo2 d2)
        {
            cout << "Caught exception of demo2 class \n";
        }
    }
}
```
