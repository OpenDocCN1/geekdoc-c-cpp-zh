void demo_str_case ::change_case()
{

int i =0;
    char ch;
    while(str[i]!=0)
    {
        if(islower(str[i]))
        {
            ch = str[i]-32;
            cout.put(ch);
        }

        else if(isupper(str[i]))
        {
            ch = str[i]+32;
            cout.put(ch);
        }

        else
            cout.put(str[i]);
        i++;
    }
}
