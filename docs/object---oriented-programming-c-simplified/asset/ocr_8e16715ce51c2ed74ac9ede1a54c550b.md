/*PROG 13.18 A TEMPLATE BASED LINKED LIST */

#include <iostream.h>
#include <typename.h>
template <class T>

class Linklist
{
        struct node
        {
                T data;
                struct node* next;
        }*start,*save,*temp;
public :
        Linklist( )
        {
                start = NULL;
        }
        void add(T d)
        {
                if(start == NULL)
                {
                        start = new node;
                        start->data = d;
                        start->next = NULL;
                }
                else
                {
                        temp = new node;
                        temp->data = d;
                        temp->next = NULL;
                        save = start;
                        while(save->next != NULL)
                                save = save->next;
                        save->next = temp;
                }
        }
}
