```
#include <string.h>

class demo
{
    char *name;
public :
    demo() {}
    demo(char*s)
    {
        name = new char[strlen(s) + 1];
        strcpy(name,s);
    }
    void show()
    {
        cout << "Name = " << name << endl;
    }
};
void main()
{
    demo*ptr[] = {
        new demo("Hari Mohan Pandey"),
        new demo("Man Mohan Pandey"),
        new demo("Vijay"),
        new demo("Ranjana"),
    };
    cout << "The names are" << endl;
    const int S = sizeof(ptr)/sizeof(demo);
    for(int i=0;i<S;i++)
    {
        ptr[i]->show();
        delete ptr[i];
    }
    getch();
}
```
