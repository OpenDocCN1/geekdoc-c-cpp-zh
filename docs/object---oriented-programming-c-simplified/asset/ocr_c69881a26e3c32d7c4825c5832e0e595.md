```
class demo
{
    static int s;
    int ns;
    public :
    void input(int x)
    {
        ns = x;
    }
    static void show()
    {
        cout << "ns=" << ns << endl;
        cout << "s=" << s << endl;
    }
};
    int demo :: s=20;
    void main()
    {
        clrscr();
        demo d1;
        d1.input(10);
        d1.show();
        getch();
    }

OUTPUT :
Error : illegal reference to data member ‘demo’ :: ns’ in a static member function.
```

# EXPLANATION :

static data members can be used in static as well as non static functions. But in static functions only static members or other static variables can be used. In the show function which is static we are using non static data members ns so compiler flashes the error.
