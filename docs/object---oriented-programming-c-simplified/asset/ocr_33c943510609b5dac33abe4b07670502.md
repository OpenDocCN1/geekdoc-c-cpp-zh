#include <conio.h>

class first
{
public :

        cout<<"First con called"<<endl;

        cout<<"First des called"<<endl;
    }
};
class second :public first
{
    char *ptr;
public :

        ptr=new char[10];
        cout<<"Second con called"<<endl;
    }
    ~second()
    {
        cout<<"Second des called"<<endl;

};
