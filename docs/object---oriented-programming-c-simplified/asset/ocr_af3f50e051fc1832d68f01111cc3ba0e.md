/* PROG 4.18 DEMO OF ARRAY OF POINTERS AND NEW OPERATOR*/

#include <iostream.h>
#include <stdlib.h>

void main()
{
    int n,i;
    cout << "How many elements you want in array\n";
    cin >> n;
    int* arr = new int [n];
    int **ptr = new int*[n];
    for(i=0;i<n;i++)
    {
        cout << "\n Enter arr[" << i << "]element : ";
        cin >> arr[i];
        ptr[i] = &arr[i];
    }
    cout << "Element\t\t Address\n";
    for(i=0;i<n;i++)
    {
        cout << *ptr[i] << "\t\t" << ptr[i] << endl;
    }
    delete arr;
    delete [ ] ptr;
}
