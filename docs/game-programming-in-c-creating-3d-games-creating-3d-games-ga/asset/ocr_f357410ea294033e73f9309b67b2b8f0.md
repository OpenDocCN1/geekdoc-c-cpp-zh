class DynamicArray
{
public:
    // Constructor takes in size of element
    DynamicArray(int size)
    : mSize(size)
    , mArray(nullptr)
    {
        mArray = new int[mSize];
    }
    // At function used to access an index
    int& At(int index) { return mArray[index]; }
private:
    int* mArray;
    int mSize;
};
