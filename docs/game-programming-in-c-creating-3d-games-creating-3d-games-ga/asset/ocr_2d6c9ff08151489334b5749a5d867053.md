bool Button::ContainsPoint(const Vector2& pt) const
{
    bool no = pt.x < (mPosition.x - mDimensions.x / 2.0f) ||
        pt.x > (mPosition.x + mDimensions.x / 2.0f) ||
        pt.y < (mPosition.y - mDimensions.y / 2.0f) ||
        pt.y > (mPosition.y + mDimensions.y / 2.0f);
    return !no;
}
