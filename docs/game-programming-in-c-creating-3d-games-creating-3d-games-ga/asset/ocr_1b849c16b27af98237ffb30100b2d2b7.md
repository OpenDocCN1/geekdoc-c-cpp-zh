// Set dx to whichever of dx1/dx2 have a lower abs
float dx = (Mathf.Abs(dx1) < Mathf.Abs(dx2)) ? dx1 : dx2;
// Ditto for dy
float dy = (Mathf.Abs(dy1) < Mathf.Abs(dy2)) ? dy1 : dy2;
// Ditto for dz
float dz = (Mathf.Abs(dz1) < Mathf.Abs(dz2)) ? dz1 : dz2;

// Whichever is closest, adjust x/y position
if (Mathf.Abs(dx) <= Mathf.Abs(dy) &&
    Mathf.Abs(dx) <= Mathf.Abs(dz))
{
    pos.x += dx;
}
else if (Mathf.Abs(dy) <= Mathf.Abs(dx) &&
         Mathf.Abs(dy) <= Mathf.Abs(dz))
{
    pos.y += dy;
}
else
{
    pos.z += dz;
}

// Need to set position and update box component
SetPosition(pos);
mBoxComp->OnUpdateWorldTransform();
}
}
