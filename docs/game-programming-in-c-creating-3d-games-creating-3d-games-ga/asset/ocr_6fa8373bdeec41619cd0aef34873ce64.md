bool AABB::Contains(const Vector3& point) const
{
    bool outside = point.x < mMin.x ||
                   point.y < mMin.y ||
                   point.z < mMin.z ||
                   point.x > mMax.x ||
                   point.y > mMax.y ||
                   point.z > mMax.z;
    // If none of these are true, the point is inside the box
    return !outside;
}
