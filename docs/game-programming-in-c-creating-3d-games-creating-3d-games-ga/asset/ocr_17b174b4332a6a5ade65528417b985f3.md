            data->mParentEdge = edge;

            if (!data->mInOpenSet)
            {
                // Compute the heuristic for this node, and add to open set
                data->mHeuristic = ComputeHeuristic(edge->mTo, goal);
                data->mInOpenSet = true;

        }
    }
}
