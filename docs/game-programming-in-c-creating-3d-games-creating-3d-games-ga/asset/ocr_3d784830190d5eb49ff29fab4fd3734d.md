```
FMOD_VECTOR VecToFMOD(const Vector3& in)
{
    // Convert from our coordinates (+x forward, +y ri
    // to FMOD (+z forward, +x right, +y up)
    FMOD_VECTOR v;
    v.x = in.y;
    v.y = in.z;
    v.z = in.x;
    return v;
}
```
