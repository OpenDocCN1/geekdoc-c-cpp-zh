DynamicArray(const DynamicArray& other)
    : mSize(other.mSize)
    , mArray(nullptr)
{
    // Dynamically allocate my own data
    mArray = new int[mSize];
    // Copy from other's data
    for (int i = 0; i < mSize; i++)
    {
        mArray[i] = other.mArray[i];
    }
}
