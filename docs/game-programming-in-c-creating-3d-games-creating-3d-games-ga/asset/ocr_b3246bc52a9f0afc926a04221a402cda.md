void InputSystem::ProcessEvent(SDL_Event& event)
{
    switch (event.type)
    {
    case SDL_MOUSEWHEEL:
        mState.Mouse.mScrollwheel = Vector2{
            static_cast<float>(event.wheel.x),
            static_cast<float>(event.wheel.y)};
        break;
    default:
        break;
    }
}
