#include <stdio.h>

int main( void )
{
    unsigned int x = 1; // set x
    unsigned int sum = 0; // set sum

    while ( x <= 10 ) { // loop while x is less than or equal to 10
        sum += x; // add x to sum
        ++x; // increment x
