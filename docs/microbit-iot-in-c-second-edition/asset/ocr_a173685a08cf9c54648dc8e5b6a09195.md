void initWiFi() {
uBit.serial.redirect(Tx, Rx);
uBit.serial.baud(115200);
uBit.serial.setRxBufferSize((uint8_t)+500);
}

int ATWiFi() {
uBit.serial.send("ATAT\r\n", SYNC_SPINWAIT);
uBit.sleep(150);
ManagedString s = uBit.serial.read(500, ASYNC);
if (DEBUG)debug("\nAT \n\r" + s + "\n\r");
  return find("OK", s);
}
