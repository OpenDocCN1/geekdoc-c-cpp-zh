            if (P0.getDigitalValue() == 0)
                break;
        }
        int buf[41];
        int j;
        disable irq();
        for (j = 0; j < 41; j++) {
            for (i = 1; i < n; i++) {
                if (P0.getDigitalValue() == 1)
                    break;
            }
            for (i = 1; i < n; i++) {
                if (P0.getDigitalValue() == 0)
                    break;
            }
            buf[j] = 0;
            if (i > threshold)
                buf[j] = 1;
        }
    }
    enable_irq();

    uint32_t byte1 = getByte(1, buf);
