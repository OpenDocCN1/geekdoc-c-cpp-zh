#include "MicroBit.h"
MicroBit uBit;
SPI(MOSI, MISO, SCK);

MicroBitPin CS(MICROBIT_ID_IO_P16, MICROBIT_PIN_P16, PIN_CAPABILITY_DIGITAL);
//Used by v1 only
intransfer(uint8_t *txBuffer, uint32_t txSize, uint8_t *rxBuffer, uint32_t rxSize)
{
    CS.setDigitalValue(0);
    uint32_t i;
    for (i = 0; i < txSize; i++)
    {
        if (i < rxSize){ spi.write(txBuffer[i]);
        } else{
            spi.write(txBuffer[i]);
        }
    }
    CS.setDigitalValue(1);
    return 0;
}
int main()
{
    CS.setDigitalValue(1);
    spi.format(8, 0);
    spi.frequency(1000000);
    uint8_t buf[1] = {0xAA};
    while (1)
    {
        CS.setDigitalValue(0);
        SPItransfer(buf, 1, 1); //comment out for V2
        // spi.write(buf, 1, 1); //uncomment for V1

        CS.setDigitalValue(1);
        uBit.serial.printf("data %X\n\r", buf[0]);
        if (buf[0] == 0xAA)
            uBit.serial.printf("data received correctly\n\r");
    }
    release fiber();
    return 0;
}
