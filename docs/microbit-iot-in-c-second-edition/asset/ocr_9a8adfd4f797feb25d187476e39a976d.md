All you have to do is read a pair of bytes from each pair of registers and thennput the results together to make 16-bit values:

```
#include "MicroBit.h"

#define LSM303 ADDR 0x32
#define LSM303 WHO 0x0F
#define LSM303 VAL 0x33
```

```
#define FX058700 ADDR 0x3C
#define FX058700 WHO 0x0D
#define FX058700 VAL 0xC7
```

```
MicroBit uBit;
```

```
int main()
{
	uBit.init();
	int uint8_t data[2];
	int int status = 0;
	uBit.compass.requestUpdate();
	int int x = uBit.compass.getX();
	uBit.serial.printf("%X %d\n\r", x);
	int y = uBit.compass.getY();
	uBit.serial.printf("%Y %d\n\r", y);
	int z = uBit.compass.getZ();
	uBit.serial.printf("%Z %d\n\r", z);
```
