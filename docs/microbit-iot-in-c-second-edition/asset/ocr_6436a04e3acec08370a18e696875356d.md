void drawPlayArea() {
  uBit.display.clear();
  int y;
  for (y = 1; y <= 5; y++) {
    uBit.display.image.setPixelValue(2, y, 255);
  }
}
```

```
int playGame() {
  int position = 4;
  int clickCount = 0;
  int t1 = uBit.systemTime();
  while (uBit.systemTime() - t1 < 20000) {
    flashMan(4, position, clickCount, 0, -1);
    if (clickCount > 10) {
      position = position - 1;
      clickCount = 0;
      if (position == 0) break;
    }
  }
  return position;
}
```

```
void endGame(int position) {
  if (position == 0) {
    moveHorizontal();
    moveVertical(0, 0);
  } else {
    moveVertical(4, position);
  }
}
