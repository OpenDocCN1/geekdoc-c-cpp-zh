+   2. (Desk Check) Determine the output produced by the following program:
#include <iostream>
using namespace std;

int main() {
    int i, j, val[3][4] = {18,6,9,52,3,15,27,6,14,25,2,10};
    for (i = 0; i < 3; ++i) {
        for (j = 0; j < 4; ++j) {
            cout << " " << val[i][j];
        }
    }
    return 0;
}
