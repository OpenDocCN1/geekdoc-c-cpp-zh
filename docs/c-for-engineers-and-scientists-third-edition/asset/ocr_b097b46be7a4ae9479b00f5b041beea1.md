# Repetition Statements

# Program 5.15

```
#include <iostream>
#include <iomanip>
#include <cmath>
using namespace std;

int main()
{
    double x, y;

    cout << "x value          y value\n";
    cout << "------------------------\n";
    cout << setflags(ios::fixed);
    cout << setflags(ios::showpoint);
    cout << setprecision(5);

    for (x = 2.0; x <= 6.0; x = x + 0.5)
{
        y = 10.0 * pow(x,2.0) + 3.0 * x - 2.0;
        cout << setw(7) << x << setw(14) << y << endl;
    }

    return 0;
}
```
