+   8. (Desk Check) The following program uses the same variable names in both the calling and called functions. Determine whether doing causes any problems for the compiler.

```
#include <iostream.h>

int time(int min, int); // function prototype

int main()
{
    int min, hour, sec;

    cout << "Enter two numbers: ";
    cin >> min, hour;
    sec = time(min, hour);
    cout << "The total number of seconds is " << sec << endl;

    return 0;
}

int time(int min, int hour)
{
    int sec;

    sec = (hour * 60 + min) * 60;

    return sec;
}
```
