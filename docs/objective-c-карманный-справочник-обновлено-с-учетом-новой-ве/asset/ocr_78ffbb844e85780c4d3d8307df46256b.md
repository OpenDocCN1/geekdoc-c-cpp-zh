# Initializing Classes

```
@implementation Init
+ (void)load
{
    NSLog("@+load called");
}
+ (void)initialize
{
    NSLog("@+initialize called");
}
-(id)init
{
    NSLog("@-init called");
    return self;
}
@end

int main(void)
{
    [NSAutoreleasePool new];
    NSLog("@-main entered");
    id init = [[Init alloc] init];
    init = [[Init alloc] init];
    return 0;
}
```
