# Synchronizing Threads

```
static NSLock *lock;
static NSMutableArray *messages;
void recordLogMessage(NSString *msg)
{
  [lock lock];
  @try
{
    [messages addObject:msg];
    NSLog("%@", msg);
  }
  @finally
  {
    [lock unlock];
  }
}
```
