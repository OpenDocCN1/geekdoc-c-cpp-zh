void getScore(int& score)
{
    cout << "Line 4: Enter course score: ";           //Line 4
    cin >> score;                                     //Line 5
    cout << endl << "Line 6: Course score is "
         << score << endl;                            //Line 6
}

void printGrade(int cScore)
{
    cout << "Line 7: Your grade for the course is "; //Line 7

    if (cScore >= 90)                                 //Line 8
        cout << "A." << endl;
    else if (cScore >= 80)
        cout << "B." << endl;
    else if(cScore >= 70)
        cout << "C." << endl;
    else if (cScore >= 60)
        cout << "D." << endl;
    else
        cout << "F." << endl;
}
