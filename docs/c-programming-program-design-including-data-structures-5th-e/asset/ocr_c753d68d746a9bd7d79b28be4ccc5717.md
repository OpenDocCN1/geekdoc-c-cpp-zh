case 'B':
    cout << "Enter the number of basic "

    cout << endl;

    cout << "Enter the number"
         << " of premium channels: ";
    cin >> numOfPremChannels;
    cout << endl;

    if (numOfBasicServConn <= 10)
        amountDue = BUS_BILL_PROC_FEES
                  + BUS_BASIC_SERV_COST
                  + numOfPremChannels *
                    BUS_COST_PREM_CHANNEL;
    else
        amountDue = BUS_BILL_PROC_FEES
                  + BUS_BASIC_SERV_COST
                  + (numOfBasicServConn - 10) *
                    BUS_BASIC_SERV_COST
                  + numOfPremChannels *
                    BUS_COST_PREM_CHANNEL;

    cout << "Account number: "
         << accountNumber << endl;
    cout << "Amount due: $" << amountDue
         << endl;
    break;
