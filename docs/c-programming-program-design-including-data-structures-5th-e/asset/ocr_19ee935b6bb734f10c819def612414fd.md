void printStars(int blanks, int starsInLine)
{
    int count;

    for (count = 1; count <= blanks; count++)                   //Line 13
        cout << ' ';                                           //Line 14
    for (count = 1; count <= starsInLine; count++)              //Line 15
        cout << "* ";                                          //Line 16
    cout << endl;
}
