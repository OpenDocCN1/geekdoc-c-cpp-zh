- 28. 考虑以下函数模板的定义：

```cpp
template <class Type>
Type funExp(Type list[], int size)
{
    int j;
    Type x = list[0];
    Type y = list[size - 1];

    for (j = 1; j < (size - 1)/2; j++)
    {
        if (x < list[j])
            x = list[j];
        if (y > list[size - 1 - j])
            y = list[size - 1 - j];
    }

    return x + y;
}
```

  - b. `cout << funExp(strList, 6) << endl;`
