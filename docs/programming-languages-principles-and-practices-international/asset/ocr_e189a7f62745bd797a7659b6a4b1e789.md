package body Queues is

   type Queuerep;
   type Queue is access Queuerep;

   type Queuerep is
   record
      data: T;
      next: Queue;
   end record;

   q: Queue;

   procedure enqueue(elem:T) is
      temp: Queue;
   begin
      temp := new Queuerep;
      temp.data := elem;

      if (q /= null) then
         temp.next := q.next;
         q.next := temp;
      else
         temp.next := temp;
      end if;
      q := temp;
