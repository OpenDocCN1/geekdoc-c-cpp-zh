```
#import <Foundation/Foundation.h>

int main (int argc, char * argv[]) {
    @autoreleasepool {
        int a = 100;
        int b = 2;
        int c = 25;
        int d = 4;

        int result = a - b; // subtraction
        NSLog("%a - b = %i", result);

        result = b * c; // multiplication
        NSLog("%b * c = %i", result);

        result = a / c; // division
        NSLog("%a / c = %i", result);

        result = a + b * c; // precedence
        NSLog("%a + b * c = %i", result);

        NSLog("%a * b + c * d = %i", a * b + c * d);
    }
    return 0;
}
```

# Program 4.2. Output

a - b = 98
b * c = 50
a / c = 4
