Furthermore, we know how to multiply these polynomials by the polynomials x^{k-j} and scalars a_{k-j,j}/(j+1). Finally, we also know how to sum these polynomials and integrate in the unit interval. This completes the algorithm for integrating p(x, y) in the unit triangle in Figure 5.5.

Here is the code that implements this algorithm:

```
template<class T>
const T
integral(const polynomial<polynomial<T> >&p){
    polynomial<T> sum(p.size()+1,0);
    polynomial<T> one(1,1);
    polynomial<T> x(2,0);
    x(1) = 1;
    polynomial<T> oneMinusX(2,1);
    oneMinusX(1) = -1;
    list<polynomial<T> > xPowers(p.size(),one);
    list<polynomial<T> > oneMinusXpowers(p.size()+1,one);
    for(int i=1; i<p.size(); i++)
        xPowers(i) = x * xPowers[i-1];
    for(int i=1; i<=p.size(); i++)
        oneMinusXpowers(i) = oneMinusX * oneMinusXpowers[i-1];
    for(int k=p.degree(); k>=0; k--)
        for(int j=0; j<=k; j++)
            sum += (p[k][j]/(j+1))
                   * oneMinusXpowers[j+1] * xPowers[k-j];
    return integral(sum);
} // integral in the triangle
```
