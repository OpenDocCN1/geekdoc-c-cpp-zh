The "-= " operator is defined in a similar way:

```
template<class T>
const sparseMatrix<T>&
sparseMatrix<T>::operator-=(const sparseMatrix<T>&M) {
    for(int i=0; i<rowNumber(); i++)
        if(this->item[i]&&M.item[i]) {
            row<T> minus = -1. * *M.item[i];
            *this->item[i] += minus;
        }
        else
            if(M.item[i]) {
                row<T> minus = -1. * *M.item[i];
                this->item[i] = new row<T>(minus);
            }
    return *this;
} // subtract a sparse matrix
```

```
template<class T>
const sparseMatrix<T>&
sparseMatrix<T>::operator*=(const T&t){
    for(int i=0; i<rowNumber(); i++)
        if(this->item[i])
            *this->item[i] *= t;
    return *this;
} // multiply by T
```

In most of this book, however, we don’t bother with declaring friend functions before the class block or with "this->" before inherited members. This is left to the reader.

Here are some nonmember arithmetic operators:

```
template<class T>
const sparseMatrix<T>
operator+(const sparseMatrix<T>&M1,
        const sparseMatrix<T>&M2) {
    return sparseMatrix<T>(M1) += M2;
} // matrix plus matrix
```

```
template<class T>
const sparseMatrix<T>
operator-(const sparseMatrix<T>&M1,
        const sparseMatrix<T>&M2) {
    return sparseMatrix<T>(M1) -= M2;
} // matrix minus matrix
```

```
template<class T>
const sparseMatrix<T>
operator*(const T&t, const sparseMatrix<T>&M) {
    return sparseMatrix<T>(M) *= t;
} // scalar times sparse matrix
```
