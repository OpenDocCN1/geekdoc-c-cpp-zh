Here is the "convDif()" function that places the discrete convection-diffusion spatial derivatives in a "difference2" object and the corresponding right-hand-side vector in a "dynamicMatrix" object:

```
template<class T>
void convDif(difference2<T>&d,dynamicMatrix<T>&f,
    double hx,double hy,double deltaT,double t){
    for(int k=0; k<d.length(); k++)
        for(int j=0; j<d.width(); j++){
            if(t>deltaT/2)f(k,j)=F(j*hx,k*hy,t-deltaT/2);
            double c1=C1(j*hx,k*hy,t)/hx;
            if(c1>0.){
                d(k,j,k,j)=c1;
                d(k,j,k,j-1)=-c1;
            }
            else{
                d(k,j,k,j)=-c1;
                d(k,j,k,j+1)=c1;
            }
            double c2=C2(j*hx,k*hy,t)/hy;
            if(c2>0.){
                d(k,j,k,j)+=c2;
                d(k,j,k-1,j)=-c2;
            }
            else{
                d(k,j,k,j)-=c2;
                d(k,j,k+1,j)=c2;
            }
        }
    }
```
