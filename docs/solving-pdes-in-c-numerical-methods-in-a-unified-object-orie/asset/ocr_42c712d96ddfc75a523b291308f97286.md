```
for(const row<T>* runner = item[i]; runner;
     runner=(const row<T>*)runner->readNext()){
    int j = runner->getColumn();
    if((j != i)&&(!coarse[j])){
        T Pij = runner->getValue();
        T AjSum =
            item[i]->rowSumCoarse(*A.item[j],coarse);

            item[i]->addCoarse(
                (Pij / AjSum) * *A.item[j],coarse);

}
```
