```
public async Task<ActionResult<IEnumerable<CustomerModel>>>
FindCustomers(
    string title = "", string firstName = "", string lastName = =
    "",
    string email = "", string phone = ""
{
    var query = from c in _context.Customers
                where EF.Functions.Like(c.Title, title)
                EF.Functions.Like(c.FirstName) &&
                EF.Functions.Like(c.LastName,
                EF.Functions.Like(c.EmailAddress,
                EF.Functions.Like(c.Phone, phone)
                select c;
    return await query.ToListAsync();
}
```

This code runs a LINQ query that the static Like function provided as part of the Entity Framework library to find all rows that match the criteria passed in as the parameters. The Like function is part of the static EF class in the Entity Framework library.
