public string WalkTree()
{
    string result = ""\n
    if (this.LeftTree is not null)
    {
        result = this.LeftTree.WalkTree();
    }

    result += $" {this.NodeData.ToString()}\}";

    if (this.RightTree is not null)
    {
        result += this.RightTree.WalkTree();
    }
    return result;
}
