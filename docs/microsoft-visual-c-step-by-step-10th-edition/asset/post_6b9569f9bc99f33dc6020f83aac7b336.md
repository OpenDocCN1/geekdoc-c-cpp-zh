```csharp
struct IntBits
{
    public bool this [ int index ]
    {
        get => (bits & (1 << index)) != 0;
    }

    set
    {
        if (value) // turn the bit on if value is true; otherwise, turn it off
            bits |= (1 << index);
        else
            bits &= ~(1 << index);
    }
}
}
```

```csharp
private static final int DEFAULT_SIZE = 100;
private int[] data;
private int head = 0, tail = 0;
private int numElements = 0;

public Queue() {
    this.data = new int[DEFAULT_SIZE];
}

public Queue(int size) {
    if (size > 0) {
        this.data = new int[size];
    } else {
        throw new IllegalArgumentException("size must be greater than zero: " + size);
    }
}

public void enqueue(int item) Exception {
    if (this.numElements == this.data.length) {
        throw new Exception("Queue Full");
    }
    this.data[this.head] = item;
    this.head = (this.head + 1) % this.data.length;
    this.numElements++;
}

public int dequeue() Exception {
    if (this.numElements == 0) {
        throw new Exception("Queue empty");
    }
    int item = this.data[this.tail];
    this.tail = (this.tail + 1) % this.data.length;
    this.numElements--;
    return item;
}
```

```csharp
Queue queue = new Queue(); // Create a new Queue
queue.Enqueue(100);
queue.Enqueue(-25);
queue.Enqueue(33);
Console.WriteLine($"{queue.Dequeue()}"); // Displays 100
Console.WriteLine($"{queue.Dequeue()}"); // Displays -25
Console.WriteLine($"{queue.Dequeue()}"); // Displays 33
```
