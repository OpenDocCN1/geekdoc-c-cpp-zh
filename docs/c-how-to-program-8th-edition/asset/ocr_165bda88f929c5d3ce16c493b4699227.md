{
    // shuffle cards randomly
    for ( int i = 0; i < numberOfCards; ++i )
    {
        int j = rand() % numberOfCards;
        Card temp = deck[ i ];
        deck[ i ] = deck[ j ];
        deck[ j ] = temp;
    } // end for
} // end function shuffle

    for ( int i = 0; i < numberOfCards; ++i )
        cout << right << setw( 5 ) << deck[ i ].face << " of "
             << left << setw( 8 ) << deck[ i ].suit
             << ( ( i + 1 ) % 2 ? '\t' : '\n' );
} // end function deal
