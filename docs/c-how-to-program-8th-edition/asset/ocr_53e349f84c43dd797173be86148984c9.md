int main()

   const char *string1 = "This is a test";

   // search for character1 in string1
   if ( strchr( string1, character1 ) != NULL )
      cout << '\'' << character1 << "' was found in \""

      cout << '\'' << character1 << "' was not found in \""
           << string1 << "\"\n";

} // end main
