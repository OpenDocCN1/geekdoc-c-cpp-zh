such as space (' '), form feed ('\f'), newline ('\n'), carriage return ('\r'), horizontal tab ('\t') or vertical tab ('\v'). Function isspace determines whether its argument is a control character such as horizontal tab ('\t'), vertical tab ('\v'), form feed ('\f'), alert ('\a'), backspace ('\b'), carriage return ('\r') or newline ('\n'). Function ispunct determines whether its argument is a printing character other than a space, digit or letter, such as $, #, %, (, ), [, ], {, }, or *. Function isprint determines whether its argument is a character that can be displayed on the screen (including the space character). Function isgraph tests for the same characters as isprint, but the space character is not included.

```cpp
// Fig. 21.20: fig21_20.cpp
// Using functions isspace, iscntrl, ispunct, isprint and isgraph.
#include <iostream>
#include <cctype> // character-handling function prototypes
using namespace std;

int main()
{
   cout << "According to isspace:\nNewline "
        << isspace( '\n' ) ? "is a" : "is not a" )
        << " whitespace character\n"
        << isspace( ' ' ) ? "is a" : "is not a" )
        << " whitespace character\n"
        << isspace( '%" ) ? "% is a" : "% is not a" )
        << " whitespace character\n";

   cout << "\nAccording to iscntrl:\nNewline "
        << iscntrl( '\n' ) ? "is a" : "is not a" )
        << " control character\n"
        << iscntrl( '$' ) ? "$ is a" : "$ is not a" )
        << " control character\n";

   cout << "\nAccording to ispunct:\n"
        << ispunct( '_' ) ? "_ is a" : "_ is not a" )
        << " punctuation character\n"
        << ispunct( ',' ) ? ", is a" : ", is not a" )
        << " punctuation character\n"
        << ispunct( '#' ) ? "# is a" : "# is not a" )
        << " punctuation character\n";

   cout << "\nAccording to isprint:\n"
        << isprint( '$' ) ? "$ is a" : "$ is not a" )
        << " printing character\n"
        << isprint( '\a' ) ? "\\a is a" : "\\a is not a" )
        << " printing character\n"
        << isprint( ' ' ) ? "' ' is a" : "' ' is not a" )
        << " printing character\n";

   cout << "\nAccording to isgraph:\n"
        << isgraph( 'Q' ) ? "Q is a" : "Q is not a" )
        << " printing character other than a space\n";
}
