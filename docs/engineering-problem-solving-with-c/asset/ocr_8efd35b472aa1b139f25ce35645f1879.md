/*-----------------------------------------*/
/* Program chapter10_3                     */
/* This program demonstrates binary file I/O */
/*-----------------------------------------*/

#include<iostream>    //Required for cout
#include<fstream>     //Required for ifstream
using namespace std;

const int MAXLEN(100);

int main()
{
   //Define file filestream
   ifstream fin("YellowBox.jpg", ios::binary);
   ofstream fout("BinaryOutput.bin", ios::binary);

   //Define character array for storing input
   char bData[MAXLEN];
   int count(0);

   //Read and write MAXLEN characters
   fin.read(bData,MAXLEN);
   fout.write(bData,MAXLEN);

   cout << fin.gcount() << " characters read..."
        << endl;

   //Read and write MAXLEN characters
   for(char ch;bData)
   {
      fin.read(&ch,1);   //read one byte...
      fout.write(&ch,l);
      ++count;
   }
   cout << count << " byte(s) read" << endl;
   cout << fin.gcount() << " character(s) read"
        << endl;
   fin.close();
   fout.close();
   return 0;
}

The function write() is called to stream the characters in bData to the file BinaryOutput.bin. The second call to the member function read() is inside a range-based for statement. The range-based for statement will execute 100 times and 1 character per iteration will be read. The variable ch is a single variable of type char, thus the address operator is required to pass the address of ch to both read() and the write() functions.
