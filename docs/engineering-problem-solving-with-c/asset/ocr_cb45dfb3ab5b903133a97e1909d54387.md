int payoff(bool move, bool opponent_move)
{
    if (move)
    {
        if (opponent_move)
        {
            return 3; // Both cooperate.
        }
        else
        {
            return 0; // I cooperate, opponent defects.
        }
    }
    else
    {
        if (opponent_move)
        {
            return 5; // I defect, opponent cooperates.
        }
        else
        {
            return 1; // Both defect.
        }
    }
}
