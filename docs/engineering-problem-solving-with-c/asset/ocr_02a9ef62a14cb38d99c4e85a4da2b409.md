1. template <typename Dtype>
Dtype mean(const Dtype x[], int n)
{
    Dtype sum = 0;
    //Determine mean value.
    for (int k=0; k<n; ++k)
        sum += x[k];
    // Return mean value.
    return sum/n;
}
```

```
2. template <typename Dtype>
Dtype median(const Dtype x[], int n)
{
    // Declare objects.
    Dtype median_x;
    int k;
    // Determine median value.
    k = floor (n/2);
    if (n%2 != 0)
        median_x = x[k];
    else
        median_x = (x[k-1] + x[k])/2;
    // Return median value.
    return median_x;
}
```

```
3. template <typename Dtype>
Dtype variance(const Dtype x[], int n)
{
    // Declare objects.
    Dtype sum(0), mu;
    // Determine variance.
    mu = mean(x,n);
    for (int k=0; k<n; ++k)
    {
        sum += (x[k] - mu)*(x[k] - mu);
    }
    // Return variance.
    return sum/(n-1);
}
```

```
4. template <typename Dtype>
Dtype std_dev(const Dtype x[], int n)
{
    // Return standard deviation.
    return sqrt(variance(x,n));
}
