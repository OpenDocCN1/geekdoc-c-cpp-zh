    public void SetLocation(int xCoord, int yCoord)
    {
        this.locX = xCoord;

    }

    public void SetColor(Color color)
    {
        if (this.shape != null)

            SolidColorBrush brush = new SolidColorBrush(color);
            this.shape.Fill = brush;
        }
