```
private void compareClick(object sender, RoutedEventArgs e)
{
    int diff = dateCompare(firstDate.Date.LocalDateTime,
secondDate.Date.LocalDateTime);
    info.Text = "";
    show("firstDate == secondDate", diff == 0);
    show("firstDate != secondDate", diff != 0);
    show("firstDate < secondDate", diff < 0);
    show("firstDate <= secondDate", diff <= 0);
    show("firstDate > secondDate", diff > 0);
    show("firstDate >= secondDate", diff >= 0);
}
```

This method runs whenever the user clicks the Compare button on the form. The expressions firstDate.Date.LocalDateTime and secondDate.Date.LocalDateTime hold DateTime values; they represent the dates displayed in the firstDate and secondDate controls on the form
