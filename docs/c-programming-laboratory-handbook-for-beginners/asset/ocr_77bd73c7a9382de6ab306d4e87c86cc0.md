void reverse(char str1[])
{
  int i,j,len,temp;
  // This loop finds the length of the string.
  for(i=0;str1[i]!='\0';i++)
  {
    ;
  }
  len=i;
  //This loop reverses the string str1 in place.
  for(j=0,i=len-1; i>j; i--,j++)
  {
    temp=str1[i];
    str1[i]=str1[j];
    str1[j]=temp;
  }
  printf("\n Reversed String: ");
  puts(str1);
}
