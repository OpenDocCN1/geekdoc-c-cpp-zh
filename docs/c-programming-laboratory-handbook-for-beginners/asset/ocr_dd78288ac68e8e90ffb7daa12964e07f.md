void Bubble_sort(int a[],int n) //Function definition.
{
    int i,j,temp;
    /*counts number of passes*/
    for (i=0;i<n-1;i++)
    {
        for(j=0;j<n-i-1;j++)
        {
            /*Compare first no. with its adjacent number*/
            if (a[j]>a[j+1])
            {
                /*Perform swapping*/
                temp=a[j];
                a[j]=a[j+1];
                a[j+1]=temp;
            }
        }
    }
}
