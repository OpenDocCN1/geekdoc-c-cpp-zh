```c
#include<stdio.h>
#include<string.h>
void main()
{
    char c;
    printf("Enter the text\n");
    c=getchar();
    printf("The output text is\n");
    while(c!='\n')
    {
        if(c==' '||c=='\t')
        {
            c=' ';
            putchar(c);
        }
        while(c==' '||c=='\t')
        {
            c=getchar();
            putchar(c);
            c=getchar();
        }
    }
    printf("\n");
}
