# Example 1.17. Declaring variables using common method.

```
using System;

namespace Chapter1 {
    class Class1 {
        static void Main() {
            // these variable were declared using the data type integer
            int Number1, Number2;

            Number1 = 1; // Their assignment values both equal one.
            Number2 = 1;

            // And, they’re both displayed using the WriteLine function.
            Console.WriteLine("The first number is " + Number1
                + "\nThe second number is " + Number2 + "\n");
        }
    }
}
```

# Example 1.18. Declaring variables using the CTS type.

```
using System;

namespace Chapter1 {
    class Class1 {
        static void Main() {
            System.Int32 Number1, Number2;

            Number1 = 1;
            Number2 = 1;

            Console.WriteLine("The first number is " + Number1
                + "\nThe second number is " + Number2 + "\n");
        }
    }
}
```
