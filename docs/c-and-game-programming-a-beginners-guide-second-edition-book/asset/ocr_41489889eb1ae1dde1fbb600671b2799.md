# **Example 5.10. Storing and retrieving data.**

```
using System;
using System.IO;

namespace Chapter5 {
    struct BankAccount {
        public string first_name;
        public string last_name;
        public long account_number;
        public decimal checking;
        public decimal savings;
        public short pin_number;
    }

    class Class1 {
        static void Main() {
            BankAccount account1;

            account1.pin_number = 0;
            account1.first_name = "";
            account1.last_name = "";
            account1.account_number = 0;
            account1.checking = 0;
            account1.savings = 0;

            char input;

            while (true) {
                Console.WriteLine("Enter <C> to create new file or <V> " +
                    "to view account");
                input = char.Parse(Console.ReadLine());

                if(Char.ToUpper(input) == 'C') {
                    CreateFile(out account1);
                } else if (Char.ToUpper(input) == 'V') {
                    ReadFile(ref account1);
                } else {
                    break;
                }
            }
        }
    }
}
```
