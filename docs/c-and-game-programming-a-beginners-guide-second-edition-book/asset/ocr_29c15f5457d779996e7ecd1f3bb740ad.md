# Example 2.25. Boolean expressions—true or false.

```
using System;

namespace Chapter2 {
    class Class1 {
        static void Main() {
            bool Variable1 = true;
            string Test;

            while (Variable1) {
                Console.Write("\n Would you like to end this loop? ");
                Test = Console.ReadLine();  // yes will break the loop

                if (Test == "yes" || Test == "YES")
                    Variable1 = false;
            } // end while loop

            Console.WriteLine ("Program complete!");
        }
    }
}
```
