static unsafe void MyFunction (void *values, char data_type) {
    switch (data_type) {
        case 's': {
            *((short *)values) += 1;
            break;
        }

        case 'i': {
            *((int *)values) -= 1;
            break;
        }

        case 'l': {
            *((long *)values) *= 2;
            break;
        }

        case 'f': {
            *((float *)values) /= 2;
            break;
        }

        case 'd': {
            *((double *)values) = *((double *)values) + 2;
            break;
        }

        case 'e': {
            *((double *)values) = *((double *)values) - 2;
            break;
        }
    }
}
