static void Display(char[,,] print) {
    const short sets = 3, rows = 5, columns = 6;
    for (int i = 0; i < sets; i++)
        for (int j = 0; j < rows; j++)
            for (int k = 0; k < columns; k++)
                Console.Write("{0}\\", print[i, j, k]);

    Console.WriteLine();
