> An interface-class reference also requires that the base class be referenced first (see Example 5.55).

# Example 5.55. Introducing interfaces.

```
using System;

namespace Chapter5 {
    interface BaseInterface {
        short Width {get; set;}
        short Length {get; set;}
    }

    public class BaseClass : BaseInterface {
        private short BIWidth;
        private short BILength;

        protected BaseClass() {}

        protected BaseClass(short Width, short y) {
            BIWidth = Width;
            BILength = y;
        }

        public short Width {
            get {return(BIWidth);}
            set {BIWidth = value;}
        }

        public short Length {
            get {return(BILength);}
            set {BILength = value;}
        }

        protected static short ReadWidth(BaseClass X) {
            return (X.Width);
        }
    }
```
