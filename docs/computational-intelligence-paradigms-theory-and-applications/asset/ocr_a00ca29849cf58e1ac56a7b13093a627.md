A genetic algorithm is set up to find the maximum of this problem.

```
```matlab
% boundaries
bounds = [-10 10];
% pop size
n = 10;
% number of iterations
numits = 100;
% numer of mutations per it
nummut = 1;

f = @multipeak;

% ezplot(f,[-10 10])
% pause

blength = bounds(2)-bounds(1);
% initial population
pop = rand(1,n)*blength + bounds(1);

for it=1:numits
    % fitness eval
    for i=1:n, fpop(i) = feval(f, pop(i)); end
    maxf(it) = max(fpop);
    meanf(it) = mean(fpop);
    % subtract lowest fitness in order to normalize
    m=min(fpop);
    fpop=fpop-m;
    cpop(1) = fpop(1);
    for i=2:n, cpop(i) = cpop(i-1) + fpop(i); end

    % SELECTION
    total_fitness = cpop(n);
    % use roulette selection (-> need pos. fitness!)
    for i=1:n
        p=rand*total_fitness;
        % now find first index
        j=find(cpop-p>0);
        if isempty(j)
            j=n;
        else
            j=j(1);
