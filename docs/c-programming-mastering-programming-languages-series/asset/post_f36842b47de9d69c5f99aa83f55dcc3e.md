// Example: Basic Exception Handling
#include <iostream>

int main() {
    try {
        // Code that may throw exceptions
        int result = 10 / 0; // Division by zero
        std::cout << "Result: " << result << std::endl; // This line won't be reached
    } catch (const std::exception& e) {
