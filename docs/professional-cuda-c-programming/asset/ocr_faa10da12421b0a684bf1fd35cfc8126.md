void printMatrix(int *C, const int nx, const int ny) {
    int *ic = C;
    printf("\nmatrix[%d%d]\n", nx, ny);
    for (int iy=0; iy<ny; iy++) {
        for (int ix=0; ix<nx; ix++) {
            printf("%3d,", ic[ix]);
        }
    }
    ic += nx;
    printf("\n");
}
