```
for (int istep = 0; istep < nsteps; istep++) {
// add a disturbance oninto gpu0 at first step
if (istep==0) {
    cudaSetDevice(gpu[0]);
    kernel_add_wavelet<<<grid,block>>>(d_u2[0],20.0,nx,ny,ngpus);
}
}
```

```
// update halo and intern asynchronously
for (int i = 0; i < ngpus++) {
    cudaSetDevice(i);
}
```

```
// compute the halo regions in the halo stream
kernel_2dfdc<<<grid, block>>>(d_u1[i], d_u2[i], nx, haloStart[i], haloEnd[i]);
```

```
// compute the internal region values in the internal stream
kernel_2dfdc<<<grid, block, 0, stream_internal[i]]>>>
d_u1[i], d_u2[i], nx, bodyStart[i], bodyEnd[i]);
}
```

```
// exchange halos in the halo stream
if (ngpus > 1) {
    cudaMemcpyAsync(d_u1[1] + dst_skip[0], d_u1[0], src_skip[0],
                    iechange, cudaMemcpyDeviceToDevice, stream_halo[0]);
    cudaMemcpyAsync(d_u1[1] + dst_skip[1], d_u1[1], src_skip[1],
                    iechange, cudaMemcpyDeviceToDevice, stream_halo[1]);
}
```

```
// synchronize for the next step
for (int i = 0; i < ngpus++) {
    cudaSetDevice(i);
    cudaDeviceSynchronize();
}
```

```
// swap global memory pointers
float tmpu0 = d_u1[i];
d_u1[i] = d_u2[i];
d_u2[i] = tmpu0;
}
```
