```
unsigned int idx = blockIdx.x * blockDim.x + threadIdx.x;
if (idx >= n) return;
```

```
// convert global data pointer to the local pointer of this block
int *idata = g_idata + blockIdx.x * blockDim.x;
```

```
// set mem by each threads
smem[tid] = data[tid];
synchreads();
```

```
// in-place reduction in shared memory
if (blockDim.x > 1024 && tid < 512) smem[tid] += 512;
syschreads();
```

```
if (blockDim.x == 512 && tid < 256) smem[tid] += 256;
syschreads();
```

```
if (blockDim.x == 256 && tid < 128) smem[tid] += 128;
syschreads();
```

```
if (blockDim.x >= 128 && tid < 64) smem[tid] += 64;
syschreads();
```

```
// unrolling warp
if (tid < 32) {
volatile int *vsm = smem;
vsm[tid] += vsm[tid + 32];
vsm[tid] += vsm[tid + 16];
vsm[tid] += vsm[tid + 8];
vsm[tid] += vsm[tid + 4];
vsm[tid] += vsm[tid + 2];
vsm[tid] += vsm[tid + 1];
}
```

// write result for this block to global mem and without shared memory
if (tid == 0) g_idata[blockIdx.x] = smem[0];
}
