Upon return this function, warp sum is saved to shared memory based on thread index\and warp size, as shown:

```
int laneIdx = threadIdx.x % warpSize;
int warpIdx = threadIdx.x / warpSize;
int mySum = warpReduce(mySum);
if (laneIdx == 0) smem[warpIdx] = mySum;
```

For block-level reduction, the block is synchronized and the same warp reduce function
u used to add each warp's sum. The final output produced by the block is then saved to
global by the first thread in the block, as shown:

```
__synctreads();
mySum = (threadIdx.x < SMEMDIM) ? smem[laneIdx] : 0;
if (warpIdx == 0) mySum = warpReduce(mySum);
if (threadIdx.x == 0) g_odata[blockIdx.x] = mySum;
```

For the grid-level reduction, g_odata is copied to the host where the final reduction
performed. Here is the complete reduceShfl kernel:

```
__global void reduceShfl(int *g_idata, int *g_odata, unsigned int n) {
    // shared memory for each warp sum
    shared int smem[SMEMDIM];

    // boundary check
    unsigned int idx = blockIdx.x*blockDim.x + threadIdx.x;
    if (idx >= n) return;

    // read from global memory
    int mySum = g_idata[idx];

    // calculate lane index and warp index
    int laneIdx = threadIdx.x % warpSize;
    int warpIdx = threadIdx.x / warpSize;

    // block-wide warp reduce
    mySum = warpReduce(mySum);

    // save warp sum to shared memory
    if (laneIdx == 0) smem[warpIdx] = mySum;

    // block synchronization
    __synctreads();

    // last warp reduce
    mySum = (threadIdx.x < SMEMDIM) ? smem[laneIdx]:0;
    if (warpIdx == 0) mySum = warpReduce(mySum);

    // write result for this block to global mem
    if (threadIdx.x == 0) g_odata[blockIdx.x] = mySum;
}
```
