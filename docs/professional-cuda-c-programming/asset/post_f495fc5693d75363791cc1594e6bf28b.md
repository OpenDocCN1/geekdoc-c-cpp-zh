
```cpp
global void readOffset(float *A, float *B, float *C, const int n,
    int offset) {
    unsigned int i = blockIdx.x * blockDim.x + threadIdx.x;
    unsigned int k = i + offset;
    if (k < n) C[i] = A[k] + B[k];
}
```

```cpp
void sumArrayOnHost(float *A, float *B, float *C, const int n,
    int offset) {
    for (int idx = offset, k = 0; idx < n; idx++, k++) {
        C[k] = A[idx] + B[idx];
    }
}

```cpp
int main(int argc, char **argv) {
    // set up device
    int dev = 0;
    cudaDeviceProp devProp;
    cudaDeviceGetProperties(&devProp, dev);
    printf("%s starting at %d\n", argv[0]);
    printf("device: %s\n", devProp.name);
    cudaSetDevice(dev);

    // set up array size
    int nElem = 1<<20; // total number of elements to reduce
    printf("array size: %d\n", nElem);
    size_t nBytes = nElem * sizeof(float);

    // set offset for summary
    int blockSize = 512;
    int offset = 0;
    if (argc > 1) offset = atoi(argv[1]);
    if (argc > 2) blockSize = atoi(argv[2]);

    // execution configuration
    dim3 block (blockSize,1);
    dim3 grid ((nElem+block.x-1)/block.x,1);

    // allocate host memory
    float *h_A = (float*)malloc(nBytes);
    float *h_B = (float*)malloc(nBytes);
    float *hostRef = (float*)malloc(nBytes);
    float *gpuRef = (float*)malloc(nBytes);

    // initialize host array
    initialData(h_A, nElem);
    memmove(h_B,h_A,nBytes);

    // summary at host side
    sumArraysOnHost(h_A, h_B, nElem, offset);

    // allocate device memory
    float *d_A,*d_B,*d_C;
    cudaMalloc((float**)&d_A, nBytes);
    cudaMalloc((float**)&d_B, nBytes);
    cudaMalloc((float**)&d_C, nBytes);

    // copy data from host to device
    cudaMempy(d_A, h_A, nBytes, cudaMemcpyHostToDevice);
    cudaMempy(d_B, h_B, nBytes, cudaMemcpyHostToDevice);

    // kernel 1:
    double iStart = seconds();
    warmup <<< grid.x, grid.y >>> (d_A, d_B, d_C, nElem, offset);
    cudaDeviceSynchronize();
    elIaps = seconds() - iStart;
    printf("warmup <<< %d, %d >>> offset %d elapsed %f sec\n", grid.x, block.x, offset, iElaps);

    iStart = seconds();
    readoff <<< grid.x, block.y >>> (d_A, d_B, d_C, nElem, offset);
    cudaDeviceSynchronize();
    elIaps = seconds() - iStart;
    printf("readoff <<< %d, %d >>> offset %d elapsed %f sec\n", grid.x, block.x, offset, iElaps);

    // copy result back to host side and check device results
    cudaMempy(gpuRef,d_C, nBytes, cudaMemcpyDeviceToHost);
    checkResult(hostRef, gpuRef, nElem-offset);

    // copy result back to host side and check device results
    cudaMempy(gpuRef,d_B, nBytes, cudaMemcpyDeviceToHost);
    checkResult(hostRef, gpuRef, nElem-offset);

    // copy kernel result back to host side and check device results
    cudaMempy(gpuRef,d_C, nBytes, cudaMemcpyDeviceToHost);
    checkResult(hostRef, gpuRef, nElem-offset);

    // free host and device memory
    cudaFree(d_A);
    cudaFree(d_B);
    cudaFree(d_C);
    free(h_A);
    free(h_B);

    // reset device
    cudaDeviceReset();
    return EXIT_SUCCESS;
}

## Global Memory Writes

Memory stores are relatively simple. The L1 cache is not used for store operations on either Fermi or Kepler GPUs, store operations are only cached in the L2 cache before being sent to device memory. Stores are preserved at a 32-byte granularity. Memory transactions can be one, or four segments at a time. For example, if two addresses fall within the same 128-byte region but not within an aligned 64-byte region, one four-segment transaction will be issued (that is, issuing a single four-segment transaction performs better than issuing two segment transactions).

## Read-Only Cache

The read-only cache was originally reserved for texture memory. For compute capability 3.5 and higher, it can also support:

- Using the intrinsic function __ldg
- Using const __restrict__ qualifiers to pointers (that is, pointers which are used exclusively to a array). nvcc will automatically direct reads from non-aligned pointers through the read-only.

```cpp
__global void copyKernel(int *__restrict__ out,
    const int *__restrict__ in) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    out[idx] = in;
}
```

```cpp
__global void copyKernel(int *out, int *in) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    out[idx] = __ldg(&in[idx]);
}
```
