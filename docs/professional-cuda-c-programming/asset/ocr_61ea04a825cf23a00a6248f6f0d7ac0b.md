The full kernel code is as follows:

```
__global void setRowReadColDynPad(int *out) {
    // dynamic shared memory
    extern __shared__ tile[];

    // mapping from thread index to global memory index
    unsigned int g_idx = threadIdx.y * blockDim.x + threadIdx.x;

    // convert idx to transposed [row, col]
    unsigned int irow = g_idx / blockDim.y;
    unsigned inticol = g_idx % blockDim.y;

    unsigned int row_idx = threadIdx.y * (blockDim.x + IPAD) + threadIdx.x;

    // convert back to shared memory index to access the transposed element
    unsigned inticol = iicol * (blockDim.x + IPAD) + irow;

    // shared memory store operation
    tile[row_idx] = g_idx;

    // wait for all threads to complete
    syncthreads();

    // shared memory load operation
    out[g_idx] = tile[icol];
}
```

- shared_store_transactions_per_request 1.000000
