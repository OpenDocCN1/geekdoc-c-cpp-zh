```
int b2 = g_data[idx+5*blockDim.x];

int b4 = g_data[idx+7*blockDim.x];
g_data[idx] = a1+a2+a3+a4+b1+b2+b3+b4;

__syncthreads();
```

```
// in-reduction in global memory
for (int stride = blockDim.x / 2; stride > 32; stride >>= 1) {
  if (tid < stride)
    idata[tid] += idata[tid + stride];
}
```

```
// synchronize within threadblock
__syncthreads();
```

```
// unrolling warp
if (tid < 32) {
  volatile int vmem = idata;
  vmem[tid] += vmem[tid + 32];
  vmem[tid] += vmem[tid + 16];
  vmem[tid] += vmem[tid + 8];
  vmem[tid] += vmem[tid + 4];
  vmem[tid] += vmem[tid + 2];
  vmem[tid] += vmem[tid + 1];
}
```

```
// write result for this block global mem
if (tid == 0) g_odata[blockIdx.x] = idata[0];
```
