The following is a row-based implementation with an unrolling factor of four. Two new array indices are introduced: one for row access, and the other for column access.

```
__global void transposeUnroll4Row(float *out, float *in, const int nx,
    const int ny) {
    unsigned int ix = blockDim.x * blockDim.x*4 + threadIdx.x;
    unsigned int iy = blockDim.y * blockDim.y + threadIdx.y;

    unsigned int ti = iy*nx + ix; // access in rows
    unsigned int to = ix*ny + iy; // access in columns

    if (ix+3*blockDim.x < nx && iy < ny) {
        out[to] = in[ti];
        out[to] + ny*blockDim.x] = in[ti+blockDim.x];
        out[to] + ny*2*blockDim.x] = in[ti+2*blockDim.x];
        out[to] + ny*3*blockDim.x] = in[ti+3*blockDim.x];
    }
```

Exchanging the row and width indices produces a column-based implementation with similar unrolling:

```
__global void transposeUnroll4Col(float *out, float *in, const int nx,
    const int ny) {
    unsigned int ix = blockDim.x * blockDim.x*4 + threadIdx.x;
    unsigned int iy = blockDim.y * blockDim.y + threadIdx.y;

    unsigned int ti = iy*nx + ix; // access in rows
    unsigned int to = ix*ny + iy; // access in columns

    if (ix+3*blockDim.x < nx && iy < ny) {
        out[ti] = blockDim.x*ny;
        out[ti] + 2*blockDim.x*ny] = in[ti+2*blockDim.x*ny];
        out[ti] + 3*blockDim.x*ny] = in[ti+3*blockDim.x*ny];
    }
```

Adding the following code to the kernel switch statement, before unrolling was added, the grid size for the switch must be adjusted.

```
case 4:
    kernel = &transposeUnroll4Row;
    kernelName = "Unroll4Row";
    grid.x = (nx+block.x*4-1)/(block.x*4);
    break;

case 5:
    kernel = &transposeUnroll4Col;
    kernelName = "Unroll4Col";
    grid.x = (nx+block.x*4-1)/(block.x*4);
    break;
```
