void tick() { // +1 second
    if(s_ >= 59) {
        s_ = 0;
        if(m_ >= 59) {
            m_ = 0;
            if(h_ >= 23) h_ = 0;
            else { h_ += 1; }
        } else { m_ += 1; }
    } else { s_ += 1; }
}
