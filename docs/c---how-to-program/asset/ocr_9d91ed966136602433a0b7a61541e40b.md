// Parameter b must be a positive integer
// to prevent infinite recursion
unsigned int mystery(unsigned int a, unsigned int b)
{
    // base case
    if (1 == b) {
        return a;
    }
    else { // recursive step
        return a + mystery(a, b - 1);
