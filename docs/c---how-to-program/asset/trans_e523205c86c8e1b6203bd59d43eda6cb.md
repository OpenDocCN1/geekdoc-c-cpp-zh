### 3.28 以下程序打印什么？

```c
#include <stdio.h>

int main( void )
{
   int x = 1;
   int total = 0;

   while ( x <= 5 ) { // loop 5 times
      total += x; // perform calculation
      printf( "%d\n", x*x);
      ++x; // increment counter x
   } // end while

   printf( "Total is %d\n", total); // display result
} // end function main
```

### 3.29 以下程序打印什么？

```c
#include <stdio.h>

int main( void )
{
   int outer_count = 1; // initialize count
   while ( outer_count <= 10 ) { // loop 10 times
      int inner_count = 1;
      while ( inner_count <= outer_count ) {
         printf( "*" );
         inner_count++;
      } // end inner while
      printf( "\n" );
      outer_count++;
   } // end outer while
} // end main
```
