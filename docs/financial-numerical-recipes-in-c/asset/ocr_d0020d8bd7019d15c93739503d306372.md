inline double hT(const double& T, const double& delta, const double& pi){
    return (1.0/pi*((1-pi)*pow(delta,T)));
};

double term_structure_class_ho_lee::d(const double& T) const{
    double d=(initial_term_structure).d(T(n+1))/(initial_term_structure).d(n);
    for (int j=1;j<=n;++j){
        d *= hT(T(n-j+1),delta,pi_) / hT(n-j,delta,pi_) ;
    };
    d *= hT(T,delta,pi_)*pow(delta,T*(n-i));
    return d;
};
