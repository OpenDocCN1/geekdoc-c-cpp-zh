**Algorithm 20** This script finds the pid of the process firefox and collects its memory usage. - Exercise 35

```
#!/bin/bash
pid=$(pidstat -O firefox | awk 'NR=4{print $4}')
printf "process Resource Utilization \\n" >> process-mid-$pid.txt
ps -p $pid -o $PID | awk 'NR=1{printf "%s \t %s \t %s \t %s \t %s \\n" $2, $3, $4, $5, $6, $1}' >> process-mid-$pid.txt
for i in {1..100};
do
  pidstat -u -p $pid | awk 'NR=2{printf "%d \t %.2f \t %.2f \t %d \t %s \\n" $2, $3, $4, $5, $6, $11}' >> process-mid-$pid.txt
  sleep 12
done
```
