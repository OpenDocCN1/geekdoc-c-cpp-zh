# **Algorithm 13** CPU Utilization Considering the State Classes - Collection Script - Exercise 27

```
top -b n 1 | awk 'NR=3{printf "%%s %%t%%s %%t%%s %%t%%s %%t%%s %%t%%s %%t%%s %%t%%s\n", $3, $5, $7, $9, $11, $13, $15, $17}' >> topcpu.txt;
for i in (1..100)
do
  top -b n 1 | awk 'NR=3{printf "%%3.2f%%t%%3.2f%%t%%3.2f%%t%%3.2f...\n", $2/100, $4/100, $6/100, $8/100, $10/100, $12/100, $14/100, $16/100}' >> topcpu.txt;
sleep 120;
done
```
