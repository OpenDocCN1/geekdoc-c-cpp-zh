**42.** To learn how nested `for` loops work, do a walk-through of the following program segments and determine, in each case, the exact output. (4, 7)

```
a.
```cpp
int i, j;

for (i = 1; i <= 5; i++)
{
    for (j = 1; j <= 5; j++)
        cout << setw(3) << i;
    cout << endl;
}
```
```

```
b.
```cpp
int i, j;
for (i = 1; i <= 5; i++)
{
    for (j = (i + 1); j <= 5; j++)
        cout << setw(5) << j;
    cout << endl;
}
```

const int M = 10;
const int N = 10;
int i, j;

for (i = 1; i <= M; i++)

    for (j = 1; j <= N; j++)
        cout << setw(3) << M * (i - 1) + j;

e.
```cpp
int i, j;

for (i = 1; i <= 9; i++)
{
    for (j = 1; j <= (9 - i); j++)
        cout << " ";
    for (j = 1; j <= i; j++)
        cout << setw(1) << j;
    for (j = (i - 1); j >= 1; j--)
        cout << setw(1) << j;
    cout << endl;
}
