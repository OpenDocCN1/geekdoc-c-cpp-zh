{
    isSorted = true; //assume that the sublist is sorted

    for (int index = 0; index < length - iteration; index++)
    {
        if (list[index] > list[index + 1])
        {
            elemType temp = list[index];
            list[index] = list[index + 1];
            list[index + 1] = temp;
            isSorted = false;
        }
    }
}
