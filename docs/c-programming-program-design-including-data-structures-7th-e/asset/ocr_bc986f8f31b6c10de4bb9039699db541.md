```
int mystery(int num)
{
    int y = 1;

    if (num == 0)
        return 0;
    else if (num < 0)
        for (int count = 0; count > num; count--)
            y = y - (num + count);
    else
        for (int count = 1; count < num; count++)
            y = y * (num - count);

    return y;
}
```
