8 EXAMPLE 2 Division of double _Complex operands could be implemented as follows.

```
#include <math.h>
#include <complex.h>
/* Divide z / ... */
double complex cdiv(double complex z, double complex w)
{
    #pragma STDC_FP_CONTRACT_OFF
    double a, b, c, d, logb, denom, x, y;
    int ilogb = 0;
    a = creal(z); b = cimag(z);
    c = creal(w); d = cimag(w);
    logb = logb(fabs(fabs(c), fabs(d)));
    if (isfinite(ilogb)) {
        ilogb = (int)ilogb;
        c = scaln(d, -ilogb);
    }
    denom = c * c + d * d;
    x = scaln(a * c + b * d) / denom, -ilogb;
    y = scaln(b * c - a * d) / denom, -ilogb;
}

/* Recover infinities and zeros that computed as NaN+Inf;NaN+; */
/* the only cases are nonzero, infinite, infinfinite, and infinite... */
if (isnan(x) && isnan(y)) {
    if ((denom == 0.0) || (isnan(b))) {
        if (!isnan(a) || isinf(b)) {
            x = copysign(INFINITY, a);
            y = copysign(INFINITY, b);
        }
    }
    else if ((isfinite(a) || isinf(b)) &&
    isfinite(c) && isfinite(d)) {
        a = copysign(c) ? 1.0 : 0.0, a;
        b = copysign(b) ? 1.0 : 0.0, b;
        x = INFINITY * (a * c + b * d);
        y = INFINITY * (b * c - a * d);
    }
    else if (logb == INFINITY) &&
    isfinite(a) && isfinite(b)) {
        c = copysign(c) ? 1.0 : 0.0, c;
        d = copysign(d) ? 1.0 : 0.0, d;
        x = 0.0 * (a * c + b * d);
        y = 0.0 * (b * c - a * d);
    }
}
```
