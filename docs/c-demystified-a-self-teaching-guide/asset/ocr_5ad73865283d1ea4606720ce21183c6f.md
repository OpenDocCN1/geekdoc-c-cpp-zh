each sale of each salesperson, and afterward displays the average sale for each salesperson. The number of iterations of the outer loop will be the number of salespeople. The number of iterations of the inner loop will be the number of sales per salesperson.

```
#include <iostream>
using namespace std;
int main(void)
{
    int persons, int numSales;
    cout << "Enter number of salespeople: ";
    cin >> persons;
    cout << "Enter number of sales per salesperson: ";
    cin >> numSales;
    for (int x = 1; x <= persons; x++)
    {
        int sale, total = 0;
        float average;
        for (int y = 1; y <= numSales; y++)
        {
            cout << "Enter sale " << y << " for salesperson " << x << ":";
            cin >> sale;
            total += sale;
        }
        average = (float)total / numSales;
        cout << "Average sales for salesperson #" << x << " is " << average << endl;
    }
    return 0;
}
```

Note If you place a break or continue keyword in the inner loop, it will affect only that inner loop, and have no effect on the outer loop.
