namespace mySpace{
    int gcd(int,int);
    int lcm(int,int);
}
```

(a) The file mySpace.h

```
#include "mySpace.h"

int mySpace::lcm(int m,int n){
    return m*n/gcd(m,n);
}
int mySpace::gcd(int m, int n){
    if(n == 0) return m;
    else return gcd(n, m % n);
}
```

(b) The file impl.cpp

```
#include <simplecpp>

    cout << lcm(36,24) << endl;
