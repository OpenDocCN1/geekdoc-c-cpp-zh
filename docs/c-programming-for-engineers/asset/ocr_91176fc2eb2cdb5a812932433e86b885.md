# 2.4. Control Structures

```
BEGIN
DECLARE age
INPUT age
IF age >= 18 THEN
    OUTPUT "OK to vote"
ELSE
    OUTPUT "Not eligible yet"
ENDIF
END
```

```
#include <stdio.h>

int main() {
    int age;

    printf("Enter your age: ");
    scanf("%d", &age);

    if (age >= 18) {
        printf("OK to vote\n");
    } else {
        printf("Not eligible yet\n");
    }

    return 0;
}
```
