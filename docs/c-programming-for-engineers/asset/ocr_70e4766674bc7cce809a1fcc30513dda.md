Example : Here's a simple implementation of a linked list-based stack in C:

```
#include <stdio.h>
#include <stdlib.h>
```

```
// Define a node structure for the linked list
struct Node {
    int data;
    struct Node* next;
};
```

```
// Define the stack structure
struct Stack {
    struct Node* top; // Top pointer points to the latest element
};
```

```
// Create a new stack
struct Stack* createStack() {
    struct Stack* stack = (struct Node*)malloc(sizeof(struct Stack));
    stack->top = NULL;
    return stack;
}
```

```
// Check if the stack is empty
int isEmpty(struct Stack* stack) {
    return stack->top == NULL;
}
```

```
// Push (add an element) to the top of the stack
void push(struct Stack* stack, int data) {
    // Create a new node
    struct Node* new_node = (struct Node*)malloc(sizeof(struct Node));
    new_node->data = data;
    new_node->next = stack->top;

```
// Update the top pointer to point to the new node
stack->top = new_node;
