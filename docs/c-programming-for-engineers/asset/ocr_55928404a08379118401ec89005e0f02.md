int main() {
    int a, b, c;

    // Request user input for the lengths of three sides of a triangle
    printf("Enter the lengths of three sides of a triangle: %%d");
    scanf("%%d %%d %%d", &a, &b, &c);

    // Check if the input forms a valid triangle
    if (a < 0 || b <= c || c <= 0) {
        printf("\Not a triangle\n");
    } else if (a >= b || b >= c || c >= a + b) {
        printf("\Not a triangle\n");
    } else {
        // Determine the type of triangle
        if (a == b && b == c) {
            printf("\nequilateral triangle\n");
        } else if (a == b || b == c || c == a) {
            printf("\nisosceles triangle\n");
        } else {
            printf("\nscalene triangle\n");
        }
    }

    return 0;
}
