fclose(fp);

Here is a simple example demonstrating how to open, write to, and close a file:

```
#include <stdio.h>

int main() {
    FILE *fp;

    // Open a file in write mode
    fp = fopen("my_file.txt", "w");

    // Check for errors during opening
    if (fp == NULL) {
        printf("Error opening file!\n");
        return 1;
    }

    // Write content to the file
    fprintf(fp, "This is some text written to the file.\n");
    fprintf(fp, "This is another line of text.\n");

    // Close the file
    fclose(fp);

    printf("File written successfully!\n");

    return 0;
}
```

This code includes error checking and demonstrates opening a file, writing to it, and closing it.
