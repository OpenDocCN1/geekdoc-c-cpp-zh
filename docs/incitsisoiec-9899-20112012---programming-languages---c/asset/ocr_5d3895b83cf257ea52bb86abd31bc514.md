EXAMPLE 2 The function f3 is similar, saves the status of the variable argument list after f2 has been called once the whole list, the trailing part of the list is gather and passed to function f4.

```
#include <stdarg.h>
#define MAXARGS 31

void f3(int n_ptrs, int f4_after, ...)
{
    va_list(ap, ap_save);
    char *array[MAXARGS];
    int ptr_no = 0;
    if (n_ptrs > MAXARGS)
        n_ptrs = MAXARGS;
    va_start(ap, f4_after);
    while (ptr_no < n_ptrs)
        array[ptr_no++] = va_arg(ap, char *);
        if (ptr_no == f4_after)
            va_copy(ap_save, ap);
    va_end(ap);
f2(n_ptrs, array);
    // Now process the saved copy.
n_ptrs = f4_after;
ptr_no = 0;
while (ptr_no < n_ptrs)
    array[ptr_no++] = va_arg(ap_save, char *);
va_end(ap_save);
f4(n_ptrs, array);
}
```
