template <typename T1, typename T2>

// 'Nested' alias
template <typename T>
    using Vec = std::vector<T>;

template <typename T1, typename T2>
    using C2 = C<T1, Vec<T2>>;

template <typename T>
    using C3 = C<T, int>;

using C4 = C<double, int>;
// Using alias template
C4 c4; // 0,0
c4.print();

int j = 3;
C3<float> c3(f, j);
c3.print(); // 1,3

// Nested alias in class C2 for readability
Vec<double> v(100);
