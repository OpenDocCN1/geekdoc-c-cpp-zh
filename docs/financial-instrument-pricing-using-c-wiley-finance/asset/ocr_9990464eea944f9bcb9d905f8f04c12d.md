{

    public:

                        yValues(std::vector<value_type>())

        {
            tValues.push_back(t); yValues.push_back(x[0]);
        }

        const std::vector<value_type>& xAxis() const
        {
            return tValues;
        }

        const std::vector<value_type>& yAxis() const
        {
            return yValues;
        }
