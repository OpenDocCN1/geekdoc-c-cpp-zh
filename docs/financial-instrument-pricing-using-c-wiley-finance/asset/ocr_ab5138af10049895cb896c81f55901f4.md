```
// Backward sweep Ux = y
result[N-1] = result[N-1]/U(N-1, N-1);
for (size_t i = N-2; i >= 0 && i < N; --i)
{
    sum = 0.0;
    for (size_t k = i+1; k < N; ++k)
    {
        sum += U(i,k)*result[k];
    }
    result[i] = (result[i] - sum) / U(i,i);
}

return result;
}
```
