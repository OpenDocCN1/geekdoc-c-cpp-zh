```
// Generalised Riccati
auto p = [=] (double x) { return P; };
auto q = [=] (double x) { return Q; };
auto r = [=] (double x) { return R; };
auto n = [=] (double x, double y)
{ return (lambda*y) / (eta - y); };

return std::shared_ptr<GeneralisedRiccati>
    (new GeneralisedRiccati (p, q, r, n));
}
```
