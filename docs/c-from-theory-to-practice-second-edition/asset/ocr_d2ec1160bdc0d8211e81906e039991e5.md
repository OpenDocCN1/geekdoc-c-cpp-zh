void show_board(int c[][COLS], int s[][COLS])
{
    int i, j;

    for(i = 0; i < ROWS; i++)
    {
        for(j = 0; j < COLS; j++)
        {
            if(s[i][j] == 1)
                printf("%d ", c[i][j]);
            else
                printf("* ");
        }
        printf("\n");
    }
}
