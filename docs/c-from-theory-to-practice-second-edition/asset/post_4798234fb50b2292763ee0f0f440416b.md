int hash_func(int key) /* We write it separately from the rest of the code to make it clear. If we want to use another hashing function, we just change that line. */

void insert_key(node **cur, int key)
{
    node *prev, *tmp, *p;

    tmp = (node*) malloc(sizeof(node));
    if(tmp == NULL)
    {
        printf("Error: Not available memory\n");
        exit(EXIT_FAILURE);
    }
    tmp->key = key;
    tmp->next = NULL;

    if(*cur == NULL) /* We check if it is the first time to insert a key in that position. If it isn't, we add the new node at the end of the list. */
        *cur = tmp;
    else
    {
        p = prev = *cur;
        while(p != NULL)
        {
            prev = p;
            p = p->next;
        }
        p = tmp;
        prev->next = p;
    }
}

node* find_key(node *p, int key)
{
    while(p != NULL)
    {
        if(p->key == key)
            return p;
        p = p->next;
    }
    return NULL;
}

{

        }
    }

## Exercise

C.14.19 Write the following function to delete the key from the array.

```c
int del_key(node **cur, int key);
```

Here is an example how to call it:

```c
pos = hash_func(key);
del_key(&hash_arr[pos], key);
```

## Answer:

```c
int del_key(node **cur, int key)
{
    node *p, *prev;
    p = prev = *cur;
    while(p != NULL)
    {
        if(p->key == key)
        {
            if(p == *cur)
                *cur = p->next;
            else
                prev->next = p->next;
            free(p);
            return 0;
        }
        prev = p;
        p = p->next;
    }
    return -1;
}
```
