int mem_cmp(const void *ptr1, const void *ptr2, size_t size)
{
    char *p1, *p2;
    p1 = (char*)ptr1;
    p2 = (char*)ptr2;
    while(size != 0)

        p1++;
        p2++;
        size--;
    }
    return 0;
}
