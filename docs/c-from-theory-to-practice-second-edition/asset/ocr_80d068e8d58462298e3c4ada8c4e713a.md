void find_book(FILE *fp, char title[])
{
    book b;
    fseek(fp, 0, SEEK_SET);
    while(1)
    {
        if(fread(&b, sizeof(book), 1, fp) != 1)
            break;
        else
        {
            if(strcmp(b.title, title) == 0)
            {
                printf("\nT:%s A:%s C:%d", b.title, b.auth,
                b.code);
                return;
            }
        }
    }
    printf("\n%s doesn't exist", title);
}
