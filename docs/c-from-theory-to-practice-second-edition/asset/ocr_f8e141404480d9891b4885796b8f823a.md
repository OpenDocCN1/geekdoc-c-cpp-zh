node* add_node(node *p, int key)
{
    if(p == NULL)
    {
        p = (node*) malloc(sizeof(node));
        if(p == NULL)
        {
            printf("Error: Not available memory\n");
            exit(EXIT_FAILURE);
        }
        p->left = p->right = NULL;
        p->key = key;
    }
    else
    {
        if(key < p->key)
            p->left = add_node(p->left, key);
        else if(key > p->key)
            p->right = add_node(p->right, key);
        else
            printf("Error: Key(%d) exists\n", key);
    }
    return p;
}
