public class MainClass {

        Book b = new Book();

        System.out.print("Price: ");
        b.prc = scan.nextFloat();

        System.out.print("Code: ");
        b.code = scan.next();

        System.out.print("Title: ");
        b.title = scan.next();

        System.out.print("Auth: ");
