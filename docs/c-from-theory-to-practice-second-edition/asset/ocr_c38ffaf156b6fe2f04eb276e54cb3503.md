    new_node = (node*) malloc(sizeof(node));
    if(new_node == NULL)
    {
        printf("Error: Not available memory\n");
        exit(EXIT_FAILURE);
    }
    *new_node = *p;
    new_node->next = NULL;

    if(head == NULL)
        head = tail = new_node; /* If the queue is empty, both head and tail pointers point to the new node. */
    else
    {
