```cpp
#include <iostream>
using namespace std;

class Student
{
public:
    int code;
    float grd;
};

int main()
{
    Student *ptr;
    int i, num, cnt;
    float grd;

    cout << "Enter number of students: ";
    cin >> num;

    try
    {
        ptr = new Student[num];
    }
    catch(bad_alloc)
    {
        cout << "Unavailable memory\n";
        exit(1);
    }
    cout << "Enter base: ";
    cin >> grd;
    cnt = 0;
    for(i = 0; i < num; i++)
    {
        cout << "Code: ";
        cin >> ptr[i].code;

        cout << "Grade: ";
        cin >> ptr[i].grd;

        if(ptr[i].grd > grd)
            cnt++;
    }
    cout << cnt << " students got more than " << grd << endl;

    delete[] ptr;
    return 0;
}
