# Code: Working with Files and Streams

```
```cpp
#include <iostream>
#include <fstream>
using namespace std;
int main()
{
    ofstream ofObj;
    string ourLine;
    ofObj.open("textFile3.txt");
    while (ofObj)
    {
        getline(cin, ourLine);
        if (ourLine == "-1")
            break;
        cout<< ourLine << endl;
    }
    ofObj.close();
    ifstream ifObj;
    ifObj.open("textFile3.txt");
    while (ifObj)
    {
        getline(ifObj, ourLine);
        cout << ourLine << endl;
    }
    ifObj.close();
    return 0;
}
```
