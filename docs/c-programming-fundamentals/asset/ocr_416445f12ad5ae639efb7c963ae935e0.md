int main()
{
    union OurUnion u1;
    u1.x = 85;
    u1.c = 'H';
    cout << "The value is : "<< u1.x << endl;
    cout << "The value is : "<< u1.c << endl;
    cout<<"Size of union  : "<<sizeof(u1)<<endl;

}
