```c
int ProcessArray(int *listA, int x, int n) (GATE 2014)
{
    int i, j, k;
    i = 0;
    j = n-1;
    do {
        k = (i+j)/2;
        if (x <= listA[k])
            j = k-1;
        if (listA[k] <= x)
            i = k+1;

}
```
