```

# define print(x) printf("%d \ x)
void P(int *y)
{
    int x = *y+2;
    Q(x);
    *y = x-1;
    print(x);
}
void main(void)
{
    x=5;
    P(&x);
    print(x);
    getchar();
}
int x;
void Q(int z)
{
    z += x;
    print(z);
}
```

a) 12 7 6
b) 22 12 11
