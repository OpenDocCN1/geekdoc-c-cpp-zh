The dynamically generated wrapper object can be treated like a visitor to the tree. The visitor accesses the nodes and brings the result back to the invoker. In the sample code, the CompositeNode structure not only defines the tree but also defines common ways to traverse the tree. It does the heavy lifting by encapsulating the tree traversal. The visitor defines how to process the single node and is responsible for bringing the result back to the invoker. In this sample, the visitor adds the value to the tree nodes and brings back the sum.

# LISTING 3-25 An example of the composite pattern

```
// define visitor interface
type IVistor<'T> =
    abstract member Do : 'T -> unit

// define a composite node
type CompositeNode<'T> =
    | Node of 'T
    | Tree of 'T * CompositeNode<'T> * CompositeNode<'T>
    with
      // define in-order traverse
      member.InOrder f =
        match with this
        | Tree(n, left, right) ->
            left.InOrder f
            f
            right.InOrder(f)
        | Node(n) -> f

      // define pre-order traverse
      member.PreOrder f =
        match with this
        | Tree(n, left, right) ->
            f
            left.PreOrder f
            right.PreOrder f
        | Node(n) -> f

      // define post-order traverse
      member.PostOrder f =
        match with this
        | Tree(n, left, right) ->
            left.PostOrder f
            right.PostOrder f
            f
        | Node(n) -> f

let invoke() =
    // define a tree structure
    let tree = Tree(1, Tree(11), Node(12), Node(13), Node(2))
```
