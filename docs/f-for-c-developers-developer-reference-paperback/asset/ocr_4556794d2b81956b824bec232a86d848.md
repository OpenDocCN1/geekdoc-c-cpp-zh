# LISTING 8-10 Simulating MapReduce on a local computer

```
let data = [| 1..100 |]

let reduceFunction list =
let r =
list
|> Seq.sum
printf "result = %A" r
r

let map () =
[0..9]
|> Seq.map (fun i -> i * 10, (i + 1) * 10 - 1)
|> Seq.map (fun a,b) ->
    printf "%A to %A: " a b
    async;
    let! sum = async { return! reduceFunction(data.[a..b]) } | Async.
StartChild
    return! sum
)

let reduce seq =
seq
|> Seq.sumBy Async.RunSynchronous

let mapReduce =
map()
|> reduce

printf "final result = %A" mapReduce
```
