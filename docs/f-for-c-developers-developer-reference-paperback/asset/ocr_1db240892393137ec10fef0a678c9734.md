# LISTING 7-55 Sum list using recursion, tail calls, and CPS

```
Recursive version
module RecursiveModule = 
let 1 = [1..1000000]

let rec sum 1 = =
match 1 with
| [] -> 0
| h:t -> h + sum t

sum 1

Tail call and CPS version
module TailCallModule = 
let 1 = [1..1000000]

let rec sum 1 acc = =
match 1 with
| [] -> acc
| h:t -> sum t (acc + h)

let res() = sum 1 0

module CPSModule = =
let 1 = [1..1000000]

let rec sum 1 cont = =
match 1 with
| [] -> cont 0
| h:t ->
let afterSum v = =
cont (h+v)
sum t afterSum

sum 1 id
```

Note The recursive version generates a stack overflow exception when processing a list.
