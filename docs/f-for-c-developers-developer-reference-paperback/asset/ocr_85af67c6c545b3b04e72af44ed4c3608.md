# LISTING 6-103 AllowIntoPattern in the CustomerOperation attribute

```
type SimpleSequence() =
  member this.For(source:seq<'a, f:'a -> seq<b>b>) =
    seq { for n in source do yield! f n }
  member this.Yield(n) = seq { yield n }
  [CustomOperation("", AllowIntoPattern=)]
  member this.where(source:seq<'a, [ProjectionParameter]>f:'a -> bool) =
    source |> Seq.filter f
nlet mySeq4 = SimpleSequence()
```

```
type SimpleSequence5() =

  member this.Return(n) = seq { yield n }
  [CustomOperation("", AllowIntoPattern=true,MaintainsVariableSpaceUsingBind=true)]
```
