# Using an Array, List, and Seq

In this section, I will present several examples related to collection data structures, such as arrays, lists, and sequences. One of the most common problems for processing these collection data structures is sorting.

Sorting data types The first example demonstrates how to use the built-in sorting functions to do sorting. This is shown in List 7-7. The portable library does not support the printf statement, so the testing of this code is performed in a console application.

# LISTING 7-7 An example of sorting data structures

```fsharp
open System

// type alias for IEnumerable
type IE = System.Collections.IEnumerable

// define record type
type MyRecord = { X : int; Y : string }

// define DU type
type MyDu =
  | Case1 of string
  | Case2 of int

// define a class
type MyClass() =
  member val MyProperty = 0 with get, set
  member val MyProperty2 = String.Empty with get, set
  override this.ToString() =
    printf("%A, %A", this.MyProperty, this.MyProperty2)

type DataSource() =
  let random = Random()

// define data sequence
let seq0 = [2; 3; 4; 5; 10; 11; 13; 14; 19]
let seq1 = seq0 | Seq.map (fun x -> float x)
let seq2 = seq0 | Seq.map (fun x -> bigint x)
let seqTuple = Seq.zip seq0 | Seq.map (printf "%A")
let seqRecord = seqTuple | Seq.map (fun (i, s) -> (i; s))
let seqDU1 = seqTuple | Seq.map (fun (i, s) -> Case2(i))
let seqDU2 = seqTuple | Seq.map (fun (i, s) -> Case1(s))
let seqClass = seqTuple | Seq.map (fun (i, s) -> MyClass(MyProperty = i, MyProperty2 = s))

// define data list; this list is an F# list type
let list0 = seq0 | Seq.toList
let list2 = seq2 | Seq.toList
let listTuple = seqTuple | Seq.toList
let listRecord = seqRecord | Seq.toList
let listDU = seqDU1 | Seq.toList
let listClass = seqClass | Seq.toList

let toString x =
  let seq = x
    |> Seq.toList
    |> List.map (fun n -> sprintf "%A" n)
  String.Join(",", seq)

member this.Seqs : list<IE> =
  [ seq0
    seq1
    seq2
    seqTuple
    seqRecord
    seqDU1
    seqDU2
    seqClass ]

member this.Lists : list<IE> =
  [ list0
    list2
    listTuple
    listRecord
    listDU
    listClass ]

member this.TestSeq() =
  let seq0 = seq0 | Seq.sort | toString
  let seq1 = seq1 | Seq.sort | toString
  let seq2 = seq2 | Seq.sort | toString

let listTuple=seqTuple | Seq.sort | toString
let listRecord=seqRecord | Seq.sort | toString
let listDU=seqDU1 | Seq.sort | toString
let listDU2=seqDU2 | Seq.sort | toString

let r =
  [ seq0
    seq1
    seq2
    seqTuple
    seqRecord
    seqDU1
    seqDU2
    seqClass ] 
  |> Array.ofList

member this.TestList() =
  let list0 = list0 | Seq.sort | toString

let list2 = list2 | Seq.sort | toString
let listTuple=listTuple | Seq.sort | toString
let listRecord=listRecord | Seq.sort | toString
let listDU=seqDU1 | Seq.sort | toString
let listDU2=seqDU2 | Seq.sort | toString

let r =
  [ list0
    list2
    listTuple
    listRecord
    listDU
    listDU2 ] 
  |> Array.ofList

member this.TestSortBy_Tuple() =
  let r = seqTuple | Seq.sortBy (fun (i, s) -> i) | toString

member this.TestSortBy_Record() =
  let r = seqRecord | Seq.sortBy (fun (n, s) -> n) | toString

member this.TestSortBy_DU() =
  let r = seqDU1 | Seq.sortBy (fun (i, s) -> n) | toString

member this.TestSortBy_Class() =
  let r = seqClass | Seq.sortBy (fun (n) -> n.MyProperty) | toString

Console application to test the portable library
```
