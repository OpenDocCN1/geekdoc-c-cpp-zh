# LISTING 8-11 MapReduce with an index range

```
let data = [| 1..100 |]

let reduceFunction a b =
let r =
    data.[a..b]
    |> Seq.sum
    printf"result = %A" r
r
let map () = =
[0..9]
|> Seq.map (fun i -> i * 10, (i + 1) * 10 - 1)
|> Seq.map (fun (a, b) ->
    printf"from A: %A" b
    async =
        let sum = async { return reduceFunction a b } |> Async.StartChild
        return! sum
    )

let reduce seq =
seq
|> Seq.sumBy Async.RunSynchronously

let mapReduce =
map()
|> reduce

printf "final result = %A" mapReduce
```
