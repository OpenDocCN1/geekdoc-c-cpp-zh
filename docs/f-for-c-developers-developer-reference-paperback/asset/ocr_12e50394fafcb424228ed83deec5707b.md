# LISTING 7-60 F# code for WebSharper ASP.NET control

```
namespace Website

open IntelliFactory.WebSharper
open IntelliFactory.WebSharper.Html

module ScriptModule =
    [JavaScript]
    let Main() =
        let input = Input [Attr.Type "Text"]
        let output = Label []
        Div [
            input
            Button [Text "Click Me!"]
            |> OnClick (fun e args ->
                output.Text <- input.Value)
        ]
ntype MyControl() =
    inherit Web.Control()

    [JavaScript]
    override this.Body = ScriptModule.Main() => _
```
