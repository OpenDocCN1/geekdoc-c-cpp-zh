let connectionString = "MyConnectionString"
let storageAccount = =
CloudStorageAccount.Parse(CloudConfigurationManager.GetSetting(connectionString))
let queueClient = storageAccount.CreateCloudQueueClient()
let queue = queueClient.GetQueueReference("myqueue")
ignore < queue.CreateIfNotNotExist()

while(true) {
  Thread.Sleep(200)
  // Peek at the next message
  let peekedMessage = queue.PeekMessage()
  if (peekedMessage != null) then {
    let msg = queue.GetMessage()
    log msg.AsString("Information")
    queue.DeleteMessage(msg)
  } else {
    log "no message" "Information"
  }
}

Microsoft.WindowsAzure.Configuration.dll needs to be added to the project references.

Before executing the code, you need to set the instance number for ConsumerRole and WorkerRole in the Configuration area in the project settings. For purposes, the ConsumerRole project instance count is set to I, so there is only one consumer removing the messages from the queue. The WorkerRole project instance count is set to 3.
