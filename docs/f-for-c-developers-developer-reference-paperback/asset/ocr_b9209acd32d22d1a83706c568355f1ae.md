let curandLogNormal(generator, n:int, mean, stddev) = =
let unitSize = Marshal.SizeOf(typeof<float32>)
let size = Size(n * unitSize)
let mutable devicePtr = Unchecked.defaultofIntPtr
let r = CUDAuntime32.cudaMalloc(&devicePtr, size)
