```
this.TriggerReset()
```

```

```

```
// clear current list
member this.Clear() =
base.Clear()
this.TriggerReset()
```

```
// insert element at the index location
member this.Insert(index, item) =
base.Insert(index, item)
this.TriggerAdd(index, item)
```

```
// insert multiple elements at the index location
member this.InsertRange(index, items) =
items | Seq.iter(fun i -> this.Insert(index+i, items))
```

```
// define an index
member this.Item
with get(i) = base.[i]
and set v =
let old = base.[i]
base[i] <- v
this.TriggerReplace(v, old, i)
```

```
// trigger add, remove, replace, and reset events
private this.TriggerAdd(item, index)
if this.IsTriggerEvent then
collectionChanged.Trigger(this,
NotifyChangedEventArgs(NotifyChangedEventArgs.Add, item,
index))
```
