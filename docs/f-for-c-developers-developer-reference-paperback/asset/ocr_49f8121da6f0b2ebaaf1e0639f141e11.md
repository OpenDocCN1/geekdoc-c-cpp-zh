member private this.SetParameter(func, size) =
if Interop.Library.SetParameter(func, size) = CUResult.Success then
else failWith("set parameter size failed")
member this.SetParameter(func, parameters) =
let mutable num = 0
for para in parameters do
match box(para) with
  |? uint32 as n ->
    num <- align(num, 4)
    if Interop.Library.cuParamSetI(func, num, n) = CUResult.Success then
    else failWith("set uint32 failed")
    num += 4
  |? float32 as f ->
    num <- align(num, 4)
    if Interop.Library.cuParamSet(func, num, f) = CUResult.Success then
    else failWith("set float failed")
    num += 4
  |? int64 as i64 ->
    num <- align(num, 8)
    let mutable i64Ref = i64
    if Interop.Library.cuParamSet(func, num, &i64Ref, 8u) = CUResult.Success
else failWith("set int64 failed")
num += 8
  |? char as ch ->
    num <- align(num, 2)
let bytes = Encoding.Unicode.GetBytes([[ch]])
