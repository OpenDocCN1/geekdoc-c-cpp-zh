let findLCA(node0, node1) root =
  let rec commonAncestor(node0, node1) root =
    match with root
    | Nil -> None
    | Node(v, left, right) ->
      match left with
      | Nil, _ -> None
      | Node(v, _, _), when v = node0 || v = node1 -> Some root
      | Node(v, _, _), when v = node0 || v = node1 -> Some root
      | _ ->
        let l = left | commonAncestor(node0, node1)
        let r = right | commonAncestor(node0, node1)
        match l, r with
        | None, _ -> None
        | Some(_), Some(_) -> Some root
        | Some(_) as n, None -> n
        | None, Some(_) as n -> n
