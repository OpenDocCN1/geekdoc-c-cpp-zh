```fsharp
let getCommonCode() = code.ToCode()

let getGPUFunctions() =
    let currentAssembly = Assembly.ExecutingAssembly()
    let gpuMethods =
        currentAssembly
        .GetTypes()
        |> List.OfArray
        |> List.filter(fun mi ->
            mi.CustomAttributes<GPUAttribute>().Length > 0)

    gpuMethods
    |> List.map(fun mi -> (mi, Expressions.TryReflectedDefinition(mi)))
    |> List.map(fun (mi, expr) -> (mi, expr))
    |> List.map(fun (mi, expr) ->
        sprintf("%s \r\n%s", getFunctionSignatureFromInfo(mi), expr))

let tempFile = "temp1.cu"

let generateToFile() =
    let gpuCode = getGPUFunctions()
    let commonCode = getCommonCode()
    let allCode = String.Join("\r\n", gpuCode)
    System.IO.File.Delete(tempFile)
    System.IO.File.WriteAllText(tempFile, allCode)
```

The generated C code will not be able to execute on the GPU. You can use NVCC to compile the C code to a PTX file. The PTX file can be loaded and executed on the GPU. The PTX code is not an executable binary. Instead, PTX file is compiled for target GPU binary at runtime. It is more like assembly language on the GPU.
