// define a DU type
type MyDU =
| Case1 of int
| Case2 of string
| Case3 of MyRecord
| Case4 of MyTuple

// define a comparison function
let inline compare x y =
  if x > y then
    printf("%A %A" x y)
  elif x=y then
    printf("%A %A" x y)
  else
    printf("%A < %A" x y)

// create records
let record1 = { X = 1; Y = "1" }
let record2 = { X = 1; Y = "2" }
let record3 = { X = 2; Y = "1" }
let record4 = { X = 1; Y = "1" }

compare record1 record2
compare record1 record3
compare record2 record3
compare record1 record4

// create tuples
let tuple1 : MyTuple = 1, "1"
let tuple2 : MyTuple = 1, "2"
let tuple3 : MyTuple = 2, "1"
let tuple4 : MyTuple = 1, "1"

compare tuple1 tuple2
compare tuple1 tuple3
compare tuple2 tuple3
compare tuple1 tuple4

// create DU Case instances
let du1 = Case1(1)
let du2 = Case1(2)
let du3 = Case1(1)

compare du1 du2
compare du1 du3
compare du2 du3

// create Case3 instances
let duRecord1 = Case3(record1)
let duRecord2 = Case3(record2)
let duRecord3 = Case3(record3)
let duRecord4 = Case3(record4)
