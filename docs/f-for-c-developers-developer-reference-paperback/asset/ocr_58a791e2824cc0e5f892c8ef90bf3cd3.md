LISTING 7-59 Recursive and CPS Fibonacci functions

```
module FibModule =
  rec fib x =
    match x with
    | when x = 1I -> 1I
    | when x = 2I -> 1I
    | -> (fib (x - 1I)) + (fib (x - 2I))
  fib 30I

module FibCPSModule =
  rec fib x =
    match x with
    | when x = 1I -> cont 1I
    | when x = 2I -> cont 1I
    | ->
      let after1 a =
        let after2 b =
          cont(a + b)
        fib (x - 2I) after2
      fib (x - 1I) after1
  fib 30I
```
