                let r = execution.Runtime.ExecuteFunction(

                    [p, p2],
                    len,
                    1)

                let r = execution.Runtime.ExecuteFunction(
                    "pascalTriangle",
                    [p2, p],
                    len,
                    1)
            r
        ) | Seq.toList
let result1, o1 = execution.Runtime.CopyDeviceToHost(p, 10)

}
