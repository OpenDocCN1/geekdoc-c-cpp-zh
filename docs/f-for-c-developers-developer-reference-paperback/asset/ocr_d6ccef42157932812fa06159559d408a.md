Listing 6-34 uses DU implement a binary tree structure and traverses it with Seq.unfold. The algorithm uses a queue to hide the unvisited node.

# LISTING 6-34 Defining a binary tree using DU and performing a layer traversal

```
type NodeType = int

type BinaryTree =
    | Nil
    | Node of NodeType * BinaryTree * BinaryTree

let tree = Node(5,
    Node(42,
        Node(3, Nil, Nil),
        Node(2, Nil, Nil)),
    Node(4,
        Node(13,
            Node(14, Nil, Nil),
            Node(16, Nil, Nil)),
        Node(12,
            Node(15, Nil, Nil),
            Node(21,
                Node(22, Nil, Nil))))

let layerTraverse state =
    match state with
    | [] -> None
    | h:t ->
        match h with
            | Nil -> None
            | Node(v, Nil, Nil) -> Some (v, t)
            | Node(v, subtree, Nil)
            | Node(v, Nil, subtree) -> Some (v, t[subtree])
            | Node(v, l, r) -> Some (v, t[l;r]))
```

[tree]
|> Seq.unfold layerTraverse
|> Seq.toList

> Note If you replace t@[subtree] and t@[1;r] with [subtree]@ and t[1;r]@t, the algorithm becomes a pre-order traversal.
