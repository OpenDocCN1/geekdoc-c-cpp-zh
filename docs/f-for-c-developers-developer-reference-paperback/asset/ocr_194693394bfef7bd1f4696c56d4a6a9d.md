# LISTING 9-21 Translating .NET types to C type

```
type Type with
member this.HasInterface(t:Type) =
this.GetInterface(t.FullName) <> null

rec translateFromNETType(t:Type) (a:string) =
    if t = typeof(int) then "int"
    elif t = typeof(float32) then "float"
    elif t = typeof(float) then "double"
    elif t = typeof(bool) then "bool"
    let elementTypeString = translateFromNETType(t.GetElementType())
    inn        printf("List<%s>", elementTypeString)
    if t.HasInterface(typeof(System.Collections.IEnumerable<_>) then
        let elementTypeString = translateFromNETType(t.GetGenericArguments()[0]) in
        printf("List<%s>", elementTypeString)
    elif t = typeof(Microsoft.FSharp.Core.uint) then printf("%s", "double"
    elif t = typeof(UInt<float32>) then printf("%s", "Float"
    let cName = "FSharp<' + t.Name
    let input = translateFromNETType(t.GetGenericArguments()[0]) in
        let out = translateFromNETType(t.GetGenericArguments()[1]) in
        printf("%s(%%s)", input, out)
    elif t = typeof(System.Void) then
        "String.Empty"
    else failwith("not supported type")
```

```
let translateFromNETTypeToFunReturn(t:Type) (a:string) =
    let r = translateFromNETType t in
    if String.IsNullOrEmpty(r) then "void"
    else r
```

```
let translateFromNETTypeLength(t:Type) c =
    if t.IsArray then
        printf(" %A_len", c)
    else
        "String.Empty"
```

```
let isValueType(t:Type) =
    if t.IsValueType then true
    elif t.HasInterface(typeof(IEnumerable<_>) then false
    else failwith("is value type failed")
```
