```
let mutable result = []
let add x =
result <- result @ [x]
let visitLeaf root =
match root with
| Nil -> ()
| Node(v, Nil, Nil) -> add v
| Node(v, left, right) ->
visitLeaf left
visitLeaf right
```

```
let visitLeftBoundary root =
match root with
| Nil -> ()
| Node(v, Nil, Nil) -> ()
| Node(v, Nil, right) ->
add v
visitLeftBoundary right
| Node(v, left, _) ->
add v
visitLeftBoundary left
```

```
let visitRightBoundary root =
match root with
| Nil -> ()
| Node(v, Nil, Nil) -> ()
| Node(v, left, Nil) ->
visitRightBoundary left
add v
| Node(v, right) ->
visitRightBoundary right
add v
```

let getBoundary root =
match root with
| Nil -> ()
| Node(v, left, right) ->
add v
visitBoundary left
visitLeaf left
visitLeaf right
visitRightBoundary right
