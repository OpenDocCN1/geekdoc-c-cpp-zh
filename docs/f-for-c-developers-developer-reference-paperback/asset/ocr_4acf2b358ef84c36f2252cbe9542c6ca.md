type UDAPointer2<'T>(UDAPointer) =
new(ptr:IntPtr) = UDAPointer2(UDAPointer(ptr))
member this.Pointer = p
member this.PointerSize = p.PointerSize
member this.Is64bit = this.PointerSize = 8
member this.Item
with get (i:float32) = failwith "code generation only"
and set (i:int)(float32) = failwith "code generation only"
member this.Set(x:float32, i:int) = failwith "code generation only"
