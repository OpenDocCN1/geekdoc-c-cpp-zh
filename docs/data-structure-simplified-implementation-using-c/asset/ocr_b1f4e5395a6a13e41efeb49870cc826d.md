//counting nodes in a linked list
template<class T>
int list<T>::count()
{
    node<T> *current;
    int c=0;
    current=first;
    while(current!=NULL)
    {
        c++;
        current=current->next;
    }
    return c;
}
//function to display linked list
template <class T>
void list<T>::display()
{
    node<T> *current;
    current=first;
    cout<<"Linked list:"<<endl;
    while(current!=NULL)
    {
        cout<<current->data<<" -> ";
        current=current->next;
    }
    cout<<endl;
}
