    //counting nodes in a linked list
    int list::count()
{
    node *current;
    int c=0;
    current=first;
    while(current->next!=first)
    {
        c++;
        current=current->next;
    }
    return c+1;
}

    void list::display()
{
    node *current;
    current=first;
    cout<<"Linked list:"<<endl;
    while(current->next!=first)
    {
        cout<<current->data<<" -> ";

    cout<<endl;
