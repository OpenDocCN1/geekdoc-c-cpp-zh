template<class t>
int dlist<t>::search(t element)
{
    int flag=0;
    dnode<t> *current;
    int b=count();
    current=first;
    cout<<"searching element "<<element<<endl;
    for(int i=1;i<=b;i++)
    {
        if(current->data==element)
        {
            flag=1;
            break;
        }
        current=current->next;
    }
    return flag;
}
