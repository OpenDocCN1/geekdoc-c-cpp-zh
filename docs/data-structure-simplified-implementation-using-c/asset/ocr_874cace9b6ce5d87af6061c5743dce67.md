template<class T>
void list<T>::reverse()
{
    node<T> *a,*b,*temp;
    a=first;
    b=a->next;
    temp=b->next;
    a->next=NULL;
    while(temp!=NULL)
    {
        b->next=a;
        a=b;
        b=temp;
        temp=temp->next;

    }
    b->next=a;
