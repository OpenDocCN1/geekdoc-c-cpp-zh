private static void DisplayHelp()
{
    // Display the command-line help
    Console.WriteLine(
        "Compress.exe / Out:< file name > / Help \n"
        + " / Priority:RealTime | High | "
        + " AboveNormal | Normal | BelowNormal | Idle");
}
