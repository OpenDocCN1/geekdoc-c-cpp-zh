```
The converter to convert a Boolean value to the `Visibility` enumeration is defined in the class `BooleanToVisibilityConverter` (code file `WPFCultureDemo\\Converters\\BooleanToVisibilityConverter.cs`):

```
```csharp
using System;
using System.Globalization;
using System.Windows;
using System.Windows.Data;

namespace CultureDemo.Converters
{
    public class BooleanToVisibilityConverter: IValueConverter
    {
        public object Convert(object value, Type targetType, object parameter,
            CultureInfo culture)
        {
            bool b = (bool)value;
            if (b)
                return Visibility.Visible;
            else
                return Visibility.Collapsed;
        }

        public object ConvertBack(object value, Type targetType,
            object parameter, CultureInfo culture)
        {
            throw new NotImplementedException();
        }
    }
}
```
