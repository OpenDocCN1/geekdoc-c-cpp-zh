The calculation is within the OnCalculate method. OnCalculate is the handler for the Click event of the Calculate button (code file UICalculator/FuelEconomyShared/FuelEconomyViewModel.cs):

```
public void OnCalculate()
{
    double fuel = double.Parse(Fuel);
    double distance = double.Parse(Distance);
    FuelEconomyType ecoType = SelectedFuelEcoType;
    double result = 0;
    switch (ecoType.Id)
    {
        case"l pk":
            result = fuel / (distance / 100);
            break;
        case"mpg":
            result = distance / fuel;
            break;
        default:
            break;
    }
    Result = result.ToString();
}
```
