INCLUDE <PI8F4321.INC>
ORG 		0x100
MULT1 	EQU 	0x15
MULT2 	EQU 	0x17
SIGN1 	EQU 	0X50
SIGN2 	EQU 	0X51
 		MOVFW 	0xFE 	; Load first 8-bit data (-2) in WREG
 		MOVWF 	MULT1 	; Save in MULT1
 		MOVWF 	0xFC 	; Load 2nd 8-bit data (-4) in WREG
 		MOVWF 	MULT2 	; Save in MULT2
 		CLRF 	SIGN1 	; Clear [SIGN1] to 0
