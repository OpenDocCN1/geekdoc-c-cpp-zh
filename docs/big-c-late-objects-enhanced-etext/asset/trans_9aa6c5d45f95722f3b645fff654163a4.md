要查找该月第一天的星期，调用此函数：

/**
计算给定日期的星期。
@param year 年份
@param month 月份（1 = 一月 ... 12 = 十二月）
@param day 日
@return 星期（0 = 星期日 ... 6 = 星期六）
*/
int day_of_week(int year, int month, int day)
{
    int y = year;
    int m = month - 1;
    if (month < 3) { y--; m = m + 12; }
    return (y + y/4 - y/100 + y/400
        + 3*m + 4 - (m - m/8) / 2 + day) % 7;
}
