String& String::operator=(const String& other)
{
    cout << "Assigning \"" << other << "\" to \""
    << *this << "\"" << endl;
    if (this != &other)
    {
        delete[] buffer;
        len = other.len;
        if (len > 0)
        {
            buffer = new char[len];
            for (int i = 0; i < len; i++)
            {
                buffer[i] = other.buffer[i];
            }
        }
        else
        {
            buffer = nullptr;
        }
    }
    return *this;
}
