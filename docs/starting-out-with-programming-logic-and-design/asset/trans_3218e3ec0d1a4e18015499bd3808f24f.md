```
147 // modifyRecord 模块允许用户修改
148 // 库存文件中的现有记录。
149 modifyRecord()
150 // 字段的变量
151 Declare String description
152 Declare Real quantity
153
154 // 用于存储搜索值的变量
155 Declare String searchValue
156
157 // 用于存储新数量的变量
158 Declare Real newQuantity
159
160 // 标记是否找到值的标志位
161 Declare Boolean found = False
162 // 声明一个输入文件。
163 Declare InputFile coffeeFile
164 // 声明一个输出文件，用于拷贝原始
165 // 文件。
166 Declare OutputFile tempFile
167 // 打开文件。
168 Open coffeeFile "coffee.dat"
169 Open tempFile "temp.dat"
170 // 获取要搜索的值。
171 Display "输入你想要更新的咖啡。"
172 Input searchValue
173 // 获取新数量。
174 Display "输入新数量。"
175 Input newQuantity
176 While not eof(coffeeFile)
177 // 从文件中读取一条记录。
178 Read coffeeFile description, quantity
179 // 将此记录写入临时文件，
180 // 如果这是要修改的记录，则写入新记录。
181 If description == searchValue Then
182 Write tempFile description, newQuantity
183 Set found = True
184 Else
185 Write tempFile description, quantity
186 End If
187 While
188 // 关闭两个文件。
189 Close coffeeFile
190 Close tempFile
191 // 删除原始文件。
192 Delete "coffee.dat"
193 // 重命名临时文件。
194 Rename "temp.dat", "coffee.dat"
195 // 表示操作是否成功。
196 If found Then
197 Display "记录已更新。"
198 Else
199 Display "在文件中未找到。"
200 End If
201 End Module
```
