```
static void Main(string[] args)
{
    Rectangle r = new Rectangle();
    r.length = 4.5;
    r.width = 3.5;
    r.Display();
    Console.ReadLine();
}
}
}
```

The following result will be produced once you execute the code shown above.

Length: 4.5

Width: 3.5

Area: 15.75
