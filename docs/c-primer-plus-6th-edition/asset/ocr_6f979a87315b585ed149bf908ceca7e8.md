```

#include <iostream>

int main() {
    using namespace std;
    char ch;
    int ctl = 0;

    cin >> ch;
    while (ch != 'q') {
        ctl++;
        cin >> ch;
    }

    int ct2 = 0;
    cin.get(ch);
    while (ch != 'q') {
        ct2++;
        cin.get(ch);
    }

    cout << "ctl1" << " = " << ctl << " " << "ct2" << " = " << ct2 << "\n";

    return 0;
}
```

What does it print, given the following input:

I see a q-Enter
