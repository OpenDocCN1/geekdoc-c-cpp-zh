```
template <class Type>
bool Stack<Type>::push(const Type &item)
{
    if (top > stacksize)
    {
      items[top++] = item;
      return true;
    }
    else
      return false;
}
```

```
template <class Type>
bool Stack<Type>::pop(Type &item)
{
    if (top > 0)
    {
      item = items[--top];
      return true;
    }
    else
      return false;
}
```

```
template <class Type>
Stack<Type>& Stack<Type>::operator=(const Stack<Type> &st)
{
    if (this == &st)
      return *this;
    delete [] items;
    stacksize = st.stacksize;
    top = st.top;
    items = new Type[stacksize];
    for (int i = 0; i < top; i++)
      items[i] = st.items[i];
    return *this;
}
```
