#include <iostream>
using std::cout;
using std::endl;
template <typename T>
class MyFriend
{
private:
    T item;
public:
    MyFriend(const T & i) : item(i) {}
    template <typename C, typename D> friend show2(C & c, D & d)
};
template <typename C, typename D> void show2(C & c, D & d)
{
    cout << c << c.item << " " << d << d.item << endl;
}
int main()
{
    ManyFriend<int> hfi1(10);
    ManyFriend<int> hfi2(20);
    ManyFriend<double> hfdb(10.5);
    cout << "hfi1, hfi2: ";
    show2(hfi1, hfi2);
    cout << "hfdb, hfi2: ";
    show2(hfdb, hfi2);

    return 0;
}
