```
Time::Time(int h, int m)
{
  hours = h;
  minutes = m;
}
```

```
void Time::AddMin(int m)
{
  minutes += m;
  hours += minutes / 60;
  minutes %= 60;
}
```

```
void Time::AddHr(int h)
{
  hours += h;
}
```

```
void Time::Reset(int h, int m)
{
  hours = h;
  minutes = m;
}
```

```
Time Time::operator+(const Time & t) const
{
  Time sum;
  sum.minutes = minutes + t.minutes;
  sum.hours = hours + t.hours + sum.minutes / 60;
  sum.minutes %= 60;
  return sum;
}
```

```
void Time::Show() const
{
  std::cout << hours << " " << minutes << " " << "minutes" << std::endl;
}
```

Like Sum(), operator() is invoked by a Time object, takes a second Time object as an argument, and returns a Time object. Thus, you can invoke the operator+() method by using the same syntax as Sum() uses:

```
total = coding.operator+(fixing); // function notation
```
