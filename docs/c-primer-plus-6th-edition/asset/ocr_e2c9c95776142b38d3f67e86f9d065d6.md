C-style string. But reading a line at a time instead of a word at time uses a different syn-tax. Listing 4.10 shows this difference.

# Listing 4.10 strtype4.cpp

```
// strtype4.cpp -- line input
#include <iostream>
#include <string> // make string class available
#include <cstring>     // C-style string library
int main() {
{
    using namespace std;
    char char[20];
    string str;
n    cout << "Length of string in char before input: ";
    cout << strlen(char) << endl;
    cout << "Length of string in str before input: ";
    cout << str.size() << endl;
    cout << "Enter a line of text:\";
    cin.getline(char, 20); // indicate maximum length
    cout << "You entered: " << char << endl;
    cout << "Enter another line of text:"
    getline(cin, str); // cin now an argument; no length specifier
    cout << "You entered: " << str << endl;
    cout << "Length of string in char after input: ";
    cout << strlen(char) << endl;
    cout << "Length of string in str after input: ";
    cout << str.size() << endl;

    return 0;
}
```
