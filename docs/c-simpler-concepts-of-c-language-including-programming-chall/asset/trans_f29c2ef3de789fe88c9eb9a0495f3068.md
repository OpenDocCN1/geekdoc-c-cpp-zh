}

void viewall()
{
    struct q *first;
    first=front;
    if(first==NULL) return;
    while (1)
    {
        printf("%d---",first->data);
        first=first->next;
        if(first==front) break;
    }
    printf("%d\n", first->data);
}

void viewrev()
{
    struct q *ptr;
    ptr=rear;
    if(ptr==NULL) return;
    while (1)
    {
        printf("%d---",ptr->data);
        ptr=ptr->prev;
        if(ptr==rear) break;
    }
    printf("%d\n",ptr->data);
}
