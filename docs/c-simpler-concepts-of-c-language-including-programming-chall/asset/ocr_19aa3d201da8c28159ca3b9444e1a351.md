# ARRAYS

```
#include<stdio.h>
#include<conio.h>
void main()
{
    int marks[10];
    float avg[10],perc;
    int i,size,sub,max_marks,sum=0;
    printf("Enter the number of students.\\n");
    scanf("%d",&size);
    printf("Enter the total number of subjects.\\n");
    scanf("%d",&sub);
    printf("Enter the maximum marks for subjects..\n");
    scanf("%d",&max_marks);

    for(i=0;i<size;i++) {
        sum+=max_marks;
    }
    printf("Total max mark = %d,\n",sum);
    for(i=0;i<size;i++) {
        printf(\\
nEnter the %d student's total marks..\\n",i+1);
        scanf("%d",&marks[i]);
        printf("Student\\\tTotal_Marks\\t\tAverage\\t\tPercentage\\n");
        for(i=0;i<=sub;i++) {
            avg[i]=marks[i]/sub;
            perc=(float)marks[i]/sum*100;
            printf("%%d\\t%%d\\t%%f\\t%%f\\n",i+1,marks[i],avg[i],perc);
        }
    }
    getch();
}
```
