# CONDITIONAL CONSTRUCTS AND LOOPING

# 2.3. MORE COMPLEX C PROGRAM

A program using all looping constructs and switch case is as follows in Program 2.12

```
#include <stdio.h>
#include <conio.h>
main()
{
    int option;
    int i;
    printf("PRINTING EVEN NUMBERS FROM 1 to 10\n");
    printf("--------------------------------\n");
    printf("Select any option from::\n1.Output using for loop\n2.output using while loop\n3.output using do-while loop\n4.quit\noutput is:\n");
    scanf("%d", &option);
    do
    {
        switch(option)
        {
            case 1:
                for(i=1;i<=10;i++)
                {
                    if(i%2==0)
                        printf("%d->", i);
                }
                printf("\n");
                break;
            case 2:
                i=1;
                while(i<=10)
                {
                    if(i%2==0)
                        printf("%d->", i);
                    i++;
                }
                printf("\n");
                break;
            case 3:
                i=1;
                do
                {
                    if(i%2==0)
                        printf("%d->", i);
                    i++;
                } while(i<=10);
                printf("\n");
                break;
        }
    } 
}
```
