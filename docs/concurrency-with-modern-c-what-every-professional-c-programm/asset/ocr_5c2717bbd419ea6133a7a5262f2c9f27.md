void Event::notify() noexcept {
    notified = true;

    void* waiter = suspendedWaiter.load();

    if (waiter != nullptr && suspendedWaiter.compare_exchange_strong(waiter, nullptr)) {
        static_cast<Awaiter*>(waiter)->coroutineHandle.resume();
    }
}
