private int width;
private int length;
public Object1( int x = 0, int y = 0) {
    length = x;
    width = y;
}
public override int area () {
    Console.WriteLine("The Object1 area is:");
    return (width * length);
}
}
class Object1Tester {
    static void Main(string[] args) {
        Object1 obj = new Object1(8, 6);
        double x = obj.area();
        Console.WriteLine("The area is: {0}",x);
        Console.ReadKey();
    }
}
}
