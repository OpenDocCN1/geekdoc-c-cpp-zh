```
private void DeliveryRadioButton_CheckedChanged(object sender,
    EventArgs e)
{
    const double DELIVERY_CHARGE = 2.00;
    if (deliveryRadioButton.Checked)
        price += DELIVERY_CHARGE;
    else
        price = DELIVERY_CHARGE;
        outputLabel.Text = "Total is " + price.ToString("C");
}
```
