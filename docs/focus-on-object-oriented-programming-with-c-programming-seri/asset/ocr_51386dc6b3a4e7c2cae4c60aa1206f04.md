```
template <class T>
Stack<T>::Stack(int max) {
    size = max;
    top = -1;
    s = new T[size];
}

template <class T>
Stack<T>::~Stack() {
    delete [] s;
}

template <class T>
void Stack<T>::push(T d) {
    s[++top] = d;
}

template <class T>
T Stack<T>::pop() {
    return s[top--];
}

template <class T>
T Stack<T>::top() {
    return s[top];
}

template <class T>
int Stack<T>::empty() {
    return top == -1;
}

template <class T>
int Stack<T>::full() {
    return top == size-1;
}

template <class T>
void Stack<T>::clear() {
    top = -1;
}
```

The Stack class template defines a family of classes when the Stack template is used to declare an object. Note that a type must be supplied to occupy the place of the dummy type T in the template declaration.
