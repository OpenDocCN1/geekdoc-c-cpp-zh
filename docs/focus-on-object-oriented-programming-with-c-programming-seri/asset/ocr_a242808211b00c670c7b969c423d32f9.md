To correct the extra calls to the initializing functions, we can simply use a colon list in the Square (derived) class constructor declaration. By specifying the proper Shape (base) class constructor, the Square (derived) constructor allows the Shape (base) class constructor to perform the initializations.

```
// shape2.cpp

#include <iostream.h>

class Shape {
private:
    int color;
    int sideCount;
public:
    Shape() {
        cout << "Shape default constructor" << endl;
        SetColor(0);
        SetSideCount(0);
    }
    Shape(int c, int sc) {
        cout << "Shape 2 arg constructor " << endl;
        SetColor(c);
        SetSideCount(sc);
    }
    void display() {
        cout << "color = " << color << endl;
        cout << "sides = " << sideCount << endl;
    }
    void SetColor(int c) {
        cout << "set color to " << c << endl;
        color = c;
    }
    void SetSideCount(int sc) {
        cout << "set side count to " << sc << endl;
        sideCount = sc;
    }
};

class Square : public Shape {
public:
    Square() : Shape(0, 4) {
        cout << "Square default constructor" << endl;
    }
    Square(int c) : Shape(c, 4) {
        cout << "Square one arg constructor" << endl;
    }
};
```
