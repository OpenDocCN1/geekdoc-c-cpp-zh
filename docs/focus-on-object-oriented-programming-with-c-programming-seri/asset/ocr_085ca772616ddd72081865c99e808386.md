# Time3 - Separation of interface and implementation.

```
// time3.h

#ifndef _TIME3_H
#define _TIME3_H

class Time3 {
private:
    int month;
    int day;
    int year;
    int hour;
    int min;
    int sec;
public:
    void display();
    void setTime(int m, int d, int y, int h, int mn, int s);
    void getTime(int& m, int& d, int& y, int& h, int& mn, int& s);
};

#endif
```

```
// time3.cpp

#include <iostream>
#include "time3.h"

using namespace std;

void Time3::display() {
    cout << (month < 10 ? "0" : "") << month << "/"
         << (day < 10 ? "0" : "") << day << "/"
         << year << " ";
    cout << ((hour == 0 || hour == 12) ? 12 : hour % 12) << ":"
         << (min < 10 ? "0" : "") << min << ":"
         << (sec < 10 ? "0" : "") << sec
         << (hour < 12 ? " AM" : " PM") << endl;
}

void Time3::setTime(int m, int d, int y, int h, int mn, int s) {
    month = m;
    day = d;
    year = y;
    hour = h;
    min = mn;
    sec = s;
}
```
