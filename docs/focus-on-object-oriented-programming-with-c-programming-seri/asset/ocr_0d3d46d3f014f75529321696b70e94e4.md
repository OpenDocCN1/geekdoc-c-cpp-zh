Similarly, we can use either the `setw()` manipulator or the `width()` member function to change the width of the field in which x appears.

Type in, compile and execute each of the two programs below.

```
// io5.cpp
#include <iostream>
#include <iomanip>
using namespace std;
int main( {
    float x, y;
    x = 8.0 / 3.0;
    y = 9.0 / 3.0;
    cout << setiosflags(ios::showpoint | ios::fixed) <<
    setprecision(3);
    cout << "\n" << setw(10) << x << "\n" << endl;
    cout << "\n" << y << "\n" << endl;
    return 0;
}
```
