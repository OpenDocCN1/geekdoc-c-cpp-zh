```
#ifndef WORD11_H
#define _WORD11_H

class Word {
private:
    char *buf;

    Word(char *s);
    static int getCount(Word &w);
```

```
// word11.cpp

#include <iostream>
#include <string.h>
#include "word11.h"

using namespace std;

int Word::wordCount = 0;

Word::Word(char *s) {
    buf = strdup(s);
    wordCount++;
}

int Word::getCount(Word &w) {
    cout << w.buf << endl;
    return wordCount;
}
```
