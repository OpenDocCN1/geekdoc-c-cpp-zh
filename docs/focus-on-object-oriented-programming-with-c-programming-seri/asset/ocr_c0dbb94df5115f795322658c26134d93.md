```
Distance Distance::addDist(Distance d) {
  Distance temp;

  temp.inches = inches + d.inches;
  temp.feet = feet + d.feet;
  if (temp.inches >= 12.0) {
    temp.inches -= 12.0;
    temp.feet += 1;
  }
  return temp;
}

Distance doubleDist(Distance d) {
  int ft = d.feet * 2;
  float in = d.inches * 2;
  if (in >= 12.0) {
    ++ft;
    in -= 12.0;
  }
  return Distance(ft, in);
}

int main() {
  Distance d1(3, 6.0), d2;
  d2 = doubleDist(d1);
  cout << "d2 = " << d2.showDist();
}
```

In main(), we now say:

```
d2 = doubleDist(d1);
```

Note: where `doubleDist()`, as a member function, takes no arguments, it takes one as a friend.

In general, the friend version of a function requires one more argument than when the function is a member.
