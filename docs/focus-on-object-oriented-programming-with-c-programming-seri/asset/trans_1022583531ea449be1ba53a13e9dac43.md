```
// dist3.cpp

#include <iostream>
#include "dist3.h"

using namespace std;

void Distance::setDist(int ft, float in) {
    feet = ft;
    inches = in; }

void Distance::getDist() {
    cout << " feet: ";
    cin >> feet;
    cout << " inches: ";
    cin >> inches;
}

void Distance::showDist() {
    cout << feet << "' " << inches << "\"";
}

void Distance::addDist(Distance d1, Distance d2) {
    inches = d1.inches + d2.inches;
    feet = d1.feet + d2.feet;
    if (inches >= 12.0) {
        inches -= 12.0;
        feet++;
    }
}

Distance Distance::addDist(Distance d) {
    Distance temp;

    temp.inches = inches + d.inches;
    temp.feet = feet + d.feet;
    if (temp.inches >= 12.0) {
        temp.inches -= 12.0;
        temp.feet += 1;
    }

    return temp;
}
```

```
// distdrv3.cpp

#include <iostream>
#include "dist3.h"

using namespace std;

int main() {
    Distance d1, d2, d3;

    d1.getDist();
    d2.getDist();
    d3 = d1.addDist(d2);

    cout << endl;
    cout << "d1 = ";
    d1.showDist();
    cout << endl << endl;

    cout << "d2 = ";
    d2.showDist();
    cout << endl << endl;

    cout << "d3 = ";
    d3.showDist();
    cout << endl << endl;
}
```

请注意，d1 和 d2 通过 addDist() 相加，返回的对象随后被赋值给 d3。

还要注意，只传递了一个 Distance 对象。在成员函数中，我们创建一个临时 Distance 对象来保存和，直到它可以返回给调用程序。和是通过相加两个距离计算得出的：

- 第一个是调用 addDist() 的对象，即 d1
- 第二个作为参数传递

最后，请注意，对 inches 和 feet 的独立引用实际上是对调用 addDist() 的对象 d1 的数据成员的引用。

我们最新的版本看起来更自然，因为赋值运算符 = 以自然的方式使用：

```
d3 = d1.addDist(d2);
```

稍后我们将看到如何使用算术 + 运算符来实现更明显的表达式：

```
d3 = d1 + d2;
```

你的程序总共应运行 10 次试验，并在 10 次试验结束时输出一对的平均百分比和同花的平均百分比（保留两位小数）。
