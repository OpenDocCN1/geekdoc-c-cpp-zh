# Implementation of dtob() Function

```
#include <BigInt.h>
BigInt *dtob(char *d) {
    BigInt *t, *ten;
    t = itob(0);
    ten = itob(10);
    while(*d) {
        BigInt *u;
        u = itob(*d++ - '0');
        bmult(t, ten, t);
        badd(t, u, t);
        bfree(u);
    }
    bfree(ten);
    return t;
}
```

# C Program Using BigInt Library

```
#include <BigInt.h>
BigInt *dtob(char *d);
int main(void) {
    BigInt *a, *b, *c, *t;
    a = dtob("12345678901234");
    b = dtob("24680135792356");
    c = itob(0);
    t = itob(300);
    badd(a, b, c);
    badd(c, t, c);
    bout(c);
    printf("\n");
    bfree(a);
    bfree(b);
    bfree(c);
    bfree(t);
}
```
