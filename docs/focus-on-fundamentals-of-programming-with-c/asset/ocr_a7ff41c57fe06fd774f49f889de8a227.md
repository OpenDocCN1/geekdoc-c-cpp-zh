// flip1.c
#include <stdio.h>
int main(void) {
    int x = 0x1234;       // 0001 0010 0011 0100
    int mask = 0x0010;    // 0000 0000 0001 0000
    if (x & mask)
        x &= 0xFFEF;      // bit is on, turn it off
    else
        x |= 0x0010;      // bit is off, turn it on
    printf("x = 0x%x\n", x); // 0001 0010 0001 0100
    return 0;
}
