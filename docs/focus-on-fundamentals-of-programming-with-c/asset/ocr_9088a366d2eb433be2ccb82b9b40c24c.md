# Example: Concatenate two files into a third file and output the third file.

```
#include <stdio.h>
#include <stdlib.h>
#define MAX 81
int main(void) {
    char line[MAX];
    FILE *fpIn, *fpIn1, *fpIn2, *fpOut;
    if ((fpIn1 = fopen("infile1.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }
    if ((fpIn2 = fopen("infile2.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }
    if ((fpOut = fopen("outfile.txt", "w"))) {
        printf("File could not be opened for output.\n");
        exit(1);
    }
    while (!feof(fpIn1)) {
        fgets(line, MAX, fpIn1);
        fputs(line, fpOut);
    }
    while (!feof(fpIn2)) {
        fgets(line, MAX, fpIn2);
        fputs(line, fpOut);
    }
    fclose(fpIn1);
    fclose(fpIn2);
    fclose(fpOut);
    if ((fpIn = fopen("outfile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }
    while (!feof(fpIn)) {
        fgets(line, MAX, fpIn);
        printf("%s", line);
    }
    return 0;
}
```
