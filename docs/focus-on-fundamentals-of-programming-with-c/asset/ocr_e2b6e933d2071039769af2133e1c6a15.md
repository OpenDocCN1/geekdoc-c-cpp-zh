We can also pass the array of structures to a function:

```
#include <stdio.h>

#define SIZE 4

typedef struct {
    char name[30];
    float gpa;
    int year;
} Student;

int main(void);
void foo(Student myClass[]);

int main(void) {
    Student myClass[SIZE] = {
        { "Haley", 3.75, 2015 },
        { "Sammie", 3.85, 2016 },
        { "Naomi", 3.25, 2017 },
        { "Garrett", 2.85, 2019 }
    };

    foo(myClass);

    return 0;
}

void foo(Student myClass[]) {
    int i;

    for (i = 0; i < SIZE; ++i) {
        printf("%10s %5.2f %5d\n",
               myClass[i].name, myClass[i].gpa, myClass[i].year);
    }
}
```

The output to the above program is shown below:

```
    Haley   3.75 2015
  Sammie   3.85 2016
   Naomi   3.25 2017
 Garrett   2.85 2019
```
