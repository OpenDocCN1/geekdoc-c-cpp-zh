This indirect component selector operator, ->, is made up of a hyphen and the greater than symbol without a space between them. It automatically first dereferences the pointer and then selects the appropriate field of the struct. Note that this operator can only be used when you have a pointer to a struct and you want to access a field of the struct. The example below shows the use of this new operator:

```
// student.c

#include <stdio.h>
#include <string.h>
#include <alloc.h>

typedef struct student {
    char name[30];
    int age;
    float gpa;
} STUDENT;

int main(void) {
    STUDENT *ps;

    ps = (STUDENT *) malloc(sizeof(struct student));
    strcpy(ps->name, "Haley Laurel");
    ps->age = 35;
    ps->gpa = 3.75;
    printf("%s is %4.2f\n", ps->name, ps->age, ps->gpa);
    free(ps);
    return 0;
}
```
