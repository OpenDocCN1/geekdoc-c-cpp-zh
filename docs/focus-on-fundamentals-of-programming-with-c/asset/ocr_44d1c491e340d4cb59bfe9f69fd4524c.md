void insertionSort(int list[], int max) {
    int i, j, temp;
    for (i = 1; i < max; i++) {
        temp = list[i];
        j = i - 1;
        while (j >= 0 && temp < list[j]) {
            list[j+1] = list[j];
            j = j - 1;
        }
        list[j+1] = temp;
    }
}
