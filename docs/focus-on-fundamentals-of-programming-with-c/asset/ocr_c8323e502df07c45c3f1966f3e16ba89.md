// arg3.c
#include <stdio.h>
#include <stdarg.h>

int main(void)
{
    void myprintf(char *fmt, ...);

    int x = -30, y = 10;
    float pi = 3.14;
    char *buf = "Hello";

    myprintf("%d %d\n", x, y);
    myprintf("%f is %s\n", x, pi, buf, y);
}

void myprintf(char *fmt, ...)
{
    char *p, *sval;
    int ival;
    double dval;
    va_list ap;

    va_start(ap, fmt);
    for (p = fmt; *p; p++) {
        if (*p != '%') {
            putchar(*p);
            continue;
        }
        switch (*++p) {
            case 'd': ival = va_arg(ap, int);
                printf("%d", ival);
                break;
            case 'f': dval = va_arg(ap, double);
                printf("%f", dval);
                break;
            case 's': for (sval = va_arg(ap, char *); *sval; sval++)
                putchar(*sval);
                break;
            default : putchar(*p);
                break;
        }
    }
    va_end(ap);
}
