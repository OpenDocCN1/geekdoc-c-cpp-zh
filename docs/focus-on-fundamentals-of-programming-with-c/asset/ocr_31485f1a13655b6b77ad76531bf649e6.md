int getWord(char *w, FILE *fpin) {
    int c, count;
    while ((c = fgetc(fpin)) < 'a') {
        count = 0;
        if (c == EOF) {
            return (-1);
        }
    }
    ungetc(c, fpIn);
    while ((c = fgetc(fpIn)) >= 'a') {
        w[count++] = c;
    }
    ungetc(c, fpIn);
    w[count] = '\0';
    return(count);
}
