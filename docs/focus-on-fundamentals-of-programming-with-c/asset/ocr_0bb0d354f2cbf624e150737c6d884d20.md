# Example: Copy the contents of one file to a second file and output the contents of the second file.

```
```c
#include <stdio.h>
#include <stdlib.h>

int main( void )
{
    FILE *fPtrIn, *fPtrOut;

    if ( ( fPtrIn = fopen( "infile.txt", "r" ) ) == NULL ) {
        printf( "File could not be opened for input.\n" );
        exit( 1 );
    }

    if ( ( fPtrOut = fopen( "outfile.txt", "w" ) ) == NULL ) {
        printf( "File could not be opened for output.\n" );
        exit( 1 );
    }

    while ( !feof( fPtrIn ) ) {
        fscanf( fPtrIn, "%d", &value );
        fprintf( fPtrOut, "%d\n", value );
    }

    fclose( fPtrIn );
    fclose( fPtrOut );

    fPtrIn = fopen( "outfile.txt", "r" );

    if ( fPtrIn == NULL ) {
        printf( "File could not be opened for input.\n" );
        exit( 1 );
    }

    while ( !feof( fPtrIn ) ) {
        fscanf( fPtrIn, "%d", &value );
        printf( "%d\n", value );
    }

    fclose( fPtrIn );

    return 0;
}
