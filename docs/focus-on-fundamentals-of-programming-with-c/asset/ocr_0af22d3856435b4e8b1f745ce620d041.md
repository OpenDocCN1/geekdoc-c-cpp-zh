We can rewrite the previous program using the ctype.h library:

```
// ascii1.c

#include <stdio.h>
#include <ctype.h>

int main(void) {
    unsigned char c;

    for (c = 0; c <= 127; ++c) {
        printf("%3d 0x%03x%c, c);
        if (isprint(c)) {
            printf(" %c", c);
        }
        if (isdigit(c)) {
            printf(" D");
        }
        if (isupper(c)) {
            printf(" UC");
        }
        if (islower(c)) {
            printf(" LC");
        }
        if (isalpha(c)) {
            printf(" L");
        }
        if (isalnum(c)) {
            printf(" AN");
        }
        if (isspace(c)) {
            printf(" S");
        }
        if (ispunct(c)) {
            printf(" P");
        }
        if (iscntrl(c)) {
            printf(" C");
        }
        printf("\n");
    }
    return 0;
}
```
