# Example: Output contents of a file to standard output requesting a filename from user.

```
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    char fileName[32];
    FILE *fp;

    printf("Enter filename: ");
    gets(fileName);
    if (!(fp = fopen(fileName, "r"))) {
        printf("File %s could not be opened.\n", fileName);
        exit(1);
    }

    while((c = getc(fp)) != EOF) {
        putchar(c);
    }
    fclose(fp);
    return 0;
}
```
