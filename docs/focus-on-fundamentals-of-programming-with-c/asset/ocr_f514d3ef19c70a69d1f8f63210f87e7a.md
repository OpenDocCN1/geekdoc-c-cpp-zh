The following four string functions copy string t into string s. Can you follow the pointers?

```
// strcpy1.c
void strcpy1(char s[], char t[]) {
    int i;
    for (i = 0; t[i] != '\0'; i++) {
        s[i] = t[i];
    }
    s[i] = '\0';
}

// strcpy2.c
void strcpy2(char *s, char *t) {
    for (; *t != '\0'; s++, t++) {
        *s = *t;
    }
    *s = '\0';
}

    while (*t) {
        *s++ = *t++;
    }
```
