{
    node*x,*y;
    y=NULL;
    x=root;
    while (x!=NULL)
    {
        y=x;
        if (z->data<x->data)
        x->x->lchild;
        else
        x->x->rchild;
    }
    z->parent=y;
    if (y==NULL)
    root=z;
    else
    {
        z->parent=y;
        if (z->data<y->data)
        y->lchild=z;
        else
        y->rchild=z;
    }
    RBInsertFixup(z) ;
}
