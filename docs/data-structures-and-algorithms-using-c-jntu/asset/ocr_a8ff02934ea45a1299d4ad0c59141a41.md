int fib(int c)
{
 int f1=-1,f2=1,f3;
 while (f2<=c)
 {
 f3=f1+f2, f1=f2, f2=f3;
 if (f3==c)
 return f3;
 if (f3<c)
 return f1;
 }
 return 0;
}
