void sort(int a[],int n)
{
 int i,j;
 for(i=n-1;i>=0;i--)
  for(j=0;j<i;j++)
   if(a[j]>a[j+1])
   {
    a[j]=(a[j]+a[j+1])-(a[j+1]=a[j]);
   }
}
void main()
{
 int i,a[30],n,low,up,search,mid;

  cin>>a[i];

 cin>>search;
 sort(a,n);
 low=0,up=n-1;
 while(low<=up)
 {
  mid=(low+up)/2;
  if(a[mid]==search)
  {

  }
  else if(a[mid]>search)
   up=mid-1;
  else
   low=mid+1;
 }

}
