/*infix to postfix conversion*/
#include<iostream.h>
#include<string.h>
#include<stdlib.h>
#include<ctype.h>
class expression
{
private:
    char infix[100];
    char stack[200];
    int top;
    int r;
    char postfix[100];
public:
    void convert();
    int input_p(char);
    int stack_p(char);
    int rank(char);
};
int expression::input_p(char c)
{
    if(c=='^+' || c=='^-')
        return 1;
    else if(c=='*' || c=='/')
        return 3;
    else if(c=='^')
        return 6;
    else if(isalpha(c)!=0)
        return 7;
    else if(c=='(')
        return 9;
    else if(c==')')
        return 0;
    else
    {
        cout<<"Invalid expression::input error\n";
        exit(0);
    }
}
int expression::stack_p(char c)
{
    if(c=='^+' || c=='^-')
        return 2;
    else if(c=='*' || c=='/')
        return 4;
    else if(c=='^')
        return 5;
