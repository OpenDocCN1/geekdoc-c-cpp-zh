/*To Sort The Given List Using MERGE SORT*/
#include<iostream.h>
#include<conio.h>
void merge sort(int a[],int lb,int mid,int ub)
{
 int i=lb,j=mid+1,k=0,b[50];
 while(i<=mid&&j<=ub)
 {
 if(a[i]<a[j])
 b[k++]=a[i++];
 else
 b[k++]=a[j++];
 }
 while(i<=mid)
 b[k++]=a[i++];
 while(j<=ub)
 b[k++]=a[j++];
 for(k=0;k<=ub-lb;k++)
 a[k+lb]=b[k];
}
