void llist<t>::push()
{
 t ele;
 cout<<"enter the element\n";
 cin>>ele;
 temp=(struct node*) malloc(sizeof(struct node));
 if(temp==NULL)
 {
 cout<<"memory allocation error\n";
 return;
 }

if(top==NULL)
{
 head->link=temp;
 temp->link=NULL;
}
else
{
 top->link=temp;
 temp->link=NULL;
}
top=temp;
}

void llist<t>::pop()
{
 t ele;
if(top==NULL)
{
 cout<<"linked list is empty\n";
 return;
}

{
 temp=temp1;
}
