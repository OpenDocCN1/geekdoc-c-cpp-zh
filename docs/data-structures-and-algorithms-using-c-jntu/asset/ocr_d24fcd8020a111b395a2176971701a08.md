{
    t ele;
    if(front==NULL)
    {
        cout<<"linked list is empty\n";
        return;
    }
    if(front==rear)
    {
        ele=front->data;
        front=rear=NULL;
    }
    else
    {
        ele=front->data;
        rear->link=front->link;
        temp=front;
        front=front->link;

}
