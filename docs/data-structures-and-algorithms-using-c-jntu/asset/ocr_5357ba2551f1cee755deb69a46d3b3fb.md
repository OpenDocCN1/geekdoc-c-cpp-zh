# Zig-zig

Before splaying:
      gX
     /  \
   pX   gX_R
  /  \
 x   pX_R
/ \
x_L x_R

After splaying:
      x
     /  \
   x_L   pX
        /  \
      x_R   gX
           /  \
         pX_R gX_R

# Zig-zag

Before splaying:
      gX
     /  \
   pX   gX_R
  /  \
pX_L   x
     /  \
   x_L  x_R

After splaying:
      x
     /  \
   pX   gX
  /  \   \
pX  x_L x_R gX_R

# Zag-zig

Before splaying:
      gX
     /  \
  gX_L  pX
        /  \
       x  pX_R
      / \
    x_L x_R

After splaying:
      x
     /  \
   gX   pX
  / \   \
gX_L x_L x_R pX_R
