int stries::search(struct trie*T,char*s)
{
int index;
if(s[0]=='\0') //Check for the empty search string.
{
cout<<"\nString Found";
return(1);
}
index=0;
if(T->link[index]==NULL)
{
cout<<"\nString Not Found";
return(0);
}
else
{
cout<<"\n"<<s[0]<<"<=>Node Found,Moving in the path";
s++;
T=T->link[index];
search(T,s);
}
return(1);
}
