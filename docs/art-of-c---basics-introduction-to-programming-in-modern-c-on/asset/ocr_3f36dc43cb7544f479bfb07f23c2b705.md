public record Token {
    // Defining the "default" value for a type is always a good idea.
    public static readonly Token InvalidToken = new('\0');

    public object V { get; }

    public Token(object V) => this.V = V;

    // Note that we do not use the positional record syntax for "Token",
    //		e.g., "record Token(object V)".
    // Why not?
    // As it turns out, when we use inheritance for records,
    // the positional primary constructors are not as convenient.
    // We will leave this as something to think about for the readers.
