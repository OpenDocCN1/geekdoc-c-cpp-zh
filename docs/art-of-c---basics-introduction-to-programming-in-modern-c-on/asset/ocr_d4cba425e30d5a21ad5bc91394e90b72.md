namespace Example;

static class Finder {
    internal static bool TryFindMax<T>(IList<T> list, out T? max) where T
: IComparable<T> {
        if (list.Count == 0) {
            max = default;
            return false;
        }

        max = list[0];
        foreach (var v in list) {
            if (v.CompareTo(max) > 0) {
                max = v;
            }
        }

        return true;
    }
}
