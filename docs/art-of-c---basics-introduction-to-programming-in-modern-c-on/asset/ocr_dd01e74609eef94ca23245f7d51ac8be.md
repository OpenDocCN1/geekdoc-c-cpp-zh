namespace Euclid;

public static class Points {
public static float Distance(IPoint p1, IPoint p2) =>
(float)Math.Sqrt((p1.X - p2.X) * (p1.X - p2.X) + (p1.Y - p2.Y) *
(p1.Y - p2.Y));
}
