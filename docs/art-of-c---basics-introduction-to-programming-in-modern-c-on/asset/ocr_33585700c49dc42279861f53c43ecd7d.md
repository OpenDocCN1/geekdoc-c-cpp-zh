// Note the order of the method parameters.
// "rhs" comes before "lhs".
// This is in order to support the (stack.Pop(), stack.Pop()) syntax
in Compute().
private static Token? ComputeExpression(Token op, Token? rhs, Token?
	lhs) {
	if (!op.IsValidOperator() || rhs is null || !rhs.IsNumber || lhs
	is null || !lhs.IsNumber) {
		throw new Exception($"Invalid expression. Op = {op}; rhs =
	{rhs}; lhs = {lhs}");
	}

	return op.Char switch {
		'+' => lhs! + rhs!,
		'-' => lhs! - rhs!,
		'*' => lhs! * rhs!,
		'/' => lhs! / rhs!,
		_ => null,
	};
}
