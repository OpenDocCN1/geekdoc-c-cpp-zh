```
         }
 
         private static Dictionary<char, string> morseCode = new() { };
         private static Dictionary<string, char> reverseCode = new() { };
 
         static Code() {
             foreach (var (k, v) in Chars.Code) {
                 ReverseCode[v] = k;
                 MorseCode[k] = v;
                 if (char.IsLetter(k)) {
                     MorseCode[key: char.ToUpper(k)] = v;
                 }
             }
         }
     }
```

static Code() {} is a "static constructor", used to initialize two static variables, morseCode and reverseCode. Note that these are variable of the "Dictionary type".
