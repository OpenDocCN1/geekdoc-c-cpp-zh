namespace ReversePolish;

public enum TokType : byte {
    INVALID = 0, // It is always a good idea to have the "default" value
    OPERATOR,
    INT,
    FLOAT,
}

public static class TokenTypes {
    // As you'll remember from the earlier lessons, this is an "extension method".
    // Since you cannot put a method in an Enum definition, we use extension methods.
    // This particullar method is used to "pretty print" the enum values.
    // The Enum.GetName<T>() method would have returned "OPERATOR",
    "INT", etc.

    // Note the use of the switch expression in this "expression body" method.
    // How did we ever program in C# without the switch expression? :) :)

    public static string ToName(this TokType t) => t switch {
        TokType.OPERATOR => "O",
        TokType.INT=> "I",
        TokType.FLOAT => "F",
        _ => "?",
    };
}
