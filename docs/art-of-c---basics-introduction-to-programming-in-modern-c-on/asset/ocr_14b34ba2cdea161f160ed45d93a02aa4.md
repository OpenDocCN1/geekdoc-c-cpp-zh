```
static string? ReadName() {
    Console.WriteLine("What is your name?");
    var name = Console.ReadLine();

    if (string.IsNullOrEmpty(name)) {
        name = "Friend";
    } else {
        name = name.ToUpper();
    }

    return name;
}
```
