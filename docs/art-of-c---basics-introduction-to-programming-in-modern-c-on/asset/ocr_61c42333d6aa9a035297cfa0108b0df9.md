namespace ReversePolish;

internal sealed class Stack {
    private readonly List<Token> list;

    internal Stack() => list = new();

    internal int Size => list.Count;

    internal Token Push(Token tok) {
        list.Add(tok);
        return tok;
    }

    internal Token? Pop() {
        if (list.Count > 0) {
            var tok = list[^1];
            list.RemoveAt(list.Count - 1);
            return tok;
        }
        return null;
    }

    internal Token? Peek() => (list.Count > 0) ? list[^1] : null;

    internal void Clear() => list.Clear();
}
