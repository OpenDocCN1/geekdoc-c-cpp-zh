```csharp
namespace ReversePolish;

public enum TokType : byte {
    INVALID = 0, // It is always a good idea to have the "default" value
    OPERATOR,
    INT,
    FLOAT,
}

public static class TokenTypes {
    // As you'll remember from the earlier lessons, this is an "extension method".
    // Since you cannot put a method in an Enum definition, we use extension methods.
    // This particular method is used to "pretty print" the enum values.
    // The Enum.GetName<T>() method would have returned "OPERATOR",
    // "INT", etc.

    // Note the use of the switch expression in this "expression body" method.
    // How did we ever program in C# without the switch expression? :) :)

    public static string ToName(this TokType t) => t switch {
        TokType.OPERATOR => "O",
        TokType.INT => "I",
        TokType.FLOAT => "F",
        _ => "?",
    };
}
```

## reverse-polish-2/Calculator.cs

```csharp
namespace ReversePolish2;

// Yes, it is "ReversePolish.Calculator". Clever, huh? ;)
// Note that the implementation is a little bit more complicated than necessary
// since we deal with two different number types, integers (long) and floats (double).
// The readers can use just one type, like double, for all numbers.
public sealed class Calculator {
    // You can call this a "composition" or "embedding" or whatever you like.
    // This kind of "code re-use" is generally preferred over inheritance.
    private Stack stack;

    // As explained in the book, you can initialize a variable in the field initializer
    // or in the constructor.
    // In general, the author prefers to do this in the constructors.
    // But, there are situations where it should be done in the field initializers.
    // Note that all fields are initialized first, in the order defined,
    // before the constructor is called.
    public Calculator() => stack = new();

    // Do you know what is missing here?
    // Yes, you are right.
    // We do NOT generally use finalizers in C#. (It's not like destructors in C++.)
    // Now, if your old C# textbook teaches finalizers, throw it away! :) :)

    // This method parses the input string into an expression,
    // and it evaluates the expression.
    // It either returns the value of the expression,
    // or it throws an exception,
    // for example, when the input string is an invalid expression in the postfix notation.
    public Number Compute(String input) {
        // Always start the computation using an empty stack!
        stack.Clear();

        var words = input.Split();
        foreach (var w in words) {
            var t = ParseWord(w);
            var num = t switch {
                Number n => n,
                OperatorToken op => ComputeExpression(op, stack.Pop(), stack.Pop()) ?? throw new Exception(),
                _ => throw new Exception($"Invalid input token: {t}"),
            };
            if (num is not null) {
                stack.Push(num);
            } else {
                throw new Exception($"Invalid input token: {t}");
            }
        }

        // The above statements are the same as this:
        // switch (t) {
        //    case Number n:
        //        stack.Push(n);
        //      break;
        //      case OperatorToken op:
        //        var rhs = stack.Pop();
        //        var lhs = stack.Pop();
        //        var res = ComputeExpression(op, rhs, lhs);
        //        if (res is not null) {
        //            stack.Push(res);
        //        }
        //        break;
        //      default:
        //        throw new Exception($"Invalid input token: {t}");
        // }
        
        return stack.Peek() ?? throw new Exception($"Failed to compute the input expression: {input}");
    }

    // Note the order of the method parameters.
    // "rhs" comes before "lhs".
    private static Number? ComputeExpression(OperatorToken op, Number? rhs, Number? lhs) {
        // Note this unusual statement. We do this here just for fun. :) 
        // But, this _can_ become a new idiom in the modern C#. Mark my
        // word. ;)
        _ = lhs ?? rhs ?? throw new Exception($"Invalid input expression. Operator = {op}");

        // This is equivalent to the following (more traditional) null-
        // check statement:
        // if ((lhs is null) || (rhs is null)) {
        //     throw new Exception($"Invalid input expression. Operator = {op}");
        // }

        // There is a lot going on in the above strange-looking
        // statement.
        // First, as stated in the book, ?? is a "null-coalescing
        // operator".
        // If the first operand is null, then it returns the second
        // argument as the value of the expression.
        // In this particular case, the second argument of the first
        // null-coalescing operator is
        //    another null-coalescing operator, whose second argument is
        //    to throw an exception.
        // In the modern C#, the throw clause is an expression, not a
        //    statement.

        // The assignment to the discard variable (_) is used to make the
        //    whole expression a statement.
        //    Otherwise, it will be syntactically invalid.

        // BTW, the assignment is also an expression in the modern C#.
        // Both the throw and the assignment expressions can be used as
        //    statements by themselves.
        // Just add the semicolon at the end. ;}

        return op.Char switch {
            '+' => lhs! + rhs!,
            '-' => lhs! - rhs!,
            '*' => lhs! * rhs!,
            '/' => lhs! / rhs!,
            _ => null,
        };
    }

    private static Token ParseWord(string word) {
        Token tok = Token.InvalidToken;
        if (long.TryParse(word, out long l)) {
            tok = new IntToken(l);
        } else if (double.TryParse(word, out double d)) {
            tok = new FloatToken(d);
        } else if (char.TryParse(word, out char c) && OperatorToken.IsValid(c)) {
            tok = new OperatorToken(c);
        }
        return tok;
    }
}
```

```csharp
namespace ReversePolish2;

// TBD: Can you make this generic? Rather than using "object V"?

public record Token {
    // Defining the "default" value for a type is always a good idea.
    public static readonly Token InvalidToken = new('\0');

    public object V { get; }

    public Token(object V) => this.V = V;

    // Note that we do not use the positional record syntax for "Token",
    // e.g., "record Token(object V)".
    // Why not?
    // As it turns out, when we use inheritance for records,
    // the positional primary constructors are not as convenient.
    // We will leave this as something to think about for the readers.
    // Not necessarily as an "exercise", but just as something to think
    // about. ;)

    public override string ToString() => $"X:{V}";
}
```

## reverse-polish-2/Number.cs

```csharp
namespace ReversePolish2;

public abstract record Number : Token {
    public static readonly Number Zero = new FloatToken(0.0);

    public Number(object V) : base(V) { }

    public static explicit operator double(Number tok) => (double)tok.V;

    public static explicit operator Number(long l) => new IntToken(l);
    public static explicit operator Number(double d) => new FloatToken(d);

    // Hmm... How do these work??? ;)

    public static Number operator +(Number lhs, Number rhs) =>
        (Number)((lhs, rhs) switch {
            (IntToken l, IntToken r) => (long)l + (long)r,
            (IntToken l, FloatToken r) => (long)l + (double)r,
            (FloatToken l, IntToken r) => (double)l + (long)r,
            _ => (double)lhs + (double)rhs,
        });

    public static Number operator -(Number lhs, Number rhs) =>
        (Number)((lhs, rhs) switch {
            (IntToken l, IntToken r) => (long)l - (long)r,
            (IntToken l, FloatToken r) => (long)l - (double)r,
            (FloatToken l, IntToken r) => (double)l - (long)r,
            _ => (double)lhs - (double)rhs,
        });

    public static Number operator *(Number lhs, Number rhs) =>
        (Number)((lhs, rhs) switch {
            (IntToken l, IntToken r) => (long)l * (long)r,
            (IntToken l, FloatToken r) => (long)l * (double)r,
            (FloatToken l, IntToken r) => (double)l * (long)r,
            _ => (double)lhs * (double)rhs,
        });

    // Note: We use the int division when both operands are integers.
    // That is, 1/3 = 0.
    public static Number operator /(Number lhs, Number rhs) =>
        (Number)((lhs, rhs) switch {
            (IntToken l, IntToken r) => (long)l / (long)r,
            (IntToken l, FloatToken r) => (long)l / (double)r,
            (FloatToken l, IntToken r) => (double)l / (long)r,
            _ => (double)lhs / (double)rhs,
        });
}
```
