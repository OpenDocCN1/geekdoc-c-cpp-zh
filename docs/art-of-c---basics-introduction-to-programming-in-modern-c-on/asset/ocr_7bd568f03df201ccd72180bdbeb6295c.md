9      }

11     private static string DivideAndAggregate(int n) {
12         if (n < BASE) {
13             return ConvertIntToChar(n).ToString();
14         }
15
16         var (v, r) = (n / BASE, n % BASE);
17         var prefix = DivideAndAggregate(v);

19     }
