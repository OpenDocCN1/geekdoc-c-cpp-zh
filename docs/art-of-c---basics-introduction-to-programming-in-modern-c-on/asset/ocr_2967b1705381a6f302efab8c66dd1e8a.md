```
var cap = 2u;
var queue = new ShortQueue<int>(cap);

Console.WriteLine($"len = {queue.Count}");

var e1 = 5;
queue.Enqueue(e1);
var e2 = 7;
queue.Enqueue(e2);

Console.WriteLine($"len = {queue.Count}");
lock (queue) {
    foreach (var item in queue) {
        Console.WriteLine($"item = {item}");
    }
}

var e3 = 9;
try {
    queue.Enqueue(e3);
    Console.WriteLine($"len = {queue.Count}");
    lock (queue) {
        foreach (var item in queue) {
            Console.WriteLine($"item = {item}");
        }
    }
} catch (Exception ex) {
    Console.WriteLine(ex);
}

var e = queue.Dequeue();
Console.WriteLine($"e = {e}");
Console.WriteLine($"len = {queue.Count}");
lock (queue) {
    foreach (var item in queue) {
        Console.WriteLine($"item = {item}");
    }
}
```
