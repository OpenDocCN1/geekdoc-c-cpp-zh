def calculate_average(numbers):
    """Calculate the average of a list of numbers"""
    if not numbers:
        return 0
    
    total = sum(numbers)
    count = len(numbers)
    average = total / count
    
    return average


# Example usage
grades = [85, 92, 78, 90, 88]
result = calculate_average(grades)
print(f"The average grade is: {result}")

def is_prime(number):
    if number < 2:
        return False
    for i in range(2, int(number ** 0.5) + 1):
        if number % i == 0:
            return False
    return True
