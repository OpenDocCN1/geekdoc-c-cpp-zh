The next function is built from the previous building blocks to create a custom function that reads data from an EEPROM at a specific address.

```
// *******************************************
// I2C_EEIN function to read from an 8-bit address Microchip
// EEPROM part (i.e. 24LC01) with 8 byte address (2k bit max)
// *******************************************
unsigned char I2C_EEIN(unsigned char address)
{
    unsigned char data;

    i2c_start();
    i2c_write(0xA0);
    i2c_write(address);
    i2c_repStart();
    i2c_write(0xA1);
    data=i2c_read(0);
    i2c_stop();
    return(data);
}
```
