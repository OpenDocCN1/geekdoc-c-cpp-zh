// ***************************
// Pulseout Command to set a port high for period of time
// ***************************

void pulseout(char pin, unsigned long per)
{
switch(pin)
{
case 0:
    TRISC7 = 0;          // Make RC7 an Output
    RC7 = 1;             // Set RC7 High
    pauseus(per);        // delay
    RC7 = 0;             // Set RC7 Low
    break;

case 1:
