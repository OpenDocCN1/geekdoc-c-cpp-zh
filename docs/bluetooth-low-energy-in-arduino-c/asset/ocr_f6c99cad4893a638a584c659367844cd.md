void onDataWrittenCallback(const GattWriteCallbackParams *params) {
    serial.printf("command written");
    if (params->handle == commandCharacteristic.getValueHandle()) {
        bleDataWritten = true;
        memcpy(
            bleCommandValue,
            (char*) params->data,
            sizeof(bleCommandValue) * params->len
        );
