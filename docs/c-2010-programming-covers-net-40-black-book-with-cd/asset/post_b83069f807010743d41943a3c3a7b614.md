The preceding listing checks the existence of the TreeView2 control. If it is found that there does not exist any node in the TreeView2 control, the GetCategories function is called to generate the first-level nodes containing the categories names. In the GetCategories function, we query the database to retrieve the records and generate the first-level tree nodes by using the node.ChildNodes.Add method, as shown in Listing 15.8:

# Listing 15.8: Showing the Code for Generating the First-Level Tree Node

```
public void GetCategories(TreeNode node)
{
    SqlCommand sqlQuery = new SqlCommand("Select CategoryName, CategoryID From Categories");
    DataSet ResultSet = null;
    ResultSet = PopulateDataFunction(sqlQuery);
    if (ResultSet.Tables.Count > 0)
    {
        foreach (DataRow row in ResultSet.Tables[0].Rows)
        {
            TreeNode NewNode = new TreeNode(row["CategoryName"].ToString(), row["CategoryID"].ToString());
            NewNode.PopulateOnDemand = true;
            NewNode.SelectAction = TreeNodeSelectAction.Expand;
            node.ChildNodes.Add(NewNode);
        }
    }
}
```

After generating the authors’ names, the code again checks for the TreeView2_TreeNodePopulate event, and this time, the GetProducts function is called to generate the second-level nodes containing the products in the respective categories whose names have been retrieved from the database by using the GetCategories function, as shown in Listing 15.9:

# Listing 15.9: Showing the Code for Retrieving the Product Names from the Database

```
public void GetProducts(TreeNode node)
{
    SqlCommand sqlQuery = new SqlCommand();
    sqlQuery.CommandText = "Select ProductName From Products where CategoryID = @categoryid";
    sqlQuery.Parameters.Add("@categoryid", SqlDbType.Int).Value = node.Value;
    DataSet ResultSet = PopulateDataFunction(sqlQuery);
    if (ResultSet.Tables.Count > 0)
    {
        foreach (DataRow row in ResultSet.Tables[0].Rows)
        {
            TreeNode NewNode = new TreeNode(row["ProductName"].ToString());
            NewNode.PopulateOnDemand = false;
            NewNode.SelectAction = TreeNodeSelectAction.None;
            node.ChildNodes.Add(NewNode);
        }
    }
}
```
