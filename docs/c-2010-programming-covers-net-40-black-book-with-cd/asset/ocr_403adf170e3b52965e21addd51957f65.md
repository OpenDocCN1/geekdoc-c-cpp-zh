        public MainWindow()
        {
            InitializeComponent();
        }
        private void element_MouseEnter(object sender, MouseEventArgs e)
        {
            ((TextBlock)sender).Background = new SolidColorBrush(Colors.LightBlue);
        }
        private void element_MouseLeave(object sender, MouseEventArgs e)
        {
            ((TextBlock)sender).Background = new SolidColorBrush(Colors.Beige);
        }
    }
}

When you run the application by pressing the F5 key, you see a window with text on it. When you move the mouse pointer over the window, the color of the window changes to light blue and when you move the mouse pointer away from the window, the color changes to beige.

# Using Triggers

You can use triggers in a WPF application to change the look and feel of the controls dynamically. A good example of this is when you move the mouse pointer over a button, the button changes its color. In WPF, you can easily do this by using the XAML code.
