<?php

function Fib($n) {
    if ($n == 0 || $n == 1) {
        $return_value = $n;
    }
    else {
        $return_value = Fib($n - 1) + Fib($n - 2);
    }

    return $return_value;
}

$num = trim(fgets(STDIN));
echo Fib($num);
