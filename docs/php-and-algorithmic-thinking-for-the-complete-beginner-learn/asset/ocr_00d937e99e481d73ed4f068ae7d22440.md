```php
<?php
define("ROWS", 30);
define("COLUMNS", 20);

$a = array(array());
for ($i = 0; $i <= ROWS - 1; $i++) {
    for ($j = 0; $j <= COLUMNS - 1; $j++) {
        $a[$i][$j] = trim(fgets(STDIN));
    }
}

$min = array();
$max = array();
for ($i = 0; $i <= ROWS - 1; $i++) {
    $min[$i] = $a[$i][0];
    $max[$i] = $a[$i][0];
    for ($j = 1; $j <= COLUMNS - 1; $j++) {
        if ($a[$i][$j] < $min[$i]) {
            $min[$i] = $a[$i][$j];
        }
        if ($a[$i][$j] > $max[$i]) {
            $max[$i] = $a[$i][$j];
        }
    }
}

for ($i = 0; $i <= ROWS - 1; $i++) {
    echo $min[$i], $max[$i];
}
?>
