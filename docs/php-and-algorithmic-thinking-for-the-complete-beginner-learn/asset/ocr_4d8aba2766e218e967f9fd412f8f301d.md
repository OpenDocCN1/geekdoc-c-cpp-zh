```php
function div_mod($a, $b, &$int_quotient, &$int_remainder, &$ret_value) {
    $ret_value = true;
    if ($b == 0) {
        $ret_value = false;
    } else {
        $int_quotient = intval($a / $b);
        $int_remainder = $a % $b;
    }
}
