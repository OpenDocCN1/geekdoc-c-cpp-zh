function my_sort(&$a, $ascending = true) {
    for ($m = 1; $m <= PEOPLE - 1; $m++) {
        for ($n = PEOPLE - 1; $n >= $m; $n--) {
            if ($ascending == true) {
                if ($a[$n] < $a[$n - 1]) {
                    my_swap($a[$n], $a[$n - 1]);
                }
            }
            else {
                if ($a[$n] > $a[$n - 1]) {
                    my_swap($a[$n], $a[$n - 1]);
                }
            }
        }
    }
}
