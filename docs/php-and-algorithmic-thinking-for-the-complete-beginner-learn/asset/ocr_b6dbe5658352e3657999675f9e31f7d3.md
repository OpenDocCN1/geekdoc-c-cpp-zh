# 8. Design the flowchart that corresponds to the following PHP script.

```
<?php
$a = trim(fgets(STDIN));
if ($a > 0) {
    for ($i = 0; $i <= $a; $i++) {
        echo 3 * $i + $i / 2;
    }
} elseif ($a == 0) {
    $b = trim(fgets(STDIN));
    while ($b > 0) {
        $b = trim(fgets(STDIN));
    }
    echo 2 * $a + $b;
} else {
    $b = trim(fgets(STDIN));
    while ($b < 0) {
        $b = trim(fgets(STDIN));
    }
    for ($i = $a; $i <= $b; $i++) {
        echo $i;
    }
}
?>
```
