        if ($weight <= 1) {
            $cost = $weight * 0.010;
        }
        elseif ($weight <= 2) {
            $cost = $weight * 0.013;
        }
        elseif ($weight <= 4) {
            $cost = $weight * 0.015;
        }
        else {
            $cost = $weight * 0.020;
        }
    }
