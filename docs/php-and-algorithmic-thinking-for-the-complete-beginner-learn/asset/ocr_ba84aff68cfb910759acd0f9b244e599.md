<?php
function formula1($x) {
    if ($x == 5) {
        echo "Error! Division by zero";
    }
    else {
        $y = 3 * $x / ($x - 5) + (7 - $x) / (2 * $x);
        echo $y;
    }
}

function formula2($x) {
    if ($x == -2) {
        echo "Error! Division by zero";
    }
    else {
        $y = (45 - $x) / ($x + 2) + 3 * $x;
        echo $y;
    }
}

echo "Enter a value for x: ";
$x = trim(fgets(STDIN));
if ($x >= 1) {
    formula1($x);
}
else {
    formula2($x);
}
?>
