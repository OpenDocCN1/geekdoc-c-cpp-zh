public static VerifyTokenResult VerifyToken(string apikey, string userid, string token, bool tmode)
{
    VerifyTokenResult res = null;

    using (TwilioCore tc = new TwilioCore())
    {
        res = tc.VerifyToken(apikey, userid, token, tmode);
    }

}
