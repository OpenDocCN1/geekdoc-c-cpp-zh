public class TwilioCore : IDisposable
{
    protected bool disposed;

    public string SID
    {
        get;
        set;
    }

    public string aTk
    {
        get;
        set;
    }

    public TwilioCore()
    {
        SID = string.Empty;
        aTk = string.Empty;
    }

    public TwilioCore(string p1, string p2)
    {
        SID = p1;
        aTk = p2;
    }

    ~TwilioCore()
    {
        Dispose(false);
    }

    public virtual void Dispose(bool disposing)
    {
        if (!disposed)
        {
            if (disposing)
            {
            }

            SID = string.Empty;
            aTk = string.Empty;
        }

        disposed = true;
    }
}
