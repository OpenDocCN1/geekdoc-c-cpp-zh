# 244 • Object-Oriented Programming with C++

```
class Student {
	char* name;
	int rollNumber;

private:
	Student() {
		name = "AlanKay";
		rollNumber = 1025;
	}
	void setNumber(int no) {
		rollNumber = no;
	}
	int getRollNumber() {
		return rollNumber;
	}
};
```

```
class AnualTest: Student {
	int mark1, mark2;

public:
	AnualTest(int ml, int m2)
	:mark1(ml), mark2(m2) {
	}
	int getRollNumber() {
		return Student::getRollNumber();
	}
};
```

```
void main()
{
	AnualTest test1(92, 85);
	cout << test1.getRollNumber();
}
```
