// allocate memory
numbers = malloc( N * sizeof(int));

// set values
int i;
for(i=0;i<N;i++){
    numbers[i] = i+3;
}

// display values
for(i=0;i<N;i++){
    printf("%d ",numbers[i]);
}
printf("\n");

// free memory
free(numbers);

return 0;
}
