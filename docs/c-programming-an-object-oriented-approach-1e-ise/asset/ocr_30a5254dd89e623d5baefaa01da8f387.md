// Recursive definition of sum (n)
int sum (int n)
{
    if (n <= 0)
    {
        return 0;
    }
    return n + sum (n - 1);
}
// Recursive definition of factorial (n)
int factorial (int n)
{
    if (n <= 1)
    {
        return 1;
    }
    return n * factorial (n - 1);
}
