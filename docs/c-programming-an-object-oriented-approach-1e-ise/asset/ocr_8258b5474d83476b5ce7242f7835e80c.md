#include <iostream>
using namespace std;

#ifndef TIME_H
#define TIME_H

class Time
{
    private:
        int hours;
        int minutes;
        int seconds;
    public:
        Time (int hours, int minutes, int seconds);
        Time ();
        ~Time ();
        void print() const;
        void tick();

    private:
        void normalize();          // Helping function
};
#endif

**Member Functions**  There are many member functions that could be applied not only on data members but also on the objects created from the class.
