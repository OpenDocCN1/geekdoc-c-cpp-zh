Integer :: Integer (int v)
try: sp (new int)
{
    *sp = v;
}
catch (...)
{
    throw;
}
// Destructor
Integer :: ~Integer ()
{
}
// Accessor function
int Integer :: getValue()
{
    return *sp;
}
