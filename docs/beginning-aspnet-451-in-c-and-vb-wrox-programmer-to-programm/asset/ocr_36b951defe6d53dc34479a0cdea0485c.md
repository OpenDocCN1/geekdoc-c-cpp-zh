VB.NET

```
Protected Sub SaveButton_Click(sender As Object, e As EventArgs)
    Handles SaveButton.Click
    Using myEntities As New WroxEntities()
Dim myReview As Review
 If _id = -1 Then Insert new item
    myReview = New Review()
    myReview.CreateDateTime = DateTime.Now
    myReview.UpdateTime = myReview.CreateDateTime
myEntities.Reviews.Add(myReview)
 Else update existing item
    myReview = (From r In myEntities.Reviews)
    Where r.Id = _id
    Select r).Single()
    myReview.UpdateTime = DateTime.Now
 End If

myReview.Title = TitleText.Text
myReview.Summary = SummaryText.Text
myReview.Body = BodyText.Text
myReview.GenreId = Convert.ToInt32(GenreList.Selected)
myReview.Authorized = Authorized.Checked

C#

```
protected void SaveButton_Click(object sender, EventArgs e)
{
    using (WroxEntities myEntities = new WroxEntities())
    {
        Review myReview = new Review();
        if (_id == -1) // Insert new item
        {
            myReview.CreateDateTime = DateTime.Now;
            myReview.UpdateTime = myReview.CreateDateTime;
            myEntities.Reviews.Add(myReview);
        }
        else // update existing item
        {
            myReview = (from r in myEntities.Reviews
                        where r.Id == _id
                        select r).Single();
        }
    }
