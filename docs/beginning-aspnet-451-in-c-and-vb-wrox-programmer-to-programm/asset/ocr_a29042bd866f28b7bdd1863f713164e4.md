    e.Values("UpdateDateTime") = DateTime.Now

```
Protected Sub DetailsView_ItemUpdated(sender As Object,
    e As DetailsViewUpdatedEventArgs Handles DetailsView.ItemUpdated
    If e.Exception Is Not Then
    EndEditing()
End If
End Sub

```
Protected Sub DetailsView_ItemUpdating(sender As Object,
    e As DetailsViewUpdatingEventArgs Handles DetailsView.ItemUpdating
    e.NewValues("UpdateDateTime") = DateTime.Now
End Sub
