9. Double-click the page in the gray, read-only area to have VS set up a handler for the Page’s Load event and add the following code:

# VB.NET

```
Protected Sub Page_Load(sender As Object, e As EventArgs) Handles Me.Load
    Using myEntities As New PlanetWroxEntities()
        Dim authorizedReviews = From review In myEntities.Reviews
                                Where review.Authorized = True
                                Order By review.CreateDateTime Descending
                                Select review

        GridView1.DataSource = authorizedReviews.ToList()
        GridView1.DataBind()
    End Using
End Sub
```

# C#

```
protected void Page_Load(object sender, EventArgs e)
{
    using (PlanetWroxEntities myEntities = new PlanetWroxEntities())
    {
        var authorizedReviews = from review in myEntities.Reviews
                                where review.Authorized == true
                                orderby review.CreateDateTime descending
                                select review;

        GridView1.DataSource = authorizedReviews.ToList();
        GridView1.DataBind();
    }
}
```
