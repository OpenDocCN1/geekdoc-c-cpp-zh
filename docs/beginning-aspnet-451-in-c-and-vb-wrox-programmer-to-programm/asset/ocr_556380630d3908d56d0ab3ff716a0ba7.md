Public Function ListView1_GetData() As IQueryable(Of Picture)
    Dim photoAlbumId As Integer = Convert.ToInt32(PhotoAlbumList.SelectedValue)
    Return From p in entities.Pictures
           where p.PhotoAlbumId = photoAlbumId
           Order By p.Id
           select p
End Function
