public void ListView_DeleteItem(int id)
{
    using (var entities = new PlanetWroxEntities())
    {
        var picture = (from p in entities.Pictures
                        where p.Id == id
                        select p).Single();
        entities.Pictures.Remove(picture);
        entities.SaveChanges();
    }
}
