public IQueryable<Picture> ListView1_GetData()
{
    int photoAlbumId = Convert.ToInt32(PhotoAlbumList.SelectedValue);
    return from p in entities.Pictures
        where p.PhotoAlbumId == photoAlbumId
        orderby p.Id
        select p;
