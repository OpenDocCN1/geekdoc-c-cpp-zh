# PROGRAMS ON STRUCTURES AND UNION

50. Create a structure employee with employee name, basic sal and netsal. Accept employee name and basic sal from the user through structure. Calculate net salary by adding DA as 11% Basic, HRA as 75% of Basic, TA as 55% of Basic, Calculate net salary and print Net salary.

```
````c
#include<stdio.h>
#include<conio.h>
struct employee{
    char emp_name[20];
    int basic_sal;
    float net_sal;
}emp;
void cal_netsalary(struct employee *x);
void main()
{
    struct employee *pt;
    pt=&emp;
    //accept details of employee
    printf("\nEnter Employee Name: ");
    scanf("%s",pt->emp_name);
    printf("\nEnter Employee Basic Salary: ");
    scanf("%d",&pt->basic_sal);
    //Calculate Net salary of Employee
    cal_netsalary(pt);
    //display Employee's salary
    printf("\n Employee's Net Salary: %f",pt->net_sal);
    getch();
}
void cal_netsalary(struct employee *x)
{
    x->net_sal = 0;
    x->net_sal+=(x->basic_sal*0.11);    //DA calculation
    x->net_sal+=(x->basic_sal*0.075);   //HRA calculation
    x->net_sal+=(x->basic_sal*0.055);   //TA calculation
}
````
