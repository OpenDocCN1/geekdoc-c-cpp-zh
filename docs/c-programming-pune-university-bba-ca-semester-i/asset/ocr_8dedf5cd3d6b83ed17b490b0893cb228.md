+   Step 4: If Num <=1

# **Program:**

```
```c
#include<stdio.h>
void main()
{
    int no, i = 2, p = 0;
    printf("Enter any no to detect prime or Not: ");
    scanf("%d", &no);
    do
    {
        if(no% i==0)
        p=1;
        i++;
    }
    while (i<no);
    if(p==0 || no==2)
        printf("\n The Given number is prime : %d", no);
    else
        printf("\n The number is NOT prime : %d", no);
}
