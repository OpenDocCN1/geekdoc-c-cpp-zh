# Example of Structures Containing Arrays

```
```c
struct Student
{
    int roll_No;
    char name[40];
    int marks[5];
    int total;
    float percentage;
};
void main()
{
    int i;
    struct Student s;
    printf("\nEnter Student Roll No: ");
    scanf("%d", &s.roll_No);
    printf("\nEnter Student Name: ");
    scanf("%s", &s.name);
    s.total = 0;
    for(i=0; i<5; i++)
    {
        printf("\n Enter Marks %d:", i+1);
        scanf("%d", &s.marks[i]);
        s.total = s.total + s.marks[i];
    }
}
