8. Program to calculate Area and Perimeter of Triangle.
(Hint : Area = L x B , Perimeter = 2(L+B))

```
~~~c
#include<stdio.h>
#include<conio.h>
void main()
{
    int length,breadth,area,peri;
    clrscr();
    printf("Enter the Length & Breath of triangle\n");
    scanf("%d %d",&length,&breadth);
    area=length * breath;
    peri=2*(length + breadth);
    printf("The area of triangle is %d",area);
    printf("\nThe perimeter of triangle is %d",peri);
    getch();
}
```
