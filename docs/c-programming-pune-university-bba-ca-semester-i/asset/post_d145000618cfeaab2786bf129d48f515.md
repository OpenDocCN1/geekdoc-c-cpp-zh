### 73. Write a program in C to read file in binary mode, count number of vowels of the file.

```c
#include<stdio.h>
void main()
{
    int cnt=0;
    FILE *fp;
    char ch;
    fp=fopen("kiran.txt","r");
    if(fp==NULL)
    {
        printf("can not open file");
    }
    else
    {
        while(1)
        {
            ch=getc(fp);
            if(ch==EOF)
            {
                break;
            }
            else
            {
                if(ch=='i'||ch=='o'||ch=='u'||ch=='a'||ch=='e'||ch=='I'||ch=='O'||ch=='U'||ch=='A'||ch=='E')
                {
                    cnt=cnt+1;
                }
            }
        }
    }
    printf("\n number of vowels are %d",cnt);
    fclose(fp);
    getch();
}
```

**Output**
Number of vowels - 4

### 76. Write a C program to read name and marks of n number of students from user and store them in a file.

```c
#include <stdio.h>
int main()
{
    char name[50];
    int marks, i, num;

    printf("Enter number of students: ");
    scanf("%d", &num);

    FILE *fptr;
    fptr = (fopen("C:\\student.txt", "w"));
    if(fptr == NULL)
    {
        printf("Error!");
        exit(1);
    }

    for(i = 0; i < num; ++i)
    {
        printf("For student%d\nEnter name: ", i+1);
        scanf("%s", name);

        printf("Enter marks: ");
        scanf("%d", &marks);
        fprintf(fptr,"\nName: %s \nMarks=%d \n", name, marks);

}
```
