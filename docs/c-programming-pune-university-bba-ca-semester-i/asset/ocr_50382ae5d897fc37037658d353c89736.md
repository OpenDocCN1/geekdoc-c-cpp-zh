```
~~~c
#include<stdio.h>
void main()
{
    int N, i, fact=1;
    printf("enter any number to calculate factorial:");
    scanf("%d", &N);
    for(i=1; i<=N; i++)
    {
        fact = fact * i;
    }
    printf("\n the factorial of given number is %d", fact);
}
~~~
```
