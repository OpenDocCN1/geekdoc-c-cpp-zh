{
        int i,n,flag = 0, mat[n][n];
        printf("Enter the value of n");
        scanf("%d",&n);
        j;
        for(i = 0; i < n; i++)
        {
                for(j = 0; j < n; j++)
                scanf("%d",&mat[i][j]);
        }
        for (i = 1; i < n; i++)
        for (j = i+1; j < i; j++)
                if (mat[i][j] != 0)
                        flag = 0;
                else
                        flag = 1;
        if (flag == 1)
                printf("Upper Triangular Matrix");
        else
                printf("Not an Upper Triangular Matrix");

}
