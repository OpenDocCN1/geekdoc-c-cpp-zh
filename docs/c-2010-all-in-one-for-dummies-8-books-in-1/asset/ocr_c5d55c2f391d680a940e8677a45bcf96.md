# Changing a WCF Service to Use ReST 765

```
<endpoint address="registration"
  binding="customBinding"
  bindingConfigurations="posBinding"
  contract="SHARPService.registration" />

<service>
  <behaviors>
    <serviceBehaviors>
      <behavior>
        <!-- To avoid disclosing metadata information, set the value below
        and remove the metadata endpoint above before deployment -->
        <serviceMetadata httpEnabled="true"/>
        <!-- To receive exception details as faults for debugging purposes, set
        the value to true. Set to false before deployment to avoid disclosing
        exception information -->
        <serviceDebug includeDetailInFaults="false" />
      </behavior>
    </serviceBehaviors>
  </behaviors>
</serviceModel>
<system.webServer>
  <modules>
    <remove managedModulesForAllRequests="true" />
  </modules>
  <connectionStrings>
    <add name="confusedEntities" connectionString="metadata=res://*/
    Sharp.sdl|res://|Sharp.sdl|provider=System.Data.SqlClient;provider connectionString="Data Source=\SQLPRESS;Server=MSSQL;DATA=C:\Program Files\Microsoft SQL Server\MSSQLDB;Integrated Security=True;Connect Timeout=30;User ID=;Password=;"
    EntityName=True;MultipleResultSet=True" providerName="System.Data.SqlClient" />
  </connectionStrings>
</configuration>
```
