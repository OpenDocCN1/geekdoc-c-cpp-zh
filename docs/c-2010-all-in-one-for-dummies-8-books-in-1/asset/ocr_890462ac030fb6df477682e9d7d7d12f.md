// An example of a prioritizable class that can be stored in
// the priority queue; any class that implements
// IPrioritizable would look something like Package.
class Package : IPrioritizable {
    private Priority _priority;
    //Constructor
    public Package(Priority priority) {
        this._priority = priority;
    }

    //Priority - Return package priority read-only.
    public Priority getPriority() {
        return _priority;
    }
}

//PackageFactory - You need a class that knows how to create a
// package of any desired type on demand; such a class
// is a classFactory
class PackageFactory {
    //Random-number generator
    Random _rand = new Random();

    //CreatePackage - The factory method selects a random priority,
    // then creates a package with that priority.
    // Could implement this as an iterator block.
    public Package CreatePackage() {
        // Need a 0, 1, or 2 (values less than 3).
        int rand = _rand.Next(3);
        // Use that to generate a new package.
        // Casting int to enum is clunky, but it saves
        // having to use ifs or a switch statement.
        return new Package((Priority)rand);
    }
}

PriorityQueue is a bit long, you need to look at each part carefully. After a look at the target class, Package, you can follow a package's journey through the Main() method near the top.

When you run PriorityQueue, it runs several times. Because it's built around random numbers, you are getting varying results on each run. Sometimes it may “receive” zero packages, for instance. (Slow days happen, I guess.)
