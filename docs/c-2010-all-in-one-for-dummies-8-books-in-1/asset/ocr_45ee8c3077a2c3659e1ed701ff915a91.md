# Touring Main ()

Before you implement the PriorityQueue class itself, it's useful to get an overview of how it works in practice in OOPs, Here's the Main() method for the PriorityQueue example:

```
static void Main(string[] args)
{
    Console.WriteLine("Create a priority queue:");
    PriorityQueue<Package> pq = new PriorityQueue<Package>();
    Console.WriteLine();
    // add a random number (0-20) of random packages to queue:
    Package pack;
    PackageFactory fact = new PackageFactory();
    // wait a random number less than 20.
    Random rand = new Random();
    int numToCreate = rand.Next(20); // Random from 0-20.
    for (int i = 0; i < numToCreate; i++)
    {
        Console.WriteLine("Generating and adding random package {0}", i);
        pack = fact.CreatePackage();
        Console.WriteLine("Added with priority {0}", pack.Priority);
        pq.Enqueue(pack);
    }

    Console.WriteLine("See what we got:");
    int count = pq.Count;
    Console.WriteLine("Packages received: {0}", count);

    Console.WriteLine("Remove a random number of packages (0-20): {0}*");
    int numToRemove = rand.Next(count);
    Console.WriteLine("Removing {0} packages, {0}", numToRemove);
    for (int i = 0; i < numToRemove; i++)
    {
        pack = pq.Dequeue();
        if (pack != null)
        {
            Console.WriteLine("tShipped package with priority {0}", pack.Priority);
        }
    }
    // see how many were "shipped."
    Console.WriteLine("Shipped {0} packages", count - pq.Count);
```
