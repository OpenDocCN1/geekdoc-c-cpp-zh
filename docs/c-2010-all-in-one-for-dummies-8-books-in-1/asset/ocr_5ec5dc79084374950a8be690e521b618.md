Console.WriteLine("\Now listed alphabetically: ");
// Array.Sort() is a method on the Array class.
// Array.Sort() does its work in-place on the planets array,
// which leaves you without a copy of the original array. The
// solution is to sort the old array to a new one and sort it.
Array.Sort(sortedNames);
// This demonstrates that (a) sortedNames contains the same
// strings as planets and (b) they're now sorted.
foreach (string planet in sortedNames) {
    Console.WriteLine("\t" + planet);
}

Console.WriteLine("\nlist by length shortest: ");
// This algorithm is called "Bubble Sort": It's the simplest
// but worst-performing. Array.Sort() method is much
// more efficient, but I couldn't use it directly to sort the
// planets in order of name length because it sorts strings,
// not their length.
int outer; // Index of outer loop
for (int outer = 0; outer < planets.Length; outer++) {
    int inner; // Index of inner loop
    // Loop from last to first: planets[4] to planets[0].
    for (int inner = planets.Length - 1; inner >= 0; inner--) {
        // On each outer loop, loop goes through all BEYOND the
        // loop goes up, for the loop, you can traverse the
        // to planets[inner]. Using this to traverse the
        // array in either direction.
        for (inner = 1; inner <= outer; inner++) {
            // Compare elements. If the earlier one is longer
            // than the later one, swap them. This shows how you can
            // swap element with another when they're out of order.
            if (planets[inner - 1] > planets[inner]) {
                // Temporarily store one planet
                string temp = planets[inner - 1];
                // Overwrite that planet with the other one.
                planets[inner - 1] = planets[inner];
                // Finally, reclaim the planet stored in temp and put
                planets[inner] = temp;
            }
        }
    }
    Console.WriteLine("\t" + planets[outer]);
}

Console.WriteLine("\nNow listed longest first: ");
// That is, just loop down through the sorted planets.
for (int i = planets.Length - 1; i >= 0; i--) {
    Console.WriteLine("\t" + planets[i]);
}

Console.WriteLine("\nPress Enter to terminate...");
Console.Read();
