6. Open App.xaml.cs.

7. Overide the OnStartup method to include the creation of the Purpose application resource by adding the code to the App class:

```
protected override void OnStartup(StartupEventArgs e) {
//create and add the Purpose application resource
string purpose = "Hello WPF World, in C#";nthis.Resources.Add("Purpose", purpose);
base.OnStartup(e);
}
```

8. Open MainWindow.xaml, and give the Grid element by name by adding the following attribute:

```
x:Name="gridLayout"
```

9. Open MainWindow.xaml.cs, and in the default constructor, add the following code:

```
//grid column components and row
this.gridLayout.ColumnDefinitions.Add(new ColumnDefinition());
this.gridLayout.RowDefinitions.Add(new RowDefinition())..Add(new RowDefinition());

//obtain label content from the application resource, Purpose
string purpose = this.FindResource("Purpose") as string;
Label lblPurpose = new Label{Content = purpose};\lblPurpose.FontSize = 25;

//add label to the grid
this.gridLayout.Children.Add(lblPurpose);

//assign attached property values
Grid.SetColumn(lblPurpose, 0);
Grid.SetRow(lblPurpose, 0);
```
