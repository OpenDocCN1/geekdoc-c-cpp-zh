# Looping Around the Iterator Block 165

// Create a collection and use two iterators to identify
// it independently (each using an iterator block).

```
{
    string[] strs = new string[] { "Joe", "Bob", "Tony", "Fred" };
    MyCollection mc = new MyCollection(strs);
    public static void Main(string[] args)
    {
        // Create the first iterator
        MyCollectionIterator m1 = mc.GetEnumerator();
        foreach (string s in m1)
        {
            // Do some useful work with each string.
            Console.WriteLine(s);
            if (s == "Tony")
            {
                // In the middle of that iteration, start a new one,
                // a second iterator, repeated for each outer loop.
                MyCollectionIterator m2 = mc.GetEnumerator();
                foreach (string s2 in m2)
                {
                    // Do some useful work with each string.
                    if (s2 == "Bob")
                    {
                        Console.WriteLine("\t" + s + " is boss's, " + s2);
                    }
                }
            }
        }

// Wait for user to acknowledge the results.
        Console.WriteLine("Press Enter to terminate...");
        Console.Read();
    }

// A simple collection of strings
public class MyCollection
{
    // Implement collection with an old-fashioned ArrayList.
    // internal, separate iterator object can access the strings.
    internal ArrayList _list = new ArrayList();
    public MyCollection(string[] strs)
    {
        foreach (string s in strs)
        {
            _list.Add(s);
        }
    }

    // As in LinkedList, returns one of your
    // iterator objects.
    public MyCollectionIterator GetEnumerator()
    {
        return new MyCollectionIterator(this);
    }
}

// The Iterator class for MyCollection
// (MyCollection is in a separate file.)
public class MyCollectionIterator
{
    // Store a reference to the collection.
    private MyCollection _mc;
    public MyCollectionIterator(MyCollection mc)
    {
        this._mc = mc;
    }
}
```
