
## 342 Looking at a Program that CAN_BE_USED_AS an Example

A class implements the Comparable<T> interface by implementing a CompareTo() method. Notice that CompareTo() takes an argument of type T, a type you supply when you instantiate the interface for a particular data type — as in this example:

```sharp
class SoAndSo : Comparable<SoAndSo> // Make me comparable.
```

It seems a little Darwin, but you could say that one Student object is "greater than" another Student object if his grade-point average is higher. (Okay, either a better student or a better apple-polisher — It doesn't matter.)

Implementing the CompareTo() method implies that the objects have a sorting order. If one student is "greater than" another, you must be able to sort the students from "least" to "greatest." In fact, most collection classes (including arrays but not dictionaries) supply a Sort() method something like this:

```sharp
void Sort(Comparable<T> objects) {
```

This method sorts a collection of objects that implement the Comparable<T> interface. It doesn't even matter which class the objects belong to. For example, they could be Student objects. Collection classes such as arrays or List<T> could even sort this version of Student:

```sharp
// Student -- Description of a student with name and grade
class Student : Comparable<Student>, IDisplayable
{
    // Constructor -- initialize a new student object.
    public Student(double grade)
        { Grade = grade;
    public double Grade { get; private set; }
    // Implement the Comparable<T> interface:
    public int CompareTo(Student other)
    {
        // Compare another object (in this class, Student objects) and
        // Decide which one comes after the other in the sorted order.
        Student rightStudent = other;
        
        // Compare the current Student (call her 'left') against the other
        // student (call her 'right').
        Student leftStudent = this;

        // Now generate -1, 0 or 1 based on the Sort criteria (the student's
        // grade). I could use class Double's CompareTo() method instead.
        {
            if (rightStudent.Grade < leftStudent.Grade)
                return -1;
        }
}
}
```

## Putting it all together

This is the moment you've been waiting for: the complete Interface program that uses the features described earlier in this chapter:

```sharp
// Interface -- Demonstrates how the interface concept can be used
// to provide an enhanced degree of flexibility in factoring
// and implementing classes.
using System;
namespace SortInterface
{
    // IDisplayable -- An object that can convert itself into a displayable
    // string format (duplicates what you can do by overriding
    // ToString(), which helps me make a point)
    interface IDisplayable
    {
        // Display -- return a string representation of yourself.
        string Display();
    }
    class Program
    {
        public static void Main(string[] args)
        {
            Console.WriteLine("Sorting the list of students:");
            // Get an unordered array of students.
            Student[] students = Student.CreateStudentList();
            // Use the IComparable<T> interface to sort the array.
            Array.Sort(students);
            // Now the IDisplayable interface to display the results
            DisplayArray(students);

            // Now sort an array of birds by name using the same routines
            // though the classes Bird and Student have no common base class.
            Console.WriteLine("sorting the list of birds:");
            Bird[] birds = Bird.CreateBirdList();
            // Notice that it's not necessary to cast the objects explicitly
            // to an array of IDisplayables (and wasn't for Students, either)
            Array.Sort(birds);
            DisplayArray(birds);
            // Wait for user to acknowledge the results.
            Console.WriteLine("Press Enter to terminate...");
            Console.Read();
        }

        // DisplayArray -- Display an array of objects that
        // implement the IDisplayable interface.
        public static void DisplayArray(IDisplayable[] displays)
        {
            foreach (IDisplayable displayable in displays)
            {
                Console.WriteLine("*", displayable.Display());
            }
        }

        // Students -- Sort students by grade
        // Student -- Description of a student with name and grade
        class Student : IComparable<Student>, IDisplayable
        {
            // Constructor -- Initialize a new object.
            public Student(string name, double grade)
            {
                Name = name;
                grade = grade;
            }
            // CreateStudentList -- To save space return a list of students.
            static string[] names = { "Peter", "Marge", "Bart", "Maggie" };
            static double[] grades = { 85, 100, 30 };
            public static Student[] CreateStudentList()
            {
                Student[] students = new Student[names.Length];
                for (int i = 0; i < names.Length; i++)
                {
                    students[i] = new Student(names[i], grades[i]);
                }
                return students;
            }
            // Access-only properties.
            public string Name { get; private set; }
            public double Grade { get; private set; }
            // Implement the IComparable<T> interface
            public int CompareTo(Student other)
            {
                // Compare another object (in this class, Student objects) and
                // Decide which one comes after the other in the sorted order.
                Student rightStudent = other;
                // Compare the current Student (call her 'left') against the other
                // student (call her 'right').
                Student leftStudent = this;
                // Now generate -1, 0 or 1 based on the Sort criteria (the student's
                // grade). I could use class Double's CompareTo() method instead.
                if (rightStudent.Grade < leftStudent.Grade)
                    return -1;
                if (rightStudent.Grade > leftStudent.Grade)
                    return 1;
                else
                    return 0;
            }
            // Display -- Implement the IDisplayable interface:
            public string Display()
            {
                string padName = Name.PadRight(9);
                return String.Format("{0}: {1}", padName, Grade);
            }
        }

        // Sort birds by their names
        // Bird -- Just an array of bird names.
        class Bird : IComparable<Bird>, IDisplayable
        {
            // Constructor -- Initialize a new Bird object.
            public Bird(string name) {}
            // CreateBirdList -- Return a list of birds to the caller;
            static string[] _birdNames = { "Robin", "Cardinal", "Bluejay", "Finch", "Sparrow" };
            public static Bird[] CreateBirdList()
            {
                Bird[] birds = new Bird[_birdNames.Length];
                for (int i = 0; i < _birdNames.Length; i++)
                {
                    birds[i] = new Bird(_birdNames[i]);
                }
                return birds;
            }
            public string Name { get; set; }
            // Implement the IDisplayable interface:
            // Compare birds by name; use the
            // Built-in String.Compare method.
            public int CompareTo(Bird right)
            {
                // Compare the 'current' bird to the 'right object' bird.
                Bird leftBird = this;
                return String.Compare(leftBird.Name, right.Name);
            }
            // Display -- Implement the IDisplayable interface.
            public string Display()
            {
                return Name;
            }
        }
    }
}
```
