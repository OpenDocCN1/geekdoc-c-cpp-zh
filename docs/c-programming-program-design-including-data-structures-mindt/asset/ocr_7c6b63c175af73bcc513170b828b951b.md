```cpp

{
    if (x > 0 && y > 0)
    {
        if (x >= y && y != 0)
        {
            cout << x % y << " ";
            recFun(x - y, y);
        }
        else if (y > x && x != 0)
        {
            cout << y % x << " ";
            recFun(y - x, x);
        }
    }

}
```
