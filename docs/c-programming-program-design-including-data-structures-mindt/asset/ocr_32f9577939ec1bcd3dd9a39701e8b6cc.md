+   27. What is the output of the following program? (6, 7, 8)

```
#include <iostream>
using namespace std;

void compute(int x, int& y, int& z);

int main()
{
    int one = 7;
    int two = 5;
    int three = 6;

    compute(one, two, three);
    cout << one << ", " << two << ", " << three << endl;

    compute(two, one, three);
    cout << one << ", " << two << ", " << three << endl;

    compute(three, two, one);
    cout << one << ", " << two << ", " << three << endl;

    compute(two, three, one);
    cout << one << ", " << two << ", " << three << endl;

    return 0;
}

void compute(int x, int& y, int& z)
{
    int w = x + y - z;

    y = w % 2;
    x = y + w;
    z = x * y;
}
```
