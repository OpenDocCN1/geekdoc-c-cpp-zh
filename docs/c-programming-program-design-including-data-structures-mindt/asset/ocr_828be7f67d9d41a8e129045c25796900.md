{
    type x = list[0];
    type y = list[size - 1];

    for (int j = 1; j < size - 1; j++)
    {
        if (x < list[j])
            x = list[j];
        if (y > list[size - 1 - j])
            y = list[size - 1 - j];
    }

    return y + x;
}
