```
proc iml;
  /* read in doses, covariance mat, models/guesstimate and model names*/
  use doses var (dose);
  read all into dose;
  use &sigma;
  read all into sigma;
  use &models;
  use all var(modelnam par par2 par3) into par;
  use &models;
  read all var(modnam) into modlist;

  /* Unity matrix */
  unity=j(&n_dose, 1, 1);

  /* Empty matrix for calculated mean vectors and contrasts */
  meanat=(&n_dose, &n_mod+1, -88);
  meanat,(1)=dose;

  /* row labels */
  contmat=j(&n_dose, &n_mod, -88);

  /* calculation of stand. mean-vector for each model */
  do j=2 to &n_mod+1;
    modnam=par[-1,1]; /* for each dose*/
    do i=1 to 3
      if modnam=1 then
        meanat[i,j]=dose[i]/par[j-1, 2] + dose[i];
      if modnam=2 then
        meanat[i,j]=exp(dose[i]/par[j-1, 2]) - 1;
      if modnam=3 then
        meanat[i,j]=dose[i];
      if modnam=4 then
        meanat[i,j]=-log(dose[i] + par[j-1, 2]);
      if modnam=5 then
        meanat[i,j]=dose[i] + par[j-1, 2]*dose[i]**2;
      if modnam=6 then
        meanat[i,j]=dose[i]*par[j-1, 3]/(par[j-1, 2)*par[j-1,
          3] + dose[i]**par[j-1], 3);
      if modnam=7 then
        maxDens=dose(par[j-1, 2)**par[j-1, 2) * (par[j-1,
          3)**par[j-1, 3]/((par[j-1, 2) + par[j-1,
          3)));
        standdose=dose[i]/par[j-1, 4];
        meanat[i,j]=1/maxDens*(standdose**j-1, 2) *
          (1 - standdose**j-1, 3]);
  end;
invsigma=inv(&sigma);

do j=2 to &n_mod+1; /* calculation of contrasts */
  mean_t=trans(meanat[,j]);
  1=mean_t*invsigma*unity;
  2=-t(unity)*invsigma*unity;
  3=1_/2
  4=3*unity;
  5=meanat(, j)-4;
  contrast=insigma_*5;
  c_norm=sqrt(ssq(contrast));
  meanat[, j-1]=&dir*contrast/c_norm;
end;
/* Export the final contrasts to dataset */
append from comat;
quit;

data &models;
set &models;
run;

data __;
set &models;
do j=1 to &n_mod;
  if i=j then
  do;
    call execute ("data optcontrasts; set optcontrasts;
    attrib col="compress(j) label="compress(modnam)";");
end;
data optcontrasts;
merge &doses(keep=dose) optcontrasts;
run;

# Example Code 7.5 Code to Fit Dose-Response Models in Example 1

********************************************************************
Fit all models compare AIC values
********************************************************************/
/* emax model */
proc nlmixed data=biom;
ods output FitStatistics=item;
parms e0=0.3 emax=1 ed50=1 sigma=1;
bounds 0.001 < ed50 < 1.5; /* bound for ed50: [0.001,1.5]*maxDose *
mn=0 + emax*dose/(dose + ed50);
model resp ~ normal(mn, sigma=2);
run;

data item;
set item(where=desc='AIC (smaller is better)');
model="emax";
drop descr;
run;

/* exponential model */
proc nlmixed data=biom;
ods output FitStatistics=fitexpo;
parms e0=0.3 e1=delta sigma=1;
bounds 0.1 < delta < 2; /* bound for delta: [0.1,2]*maxDose *
mn=0 + e1*(exp(dose/delta)-1);
model resp ~ normal(mn, sigma=2);
run;

data fitexpo;
set fitexpo(where=desc='AIC (smaller is better)');
model="expo";
drop descr;
run;

/* linear model (could also be fit with other procedures!) */
proc nlmixed data=biom;
ods output FitStatistics=fitlin;
parms e0=0.3 delta=1;
mn=0 + delta*dose;
model resp ~ normal(mn, sigma=2);
run;

data fitlin;
set fitlin(where=desc='AIC (smaller is better)');
model="lin";
drop descr;
run;

/* quadratic model (could also be fit with other procedures!) */
proc nlmixed data=biom;
ods output FitStatistics=fitquad;
parms e0=0.3 b1=1 b2=1 sigma=1;
mn=e0 + b1*dose + b2*dose^2;
model resp ~ normal(mn, sigma=2);
run;

data fitquad;
set fitquad(where=(descr='AIC (smaller is better')));
model="quad";
drop descr;
run;

/* Emax model */
proc nlmixed data=biom;
ods output FitStatistics=fitsig;
parms emax=0.3 ed50=1 h=1 sigma=1;
bounds 0.001 < ed50 < 1.5, 0.5 < h < 1;
mn=0 + emax*dose/(dose + ed50);
model resp ~ normal(mn, sigma=2);
run;

data fitsig;
set fitsig(where=(descr='AIC (smaller is better')));
model="emax";
drop descr;
run;

ata itemexpo fitlin fitquad;
/* compare AICs of different models */
proc print data=aics;
run;

/* emax model has best AIC value */

Example Code 7.6 Code to Produce Plot for Example 1

****************************************************************
Create a plot showing the results
****************************************************************
/* data set containing where to predict dose-response curves */
data biompredict;
do i=1 to 100;
    dosepred=(i-1)/99;
    output;
end;
drop i;
run;
data biompredict;
merge biompredict;
run;

/* model fit */
%let deltiff=0.4;

/* fit emax and predict at specified grid of dose values */
proc nlmixed data=biompredict;
    ods output FitStatistics;
    parms emax=0.3 ed50=1 sigma=1;
    bounds 0.001 < ed50 < 1.5;
    mn=0+ emax*dose/(dose + ed50);
    model resp ~ normal(mn, sigma=2);
    predict emax*dosepred/(dosepred + ed50) out=preout;
    estimate 'TD' &deltiff* ed50/(emax &deltiff);
run;

/* create adjusted raw means for plot */
proc sql;
    create table adj_means as
    select distinct(dose), mean(resp) as mean_resp
    from biom
    group by dose;
run;
data adj_means;
    set adj_means;
    if dose=0 then mean0=mean_resp;
    retain mean0;
    if dose ne 0 then diff=mean_resp-mean0;
run;
data preout;
    set preout adj_means;
    recline=0.4;
run;

/* plot dose-response curve confidence intervals and adjusted raw means */
proc sgplot data=preout;
    xaxis grid label="Dose";
    yaxis grid label="Response (difference to placebo)";
    band x=dosepred lower=upper upper=upper /transparency=0.5;
```
