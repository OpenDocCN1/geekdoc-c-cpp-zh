In addition, the prior probability of each skeleton is stored in the %priormodel data set for which we equal probabilities as a default. The %BMACRM macro has the same output as the %CRM macro. To illustrate this we consider the above example with seven cohorts of patients treated in the trial and the goal is to determine the dose assignment for the next cohort.

# Example Code 5.2 Bayesian Model Averaging CRM Macro

```
%macro BMACRM(currentdata,skeleton,priormodel,target=0.3,toxstop=0.9);
proc iml;
  use &currentdata;
  use all;
  read all;
  use &skeleton;
  read all into skeleton;
  use &priormodel;
  read all;

nmodel=now(priormodel);
ndose=now(skeleton);

dosecurr=doselevel(nrow(doselevel));\stop=0;
/* posterior-likelihood * prior */
start posttoxf(alpha) global(DLT,doselevel,p,cohortsize);
  sigma=2;
  lik=1;
  nn=nrow(DLT);
  do k=1 to nn;
    p=p(d[k])**#exp(alpha);
    lik=lik*(pi(d[k]))##(1-pi))##(cohortsize-l[k]);
  end;
  lik=lik*exp(-0.5*alpha/alpha/sigma2);
  return(link);
finish;

/* used to calculate the posterior mean of pi i */
start posttoxf(alpha) global(DLT,doselevel,p,cohortsize);
  post=p[j]##exp(alpha)*posterior(alpha);
  return(post);
finish;

marginal=();
povertox=(0,ndose,nmodel);
 md=1 to nmodel;
  p=skeleton,[md];
  lb=log(&target)/log(p[1]);
  call quad(marginal,"posterior",.M .PP) eps=1E-4 peak=-3 scale=0.5
  /* default:PEAK-1
  marginal=marginal/marginal;

do j=1 to ndose;
  call quad(pj,"posttoxf",.M .PP) eps=1E-4 peak=-3 scale=0.5;
  pi-pj=marginal;
  pjm[m,md]=pi;
end;

call quad(povertox,"posterior",-35|lb) eps=1E-4 peak=-3 scale=0.5;
```
