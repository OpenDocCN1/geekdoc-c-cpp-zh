## 6.4.3 Binary Endpoint

The sample SAS code for the simulation and analysis of a binary endpoint are provided here. The reader can use Example Code 6.6 Program to Simulate Binary Data Using the Emax Model to simulate a data set based on the above model.

Example Code 6.6 Program to Simulate Binary Data Using the $E_{max}$ Model

```sas
data dose;
  input dose@;
  datalines;
  0 2.5 10 20
run;

%macro sql(n, e0=-1.39, emax=3.54, ed50=5.5, h=1);
  proc sql noprint;
    select count(dose) into:ngprp
    from dose;
quit;
%let ngprp = &ngprp;
data null;
  set dose;
  pai=exp(e0+(emax.**dose**&h))/(ed50.**dose**&h))/
  (1+exp(e0+(emax.**dose**&h))/(ed50.**dose**&h))));
  call synonym(compress("pai"||put(n_, 1)), compress(pai) );
  call synonym(compress("dose"||put(n_, 1)), compress(dose) );
run;
%do i=1 to &ngprp; %put Group &i. **** &pai&;. %end;

proc im;
  prob= %do i=1 to &ngprp; &pai&;. %end;
  p = repeat(prob,n); /* repeat row times: n/group *
  call srandseed(321); /* to set seed number; *
  x = rand("Bernoulli", p);
  /* create MyData set *
  append from x;
  /* write data in vectors *
  close MyData;
  /* close the data set *
quit;
data mydata;
  set mydata;
  id_n_;
run;
nproc transpose data=mydata out=mydata;
  var col-colngp;
  by id;
run;
data xdose;
/* add the COLn to retrieve the dose */
set dose;
  _name_ = compress("COL" || put(n_, 1));
```

```sas
run;
proc sql noprint; /* retrieve the dose for the different groups*/
create table dstbin as

from mydata left join xdose b

```

## Example Code 6.7 4-parameter Binary Emax Model Fitting Using proc nlmixed

*****specify that ed50 must be positive;
bounds ed50>0;

*****define models;
if dose=0 then eta = e0;
else eta = e0 + ((emax*dose**h)/(ed50**h+dose**h));nexpeta = exp(eta);

 count = binomial(n, p);
