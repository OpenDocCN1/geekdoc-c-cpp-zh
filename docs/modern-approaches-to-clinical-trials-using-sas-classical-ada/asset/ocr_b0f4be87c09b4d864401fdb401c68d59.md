```
proc transpose data=quant out=tquant;
run;
data dosevar;
  do i=1 to 101;
    dose=(i-1)/100*30;
    output;
  end;
  drop i;
run;
```

```
data tquant;
  set tquant;
  set dosevar;
run;
```
