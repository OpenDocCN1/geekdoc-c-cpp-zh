Example Code 11.3 SimGama Macro for Computing Required Sample Size

```
%macro simgamma(seed1, seed2, mux, muy, stdx, stdy, simN=10000, alpha=0
data temp;
ax=round((mux**2)/($sdx**2), 0.0001);
bx=round(mux/($sdx**2), 0.0001);
ay=round((mux**2)/($stdy**2), 0.0001);
by=round(muy/($stdy**2), 0.0001);
run;
data _null;
set temp;
  call symput('ax', compress(trim(ax)));
  call symput('bx', compress(trim(bx)));
  call symput('ay', compress(trim(ay)));
  call symput('by', compress(trim(by)));
run;
data x;
set temp;
  call streaminit(&seed1);
  do i=1 to &simN
    x=rand('gamma', ax);
    x=x/bx;
    output;
end;
run;
data y;
set temp;
  call streaminit(&seed2);
  do i=1 to &simN
    y=rand('gamma', ay);
    y=y/by;
    output;
end;
run;
data one;
merge x y;
  if x<y then flag=1;
  else flag=0;
run;
proc means noprint data=one;
  var flag;
  output out=pl mean=pl;
run;
data _null;
```
