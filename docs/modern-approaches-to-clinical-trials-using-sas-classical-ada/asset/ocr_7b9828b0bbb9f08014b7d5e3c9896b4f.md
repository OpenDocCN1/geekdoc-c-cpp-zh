```
* f(x)=1-x <- probability of assignment to A */
* g(x)=x <- probability of assignment to B */
* where x <- (diff in num of subjects in trt)/(total num of subjects) */

%macro BCDandomize(n1, n2, seed);
data Assignment;
%let numer=EVAL(&n1+&n2);
%let denom=EVAL(&n1+&n2);

* assign to B with probability x, A with prob 1-x */
%if %SYSEVAL(&n1&=0) & %SYSEVAL(&n2&=0) %then let DiffRatio = %SYSEVAL(
%let AllocProb=%SYSEVAL(1/2-&DiffRatio/2);
%let RandNum=%SYFUNC(runani(&seed)); *uniform[0,1] random var;
%if %SYSEVAL(&RandNum <&AllocProb) %then let Assign-A=;
%else let Assign-B=;

* create subject num var _ */
%let AssignNum = %EVAL(&n1+&n2+&1);
groupA = &n1;
groupB = &n2;
allopB = &AllocProb;
newGroup = "Assign";

label group groupA = "Number in arm A"
groupB = "Number in arm B"
allop = "Allocation probability to arm A"
newGroup = "Next assignment"
run;

proc print data=Assignment label noobs;
run;
%end BCDandomize;

%BCDandomize(n1=0,n2=0,seed=123);
%BCDandomize(n1=5,n2=10,seed=123);
```
