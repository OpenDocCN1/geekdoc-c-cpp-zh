with mean 0) has been implemented using the same assumed standard deviation that was used for the $E_0$ parameter.

The information on the $ED_{50}$ parameter only comes from non-clinical studies. A uniform (0, 12) scaled to just a little more than the dose range will be used as a non-informative prior distribution.

Example Code 8.4 Simulated Data for Binary Endpoint is used to simulate the data for this study, including a 15% dropout rate.

# Example Code 8.4 Simulated Data for Binary Endpoint

```
%macro binmaxsim(ntrt=0,p1,p2,p3,p4,p5,p6,p7,p8,npergrp,dpdrop,s
data sims
  data sims (Keep = subject treatment dose response);
  format treatment $88;
  subject = 0;
  do trt = 0 to &ntrt;
    do i = 1 to &npergrp;
      subject = subject + 1;
      if trt = 0 then do;
        treatment = "Placebo";
        dose = 0;
      end;
      else do;
        treatment = "Active";
        dose = 3*(&ntrt -1);
      end;
      response = ranbin(floor(&seed), 1, &p&trt);
      dropout = ranbin(floor(&seed), 1, &pdrop);
      if dropout = 0 then output;
    end;
  end;
run;

proc freq data = sims noprint nopercut out=sum_dt outpct;
  table response * dose dose count total pct_col;
run;
data sum_dt(keep = response dose count total pct_col);
  set sum_dt;
  total = round(count / pct_col)*100,1);
  if response = 1;
run;
%mend;
```
