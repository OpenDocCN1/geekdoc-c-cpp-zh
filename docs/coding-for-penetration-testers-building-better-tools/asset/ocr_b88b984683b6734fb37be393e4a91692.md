my $oct4;

sub get_ip_address {
my $rawaddress;
my $retval;
my $oct;
my $counter;
$rawaddress = shift;
while ($counter<4) {
$oct = $rawaddress % 2**8; #get the rightmost 8 bits
$retval = $oct "." $retval;
$rawaddress = int($rawaddress / 2**8); #get the next 8 bits
$counter++;
chop $retval;
if ($retval =~ m/\.(255|0)$/) { # skip 0 & 255 addresses
return 0;
