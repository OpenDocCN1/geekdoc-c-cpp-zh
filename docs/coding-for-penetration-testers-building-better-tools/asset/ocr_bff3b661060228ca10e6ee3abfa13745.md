sub get_raw_address {
  my $ipaddress;
  my $oct1;
  my $oct2;
  my $oct3;
  my $oct4;
  my $retval;
  $ipaddress = shift;
  ($oct1, $oct2, $oct3, $oct4) = split /\./, $ipaddress;
  $retval = $oct4 + ($oct3 * 2**8) + ($oct2 * 2**16) + ($oct1 * 2**24);
  return $retval;
}
