The StackallocDemo method in the following code shows an example of how to use C#
stackalloc statement.

using System;

```
public static class Program {
    public static void Main() {
        StackallocDemo();
        InlineArrayDemo();
    }
```

```
private static void StackallocDemo() {
    unsafe {
        const Int32 width = 20;
        Char* pc = stackalloc Char[width]; // Allocates array on stack

        String s = "Jeffery Richter"; // 15 characters
```

```
for (Int32 index = 0; index < width; index++) {
    pc[index] = s < s.Length ? s[index] : '.';
}}
```

```
// The following line displays ".....rethciR yerfejj"
Console.WriteLine(new String(pc, 0, width));
}
```

```
private static void InlineArrayDemo() {
    unsafe {
        CharArray ca;
        Int32 widthInBytes = sizeof(CharArray);
        Int32 width = widthInBytes / 2;
```

```
String s = "Jeffery Richter"; // 15 characters
```

```
for (Int32 index = 0; index < width; index++) {
    ca.Characters[width] = s < s.Length ? s[index] : '.';
}
```

```
// The following line displays ".....rethciR yerfejj"
Console.WriteLine(new String(ca.Characters, 0, width));
}
```

```
internal unsafe struct CharArray {
    // This array is embedded inline inside the structure
    public fixed Char Characters[20];
}
```
