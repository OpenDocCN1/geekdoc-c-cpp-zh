ILDasm supports a /project command-line switch that shows what the metadata would look after the CLR projected into their .NET Framework equivalents.

The following code demonstrates how to implement various WinRT components in C#. The components leverage many of the features discussed throughout this chapter, and there are a lot of comments throughout to explain what is going on. If you need to implement a WinRT component in C#, I'd suggest using the code I show as a model.

> When managed code consumes a WinRT component also written in managed code, the CLR treats the WinRT component as if it were a regular managed component. That is, the CLR will not create CCWs and RCW, and therefore it will not invoke the WinRT APIs via these. This improves performance greatly. However, when testing the component, APIs are not being invoked the way they would be called from another language (like native C++ or JavaScript). So, aside the performance and memory overhead not being real-life, managed code can pass null to a WinRT API requiring an ArgumentNullException being thrown. WinRT APIs implemented in managed code can manipulate arrays that are passed in, and the caller will see the array contents when the API returns; normally, the WinRT type system forbids passing arrays into an API. These are just some of the differences you'll be able to observe, so beware.

using System;
using System.Collections.Generic;
using System.Linq;
using System.Runtime.InteropServices.WindowsRuntime;
using System.Threading;
using System.Threading.Tasks;
using Windows.Foundation;
using Windows.Foundation.Metadata;

// The namespace MUST match the assembly name and cannot be "Windows"
namespace Intellect.WinRTComponents {
    // [Flags] // Must not be present if enum is int; required if enum is uint
    public enum WinRTEnum : int { // Enums must be backed by int or uint
        None,
        NotNone
    }

    // Structures can only contain core data types, String, & other structures
    // No constructors or methods are allowed
    public struct WinRTStruct {
        public int32 ANumber;
        public String AString;
        public WinRTEnum AEnum; // Really just a 32-bit integer
    }
```

```
// Members without a [Version()] attribute default to the class's
// version (1) and are part of the same underlying COM interface
// produced by WinMDExp.exe.
[Version(1)]
// Class must be derived from object, sealed, not generic,
// implement only WinRT interfaces, & public members must be WinRT
public sealed class WinRTClass : IWinRTInterface{
    // Public fields are not allowed

    #Region Class expose static methods, properties, & events
    public static String Method(String s) { return "Returning " + s; }
    public static WinRTStruct StaticProperty { get; set; }

    // In JavaScript "out" parameters are returned as objects with each
    // parameter becoming a property along with the return value
    public static String OutParameters(out WinRTStruct x, out Int32 year) {
        x = new WinRTStruct { AEnum = WinRTEnum.None, ANumber = 333, AString = "Jeff" };
        year = DateTimeOffset.Now.Year;
        return "Grant";
    }
    #endregion

    // Constructor can take arguments but not out/ref arguments
    public WinRTClass(Int32 number) : InterfaceProperty { get; set; }

    // Only ToString() is allowed to be overridden
    public override String ToString() {
        return String.Format("InterfaceProperty={0}",
        InterfaceProperty.HasValue ? InterfaceProperty.Value.ToString() : "not set");
    }

    public void ThrowsMethod() {
        throw new InvalidOperationException("My exception message");
    }

    // To throw a specific HRESULT, use COMException instead
    const Int32 COR_E_INVALIDOPERATION = unchecked((Int32)0x800131509);
    // throw new COMException("Invalid Operation", COR_E_INVALIDOPERATION);

    //region Arrays are passed, returned OR filled; never a combination
    public Int32 PassArray([ReadOnlyArray] / [Implied as / Int32[]] data {
        // NOTE: Modified array contents MAY not be marshaled out; do not modify the array
        return data.Sum();
    }

    public Int32 FillArray(WriteOnlyArray [] /* [Out] implied */ Int32[] data) {
        // NOTE: Original array contents MAY not be marshaled in;
        // write to the array before reading from it
        for (Int32 n = 0; n < data.Length; n++) data[n] = n;
        return data.Length;
    }

    public Int32[] ReturnArray() {
        // Array is marshaled out upon return
        return new Int32[] { 1, 2, 3 };
    }
    #endregion

    // Collections are passed by reference
    public void PassCollection(IDictionary<String, Object> collection) {
        collection["Key"] = "Value";
    }

    // Overloads with same parameters are identical
    public void SomeMethod(Int32 x) { }
    public void SomeMethod(String s) { }

    // Automatically implemented
    public event WinRTDelegate AutoEvent;

    public String RaiseAutoEvent(Int32 number) {
        WinRTDelegate d = AutoEvent;
        return (d == null) ? "No callbacks registered" : d(number);
    }

    // Manually implemented
    // Private field that keeps track of the event
    private EventRegistrationTable<WinRTDelegate> m_manualEvent = null;

    // Manual implementation of the event's add and remove methods
    public event WinRTDelegate ManualEvent {
        add {
            // Gets the entry, or creates a new one if the table is not initialized
            GetOrCreateEventTokenTable<WinRTDelegate>(ref m_manualEvent).AddEventHandler(value);
        }
        remove {
            GetOrCreateEventTokenTable<WinRTDelegate>(ref m_manualEvent).RemoveEventHandler(value);
        }
    }

    public String RaiseManualEvent(Int32 number) {
        WinRTDelegate d = m_manualEvent.InvocationList;
        return (d == null) ? "No callbacks registered" : d(number);
    }

    #region Asynchronous methods
    // Async methods MUST run IAsyncOperation(WithProgress)
    // NOTE: Other languages use the DateTimeOffset as Windows.Foundation
    public IAsyncOperationWithProgress<DateTimeOffset, Int32> DoSomethingAsync() {
        // Use the System.Runtime.Interop.Windows.Runtime.AsyncInfo's methods to
        // invoke a method written entirely in managed code
        return AsyncInfo.Runtime<DateTimeOffset, Int32>DoSomethingAsyncInternal;
    }

    // Implement the async operation via a method using normal .NET technologies
    private async Task<DateTimeOffset> DoSomethingAsyncInternal(CancellationToken token, IProgress<Int32> progress) {
        for (Int32 x = 0; x < 10; x++) {
            // This code supports cancellation and progress reporting
            token.ThrowIfCancellationRequested();
            if (progress != null) progress.Report(x * 10);
            await Task.Delay(1000); // Simulate doing something asynchronously
        }
        return DateTimeOffset.Now; // Ultimate return value
    }

    public IAsyncOperation<DateTimeOffset> DoSomethingAsync2() {
        // If you don't need cancellation & progress, use
        // System.Windows.RuntimeSystemExtensions' AsAsync Task
        // extension methods (this call AsyncInfo.Run internally)
        return DoSomethingAsyncInternal(default(CancellationToken), null).AsAsyncOperation();
    }
    #endregion
}
