public static Task Log(Task task, String tag = null, String callerName = null, String callerPath = null, Int32 callerLineNumber = -1) {
    if (LogLevel == LogLevel.None) return task;
    var logEntry = new LogEntry {
        Task = task,
        LogTime = DateTime.Now,
        Tag = tag,
        CallerName = callerName,
        CallerPath = callerPath,
        CallerLineNumber = callerLineNumber
    };
    s_log[task] = logEntry;
    task.ContinueWith(t => { LogEntry entry; s_log.Remove(task, out entry); }, TaskContinuationOptions.ExecuteSynchronously);
    return task;
}
