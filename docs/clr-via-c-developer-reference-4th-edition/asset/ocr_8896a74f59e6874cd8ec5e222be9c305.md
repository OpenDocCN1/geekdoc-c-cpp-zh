To drive home the performance benefits of generics, I wrote a program that tests the performance of the generic List algorithm against the non-generic ArrayList algorithm. In fact, I tested the performance of these two algorithms using both value type objects and reference type objects. Here is the program itself.

```
using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
```

```
public static class Program {
    public static void Main() {
        ValueTypePerf();
        ReferenceTypePerf();
    }

    private static void ValueTypePerf() {
        const Int32 count = 1000000000;

        using (new OperationTimer("List<Int32>")) {
            List<Int32> list = new List<Int32>();
            for (Int32 n = 0; n < count; n++) {
                list.Add(n); // No boxing
                Int32 x = list[n]; // No unboxing
            }
            list = null; // Make sure this gets garbage collected
        }

        using (new OperationTimer("ArrayList of Int32")) {
            ArrayList list = new ArrayList();
            for (Int32 n = 0; n < count; n++) {
                list.Add(n); // Boxing
                Int32 x = (Int32)list[n]; // Unboxing
            }
            list = null; // Make sure this gets garbage collected
        }
    }

    private static void ReferenceTypePerf() {
        const Int32 count = 1000000000;

        using (new OperationTimer("List<String>")) {
            List<String> list = new List<String>();
            for (Int32 n = 0; n < count; n++) {
                list.Add("X"); // Reference copy
                String x = list[n]; // Reference copy
            }
            list = null; // Make sure this gets garbage collected
        }

        using (new OperationTimer("ArrayList of String")) {
            ArrayList list = new ArrayList();
            for (Int32 n = 0; n < count; n++) {
                list.Add("X"); // Reference copy
                String x = (String)list[n]; // Cast check & reference copy
            }
        }
    }
}
```
