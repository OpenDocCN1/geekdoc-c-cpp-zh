drawing. Unlike MouseMove events, which occur continuously as the user moves the\mouse, the system generates a MouseDown event only when a mouse button is pressed
gates a MouseUp event only when a mouse button is released.

// Fig. 14.38: PainterForm.cs
// Using the mouse to draw on a Form.
using System;
using System.Drawing;
using System.Windows.Forms;
nnamespace Painter
{
    // creates a Form that's a drawing surface
    public partial class PainterForm : Form
    {
        bool ShouldPaint() { get; set; } = false; // whether to paint

        // default constructor
        public PainterForm()
        {
            InitializeComponent();
        }

        // should paint when mouse button is pressed down
        private void PainterForm_MouseDown(object sender, MouseEventArgs args)
        {
            // indicate that user is drawing the mouse
            ShouldPaint = true;
        }

        // stopping when mouse button is released
        private void PainterForm_MouseUp(object sender, MouseEventArgs args)
        {
            // indicate that user released the mouse button
            ShouldPaint = false;
        }

        // draw circle whenever mouse moves with its button down
        private void PainterForm_MouseMove(object sender, MouseEventArgs args)
        {
            if (ShouldPaint) // check if mouse button is being pressed
            {
                // draw a circle where the mouse pointer is present
                using (Graphics graphics = CreateGraphics())
                {
                    graphics.FillEllipse(new SolidBrush(Color.BlueViolet), e.X, e.Y, 4, 4);
                }
            }
        }
    }
