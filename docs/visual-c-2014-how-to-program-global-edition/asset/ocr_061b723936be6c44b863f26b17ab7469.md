# 17.3 Creating a Sequential-Access Text File 745

```
namespace CreateFile {
    public partial class CreateFileForm : BankUIForm {
        private StreamWriter fileWriter; // writes data to text file

        // parameter constructor
        public CreateFile() {
            Initialize();
        }

        // event handler for Save Button
        private void saveButton_Click(object sender, EventArgs e) {
            // create show dialog box enabling use to save file
            DialogResult result; // result of SaveFileDialog
            string fileName; // name of file containing data
            using (var chooser = new SaveFileDialog()) {
                chooser.CheckFileExists = false; // let user create file
                result = chooser.ShowDialog();
                fileName = chooser.Name; // name to save data
            }
            // ensure that user clicked "OK"
            if (result == DialogResult.Ok) {
                MessageBox.Show("Invalid File Name", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            } else {
                // save via FileStream
                try {
                    // open file with write access
                    var output = new FileStream(fileName, FileMode.OpenOrCreate, FileAccess.Write);
                    // sets file to where data is written
                    fileWriter = new StreamWriter(output);

                    // disable Save button and enable Enter button.
                    saveButton.Enabled = false;
                    enterButton.Enabled = true;
                } catch (IOException) {
                }
            }
        }
    }
}
```
