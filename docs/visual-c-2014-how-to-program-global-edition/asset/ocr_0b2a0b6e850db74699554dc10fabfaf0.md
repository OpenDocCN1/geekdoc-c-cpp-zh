            // check whether specified path exists
            if (!string.IsNullOrEmpty(TextBox.Text) &&
                Directory.Exists(TextBox.Text))
            {
                // show error if user does not specify valid directory
                MessageBox.Show("Invalid Directory", "Error",
                    MessageBoxButtons.Ok, MessageBoxIcon.Error);
            }
            else
            {
                // directory to search; if not specified use current directory
                string currentDirectory =
                    !string.IsNullOrEmpty(TextBox.Text) ? 
                    TextBox.Text : Directory.GetCurrentDirectory();

                TextBox.Text = currentDirectory; // show directory

                TextBox.Clear();
