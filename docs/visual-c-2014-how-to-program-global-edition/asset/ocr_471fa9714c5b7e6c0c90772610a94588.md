// displays results and time in milliseconds

```
static void DisplayResults(
    int min, int max, double average, double time)
{
    Console.WriteLine("Min: {min}\nMax: {max}\n" + 
    "$Average: {average:F}\nTotal time in milliseconds: {time:F}");
}
```
