// Fig. 19.14: StackInheritanceTest.cs
// Testing class StackInheritance.
using System;
using StackInheritanceLibrary;
using LinkedLibrary;

// demonstrate functionality of StackInheritance
class StackInheritanceTest
{
    static void Main()
    {
        StackInheritance stack = new StackInheritance();

        // create objects to store in the stack
        bool aBoolean = true;
        char aInchar = '$';
        int aInteger = 34567;
        string aString = "hello";

        // use method to add items to stack
        stack.Push(aBoolean);
stack.Display();
stack.Push(aInchar);
stack.Display();
stack.Push(aInteger);
stack.Display();
stack.Push(aString);
stack.Display();

        // remove items from stack
try
        {
            object removedObject = stack.Pop();
            Console.WriteLine("{removedObject} popped");
            stack.Display();
        }
        catch (EmptyListException emptyList)
        {
            // if exception occurs, write stack trace
            Console.WriteLine(emptyList.StackTrace);
        }
}
