    private void GetInformation(string fileName)

        outputTextBox.Clear();
        // output if file or directory exists
        outputTextBox.Append("{fileName} exists\n");
        // output when file or directory was created
        outputTextBox.Append("{Created: {File.CreationTime(fileName)}\n" + 
            Environment.NewLine);
        // output when file or directory was last modified
        outputTextBox.Append("{Last modified: {File.GetLastTime(fileName)}\n" + 
            Environment.NewLine);
        // output when file or directory was last accessed
        outputTextBox.Append("{Last accessed: {File.GetLastAccessTime(fileName)}\n" +
