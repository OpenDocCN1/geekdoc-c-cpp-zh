```csharp
// 图 7.17: FactorialTest.cs
// 递归阶乘方法。
class FactorialTest {
    static void Main() {
        // 计算 0 到 10 的阶乘
        for (long counter = 0; counter <= 10; ++counter) {
            Console.WriteLine($"{counter}! = {Factorial(counter)}");
        }
        // 方法 Factorial 的递归声明
    }
    static long Factorial(long number) {
        // 基础情况
        if (number <= 1) {
            return 1;
        } else {
            return number * Factorial(number - 1);
        }
    }
}
```
