public partial class MenuTestForm : Form {
    // constructor
    public MenuTestForm() {
        InitializeComponent();
    }
    // display MessageBox about ToolStripMenuItem is selected
    private void aboutToolStripMenuItem_Click(object sender, EventArgs e) {
        MessageBox.Show("This is an example\nof using menus.", "About", MessageBoxButtons.OK, MessageBox.Information);
    }
    // exit program When ToolStripMenuItem is selected
    private void exitToolStripMenuItem_Click(object sender, EventArgs e) {
        Application.Exit();
    }
    // reset checkmarks for Color ToolStripMenuItem
    private void ClearColor() {
        // clear all checkmarks
        blackToolMenuItem.Checked = false;
        blueToolMenuItem.Checked = false;
        redToolMenuItem.Checked = false;
        greenToolMenuItem.Checked = false;
    }
    // update menu state and color display black
    private void blackToolMenuItem_Click(object sender, EventArgs e) {
        // reset checkmarks for Color ToolStripMenuItem
        ClearColor();
        // set color to Black
        displayLabel.ForeColor = Color.Black;
        blackToolMenuItem.Checked = true;
    }
    // update menu state and color display blue
    private void blueToolMenuItem_Click(object sender, EventArgs e) {
        // reset checkmarks for Color ToolStripMenuItem
        ClearColor();
        // set color to Blue
        displayLabel.ForeColor = Color.Blue;
    }
}
}```
