public void Display()
{
    if (IsEmpty())
    {
        Console.WriteLine($"Empty {name}");
    }
    else
    {
        Console.Write($"{The {name} is: ");
        ListNode current = firstNode;
        // output current node data not at end of list
        while (current != null)
        {
            Console.Write($"{current.Data}");
            current = current.Next;
        }
        Console.WriteLine("\n");
    }
}
