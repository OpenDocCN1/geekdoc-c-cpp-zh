            // argument ensures that if the thread gets the processor

            m_cv.wait(dataLock, [&]() {return m_occupied;});

        m_occupied = false;

            "Consumer reads "s + std::to_string(m_buffer),
            m_buffer, m_occupied, std::string(34, '-'));

    m_cv.notify_one();

}
