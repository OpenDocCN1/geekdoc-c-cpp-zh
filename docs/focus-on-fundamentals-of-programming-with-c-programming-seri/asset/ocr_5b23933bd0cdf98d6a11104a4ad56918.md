# Example: Output the contents of a file to standard output.

```
// file2.c

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    FILE *fp;

    fopen_s(&fp, "infile.txt", "r");
    if (fp == NULL) {
        printf("File could not be opened.\n");
        exit(1);
    }

    while((c = getc(fp)) != EOF) {
        putchar(c);
    }

    fclose(fp);
    return 0;
}
```
