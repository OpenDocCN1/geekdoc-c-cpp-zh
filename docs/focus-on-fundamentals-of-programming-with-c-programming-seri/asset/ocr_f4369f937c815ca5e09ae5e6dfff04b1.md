```
// list.h

#ifndef LIST_H

    int info;
    struct node *next;

NODEPTR getNode(void);
```

Note the placement of our NODEPTR struct definition and function prototypes in the header file. To prevent the compiler from looking at this header file more than one time we use a preprocessor conditional compilation command to tell the compiler to only bring this code into the project once:

```
#ifndef LIST_H
#define __LIST_H
```
