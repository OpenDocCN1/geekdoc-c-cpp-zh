See if you can figure out the output to the following program:

```
// doublep.c

#include <stdio.h>

int main(void);
void foo(char *p);
void foo2(char **q);

int main(void) {
    char *p = "Hello";
    char *q = "Bye";

    printf("Function: main() ----> p = %s\n", p);
    foo(p);
    printf("Function: main() ----> p = %s\n", p);
    printf("Function: main() ----> q = %s\n", q);
    foo2(&q);
    printf("Function: main() ----> q = %s\n", q);
    return 0;
}

void foo(char *pp) {
    ++pp;
    printf("Function: foo() ----> pp = %s\n", pp);
}

void foo2(char **qq) {
    ++*qq;
    printf("Function: foo2() ----> *qq = %s\n", *qq);
}
```

The output produced is as follows:

```
Function: main() ----> p = Hello
Function: foo() ----> pp = ello
Function: main() ----> p = Hello
Function: main() ----> q = Bye
Function: foo2() ----> *qq = ye
Function: main() ----> q = ye
```
