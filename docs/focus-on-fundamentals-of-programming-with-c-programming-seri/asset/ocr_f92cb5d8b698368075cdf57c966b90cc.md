#include <stdio.h>
#define MIN 1
#define MAX 15
int main(void) {
 int i;
 printf("%10s %10s %10s\n", "Value", "Square", "Cube");
 printf("%10s %10s %10s\n", "-----", "------", "-----");
 for (i = MIN; i <= MAX; ++i) {
   printf("%10d %10d %10d\n", i, i * i, i * i * i);
 }
 return 0;
}
