See if you can follow the pointers in the following program.

```
// dynamic.c

#include <stdio.h>
#include <alloc.h>

int main(void) {
    int *p, *q;

    p = (int *) malloc(sizeof(int));
    *p = 3;
    q = (int *) malloc(sizeof(int));
    *q = 5;
    printf("*p = %d\t*q = %d\n", *p, *q);
    free(p);
    p = q;
    printf("*p = %d\t*q = %d\n", *p, *q);
    *q = 7;
    printf("*p = %d\t*q = %d\n", *p, *q);
    q = (int *) malloc(sizeof(int));
    *q = 9;
    printf("*p = %d\t*q = %d\n", *p, *q);
    *p = *q;
    printf("*p = %d\t*q = %d\n", *p, *q);
    free(p);
    p = q;
    printf("*p = %d\t*q = %d\n", *p, *q);
    free(p);
    printf("*p = %d\t*q = %d\n", *p, *q);
    return 0;
}
```

The output to the above program is:

```
*p = 3      *q = 5
*p = 5      *q = 5
*p = 7      *q = 7
*p = 7      *q = 9
*p = 9      *q = 9
*p = 9      *q = 9
```
