# Example: Output contents of a file to standard output requesting a filename from user.

```
// file3.c

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    char fileName[32];
    FILE *fp;

    printf("Enter filename: ");
    gets(fileName);
    fopen_s(&fp, fileName, "r");
    if (fp == null) {
        printf("File %s could not be opened.\n", fileName);
        exit(1);
    }

    while((c = getc(fp)) != EOF) {
        putchar(c);
    }
    fclose(fp);
    return 0;
}
```
