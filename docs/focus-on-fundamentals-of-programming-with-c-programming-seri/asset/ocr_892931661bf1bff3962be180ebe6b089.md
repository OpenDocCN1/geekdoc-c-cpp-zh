test.c
#include <header.h>
#define TOTAL x + y
#define MESSAGE "ON A CLEAR DAY YOU CAN C FOREVER"
#define FORMAT1 "%s\n"
#define FORMAT2 "%d %d\n"
int main(void) {
    int x, y, z, i, sum;
    printf(FORMAT1, MESSAGE);
    printf(NEWLINE);
    sum = ZERO;
    for (i = MIN; i <= MAX; ++i) {
        sum += i;
    }
    x = MAX;
    y = MIN;
    z = TOTAL;
    printf(FORMAT2, sum, z);
    return 0;
}
