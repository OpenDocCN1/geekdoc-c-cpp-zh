```
// trace2.c
#include <stdio.h>
int main(void) {
    int a, b;
    a = 1;
    b = -1;
    printf("a = %d   b = %d\n", a, b);
    a = a + b;
    b = a - b;
    a = a - b;
    printf("a = %d   b = %d\n", a, b);
    return 0;
}
```

Here's the output that's produced:

```
a = 1   b = -1
a = -1  b = 1
```
