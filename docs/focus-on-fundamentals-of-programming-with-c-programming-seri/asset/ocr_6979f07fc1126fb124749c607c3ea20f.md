An alternate approach to dynamically create an array of items on the heap is as follows:

```
// darray2.c
#include <stdio.h>
#include <alloc.h>
typedef struct student {
    char name[30];
    int age;
    float gpa;
} STUDENT;
void initialize(STUDENT *pstud);
int main(void) {
    int i;
    STUDENT *pstud;
    pstud = (STUDENT *) malloc(5 * sizeof(struct student));
    initialize(pstud);
    for (i = 0; i < 5; i++) {
        printf("%s %d %4.2f\n", pstud[i].name, pstud[i].age,
               stud[i].gpa);
    }
    free(pstud);
    return 0;
}
```
