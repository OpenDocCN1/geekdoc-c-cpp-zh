# We can also count newlines:

```
// count.c
#include <stdio.h>
int main(void) {
    int c, count = 0;
    while ((c = getchar()) != EOF) {
        if (c == '\n') {
            ++count;
        }
    }
    printf("%d\n", count);
    return 0;
}
```

# Here's a program to map a program's input to lower case:

```
// map1.c
#include <stdio.h>
#include <ctype.h>
int main(void) {
    int c;
    while ((c = getchar()) != EOF) {
        printf("%c", isupper(c) ? tolower(c) : c);
    }
    return 0;
}
```

# Here's an expression that causes blank characters in the input stream to be skipped:

```
// skipBlanks.c
while ((c = getchar()) == ' ') {
    ;
}
```
