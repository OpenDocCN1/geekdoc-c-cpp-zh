In the above program, note that we use the direct component selector (.) operator rather than the indirect component selector (->) operator to access the fields of the struct. This is due to the fact that we do not have an array of pointers to structs this time, but rather the structs themselves are contained within the array.

Our final example also creates a dynamic array on the heap. This time, however, not only is an array of pointers created on the heap, but the student structs that the pointers point to are also created on the heap.

// darray3.c

#include <stdio.h>
#include <alloc.h>

typedef struct student {
    char name[30];
    int age;
    float gpa;
} STUDENT;

void initialize(STUDENT **pstud);

int main(void) {
    int i;
    STUDENT **pstud;

    pstud = (STUDENT **) malloc(5 * sizeof(STUDENT *));
    for (i = 0; i < 5; i++) {
        pstud[i] = (STUDENT *) malloc(sizeof(struct student));
        initialize(pstud);
        for (i = 0; i < 5; i++) {
            printf("%s %d %4.2f\n", pstud[i]->name, pstud[i]->age,
                   pstud[i]->gpa);
        }
    }

    for (i = 0; i < 5; i++) {
        free(pstud[i]);
    }
    free(pstud);
    return 0;
}
