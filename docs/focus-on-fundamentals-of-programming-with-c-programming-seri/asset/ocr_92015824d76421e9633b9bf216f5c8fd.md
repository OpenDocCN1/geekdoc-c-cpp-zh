```
// xor3.c

#include <stdio.h>

int main(void) {
    int x = 0x1234;       // 0001 0010 0011 0100
    int y = 0xABCD;       // 1010 1011 1100 1101

    x ^= y;
    y ^= x;
    x ^= y;

    printf("x = 0x%x\n", x);
    printf("y = 0x%x\n", y);
    return 0;
}
```

The output produced is:

```
x = 0xABCD
y = 0x1234
```
