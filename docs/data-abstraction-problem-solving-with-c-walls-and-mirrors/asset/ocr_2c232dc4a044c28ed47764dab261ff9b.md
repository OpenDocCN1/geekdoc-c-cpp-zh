HashedDictionary's definition of remove follows:

```
template <class HashedKey, class ItemType>
b HashedDictionary<HashedKey, ItemType>::remove(const KeyType& searchKey) {
    bool itemFound = false;
    // Compute the hashed index into the array
    int itemHashIndex = getHashIndex(searchKey);
    if (hashTable[itemHashIndex] != nullptr) {
        // Special case - first node has target
        if (searchKey == hashTable[itemHashIndex]) {
            HashedEntry<HashedKey, ItemType>* entryToRemove = hashTable[itemHashIndex];
            hashTable[itemHashIndex] = hashTable[itemHashIndex]->getNext();
            delete entryToRemove; // For safety
            itemFound = true;
        }
        else // Search the rest of the chain
            HashedEntry<HashedKey, ItemType>* prevPtr = hashTable[itemHashIndex];
            HashedEntry<HashedKey, ItemType>* curPtr = prevPtr->getNext();
            while ((curPtr != nullptr) && !itemFound) {
                // Find item in chain to remove that
                if (searchKey == curPtr->getKey()) {
                    prevPtr->setNext(curPtr->getNext());
                    delete curPtr;
                    curPtr = nullptr; // For safety
                    itemFound = true;
                }
                else // Look at next entry in chain
                    prevPtr = curPtr;
                    curPtr = curPtr->getNext();
                } // end if
            } // end while
        } // end if
    }
    return itemFound;
} // remove
```
