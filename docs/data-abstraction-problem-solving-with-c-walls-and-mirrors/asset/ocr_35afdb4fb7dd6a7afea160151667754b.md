# LISTING 9-2 The header file for the class LinkedList

```
/** ADT list: Link-based implementation.
 * @file LinkedList.h */

#ifndef _LINKED_LIST
#define _LINKED_LIST

#include "ListInterface.h"
#include "Node.h"
#include "PrecondViolatedExec.h"

template<class ItemType>
class LinkedList : public ListInterface<ItemType>\{
private:
    Node<ItemType>* headPtr; // Pointer to first node in the chain
    // (contains the first entry in the list)
    // Current count of list items
    int itemCount;
    // Locate a specified node in a linked list.
    // @pre position is the number of the desired node;
    // position >= 1 and position <= itemCount;
    // @post The node is found and a pointer to it is returned.
    // @param position The number of the node to locate.
    // @return A pointer to the node at the given position.
    Node<ItemType>* getNode(int position) const;
public:
    LinkedList();
    LinkedList(const LinkedList<ItemType>&& aList);
    virtual ~LinkedList();
    bool isEmpty() const;
    int getLength() const;
    bool insert(int position, const ItemType& newEntry);
    bool remove(int position);
    void clear();
    /** @throw PrecondViolatedExec if position < 1 or 
     *      position > getLength(). */
    ItemType getEntry(int position) const throw(PrecondViolatedExec);
    /** @throw PrecondViolatedExec if position < 1 or 
     *      position > getLength(). */
    void setEntry(int position, const ItemType& newEntry)
        throw(PrecondViolatedExec);

}; // end LinkedList

#include "LinkedList.cpp"
#endif
```
