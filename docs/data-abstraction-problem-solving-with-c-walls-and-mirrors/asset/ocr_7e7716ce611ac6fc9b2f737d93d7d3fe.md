# LISTING 3-2 A program that tests the core methods of the class ArrayBag

```
#include <iostream>
#include <string>
#include "ArrayBag.h"

using namespace std;

void displayBag(ArrayBag<string>& bag)
{
    cout << "The bag contains " << bag.getCurrentSize()
    cout << " items:" << endl;
    vector<string> bagItems = bag.toVector();
    int numberOfEntries = (int)bagItems.size();
    for (int i = 0; i < numberOfEntries; i++)
    {
        cout << bagItems[i] << " ";
    }
    cout << endl << endl;
} // displayBag

void tester(ArrayBag<string>& bag)
{
    cout << "isEmpty returns " << bag.isEmpty()
    cout << "; should be 1 (true)" << endl;
    displayBag(bag);

    string items[] = {"one", "two", "three", "four", "five", "one"};
    cout << "Add 6 items to the bag:" << endl;
    for (int i = 0; i < 6; i++)
    {
        bag.add(items[i]);
    }
    // end for
    displayBag(bag);

    cout << "isEmpty returns " << bag.isEmpty()
    cout << "; should be 0 (false)" << endl;
    cout << "getCurrentSize returns " << bag.getCurrentSize()
    cout << "; should be 6" << endl;

    cout << "Try to add another entry: " << \"extra\" << returns "
    bag.add("extra")" << endl;
} // end tester

int main()
{
    ArrayBag<string> bag;
    cout << "Testing the Array-Based Bag:" << endl;
    cout << "The initial bag is empty." << endl;
    tester(bag);
    cout << "All done!" << endl;

    return 0;
} // end main
```
