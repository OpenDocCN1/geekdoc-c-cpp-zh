```

The C++ program for this algorithm is shown here.

# project_13_1_2c

```
```cpp
#include <iostream>
using namespace std;
int main() {
    int digit1, digit2, digit3, digit4, number, r, sum;

    cout << "Enter a four-digit integer: ";
    cin >> number;

    digit4 = number % 10;
    r = (int) (number / 10);

    digit3 = r % 10;
    r = (int) (r / 10);

    digit2 = r % 10;
    digit1 = (int) (r / 10);

    sum = digit1 + digit2 + digit3 + digit4;
    cout << sum;
    return 0;
}
```
