#include <stdio.h>
#include <string.h>

struct Address
{
    char street[50];
    char city[50];
};

struct Employee
{
    int id;
    char name[20];
    char title[20];
    struct Address address;
} emp;

int main()
{
    struct Employee emp =
    {1000, "John Smith", "Developer", "123 Main Street", "Chicago"};

    printf("Emp Id: %d \n",  emp.id);
    printf("Name:  %s \n",  emp.name);
    printf("Title: %s \n\n", emp.title);
