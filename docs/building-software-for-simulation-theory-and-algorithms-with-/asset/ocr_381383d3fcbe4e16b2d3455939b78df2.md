The implementations of the exclusive-or and memory models follow directly from their definitions. Note that the *delta* methods of both classes check that the input bag contains the proper number of elements. This runtime test is helpful for debugging and, even when inactive (i.e., when compiling with NDEBUG defined), is a reminder that the model accepts a restricted range of input. The implementations of the exclusive-or and memory models are listed below.

# **Exclusive-Or and Memory Models**

```
    #ifndef __xor_memory_models_h_
    #define __xor_memory_models_h_

    #include "adts.h"
    #include <cassert>

    // Structure used for input from and output to the models
    struct port_value_t
    {
        char port; // Port on which the I/O appears
        bool bit;  // Bit appearing at the port
        bool operator<(const port_value_t& other) const { // For the STL
            return bit < other.bit;
        }
    };

    // Definition of the exclusive-or model
    class Xor: public adts::Atomic<port_value_t>
    {
    public:
        Xor():adts::Atomic<port_value_t>(),s(false){} // State is 0
        // State transition function
        void delta(const adts::Bag<port_value_t>& xb)
        {
            // Make sure the input bag is acceptable
            assert(xb.size() == 2);
            // Process the two input events
            adts::Bag<port_value_t>::iterator iter = xb.begin();
            bool x1 = (*iter).bit, x2 = (*(++iter)).bit;
            s = (x1 && !x2) || (x2 && !x1);
        }
        // Output function
        void output_func(adts::Bag<port_value_t>& yb)
        {
            port_value_t y;
            y.bit = s;
            yb.insert(y);
        }
    };
```
