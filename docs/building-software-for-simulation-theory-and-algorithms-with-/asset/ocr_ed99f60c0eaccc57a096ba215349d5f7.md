#include "adtts.h"
#include "n1.h"
#include <iostream>
using namespace std;

class MachineListener: public adtts::EventListener<port_value_t>
{
  public:
    MachineListener(adtts::Dtss<port_value_t>* top):top(top){}
    void outputEvent(adtts::Event<port_value_t> x, unsigned int t)
    {
        if (x.model == top) cout << x.value.bit << " ";
    }
    void stateChange(adtts::Atomic<port_value_t>*,unsigned int){}
  private:
    adtts::Dtss<port_value_t>* top;
};

int main()
{
    // Create the model
    N1* m = new N1();
    // Setup the simulator and output listener
    MachineListener* l = new MachineListener(m);
    adtts::Simulator<port_value_t>* sim =
        new adtts::Simulator<port_value_t>(m);
    sim->addEventListener(l);
