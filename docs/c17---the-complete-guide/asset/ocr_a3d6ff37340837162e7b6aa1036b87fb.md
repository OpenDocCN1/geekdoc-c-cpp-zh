```
lib/circle.hpp
#ifndeF CIRCLE_HPP
#define CIRCLE_HPP

#include "coord.hpp"
#include <iostream>

class Circle {
    private:
        Coord center;
        int rad;
    public:
        Circle (Coord c, int r)
            : center{c}, rad{r} {
        }

        void move(const Coord& c) {
            center += c;
        }

        void draw() const {
            std::cout << "circle at " << center
                      << " with radius " << rad << '\n';
        }
};

#endif
```

We can now put elements of these types into a collection by creating corresponding objects and passing them by value into a container:

```
std::vector<GeoObj> createFigure()
{
    std::vector<GeoObj> f;
    f.push_back(Line{Coord{1,2},Coord{3,4}});
    f.push_back(Circle{Coord{5,5},2});
    f.push_back(Rectangle{Coord{3,3},Coord{6,4}});
    return f;
}
```

This code would not be possible with polymorphism using inheritance because then the types would have to have GeoObj as a common base class and we would need a vector of pointers of GeoObj elements, and, due to using pointers, we would have to create the objects with new so that we have to track when to call delete or use smart pointers (unique_ptr or shared_ptr).
