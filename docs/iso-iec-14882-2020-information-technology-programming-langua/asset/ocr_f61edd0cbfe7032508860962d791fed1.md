```
struct link {
    link* next;
};

link* first;

void h(link*& p) { // p is a reference to pointer
    p->next = first;
    first = p;
    p = 0;
}

void k() {
    link* q = new link;
    h(q);
}
```

declares p to be a reference to a pointer to link so h(q) will leave q with the value zero. See also 9.4.4. — end example
