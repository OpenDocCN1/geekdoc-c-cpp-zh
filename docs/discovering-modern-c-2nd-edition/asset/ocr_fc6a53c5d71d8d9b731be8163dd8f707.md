struct uncopyable
{
    uncopyable(int v) : v{v} {}
    uncopyable(const uncopyable&) = delete;
    uncopyable(uncopyable&&) = default;

    uncopyable& operator+=(const uncopyable& rhs) &
    {
        v+= rhs.v;
        return *this;
    }

    operator int() const { return v; }
    int v;
};
