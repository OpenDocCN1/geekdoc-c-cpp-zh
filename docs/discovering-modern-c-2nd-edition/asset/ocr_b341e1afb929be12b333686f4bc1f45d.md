```
#include <iostream>
#include <limits>

using namespace std;

template <typename T>
inline void test(const T& x)
{
    cout << "x = " << x << " (";
    auto oldp = cout.precision(numeric_limits<T>::digits10 +
1);
    cout << x << ") " << endl;
    cout.precision(oldp);
}

int main()
{
    test(1.f/3.f);
    test(1./3.0);
    test(1./3.01);
}
```
