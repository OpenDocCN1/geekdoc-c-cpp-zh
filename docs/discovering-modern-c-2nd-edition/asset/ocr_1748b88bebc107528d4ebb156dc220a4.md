```
template <int N>
struct fsize_assign
{
    template <typename Target, typename Source>
    static void eval(Target& tar, const Source& src)
    {
        if constexpr (N > 0)
        fsize_assign<N-1>::eval(tar, src);
        tar[N]= src[N];
    }
};
```
