#include <exception>

struct io_error
    : std::runtime_error
{
    io_error(const string& message = "I/O error")
        : std::runtime_error(message) {}
};

struct cannot_open_file
    : io_error
{
    cannot_open_file(const string& fname)
        : io_error("File '" + fname + "' not found or no permission") {}
};
