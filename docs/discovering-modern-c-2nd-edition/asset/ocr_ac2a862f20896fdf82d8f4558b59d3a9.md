template <unsigned N, typename F, typename T>
class nth_derivative
{
    using prev_derivative= nth_derivative<N-1, F, T>;
public:
    nth_derivative(const F& f, const T& h)
    : h{h}, fp{f, h} {}

    T operator()(const T& x) const
    {
        return ( fp(x+h) - fp(x) ) / h;
    }

private:
    T h;
    prev_derivative fp;
};
};
