template <typename Matrix>
typename Matrix::value_type
frobenius_norm(const Matrix& A)
{
    using std::abs; using std::sqrt;
    using value_type = typename Matrix::value_type;
    value_type ss = 0;
    on_each_nonzero(A, [&ss](value_type x) { ss+= abs(x) * abs(x); });
    return sqrt(ss);
}
