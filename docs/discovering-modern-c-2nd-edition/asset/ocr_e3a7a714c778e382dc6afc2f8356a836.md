template <std::size_t N>
class array_lock_guard
{
public:
    array_lock_guard(std::array <std::mutex, N>& mutexes)
        : mutexes[mutexes]
    {
        // lock them all
    }

    ~array_lock_guard()
    {
        for (auto& m : mutexes)
            m.unlock();
    }

private:
    std::array<std::mutex, N>& mutexes;
};

template <std::size_t ...I>
void lock_all(std::index_sequence <I...>)

array_lock_guard(std::array <std::mutex, N>& mutexes)
