```
template <typename T>
concept Returnable=
    std::is_move_constructible_v<T>
    || std::is_copy_constructible_v<T>;

template <typename Iter, typename Value>
concept Summable=
    requires (Iter it, Value init)
    {
        requires std::input_iterator<Iter>;
        requires Returnable<Value>;
        init+= *it;
    };
```
