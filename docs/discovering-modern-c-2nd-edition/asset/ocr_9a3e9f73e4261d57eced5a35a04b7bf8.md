template <typename Iter, typename Distance>
inline void advance_aux(Iter& i, Distance n, input_iterator_tag)

    for (; n > 0; --n)

template <typename Iter, typename Distance>
inline void advance_aux(Iter& i, Distance n,
    bidirectional_iterator_tag)
{
    if (n >= 0)
        for (; n > 0; --n) ++i;
    else
        for (; n < 0; ++n) --i;
}
