template <int Row, int Col>
struct fsize_mat_vec_mult_aux
{
    template <typename Matrix, typename VecIn, typename Tmp>
    static void eval(const Matrix& A, const VecIn& v, Tmp& w)
    {
        if constexpr (Col == 0) {
            w = A[Row][0] * v[0];
        } else {
            fsize_mat_vec_mult_aux<Row, Col-1>::eval(A, v, w);
            w += A[Row][Col] * v[Col];
        }
    }
};
};
