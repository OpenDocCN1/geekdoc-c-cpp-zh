class rational
{
public:
    std::weak_ordering operator<=>(const rational& r2) const {
        using wo = std::weak_ordering;
        int cmp = p * ll(r2.q) - r2.p * ll(q);
        if (cmp == 0)
            return wo::equivalent;
        else if (cmp < 0)
            return wo::less;
        else
            return wo::greater;
    }
}
