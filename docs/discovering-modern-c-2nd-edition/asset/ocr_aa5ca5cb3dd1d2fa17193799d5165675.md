double one_norm(int size, double *vp)
{
    double sum= 0;
    for (int i = 0; i < size; i++)
        sum+= fabs(vp[i]);

    return sum;
}

double dot(int size, double *vp, double *wp)
{
    double sum= 0;
    for (int i = 0; i < size; i++)
        sum+= vp[i] * wp[i];
