namespace views {
struct enumerate_t
{
    template <typename Range>
    enumerate_view<Range> operator()(Range&& r) const noexcept
    {
        return enumerate_view<Range>{std::forward<Range>(r)};
    };
};
static constexpr enumerate_t enumerate;
}
