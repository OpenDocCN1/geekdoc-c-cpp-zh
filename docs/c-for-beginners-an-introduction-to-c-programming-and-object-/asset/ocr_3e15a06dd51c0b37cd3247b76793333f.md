using namespace std;

void swap_by_value (int, int);
void swap_by_reference (int &, int &);

void swap_by_value (int x, int y)
{
    int temp = x;
    x = y;
    y = temp;
    //Display x and y
    cout << endl << endl << "Inside swap_by_value" << endl
         << endl
         << "x: " << x << "    y: " << y << endl;
}
void swap_by_reference (int &x, int &y)
{
    //Swap x and y
    int temp = x;
    x = y;
    y = temp;

         << endl
