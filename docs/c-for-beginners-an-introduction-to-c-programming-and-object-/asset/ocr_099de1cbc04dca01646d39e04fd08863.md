using namespace std;

class Test
{
    int x, y;
public:
    void setData (int , int);
    void displayData ();
    //Operator overloading as follows:
    Test operator+(Test Right)
    {
        Test Result;
        Result.x = x + Right.x;
        Result.y = y + Right.y;
        return Result;
    }
};

//setData definition
void Test :: setData (int a , int b)
{
    x = a;
    y = b;
}

//displayData definition
void Test :: displayData ()
{
    cout << endl << endl
         << "x: " << x << " y: " << y ;
}

int main ()
{
    // create two objects of Test
    Test t1, t2;
    cout << endl << endl
         << "Enter values ...";
    //first values of t1 and t2
    t1.setData ( , );
    t2.setData ( , );
    //Display values of t1 and t2
    cout << endl << endl
         << "...";
    t1.displayData ();
    cout << endl << endl
         << "...";
    t2.displayData ();
}
