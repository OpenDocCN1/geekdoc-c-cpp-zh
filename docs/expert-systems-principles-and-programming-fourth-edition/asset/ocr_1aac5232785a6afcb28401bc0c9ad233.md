(defmethod check-input ((?question STRING)
                             (?value1 NUMBER)
                             (?value2 NUMBER))
  (printout t ?question " (" (float ?value1)
           "--" (float ?value2) ") ")
  (bind ?answer (read))
  (while (or (not (numberp ?answer))
             (< ?answer ?value1)
             (> ?answer ?value2))
    (printout t ?question " (" (float ?value1)
             "--" (float ?value2) ") ")
    (bind ?answer (read)))
  (return ?answer))
