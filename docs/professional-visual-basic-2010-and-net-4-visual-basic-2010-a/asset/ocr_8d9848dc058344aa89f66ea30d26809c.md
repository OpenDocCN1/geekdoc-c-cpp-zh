```
Protected Sub SubmitButton_Click(ByVal sender As Object, ByVal e As EventArgs)
    Handles SubmitButton.Click
    ResultsLabel.Text = String.Format("You entered Name: {0} and Email: {1}",
        NameTextBox.Text, EmailTextBox.Text)
End Sub
```
