    'retrieves the list of authors

Private Sub InitializeList()
    Me.AuthorsList.Items.Clear()

    Dim authors = From a In context.authors
                  Where a.state = "CA"
                  Order By a.city Select a
    For Each author In authors
        Me.AuthorsList.Items.Add(author)
    Next
End Sub
