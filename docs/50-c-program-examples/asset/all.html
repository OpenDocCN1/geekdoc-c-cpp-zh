24. Operator Overloading:

#include <iostream>

class Complex {

public:

double real;

double imag;

Complex(double r, double i) : real(r), imag(i) {}

Complex operator + (const Complex& other) {

return Complex(real + other.real, imag + other.imag);

}

void display() {

std::cout << real << " + " << imag << "i" << std::endl;

}

};

int main() {

Complex c1(2.0, 3.0);

Complex c2(1.0, 1.0);

Complex result = c1 + c2;

result.display();

return 0;

}

28. Write and Read Binary Data:

#include <iostream>

#include <fstream>

// Writing binary data

int data[] = {1, 2, 3, 4, 5};

std::ofstream outfile("data.bin", std::ios::binary);

if (outfile.is_open()) {

outfile.write(reinterpret_cast<char*>(data), sizeof(data));

outfile.close();

std::cout << "Binary data written successfully." << std::endl;

} else {

std::cerr << "Unable to open file." << std::endl;

}

// Reading binary data

int readData[5];

std::ifstream infile("data.bin", std::ios::binary);

if (infile.is_open()) {

infile.read(reinterpret_cast<char*>(readData), sizeof(readData));

infile.close();

for (int i = 0; i < 5; ++i) {

std::cout << readData[i] << " ";

}

std::cout << std::endl;

} else {

std::cerr << "Unable to open file." << std::endl;

}

return 0;

}

36. C++ Standard Library Algorithms:

#include <iostream>

#include <vector>

#include <algorithm>

int main() {

std::vector<int> numbers = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5};

// Sort the vector

std::sort(numbers.begin(), numbers.end());

// Print sorted vector

for (int n : numbers) {

std::cout << n << " ";

}

std::cout << std::endl;

// Find the position of 5 in the sorted vector

auto pos = std::lower_bound(numbers.begin(), numbers.end(), 5);

if (pos != numbers.end()) {

std::cout << "Found 5 at position " << (pos - numbers.begin()) << std::endl;

} else {

std::cout << "5 not found" << std::endl;

}

return 0;

}

50. C++ Custom Iterators:

#include <iostream>

#include <vector>

class MyIterator {

private:

int index;

std::vector<int>& data;

public:

MyIterator(std::vector<int>& arr) : index(0), data(arr) {}

bool hasNext() {

return index < data.size();

}

int next() {

return data[index++];

}

};

int main() {

std::vector<int> numbers = {1, 2, 3, 4, 5};

MyIterator iterator(numbers);

while (iterator.hasNext()) {

std::cout << iterator.next() << " ";

}

std::cout << std::endl;

return 0;

}
