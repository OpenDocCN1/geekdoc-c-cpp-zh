{
    int choice;
    printf("1.Add\n");
    printf("2.Subtract\n");
    printf("3.Multiply\n");
    printf("4.Divide\n");
    printf("5.Exit\n");
    printf("Enter your choice : ");
    scanf("%d",&choice);
    return choice;
}
float add(float a,int b)
{
    return a+b;
}
float sub(float a,int b)
{
    return a-b;
}
float mul(float a,int b)
{
    return a*b;
}
float div(float a,int b)
{
    return a/b;
}

This program could be written using a switch but writing it using function pointers increases efficiency. Function pointers are generally used in compilers, interpreters and database programs.
