```
int passmarks;
switch(s)
{
    case hindi:
    case english:
        passmarks=25;
        break;
    case maths:
        passmarks=40;
        break;
    case physics:
    case chemistry:
        passmarks=35;
        break;
    default:
        passmarks=33;
}
```

Suppose we need a function that returns number of days in a month. It can be written as-

```
enum month m;
int days;
if( m==Apr || m==Jun || m==Sep || m==Nov )
    days=30;
if( m==Jan || m==Mar || m==May || m==Jul || m==Aug || m==Oct || m==Dec)
    days=31;
if( m==Feb )
    days=28 ;
```
