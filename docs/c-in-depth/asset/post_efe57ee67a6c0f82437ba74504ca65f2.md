## Problem 8

Write a function that converts a string to an integer (string contains integer in decimal number system).

```c
/*P9.30*/
#include<stdio.h>
#include<string.h>
main()
{
    char str[20];
    printf("Enter a number : ");
    scanf("%s",str);
    printf("%d\n",str_to_i(str));
}
int str_to_i(char str[])
{
    int i,num=0;
    if(str[0]=='-')
        i=1;
    else
        i=0;
    while(i<strlen(str))
        num=num*10+(str[i++]-48);

    if(str[0]=='-')
        return -num;
    else
        return num;
}
```

void i_to_str(int num,char str[],int b)
{
    int i=0,temp,rem,j;
    while(num>0)
    {
        rem=num%b;
        num/=b;
        if(rem>9 && rem<16)
            str[i++]= rem-10+'A';
        else
            str[i++]=rem+'0';
    }
    str[i]='\0';
    for(i=0,j=strlen(str)-1;i<j;i++,j--) /*Reverse the string*/
    {
        temp=str[i];
        str[i]=str[j];
        str[j]=temp;
    }
}
void f_to_str(float num,char str[])

    int i,k;
    float f;
    i=num;
    i_to_str(i,str,10);
    str[strlen(str)+1]='\0';
    str[strlen(str)]='.';

    k=f*10000; /*we'll get the float value upto 4 decimal places*/
