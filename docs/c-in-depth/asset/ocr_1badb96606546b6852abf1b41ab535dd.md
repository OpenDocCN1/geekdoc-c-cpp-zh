```
addafter(int data, int pos)

struct node *tmp, *q;
int i;
q=start;
for(i=0;i<pos-1;i++)
{
    q=q->link;
    if(q==NULL)
    {
        printf("There are less than %d elements", pos);
        return;
    }
}
/*End of for*/
tmp=malloc(sizeof(struct node));
tmp->link=q->link;
tmp->info=data;
q->link=tmp;
/*End of addafter()*/
```

del(int data)

struct node *tmp,*q;
if(start->info==data)
{
    tmp=start;
    start=start->link; /*First element deleted*/
    free(tmp);
    return;
}
q=start;
while(q->link->link!=NULL)
{
    if(q->link->info==data)/*Element deleted in between*/
    {
        tmp=q->link;
        q->link=tmp->link;
        free(tmp);
        return;
    }
    q=q->link;
}
/*End of while */
