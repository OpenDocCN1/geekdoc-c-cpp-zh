/*P7.26 Program to print the spiral matrix*/
#include<stdio.h>
#define MAX 20
main()
{
    int n,i=0,j=0,l,u,num=1,arr[MAX][MAX]={0};
    printf("Enter value of n : ");
    scanf("%d",&n);
    l=0;u=n-1;
    for(num=1;num<=n*n;num++)
    {
        arr[i][j]=num;
        if(i==l&&j<u)
            j++;
        else if(j==u&&i<u)
            i++;
        else if(i==u&&j>l)
            j--;
        else if(j==l&&i>l)
            i--;
        if(arr[i+1][j]!=0)      /*If square already filled*/
        {
            l++;
            u--;
            i++;
            j++;
        }
    }
    /*End of for*/
    for(i=0;i<n;i++)
    {
        for(j=0;j<n;j++)
            printf("%4d",arr[i][j]);
        printf("\n");
    }
}
