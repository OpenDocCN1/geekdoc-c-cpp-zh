swapvar(int a,int b)
{
    int temp;
    temp=a,a=b,b=temp;
}
swaparr(int arr1[5],int arr2[5])
{
    int i,temp;
    for(i=0;i<5;i++)
    { temp=arr1[i], arr1[i]=arr2[i], arr2[i]=temp; }
}
