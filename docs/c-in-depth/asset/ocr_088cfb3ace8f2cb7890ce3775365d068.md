# 6. Cmpdate.c

int cmpdate(char *date1, char *date2)
Input:
- date1: First date in dd/mm/yyyy format
- date2: Second date in dd/mm/yyyy format
Return value:
- -1: date1 > date2
- 0: date1 == date2
- 1: date1 < date2
