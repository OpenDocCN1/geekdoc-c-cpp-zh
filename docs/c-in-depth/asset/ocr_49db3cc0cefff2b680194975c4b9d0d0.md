# Problem 2

Write a program to remove all comment lines from a syntax error free C source program.

```
/*P11.29*/
#include<stdio.h>
main()
{
    FILE *fp1,*fp2;
    char name[50];
    int c1,c2,found='n';
    printf("Enter the file name : ");
    scanf("%s",name);

    if((fp1=fopen(name,"r"))==NULL)
    {
        printf("Error in opening file\n");
        exit();
    }

    fp2=fopen("c:\\new.c","w");
    c1=fgetc(fp1);
    c2=fgetc(fp1);

    while(c2!=EOF)
    {
        if(c1=='/'&&c2=='*')
            found='y';
        if(found=='n')
            fputc(c1,fp2);
        if(c1=='*'&&c2=='/')
        {
            found='n'
            c2=fgetc(fp1);
        }
        c1=c2;
        c2=fgetc(fp1);
    }
    fclose(fp1);
    fclose(fp2);
}
```
