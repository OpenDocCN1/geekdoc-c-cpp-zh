# Structure and Union

```
```c
#include<string.h>
main()
{
union tag{
char name[15];
int age;
}rec;
strcpy(rec.name, "Somalika");
rec.age=23;
printf("Name = %s\n",rec.name);
}
```
```

```
```c
struct{
char a[20];
int b;
union{
double c;
struct{
char d[15];
float e;
}x;
}z;
}y;
main()
{
printf("%u %u %u\n",sizeof(z.y.x),sizeof(z.y),sizeof(z));
}
