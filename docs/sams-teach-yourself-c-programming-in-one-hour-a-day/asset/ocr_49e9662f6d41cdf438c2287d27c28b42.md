# Constants

# LISTING 3.2 const.c - A Program That Demonstrates the Use of Variables and Constants

```
/* Demonstrates variables and constants */
#include <stdio.h>
/* Define a constant to convert a number of laps to miles */
#defineAPS_PER_MILE 4
/* Define a constant for the current year */
int CURRENT_YEAR = 2013;
/* Declare the needed variables */
float miles_covered;
int laps_run, year_of_birth, current_age;
int main ( {
    /* Input data from user */
    printf("How many laps did you run: ");
    scanf("%d", &laps_run);
    printf("Enter your year of birth: ");
    scanf("%d", &year_of_birth);
    /* Perform conversions */
    miles_covered = (float)laps_run/APS_PER_MILE;
    current_age = CURRENT_YEAR - year_of_birth;
    /* Display results on the screen */
    printf("\nYou ran %.2f miles.,\n", miles_covered);
    printf("\nNot bad for someone turning %d this year!\n", current_age);
    return 0;
}
```

# Output

How many laps did you run: 7
Enter your year of birth: 1975

You ran 1.75 miles.
