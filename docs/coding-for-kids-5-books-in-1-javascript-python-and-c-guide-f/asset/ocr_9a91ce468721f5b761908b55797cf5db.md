vowels = 'aeiou'

# returns the count of vowels in text
def count_vowels(text):
    count = 0
    # convert text to all lowercase so it compares correctly
    # to vowels
    text = text.lower()
    for char in text:
        if char in vowels:
            count += 1
    return count
