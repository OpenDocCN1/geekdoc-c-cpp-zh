# **list[starting_index: ending_index]** – slicing

operator, slices the list at starting index and ends at one element before ending index.

list1 = ['Python', 'C++', 'C', 'Java Script']  
list1[0:2] = ['Python', 'C++']  
list1[0:-1] = ['Python', 'C++', 'C']

# **list.append(elem)** – *Append element to list.*

list1 = [1, 2, 3]  
list1.append(4)  
list1 value is [1, 2, 3, 4]

# **list.clear()** – Clears list values .

list1 = [1,2,3]  
list1.clear()  
list1 value is []

# **list.count(elem)** – Count how many times element elem is on list .

list1 = [1,2,2,2,3,4]  
list1.count(2) outputs 3
