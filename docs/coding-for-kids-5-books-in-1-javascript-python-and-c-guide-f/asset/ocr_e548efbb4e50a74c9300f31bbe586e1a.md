# Example 1: Comparisons and logical operators

This snippet illustrates some examples using comparison and logical operators.

```
```python
# comparison operators

x = 0
y = 10

# number comparisons
print("y == x is", y == x)
print("y > x is", y > x)
print("y < x is", y < x)
print("y <= x is", y <= x)

print()

# logical operators
print("(not True) is", not True)
print("(True and True) is", True and True)
print("(True and False) is", True and False)
print("(True or False) is", True or False)
print("(False or False) is", False or False)

# Output:
y == x is False
y > x is True
y < x is False
y <= x is False

(not True) is False
(True and True) is True
(True and False) is False
(True or False) is True
(False or False) is False
