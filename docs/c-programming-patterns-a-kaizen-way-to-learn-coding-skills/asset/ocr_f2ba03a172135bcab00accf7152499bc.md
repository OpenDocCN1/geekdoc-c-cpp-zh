# NumberPatternIC_3.c

```
#include <stdio.h>
int main() {
    int totalRows = 7; //number of rows to display

    for( int row = 1 ; row <= totalRows ; row++ ) {
        int upper_limit = ( totalRows + 1 - row );
        for(int col = 1 ; col <= upper_limit ; col++ ) {
            printf( "%d ", upper_limit );
        }
        printf("\n");// moves control to the next line
    }

    return 0;
}
```
