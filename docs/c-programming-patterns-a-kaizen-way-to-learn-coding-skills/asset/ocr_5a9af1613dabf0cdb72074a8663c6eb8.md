```c
#include <stdio.h>
int main() {

    int totalRows = 5;  //number of rows to display

    for( int row = totalRows ; row >= 1; row -- ) {

        // row value to be printed
        for(int col = 1 ; col <= row ; col++) {
            printf( "%d ", row );
        }// inner for-loop

        printf("\n");// moves control to the next line

    }// outer for-loop

    return 0;

}
