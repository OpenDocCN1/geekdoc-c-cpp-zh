# StatementIfElseIf.c

```
#include <stdio.h>
int main() {
    int num1 = 80;
    int num2 = 40;
    if( num1 > num2 ) {
        printf("%d is greater than %d",num1, num2);
    }else if( num1 < num2 ) {
        printf("%d is less than %d",num1, num2);
    }else {
        printf("%d equals to %d" + num2);
    }
    return 0;
}
```

# Output

80 is greater than 40
