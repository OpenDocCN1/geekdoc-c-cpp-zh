#include <stdio.h>
int main() {
    int totalRows = 10; //number of rows to display
    for ( int row = 1; row <= totalRows; row++ ) {
        for ( int col = 1; col <= row; col++ ) {
            // added extra space for clarity in output.
            printf("%d ", col );
        } // end inner for-loop
        printf("\n");// moves control to the next line
    } // end outer for-loop
    return 0;
}
