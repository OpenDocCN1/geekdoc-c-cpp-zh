```c
#include <stdio.h>
int main() {

    int  totalRows = 5;  //number of rows to display
    for(int row = 1; row <= totalRows; row ++) {

        //prints variable col value for (row >= col >=1)
        for(int col =totalRows; col >= 1; col--) {
            if( col<=row ) {
                printf("%d ", row );
            }else {
                printf("  ");
            }
        }

        printf("\n");// moves control to the next line
    }
    return 0;
}
