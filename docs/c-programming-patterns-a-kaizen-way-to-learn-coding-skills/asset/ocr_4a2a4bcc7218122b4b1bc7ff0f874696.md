# NumberPattern3D.C

```c
#include <stdio.h>
int main() {

    int totalRows = 7;  //number of rows to display

    for( int row = totalRows ; row >=1 ; row -- ) {

        for( int col = totalRows; col >= 1; col-- ) {

            if( col <= row ) {
                // row value to be printed
                printf("%d ", row );
            }else {
                // spaces to be printed
                printf( "  " );
            }

        }

        printf("\n");// moves control to the next line
    }

    return 0;

}
