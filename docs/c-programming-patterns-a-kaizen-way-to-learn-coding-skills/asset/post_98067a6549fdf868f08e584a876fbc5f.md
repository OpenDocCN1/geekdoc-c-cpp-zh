
## Strings

It is common programming practice to modify or manipulate text in some way. Some of the examples are dynamically creating web-page url addresses or file paths, extracting specific information from different kinds of file like text files/PDFs/HTML, developing web-scraping tools that extract text from web-pages. Every programming language provides in-built string libraries for string manipulation. We will experiment with some of commonly used in-built string methods or functions.

## String concatenation

We can output a resultant string by joining one string with another using in-built strcat method or function and this process is known as concatenating strings. The string library to use is;

```
#include <string.h>

char *strcat( char *target, const char *source );
```

This method appends source_string to the target string.

## StringConcat.c

```
#include <stdio.h>
#include <string.h>

int main() {
    char name[20] = "Steve" ; // target string
    // appends target string to source string
    strcat( name , " Jobs");
    printf("%s \n", name);
    return 0;
}
```

## Output

Steve Jobs

```
#include <stdio.h>
#include <string.h>
int main() {
    int totalRows = 5; // number of rows to display
    char line[44]={" "}; // initialize with blank string.
    // Column level Repetitive Action :
    for( int row = 1 ; row <= totalRows ; row ++ ) {
        strcat( line , "* " ); // append strings
        // Print string and move cursor to the next row
        printf("%s\n",line);
    }
    return 0;
}
```

## StringPatternG1.c

```c
#include <stdio.h>
int main() {
    int totalRows = 5;  // number of rows to display
    int index = -1;  // array's index to increment by 1.
    char line[44]=""; // initialize with blank string.

    for( int row = 1; row <= ( 2 * totalRows - 1 ); row++ ) {
        // Column level Repetitive Action :
        // expanding array by filling it up using '*' char.
        if ( row <= totalRows ){
            line[++index] = '*';
        } else {
            // replacing '*' char from the end of the array by space.
            // decrement index only when row > totalRows
            line[index--] = ' ';
        }
        // Print string and move cursor to the next row
        printf("%s\n",line);
    }
    return 0;
}
```

```
#include <stdio.h>
```

```
int main() {
    char string[44] = "Programming";
    char spaces[20] = {" "};
    for( int index = 0 ; string[index]!='\0' ; index++) {
        strcat( spaces , " " ); // append strings
        char ch = string[index] ;
        printf("%c",ch);
        printf("%s%s%s",spaces, spaces);
        // Print last char and move cursor to the next row
        printf("%c\n",ch);
    }
    return 0;
}
```

## PatternBySubString.c

```
#include <stdio.h>
#include <stdbool.h> //C99
int main() {

    char string[44] = "Programming";
    char line[44] = {""}; // initialize with blank string.
    int chars = -1;
    int index = -1;
    bool isLast = false;
    for( index = 0 ; (chars > 0 || string[index] != '\0'); index++ ) {
        // Column level Repetitive Action :

        // expanding array by filling it up using array char.
        line[index] = string[index];
        if(string[index] == '\0'){
            isLast = true;
        }
        if ( isLast ) {
            line[chars--] = ' ';
        }else{
            chars ++;
        }
        // Print string and move cursor to the next row
        printf("%s\n",line);

    }
    return 0;
}
```
