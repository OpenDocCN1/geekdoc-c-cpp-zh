ExerciseGB4. Generating pattern using character "*" by printing it for every (row, col) iteration values.

# RowColumnStars.c

```
#include <stdio.h>
int main() {
    int TOTAL_ROWS = 10;
    int TOTAL_COLUMNS = 10;
    for( int row = 1 ; row <= TOTAL_ROWS ; row++ ) {
        for( int col = 1 ; col <= TOTAL_COLUMNS ; col++ ) {
            printf("* ");
        }
        // moves the control to the next line
        // printing of "*" characters starts from next line
        printf("\n");
    }
    return 0;
}
```
