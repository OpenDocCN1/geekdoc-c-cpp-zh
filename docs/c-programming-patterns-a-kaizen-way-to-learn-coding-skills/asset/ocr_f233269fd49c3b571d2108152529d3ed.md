NumberPattern1A.c

```c
#include <stdio.h>
int main() {

    int totalRows = 5; //number of rows to display

    for (int row = 1; row <= totalRows; row++) {

        for (int col = 1; col <= row; col++) {
            // added extra space for clarity in output.
            printf("%d ", row);
        }

        printf("\n");// move to the next line

    }

    return 0;
}
