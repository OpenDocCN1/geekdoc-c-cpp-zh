```
#include <iostream>

using namespace std;

class Beta {    // incomplete class declaration allows class Alpha
               // to refer to class Beta

class Alpha {
    private:
        char *s1;
    public:
        Alpha() {
            s1 = new char[strlen("Testing")+1];
            strcpy(s1, "Testing");
        }
        friend void show(Alpha a, Beta b);
};
 Beta {
    private:
        char *s2;
    public:
        Beta() {
            s2 = new char[strlen("one, two, three")+1];
            strcpy(s2, "one, two, three");
        }
        friend void show(Alpha a, Beta b);
};
void show(Alpha a, Beta b) {
    cout << a.s1 << b.s2;
}

int main() {
    Alpha a;
    Beta b;
    show(a, b);
    return 0;
}
```

Note that a friend function declaration must appear inside the class declaration to which it is a friend. The function is prefaced by the keyword friend and can appear in either the public or private part of the class without affecting its meaning.

Friend functions are not member functions and therefore do not implicitly have this pointer passed as their first argument.
