Type in, compile and execute each of the two programs below:

```
// io3.cpp

#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    float x, y;

    x = 8.0 / 3.0;
    y = 9.0 / 3.0;
    cout << setiosflags(ios::showpoint | ios::fixed)
         << setprecision(3);
    cout << "(" << x << ")" << endl;
    cout << "(" << y << ")" << endl;
    return 0;
}
```

```
// io4.cpp

#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    float x, y;

    x = 8.0 / 3.0;
    y = 9.0 / 3.0;

    cout << setiosflags(ios::showpoint | ios::fixed);
    cout << "(" << x << ")" << endl;
    cout << "(" << y << ")" << endl;
    return 0;
}
```

Each of these programs will alter the output as follows:

(2.667)
(3.000)
