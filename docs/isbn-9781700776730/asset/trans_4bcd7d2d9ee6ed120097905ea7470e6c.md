请记住，在模板类定义之外定义的成员本身也是模板。类作用域标识符必须包含模板参数：

```cpp
template <class K, class D>
Record<K, D> :: Record(K kx, D dx) {
    key = kx;
    data = dx;
}

// record2.cpp
class String {};

template <class K, class D>
class Record {
private:
    K key;
    D data;
public:
    Record() { }
    Record(K kx, D dx) {}
    K getkey() { return key; }
    D getdata() { return data; }
};
```
