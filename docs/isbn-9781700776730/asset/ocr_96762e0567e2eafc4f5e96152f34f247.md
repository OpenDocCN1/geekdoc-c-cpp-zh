// time2.cpp

#include <iostream>

using namespace std;

class Time2 {
private:
    int month;
    int day;
    int year;
    int hour;
    int min;
    int sec;
public:
    void display();
    void setTime(int m, int d, int y, int h, int mn, int s);
    void getTime(int& m, int& d, int& y, int& h, int& mn, int& s);
};

void Time2::display() {
    cout << (month < 10 ? "0" : "") << month << "/"
         << (day < 10 ? "0" : "") << day << " ""
         << year << " ""
    cout << ((hour == 0 || hour == 12) ? 12 : hour % 12) << ":"
         << (min < 10 ? "0" : "") << min << ":"
         << (sec < 10 ? "0" : "") << sec
         << (hour < 12 ? " AM" : " PM") << endl;
}

void Time2::setTime(int m, int d, int y, int h, int mn, int s) {
    month = m;
    day = d;
    year = y;
    hour = h;
    min = mn;
    sec = s;
}

void Time2::getTime(int& m, int& d, int& y, int& h, int& mn, int& s) {
    m = month;
    d = day;
    y = year;
    h = hour;
    mn = min;
    s = sec;
}
