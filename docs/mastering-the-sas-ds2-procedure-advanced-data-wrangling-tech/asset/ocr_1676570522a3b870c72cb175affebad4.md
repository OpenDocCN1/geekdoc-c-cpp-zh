Here is an example that shows similar processing, loads data from a data table, moving the data from a matrix to a VARARRAY with method, and writing the results out to the DS2 program result set:

```
proc ds2;
data;
  dcl varchar(7) Timing;
  dcl package matrix m(2,2);
  vararray double a[2] vl2;
  vararray double v[2, 2] w x y z;
  dcl double b[2,2];
  drop v1 v2;
  method run();
  /* Read data, load each row of matrix from vararray*/
    set x;
    m.in(a, n);
    m.tovarray(b);
  end;
  method term();
  Timing='Before:';
    m.tovarray(v);
    output;
  Timing='Trans:';
    m=m.trans();
    m.tovarray(v);
    output;
  Timing='SQRT:';
    m=m.sqrt();
    m.tovarray(v);
    output;
  end;
ata;
run;
quit;
```
