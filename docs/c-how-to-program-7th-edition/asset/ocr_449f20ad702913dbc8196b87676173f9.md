double GradeBook::getAverage()
{
    int total = 0; // initialize total

    // sum grades in array
    for ( int grade = 0; grade < students; grade++ )
        total += grades[ grade ];

    // return average of grades
    return static_cast<double>( total ) / students;
} // end function getAverage
