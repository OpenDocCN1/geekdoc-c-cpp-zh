# 9.3 Adding Weapon Control to the AI Controller

The BaseArmedEnemy.cs script is intended to be added to a player to take control of the weapons system and deal with firing. It does not specifically reference or require the AI controller, and it works independently, relying only on the Standard_SlotWeaponController.cs script to operate the weapon and a separate class named AIAttackStates.
A AIAttackStates.cs script stores an enumerated list of the possible states for the armed enemy to take:

```c
using UnityEngine;

namespace AIAttackStates
{
    public enum AIAttackState
    {
        random_fire,
        look_and_destroy,
        no_attack,
    }
}
```

Below is the BaseArmedEnemy.cs script:

```c
using AIAttackStates;
using UnityEngine;
using System.Collections;
public class BaseArmedEnemy : ExtendedCustomMonoBehavior
{
    [System.NonSerialized]
    public bool doFire;

    public bool onlyFireWhenOnscreen=50;
    public int pointsValue=10;
    public int thisEnemyObjectStrength=1;
    public bool thisGameObjectShouldFire;

    // we use a renderer to test whether or not the ship is on screen
    public Renderer rendererToTestAgainst;

    public StandardSlotWeaponController weaponControl;
    public GameObject mesh_parentGO;

    private bool fireCan;
    public float fireDelayTime =1f;

    BasePlayerManager myPlayerManager;
    public BaseUserManager myDataManager;
    public bool isBoss=false;

    public int tempINT;
    // default action is to attack nothing
    public AIAttackState currentState=AIAttackState.random_fire;

    public string tagOfTargetsToShootAt;

    public void Start ()
    {
        // now call our script-specific init function
        InitThis();
    }

    public void InitThis()
    {
        // cache our transform
        myTransform= transform;
        // cache our gameObject
        myGO= gameObject;

        if (weaponControl==null) {
            // try to find weapon control on gameObject
            weaponControl= myGO.GetComponent<StandardSlotWeaponController>();
        }

        if (rendererToTestAgainst==null) {
            // find renderer in children
            rendererToTestAgainst= myGO.GetComponentInChildren<Renderer>();
        }

        if (myPlayerManager==null) {
            // find player manager
            myPlayerManager= myGO.GetComponent<BasePlayerManager>();
        }

        if (myDataManager==null) {
            // find data manager
            myDataManager= myGO.GetComponent<BaseUserManager>();
        }
    }

    public virtual void Update()
    {
        // use a flag to disable firing
        if (thisGameObjectShouldFire && fireCan) {
            // check if we are on screen
            if (onlyFireWhenOnscreen || rendererToTestAgainst.IsVisibleFrom(Camera.main)) {
                fireCan = false;
                if (currentState == AIAttackState.random_fire) {
                    // random fire
                    if (Random.Range(0,100) > 98) {
                        weaponControl.Fire();
                    }
                } else if (currentState == AIAttackState.look_and_destroy) {
                    // look and destroy
                    RaycastHit hit;
                    if (Physics.Raycast(myTransform.position, myTransform.forward, out hit, 100)) {
                        if (hit.collider.CompareTag(tagOfTargetsToShootAt)) {
                            weaponControl.Fire();
                        }
                    }
                }
            }
        } else {
            // reset flag to true
            if (!thisGameObjectShouldFire) {
                fireCan = true;
            }
        }
    }
}
```
