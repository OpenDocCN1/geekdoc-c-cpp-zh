    steer= Mathf.Clamp ( default_input.GetHorizontal() , -1, 1 );

    motor= Mathf.Clamp(default_input.GetVertical() , 0, 1 );

void Hit()
{
    // reduce lives by one
    DataManager.ReduceHealth(1);

    if(DataManager.GetHealth()<1) // destroyed
    {
        isRespawning=true;
    }

    // blow up!
    myBody.AddosionForce(myBody.mass * 2000, myBody.position, myBody.angularVelocity-new Vector3(Random.Range(-100,100), Random.Range(-100,100), Random.Range(-100,100)));

    // tell game controller to do a nice explosion
    CameraController_MVD.Instance.PlayerBigHit(myTransform);

    // respawn
    Invoke("Respawn",4f);

    // reset health to full
    DataManager.SetHealth(startHealthAmount);
}

else {
    // tell game controller to small scale hit
    GameController_MVD.Instance.PlayerHit(myTransform);
}

void Respawn()
{
    // reset the 'we are respawning' variable
    isRespawning=false;

    // reset our velocities so we don't reposition
    myBody.velocity+=Vector3.zero;
    myBody.angularVelocity-=Vector3.zero;

    // get the waypoint to respawn at the race controller
    tempTR=raceController.GetRespawnWaypoint();
    tempVEC=tempTR.position;

    // cast a ray down from the waypoint to find the ground
    RaycastHit hit;
    if(Physics.Raycast(tempVEC + (Vector3.up * 300), -Vector3.up, out hit)) {
        tempVEC.y=hit.point.y+15;
    }
    // reposition the player at tempVEC (the waypoint position
    // with a corrected y value via raycast)
    // and also set the player rotation to the waypoint's
    // rotation that we are facing in the right direction after respawning
    myTransform.rotation=tempTR.rotation;
    myTransform.position=tempVEC;

    // we need to be invulnerable for a little while
    MakeInvulnerable();

    Invoke("MakeInvulnerable",3);

    // revert to the first weapon
    if(weaponControl!=null) {
        weaponControl.SetWeaponSlot(0);
    }
}

After the variable declarations, the Start() function overrides the Start() the BaseVehicle class. Even though this overridden version features no actual functionality, it is important to do so since the original function (the we are overriding) called Init immediately, and we do not want to do there. The Init() function will be by the game controller when it is ready:
