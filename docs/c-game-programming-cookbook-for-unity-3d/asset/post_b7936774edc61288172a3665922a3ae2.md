
## 11.6 Player Controller

Lazer Blast uses an animated human model from the Unity3d Angry Bots tutorial, included with Unity. It uses a class that derives from the top-down controller script shown in Chapter 5, adding custom game-specific scripting to it and a separate class to take care of animating the character model.

The player class _LBS derives from the controller script BaseTopDown.cs, a modified version of the third-person controller script included with Unity that uses Unity's built-in character controller for physics. BaseTopDown.cs is detailed in full in Chapter 5 (Building Player Movement Controllers).

Player_LBS.cs looks like this:

```csharp
using UnityEngine;
using System.Collections;

public class Player_LBS : BaseTopDown
{
    private bool isInvulnerable;
    private bool isRespawning;

    public BasePlayerManager myPlayerManager;
    public BaseUserManager myDataManager;

    public bool godMode = false;
    public GameObject meshGO;

    public Standard.SlotWeaponController weaponControl;
    public bool canFire;

    public bool isFinished;

    public override void Init()
    {
        base.Init();

        // do god mode, if needed
        if(godMode)
        {
            MakeVulnerable();
        } else {
            MakeInvulnerable();
        }

        // start out with no control from the player
        canControl = false;

        // get a ref to the weapon controller
        weaponControl = myGO.GetComponent<Standard.SlotWeaponController>();

        // if a player manager is not set in the editor, let's try
        if(myPlayerManager == null)
            myPlayerManager = myGO.GetComponent<BasePlayerManager>();
    }
}
```

```csharp
// set up the data for our player
myDataManager = myDataManager.DataManager;
myDataManager.SetName("Player");
myDataManager.SetHealth(3);

isFinished = false;

// tell game control to update the lives UI
GameController_LBS.Instance.UpdateLivesPl(myDataManager.GetHealth());
}

public override void GetInput()
{
    if(isFinished || isRespawning)
    {
        horz = 0;
        vert = 0;
        return;
    }

    // drop out if we're not supposed to be controlling this player
    if(!canControl)
        return;
}

// grab inputs from the input provider
horz = Mathf.Clamp(default_input.GetHorizontal(), -1, 1);
vert = Mathf.Clamp(default_input.GetVertical(), -1, 1);

// fire if we need to
if(default_input.GetFire() && canFire)
{
    // tell weapon controller to deal with firing
    weaponController.Fire();
}

public void GameStart()
{
    // this function is called by the game controller to tell us
    // when we can start moving
    canControl = true;
}

void LostLife()
{
    isRespawning = true;

    // blow us up!
    GameController_LBS.Instance.PlayerHit(myTransform);

    // reduce lives by one
    myDataManager.ReduceHealth(1);
    myController_LBS.Instance.UpdateLivesPl(myDataManager.GetHealth());

    if(myDataManager.GetHealth() < 1) // game over
    {
        // stop movement, as long as rigidbody is not
        // kinematic (otherwise it will have no velocity and
        // we will generate an error message trying to set it)
        if (!PlayerController.rigidbody.isKinematic)
            PlayerController.rigidbody.velocity = Vector3.zero;

        // hide ship body
        theMeshGO.SetActive(false);

        // disable and hide weapon
        weaponControl.DisableCurrentWeapon();

        // do anything we need to do at game finished
        PlayerFinished();
    }
    else {
        // hide ship body
        theMeshGO.SetActive(false);

        // disable and hide weapon
        weaponControl.DisableCurrentWeapon();

        // respawn
        Invoke("Respawn", 2f);
    }
}

void Respawn() 
{
    // reset respawning flag
    isRespawning = false;

    // make the player invulnerable for a while
    MakeInvulnerable();

    // show ship body
    theMeshGO.SetActive(true);

    // show the weapon
    weaponControl.EnableCurrentWeapon();
}

void OnEnter(Collision collision)
{
    // if the player is invulnerable, ignore the collision
    if (isInvulnerable) return;

    // check if we collided with a projectile
    if (collision.gameObject.CompareTag("Projectile"))
    {
        LostLife();
    }
}

void MakeVulnerable() 
{
    isInvulnerable = true;
}

void MakeInvulnerable() 
{
    isInvulnerable = false;
}
```
