void InitAI()
{
// cache reference to the AI controller
AIController=myGO.GetComponent<BaseAIController>();

// check to see if we found an AI component, if
// not add one here
if(AIController==null)
    AIController=myGO.AddComponent<BaseAIController>();

// initialize the AI controller
AIController.Init();

// tell the AI controller to go into waypoint mode
AIController.SetAIState(AIStates.steer_to_waypoint);

// disable our default input
default_input.enabled=false;

// add an AI weapon controller
gunControl=myGO.GetComponent<BaseEnemy>();

// if we don't already have a gun controller, let's add one
// to stop breaking but warn about it it may be fixed
if(gunControl==null)
{
gunControl=myGO.AddComponent<BaseEnemy>();
// tell gun controller to do 'look' and destroy
gunControl.currentState=AttackStates.look_and_destroy;
}

void AddBattleController()
{
if(myGO==null)
myGO=gameObject;
// add a battle controller script to our object
battleControl=myGO.AddComponent<BattleController>();

// grab an ID from the battle control script so we don't
// have to look it up whenever we communicate with
// the global battle controller
id=battleControl.GetID();

// we are going to use the same id for this player
// and its weapons
weaponControl.SetOwner(id);

// set a repeating invoke to update our position, rather
// than calling it single tick or update
InvokeRepeating("UpdatePosition",0, 0.5f);

public void UpdateBattlePosition()
{
// grab our score board position from the global battle manager
GlobalBattleManager.Instance.GetPosition(id);
}

public override void UpdatePhysics()

if(canControl)

if(isFinished)
myBody.velocity *=0.99f;

// if we are moving slow, we apply some extra force to

// no 'ground' it will apply it in the air, too (bad!)
if(mySpeed < TorqueHelperMaxSpeed)
{
myBody.AddTorque(new Vector3(0, steer * myBody.mass * TorqueHelper, 0));
}

public override void LateUpdate()
{
// get the state of the battle from the battle controller
isGameRunning=battleControl.battleRunning;

if(isGameRunning)
{
// check to see if we've crashed and are upside down
Check_Car_Is_Flipped();
}

// make sure that gunControl has been told it can fire
if(isAIControlled)
gunControl.canControl=true;

// we check for input in LateUpdate because Unity recommends this
if(temp.layer==17 || temp.layer==9)
{
// grab a ref to the projectile's
aProj=tempGO.GetComponent<ProjectileController>();
    
// quick check to make sure that this projectile was not launched by this player
if (aProj.ownerTypeId != id ) 
{
// tell the hit function about this collision, passing in the
// gameObject so that we can get to its projectile controller
// script and find out more about where it came from
Hit();
// tell our battle controller that we got hit
battleControl.Fragged();
}
    
// tell the global battle controller who fragged us
GlobalBattleManager.Instance.RegisterFrag(aProj.ownerTypeId);
}

public void OnTriggerEnter( Collider other )
{
// check to see if the trigger uses any of the layers where
// we want to automatically respawn the player on impact
int objLayerMask = (1 << other.gameObject.layer);
    
if ((respawnLayerMask.value & objLayerMask) > 0)
{
    Respawn();
}
}

void Hit()
{
// reduces lives by one
myDataManager.ReduceHealth(1);
    
    if (myDataManager.GetHealth()<1) // destroyed
    {
        isRespawning=true;
        
        // blow up! (apply a force to affect everything in
// the vicinity and let's get the model spinning
// too, with angular velocity)
        myBody.AddExplosionForce(myBody.mass * 2000f,
        myBody.position, 100);
        myBody.angularVelocity=new Vector3( Random.Range(-100,100), Random.Range (-100,100), Random.Range(-100,100) );
    }
// tell game controller to do a nice big explosion
GameController_TB.Instance.PlayerBigHit( myTransform );

// respawn
Invoke("Respawn",4f);

// reset health to full
myDataManager.SetHealth(startHealthAmount);
}

void Respawn()
{
// reset the 'we are respawning' variable
isRespawning=false;

// reset our velocities so that we stop spinning
myBody.velocity=Vector3.zero;
myBody.angularVelocity=Vector3.zero;

// reposition the player at tempVEC (the waypoint position)
// and also set the player to the waypoint's rotation so
// we are facing in the right direction after respawning
myTransform.eulerAngles=respawnRotation;
myTransform.position=tempVEC;
}
