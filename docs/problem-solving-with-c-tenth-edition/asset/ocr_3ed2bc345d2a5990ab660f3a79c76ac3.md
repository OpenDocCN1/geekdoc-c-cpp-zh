#include <iostream>
#include <thread>

using std::cout;
using std::endl;
using std::thread;

class Runnable
{
public:
    Runnable();
    Runnable(int *target, int *results, int num, int start, int end);
    void operator()();
private:
    int *target, *results;
    int num, start, end;
};

Runnable::Runnable()
{
    target=nullptr;
    results=nullptr;
    num=0;
    start=0;
    end=0;
}

Runnable::Runnable(int *target, int *results, int num, int start,
                   int end)
{
    this->target = target;
    this->results = results;
    this->num = num;
    this->start = start;
    this->end = end;
}

void Runnable::operator()()
{
    int min = target[start];
    for (int i=start+1; i<=end; i++)
    {
        if (target[i]<min)
            min = target[i];
    }
}
