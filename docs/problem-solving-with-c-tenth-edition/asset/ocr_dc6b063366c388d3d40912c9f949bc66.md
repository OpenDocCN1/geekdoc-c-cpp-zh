```
#include <iostream>
#include <string>
using namespace std;

class NegativeNumber
{
public:
    NegativeNumber();
    NegativeNumber(string takeMeToYourCatchBlock);
    string getMessage();
private:
    string message;
};

class DivideByZero
{};

int main()
{
    int jemHadar, klingons;
    double portion;

    try
    {
        cout << "Enter number of JemHadar warriors:\n";
        cin >> jemHadar;
        if (jemHadar< 0)
            throw NegativeNumber("JemHadar");

        cout << "How many Klingon warriors do you have?\n";
        cin >> klingons;
        if (klingons< 0)
            throw NegativeNumber("Klingons");
        if (klingons != 0)
            portion = jemHadar/static_cast<double>(klingons);
        else
            throw DivideByZero();
        cout << "Each Klingon must fight "
             << portion << " JemHadar.\n";
    }
    catch(NegativeNumber e)
    {
        cout << "Cannot have a negative number of "
             << e.getMessage() << endl;
    }
```

Although not done here, exception classes can have their own interface and implementation files and can be put in a namespace. This is another toy example.
