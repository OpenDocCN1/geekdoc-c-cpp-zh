+   4. and 5. The answer to 4 is `writeUp(int n)`; The answer to 5 is `writeDown(int n)`;

```
#include <iostream>
using namespace std;
void writeDown(int n)
{
    if (n >= 1)
    {
        cout << n << " ";
        writeDown(n - 1);
    }
}

void writeUp(int n)
{
    if (n >= 1)
    {
        writeUp(n - 1);
        cout << n << " ";
    }
}
```
