So let's read the file a line at a time, storing each into a list as we go.

+   - Make an empty list
+   - Open the file

def read_file(filename):
    """Read a file from disk"""
    lines = []

    with open(filename) as f:
        for line in f:
            lines.append(line)

    return lines

if filename is not None:
    lines = read_file(filename)
else:
    lines = []
