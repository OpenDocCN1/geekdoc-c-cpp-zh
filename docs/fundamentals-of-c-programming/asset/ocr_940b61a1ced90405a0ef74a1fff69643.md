/*
 * print
 * Prints the contents of an integer vector
 * a is the vector to print.
 * a is not modified.
 */
void print(const std::vector<int>& a) {
    int n = a.size();
    std::cout << '{';
    if (n > 0) {
        std::cout << a[0];      // Print the first element
        for (int i = 1; i < n; ++i)
            std::cout << ', ' << a[i];  // Print the rest
    }
    std::cout << '}';
}
