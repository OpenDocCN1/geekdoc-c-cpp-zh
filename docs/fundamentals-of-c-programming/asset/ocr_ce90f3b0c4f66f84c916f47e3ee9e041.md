    // Make p point to a dynamically created a widget object
    auto p = std::make_shared<Widget>();
    std::cout << p->id << " \n";
    p->id = 25;
    std::cout << p->id << " \n";
    std::cout << "---- Leaving Test 1 ----\n";
}

    q = nullptr; // Make q point to nothing
    std::cout << "---- Leaving Test 2 ----\n";
}
