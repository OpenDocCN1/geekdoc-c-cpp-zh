# Listing 17.8: testtext.cpp

```
#include <iostream>
#include "text.h"
#include "fancytext.h"
#include "FixedText.h"

int main() {
    Text t1("plain");
    FancyText t2("fancy", "<'", "'>", "***");
    FixedText t3;
    std::cout << t1.get() << " ' \n";
    std::cout << t2.get() << " ' \n";
    std::cout << t3.get() << " ' \n";
    std::cout << "-----------------------------------\n";
    t1.append("A");
    t2.append("A");
    t3.append("A");
    std::cout << t1.get() << " ' \n";
    std::cout << t2.get() << " ' \n";
    std::cout << t3.get() << " ' \n";
    std::cout << "-----------------------------------\n";
    t1.append("B");
    t2.append("B");
    t3.append("B");
    std::cout << t1.get() << " ' \n";
    std::cout << t2.get() << " ' \n";
    std::cout << t3.get() << " ' \n";
}
```
