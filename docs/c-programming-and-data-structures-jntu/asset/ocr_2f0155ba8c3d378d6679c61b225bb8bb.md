```
# 3. /* Write a program to print the signed and unsigned numbers by using C99 standards*/

```
```c
#include<stdio.h>
int main(void)
{
int i= -99;
{
for(int i=0; i<10; i++) //declare i within for
printf("%d",i);
printf("\n");
}
printf("Value of i is: %d",i); //display-99
return 0;
}
```
