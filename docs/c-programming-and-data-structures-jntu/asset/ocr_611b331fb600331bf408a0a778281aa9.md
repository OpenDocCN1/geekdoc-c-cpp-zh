int priority(char c)
{
 if (c == '^') return 3;
 if (c == '*' || c == '/' || c == '%') return 2;
 else if (c == '+' || c == '-') return 1;
 else return 0;
}

void ConvertToPostfix(void)
{
 char opr;
 while (*s != '\0')
 {
 if (*s == '*' || *s == '/' || *s == '%')
 {
 s++;
 continue;
 }
 if (isdigit(*s) || isalpha(*s))
 {
 while (isdigit(*s) || isalpha(*s))
 {
 *t = *s;
 s++;
 t++;
 }
 }
 if (*s == '^')
 {
 PushOnStack(*s);
 s++;
 }
 if (*s == '*' || *s == '/' || *s == '%' || *s == '+' || *s == '-' || *s == '^')
 {
 if (top != -1)
 {
 opr = PopFromStack();
 while (priority(opr) >= priority(*s))
 {
 *t = opr;
 t++;
 opr = PopFromStack();
 }
 }
 PushOnStack(*s);
 }
 else
 PushOnStack(*s);
 s++;
 }
 if (*s == '^')
 {
 opr = PopFromStack();
 }
 }
