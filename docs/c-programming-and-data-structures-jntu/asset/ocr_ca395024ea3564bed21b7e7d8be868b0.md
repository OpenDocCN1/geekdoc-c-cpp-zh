void smerge_sort(int *a, int *b, int *c, int n1, int n2, int *n)
{
int i=0, j=0, k=0;
while(i<n1 && j<n2)
{
if(a[i]<b[j])
{
c[k]=a[i];
i++;
}
else
{
c[k]=b[j];
j++;
}
k++;
}
while(i<n1)

c[k]=a[i];
i++;
k++;
}
while(j<n2)
{

k++;
}
*n=k;
}
