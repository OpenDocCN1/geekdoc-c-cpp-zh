{
    int r1 = node[x].point;
    int r2 = -1;

    while (r1 >= 0 && node[r1].point != y)
    {
        r2 = r1;
        r1 = node[r1].next;
    }

    if (r1 >= 0)
    {
        /* r1 points to an arc from node[x] to node[y] */
        if (r2 < 0)
            node[x].point = node[r1].next;
        else
            node[r2].next = node[r1].next;

        free (node[r1]);
        return;
    }

    /* do nothing if the arc if not found */
}
