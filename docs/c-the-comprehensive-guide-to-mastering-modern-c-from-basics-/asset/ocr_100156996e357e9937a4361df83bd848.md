```
// modern101.cpp
#include <iostream>
int fibonacci(int n) {
    return n <= 1 ? n : fibonacci(n-2) + fibonacci(n-1);
}
int main() {
    std::cout << "Which Fibonacci number? ";
    int n = 0;
    std::cin >> n;
    std::cout << "Fib(" << n << =") " << fibonacci(n) << "\n";
}
```
