r.second). Naturally, in a set, this is either zero or one. Since C++20, there is also contains, which returns a bool. All searches are guaranteed to take at most log n steps.

Note the somewhat asymmetric behavior of upper bound: when you search in the set s with elements (1,2,3) using it= s.upper_bound(2), it points to 3, because it is the first element greater than 2. In contrast, it= s.lower_bound(2) returns an iterator to 2, because it is the last element not less than 2. they are exactly equal_range returns a pair. This is illustrated in Listing 24.56.

#include <set>
#include <iostream>
using std::ostream; using std::set;
void search(const set<int>&data, int what, ostream& os) {
  // contains
  auto inside = contains(what); // C++20
  os << "inside: " << (what ? "yes." : "no."); // check contains
  auto where = data.find(what);
  if (where != data.end()) {
    os << "found: " << *where << "";
  } else {
    os << "not found.";
  }
  auto lo = data.lower_bound(what);
  if (lo != data.end()) {
    os << "lo: " << *lo;
  } else {
    os << "lo:--";
  }
  auto up = data.upper_bound(what);
  if (up != data.end()) {
    os << "up: " << *up;
  } else {
    os << "up:--";
  }
  // [lo,up) is the same as what equal_range would have returned
  os << "Range:";
  for(auto i=lo; i!=up; ++i) {
    os << *i << ",";
  }
  os << ")";
  // count
  os << "C: " << data.count(what) // count hits
  os << "\n";
}
set<int> data={10, 20, 30, 40, 50, 60};
search(data, 20); // 20 in:yes. found:20 up:30 range:{20} C:1
search(data, 25); // 25 in:no. lo:30 up:30 range:{} C:0
search(data, 10, cout); // 10 in:yes. found:10 up:20 range:{10} C:1
search(data, 60, cout); // 60 in:yes. found:60 up: range:{60} C:1
search(data, 5, cout); // 5 in:no. lo:10 up:10 Range:{} C:0
