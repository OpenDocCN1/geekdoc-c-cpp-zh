```cpp
#ifndef PERSON_HPP
#define PERSON_HPP
#include <string>
#include <iostream> // ostream
struct Person {
    std::string name;
    int age;
    std::string city;
    std::ostream& print(std::ostream& os);
};
#endif // PERSON_HPP
```

```cpp
#include "person.hpp"
#include <format>
using std::ostream; using std::format;
ostream& Person::print(ostream& os) {
    return os << format("{} {}", name, age);
}
```
