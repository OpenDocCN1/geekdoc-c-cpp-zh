Therefore, a constval function can only perform constexpr computations. Such a function can be quite complex in itself (you can follow the Compiler Explorer link in the next listing to see the machine code and confirm that the calculated primes are part of the compiled program).

```
// https://godbolt.org/Wbr7n8M4q
#include <iostream>
#include <array>
constexpr bool isPrime(int n) { // computable at compile-time
if (n < 2) return false; // 0 and 1 are not prime
for (int i = 2; i*i <= n; i += i>2 ? 2 : 1) { // 2,3,5,7,9,11,13,15...
if (n % i == 0) return false;
}
return n > 1; // for 0 and 1
}

template<int Num>
constval std::array<int, Num> primeNumbers() { // only at compile-time
std::array<int, Num> primes{0};
int idx = 0;
for (int val = 1; idx < Num; ++val) {
if (isPrime(val)) primes[idx++] = val;
}
return primes;
}

int main() {
// initialize with prime numbers
auto primes = primeNumbers<100>(); // 1000000 doesn't work
for (auto v : primes) {
std::cout << v << ' ';
}
std::cout << "\n";
}
```
