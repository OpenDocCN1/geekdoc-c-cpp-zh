#include <iostream>  // container vector
#include <vector>
#include <string>
int inputUpTo(int argc, const char* argv[]) {
    /* Determine number * */
    int upTo = 0;  // introduce new variable
    if(argc<=1) { // if statement with then and else
block
        std::cout << "Up what number do you want to calculate prime
numbers? " <<;
        if(!(std::cin >> upTo)) { // check return value
            return -1; // error in user input
        }
    } else {
        upTo = std::stoi(argv[1]);  // else part of the if statement
    }
    return upTo;
}
std::vector<int> primes{2}; // new vector<int> with
initialization
bool isPrime(int n) {
    /* primes are sorted in ascending order */ // range-based for loop
    for(int factor : primes) {
        if(factor > n) // too large to be a divisor at
all?
            return true; // then terminate inner loop
arly
        if(n%factor==0) // is divisor?
            return false; // then exit
    }
    return true;
}
void calculatePrimesUpTo(int upTo) {
    /* Primes calculation */
    /* vector must contain {2} at this point */
    for(int n=3; n<=upTo; n+=2) {\ // standard for loop
        if(isPrime(n)) {
            primes.push_back(n);\ // is prime - mark as divisor and
result
        }
    }
}
void outputPrimes() {
    for(int prime : primes) { // range based, over all elements
        std::cout << prime << " ";
    }
}
