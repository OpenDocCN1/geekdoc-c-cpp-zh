#include <set> // set, multiset

#include <string>
#include <chrono> // Time measurement
using std::chrono;
using namespace std::chrono;

long milliSince(steady_clock::time_point start) {
    return duration_cast<milliseconds>(steady_clock::now()-start).count();
}

const size_t ITERATIONS = 108000;
template <typename Gen, typename Type>
requires(Gen gen, size_t n) -> std::same_as<int> &&
requires(Gen gen, size_t n) { { gen(n); } -> std::same_as<int>;
std::size_t as(typename Type::value_type int);

void timeFunc(std::string name, Type data, Gen gen) {
    auto start = steady_clock::now();
    for (size_t i=0; i<ITERATIONS; ++i) {
        data.insert(gen(i));
    }
    cout << name << " : " << milliSince(start) << " ms" << std::endl;
} // always generates the same number
int scatter(size_t n) { return rand(n); } // generates different numbers
struct BadHash {\ // worst possible hash function as a function
    size_t operator()(int) const { return 0; }
};

int main() {
    std::multiset<int> same; // same, &same
    timeFunc("multiset", same, scatter); // scatter, &scatter
    std::multiset<int> set; // same, &same
    timeFunc("set", set, scatter); // scatter, &scatter
    std::unordered_set<int> unordered_set; // unordered, &same
    timeFunc("unordered_set", unordered_set, scatter); // scatter, &scatter
    std::unordered_multiset<int> unordered_multiset; // unordered, &same
    timeFunc("unordered_multiset", unordered_multiset, scatter); // scatter, &scatter
    std::unordered_multiset<int> badhash; // unordered, &same
    timeFunc("unordered_multiset", unordered_multiset, badhash); // scatter, &scatter
}
