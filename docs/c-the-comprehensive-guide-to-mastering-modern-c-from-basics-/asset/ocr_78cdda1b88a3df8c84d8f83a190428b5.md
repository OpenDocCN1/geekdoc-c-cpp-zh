will find functions that mainly query something from a range. With find_if, for example, you can search for the first occurrence of an element with a certain property in a range. You test this with a predicate, a function that returns bool. In the case of find_if, this is a unary predicate, meaning it is a function with one argument.

```
// https://godbolt.org/z/W9n7sTovz
#include <algorithm>
#include <iostream>
#include <vector>
#include <string>
#include <string_view>
#include <ranges>

std::vector; using std::string; using std::string_view;
using namespace std::literals; using std::find_if;
vector<string> split(string_view s) {
    vector<string> result;
    auto it = s.begin();
    while (it != s.end()) {
        // normal marker
        it = find_if(it, s.end(), [](char c) { return c== ' '; });
        // whitespace
        it = find_if(it, s.end(), [](char c) { return c== ' '; });
        result.push_back(string(it)); // Copy to result
        it++;
    }
    return result;
}
```
