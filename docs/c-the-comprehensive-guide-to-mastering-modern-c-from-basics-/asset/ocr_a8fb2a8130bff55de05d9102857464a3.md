```
void testAdd_nodups() {
    UnitTest inst();
    BOOST_REQUIRE_EQUAL(inst.size(), 0u); /* arrange */
    inst.add("*"); /* act */
    BOOST_CHECK_EQUAL(inst.size(), 1u); /* assert */
    inst.add("ENTRY", "entry");
    BOOST_CHECK_EQUAL(inst.size(), 2u); /* assert */
    inst.add("OTHER", "other");
    BOOST_CHECK_EQUAL(inst.size(), 3u); /* assert */
}
```
