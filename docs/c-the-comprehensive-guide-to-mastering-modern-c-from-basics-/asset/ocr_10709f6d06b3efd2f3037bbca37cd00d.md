has to ask whether the LSP is being violated. Often, the matter is not so simple, and only common sense helps.

One example, or even the classic example, of LSP violation revolves around the geometric shapes of the rectangle and the square. Intuitively, one might build a class hierarchy where Rectangle is the base class and Square is the derived class, a square is is-a rectangle, right?

```
// https://ggdbolt.org/z/rc67xWfT
#include <iostream>
struct Point { int x, y; };
class Rectangle {
protected:
    Point origin; int width; int height;
public:
    Rectangle(Point o, int w, int h) : origin{o}, width{w}, height{h} {}
    virtual void setHeight(int height) { height = height; }
    virtual int getHeight() const { return height; }
    virtual void setWidth(int width) { width = width; }
    virtual int getWidth() const { return width; }
    virtual int getArea() const { return width_ * height_; }
};
class Square : public Rectangle {
public:
    Square(Point o, int wh, int wh) : Rectangle(o, wh, wh) {}
    void setHeight(int wh) override { width = wh; height_ = wh; }
    void setWidth(int wh) override { width = wh; height_ = wh; }
};
void areaCheck(Rectangle& rect) {
    rect.setWidth(5);
    rect.setHeight(4);
    auto areaValue = rect.getArea();
    if(areaValue != 20) {
        std::cout << "error!\n";
    } else {
        std::cout << "all fine\n";
    }
}
int main() {
    Rectangle rect{{0,0}, 0, 0};
    areaCheck(rect); // Output: all fine
    Square square{{0,0}, 0, 0};
}
```
