```
#include <iosstream> // left, right, internal
#include <iostream>
#include <iomanip>   // setw
using std::cout; using std::cin; using std::ostream;
std::ostream& tabl(std::ostream& os) {
    os << '\t';
    return os;
}

std::ostream& firstNum(std::istream& is) {
    char ch;
    is.get(ch);
    if (ch >= '0') && (ch <= '9')) {
        std::cin.putback(ch);
    }
    return is;
}

int main() {
    int val=0;
    cout << "Text1" << tabl << "Text2" << endl; // Output: Text1
    cout << "Make an input: ";
    cin >> firstNum >> val;
    cout << val << std::endl; // Output: 12345
}
```
