iterators, use a vector<bool>. An example with some
but not all—operations is shown in Listing 24.90.

```
// https://godbolt.org/z/5e3c8KK8j
#include <bitset>
#include <iostream>
using std::cout;
int main() {
    std::bitset<8> bits;
    bits.set(4);                // 8 bits densely packed
    cout << bits << '\n';      // 5th bit to 1
    bits.flip();                // 00010000
    cout << bits << '\n';      // invert all bits
    bits.set();                 // 11011111
    cout << bits << '\n';      // set all bits to 1
    bits.flip();                // invert 2nd bit
    cout << bits << '\n';      // 11111101
    bits.reset();               // set all bits to 0
    bits.set(4);                // 5th bit to 1
    cout << bits << '\n';      // 00010000
    bits.flip();                // invert all bits
    cout << bits << '\n';      // 11101111
    bits.set();                 // set all bits to 1
    cout << bits << '\n';      // invert 2nd bit
    bits.flip();                // 11011101
    bits.flip(6);               // invert 7th bit
    cout << bits << '\n';      // 10011101
    // Bitwise operations
    std::bitset<8> jack("....####", 8, '.\', '#');
    cout << jack << '\n';       // 00000111
    cout << (bits & jack) << '\n';  // 00000101
    cout << (bits | jack) << '\n';  // 10111101
    cout << (bits ^ jack) << '\n';  // 10101000
    // other integer types
    std::bitset<64> b(0x123456789abcdef0L);
    cout << b << '\n';          // 00001001100110001010111110001001101010111111111110000
    cout << std::hex << b.to_ull() << '\n'; // convert
    // 123456789abcdef0
}
```
