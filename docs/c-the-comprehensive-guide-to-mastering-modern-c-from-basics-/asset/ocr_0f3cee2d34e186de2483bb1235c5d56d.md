auto max0f2(int a, int b) {
    return a<b ? b : a; // one return: the compiler determines int
}

auto min0f3(int a, int b, int c) {
    if(a-b) return a<c ? a : c;
    else return b<c ? b : c;
}

auto median0f3(int a, int b, int c) {
    // more complex, but no problem for the compiler
    return min0f3(a,b), max0f2(b,c), max0f2(a,c));
}
