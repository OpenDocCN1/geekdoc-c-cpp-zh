+   "sv as a suffix
The standard library also provides the user-defined
""s literal to create a string_view. The advantage over
""s for string is that it does not create a new string,
but only a view on the const char*—almost nothing
needs to be copied or created.

```
// https://godbolt.org/z/P7zbrh7ld
#include <vector>
#include <string>
#include <iostream>
#include <string_view>

using std::string; using std::string_view; using std::vector; using
std::cout;

int get_len(string_view str) { return str.size(); } // string_view as
parameter

int main() {
    string s1 = "Hello"; // simple with string literal
    string s2{'H','e','l','l','o'}; // or with list of char
    using namespace std::literals;\ // for "s-suffix and "sv-\-
suffix
    auto s3 = "Hello"; // even simpler with real string literal
    vector<char> v1("Hello"); // no vector string with
Literal
    vector<char> v2{'H','e','l','l','o'}; // list of char is okay
    cout << s1 << s2 << s3 << '\n'; // Output of string works
    cout << v1 << v2 << '\n'; // vector has no output

    const auto str = "String"; // Stringliteral
    const auto strv = "String-View"; // String-View literal
    cout << "Length of str is " << get_len(str) << '\n'; // Output:
    cout << "Length of strv is " << get_len(strv) << '\n'; // Output: 
}
```
