#include <unordered_set>

using std::string; using std::unordered_set; using std::cout;
Word {
    string word;
    size_t size;
    Word(const string& w, size_t row) : word(w), size(row) {}
    friend bool operator==(const Word& a, const Word& b) {
        return a.word == b.word;
    } // ignores row
};
namespace std {
template<class T> struct hash<Word { // ignores row
    std::hash<T> ihash;
    std::size_t operator(const T& w) const {
        return ihash(w.word);
    }
};
}
struct ExactHash { // includes row
    std::hash<string> ihash;
    std::size_t size;
    bool operator(const Word& a) const {
        return ihash(a.word) ^ ihash(a.row);
    }
};
struct ExactEqual { // includes row
bool operator(const Word& a, const Word& b) const {
    return std::tie(a.word, a.row) == std::tie(b.word, b.row);
};
