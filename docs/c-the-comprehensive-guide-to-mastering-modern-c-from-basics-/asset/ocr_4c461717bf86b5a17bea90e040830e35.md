# Listing 24.53 You can return the value when inserting sorted ranges.

The return value of the insert method with a position hint returns an iterator for associative containers as well as for sequence containers. This allows you to write functions that work with both container types.

```
// https://godbolt.org/z/h1Gv86jv
#include <set>
#include <vector>
#include <iostream>
using std::cout; using std::ostream; using std::set; using std::vector;

template<typename Container>
void insertFive(Container& c, int a, int b, int c, int d, int e) {
    auto it = c.end();
    for(int x : { a, b, c, d, e }) {
        it = c.insert(it, x); // works with vector, set etc.
    }
}

int main() {
    vector<int> dataVec;
    insertFive(dataVec, 0, 2, 2, 0, 4);
    for(auto e : dataVec) cout << e << " ";
    cout << "\n"; // Output: 4 0 2 2 0
    set<int> dataSet;
    insertFive(dataSet, 9, 4, 2, 2, 0);
    for(auto e : dataSet) cout << e << " ";
    cout << "\n"; // Output: 0 2 4 9
```
