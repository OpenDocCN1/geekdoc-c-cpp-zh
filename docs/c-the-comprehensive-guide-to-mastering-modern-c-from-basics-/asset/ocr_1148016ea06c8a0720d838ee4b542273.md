// https://godbolt.org/z/aGd56xxzsT
#include <string>           // string, stoi
#include <iostream>        // cin, ostream
#include <format>
using std::ostream; using std::format;
class Year { /* Helper types for safe date */
    int value_; // e.g., 2024
public:
    Year(int v) : value_{v} {}
    int value() { return value_; };
};
class Month {
    int value_; // 1.12
public:
    Month(int v) : value_{v} {}
    int value() { return value_; };
};
class Day {
    int value_; // 1.31
public:
    Day(int v) : value_{v} {}
    int value() { return value_; };
};
/* type-safe constructing date */
class Date {
    Year year_;
    Month month_ = 1;
    Day day_ = 1;
public:
    Date(int y) : year_{y} // 1-argument constructor
                  // sets 1st January of the specified
    Date(Year y, Month m, Day d) // 3-argument constructor
        : year_{y}, month_{m}, day_{d} {}
    ostream& print(ostream& os) { // e.g., 2024-06-27
};
ostream&::print(ostream& os) { // e.g., 2024-06-27
    return os << format("{:04}-{:02}-{:02}",
                        year_.value(), month_.value(), day_.value());
}
