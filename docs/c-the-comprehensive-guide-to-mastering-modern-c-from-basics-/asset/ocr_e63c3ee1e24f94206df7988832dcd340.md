/* Print matrix */
template<class T>
void printMatrix(ostream&os, const valarray<T>& a, size_t n) {
  for(size_t i = 0; i < (n*n); ++i) {
    os << setw(3) << a[i];      // Print value
    os << ((i+1)%n ? '' : '\n');   // next line?
  }
}
