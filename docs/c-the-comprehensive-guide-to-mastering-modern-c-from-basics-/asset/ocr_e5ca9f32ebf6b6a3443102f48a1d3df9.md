#include <iostream>
#include <string>
#include <algorithm> // min, max
#include <tuple>
using std::tuple; using std::tuple; std::get; std::cout;
using std::string;
tuple<int,int,int> arrange(int a, int b, int c) {
    using std::min; using std::max;
    auto x = min(a,b,c);
    auto y = max(min(a,b), max(a,b,c));
    auto z = max(a,b,c);
    return make_tuple(x, y, z);
}
auto auto print(int year) {
    using namespace std::literals;
    if(year >= 2021)
        return std::make_tuple("Joseph", "Biden", "Democratic"s, 1942);
    if(year >= 2017)
        return std::make_tuple("Donald", "Trump", "Republican"s, 1946);
    if(year >= 2009)
        return std::make_tuple("Barack", "Obama", "Democratic"s, 1961);

    return std::make_tuple("s", "s", "s", 0); // Default case
}
int main() {

You combine multiple values of different types into a tuple using auto, as with arrange and zs, feel free to write out the tuple type. However, I recommend working with auto, as shown with presidentand ps. Here it is actually a tuple<string, string,int>, but you can save yourself the typing with auto.
