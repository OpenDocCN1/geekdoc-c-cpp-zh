    /* Primes calculation */
    /* vector must contain {2} at this point */
    for(int n=3; n<upTo; n+=2) {
        for(int factor=3;

                goto prime; // using goto, because a break ...

                goto notPrime; // Target of the jump after push_back
        }

}
