import java.util.Random;

class ProcessAccounts {

        Account anAccount;
        for (int i = 0; i < 3; i++) {
            anAccount = new Account();
            anAccount.lastName = "" +

                (char) (myRandom.nextInt(26) + 'a');

            anAccount.balance = myRandom.nextInt(10000);
