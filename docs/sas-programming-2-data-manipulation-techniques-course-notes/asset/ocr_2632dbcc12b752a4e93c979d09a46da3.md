# 6. Using Arrays for Repetitive Computations

data special offer;
set orion.orders midyear;
array mon{*} month1-month3;
keep Total Sales Projected Sales Difference;
Total Sales=sum(of month1-month6);
do i=1 to 3;
mon{i}=mon{i} *.90;
end;
Projected_Sales=sum(of month1-month6);
Difference=Total_Sales-Projected_Sales;
