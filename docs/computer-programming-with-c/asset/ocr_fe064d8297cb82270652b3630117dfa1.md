#include<iostream.h>
class ShoppingCart
{
private:
    int count;
    int list[100];
public:
    ShoppingCart()
    {
        count=0;
    }
    void addItem()
    {
        int id;
        cout<<"Enter the Item identification of the item to be
added"<<endl;
        cin>>id;
        list[count]=id;
        count++;
    }
    void display()
    {
        int i;
        cout<<"The items in the list are"<<endl;
        for(i=0;i<count;i++)
        {
            cout<<list[i]<<endl;
        }
    }
}
void main()
{
ShoppingCart c1;
ShoppingCart c2;
c1.addItem();
c1.addItem();
c1.addItem();
c2.addItem();
c2.addItem();
cout<<"------------Cart C1----------------"<<endl;
c1.display();
cout<<"------------Cart C2----------------"<<endl;
c2.display();
}
