Given below is the full source code to check if the string is palindrome or not.

```
```c
#include<stdio.h>
void main()
{
    int len=0,i,k,flag=0;
    char a[100],b[100];
    printf("Enter a string:");
    scanf("%s",a);
    for(i=0;a[i]!='\0';i++)
    {
        len++;
    }
    k=len-1;
    for(i=0;k>=0;i++)
    {
        b[i]=a[k];
        k--;
    }
    b[len]='\0';
    for(i=0;a[i]!='\0';i++)
    {
        if(a[i]!=b[i])
        {
            flag=1;
            break;
        }
    }
    if(flag==0)
    {
        printf("The string is a Palindrome\n");
    }
    else
    {
        printf("The string is not a Palindrome\n");
    }
}
```
