    {
        continue;
    }
    else
    {
        /*take the variable i to the beginning of a particular word*/
        while(a[i]!=' ' && i>=1)
        {
            i--;
        }
        //print the word in forward direction
        while (i< len && a[i]!=' ')

            printf("%c",a[i]);
            i++;
        }
        printf(" "); //print the space

}
