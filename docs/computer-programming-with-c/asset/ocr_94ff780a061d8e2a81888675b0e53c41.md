The following is the full source code defining a simple clone() function to create copy of object using this keyword:

```
```cpp
#include<iostream.h>
class Data
{
    private:
        int a;
    public:
    void set(int t)
    {
        a=t;
    }
    Data clone()
    {
        return *this;
    }

    void output()
    {
        cout<<"a="<<a<<endl;
    }
};
void main()
{
Data o1,o2,o3;
o1.set(10);
o2=o1.clone();
cout<<"******o1******"<<endl;
o1.output();
cout<<"******o2******"<<endl;
o2.output();
o3=o2; //another way to clone objects
cout<<"******o3******"<<endl;
o3.output();
}
```
