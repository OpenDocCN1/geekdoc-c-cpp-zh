If the user enters a digit from 0 to 9, we will not throw any exception but print the success message. This means that it is possible for `check()` function to throw any one of the three exceptions depending on the user input. ```cpp
#include<iostream.h>
class UpperCaseException { };
class LowerCaseException { };
class SpecialException { };
void check(char ch) throw (UpperCaseException, LowerCaseException,
        SpecialException)
{
    if(ch>=48 && ch<=57)
    {
        cout<<"Correct input - check successful"<<endl;
    }
    else if(ch>=65 && ch<=90)
    {
        throw UpperCaseException();
    }
    else if(ch>=97 && ch<=122)
    {
        throw LowerCaseException();
    }
    else
    {
        throw SpecialException();
    }
}
void main()
{
    char ch;
    cout<<"Enter a character"<<endl;
    cin>>ch;
    try
    {
        check(ch);
    }
    catch( ...)
    {
        cout<<"Invalid Input"<<endl;
    }
}
