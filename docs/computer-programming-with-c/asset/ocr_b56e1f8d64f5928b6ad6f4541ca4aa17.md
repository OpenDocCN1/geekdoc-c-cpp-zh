Note that the program should count the presence of character in upper or lowercase in the input sentence immaterial of in what case the user inputs the character.

```
#include<ctype.h>
#include<stdio.h>
void main()
{
    int i,count=0;
    char a[100];
    char ch;
    printf("Enter a sentence:");
    gets(a);
    printf("Enter a character:");
    scanf("%c",&ch);
        for(i=0;a[i]!='\0';i++)
        {
            if(toupper(a[i])==toupper(ch))
            {
                count++;
            }
        }
    printf("Character %c occurs %d times in the sentence.\n",ch,count);
}
```
