int evalPost(char postExp[])
{
    int i = 0, op1, op2;
    char ch;
    while((ch = postExp[i++]) != '\0')
    {
        if(ch == ' ')
            continue;
        if((tolower(ch) >= 'a') && (tolower(ch) <= 'z'))
            push(value(ch));
        else if(ch=='+' || ch=='-' || ch=='*' || ch=='/' || ch=='%')
        {
            if(pop(&op2) || pop(&op1))
                printError();
            push(calc(op1, op2, ch));
        }
    }
    pop(&op1);
    if(!pop(&op2))
        printError();
    return op1;
}
void printError()
{
    printf("The given postfix expression is Wrong");
    exit(0);
}
