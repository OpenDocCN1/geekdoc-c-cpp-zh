int push(int value)
{
    extern int top, stack[];
    if(isFull())
    return -1;
    stack[++top] = value;
    return 0;
}

```c
int pop(int *value)
{
    extern int top, stack[];
    if(isEmpty())
    return -1;
    *value = stack[top--];
    return 0;
}
```

```c
int peek(int *value)
{
    extern int top, stack[];
    if(isEmpty())
    return -1;
    *value = stack[top];
    return 0;
}
```

```c
int isEmpty()
{
    extern int top, stack[];
    if(top == -1)
    return 1;
    else
    return 0;
}
