void insertEdge(int vStart, int vEnd)
{
    vnode *pv;
    enode *te, *pe;
    for(pv = adjList; pv != NULL && pv->vName != vStart; pv = pv->vlink);
    if(pv == NULL)
        return;
    te = getNode();
    printf("Enter Edge Weight from V%d to V%d : ", vStart, vEnd);
    scanf("%d",&te->eWeight);
    te->vName = vEnd;
    if(pv->elink == NULL)
    {
        pv->elink = te;
        return;
    }
    for(pe = pv->elink; pe->elink != NULL; pe = pe->elink);
    pe->elink = te;
}
