void inToPost(char inStr[], char postStr[])
{
    char ch, item;
    int i = 0, st = 0, spr, ipr;
    push('\\0');
    while((ch = inStr[st++]) != NULL)
    {
        if(tolower(ch) >= 'a' && tolower(ch) <= 'z')
            postStr[i++] = ch;
        else if(ch == '(')
            push(ch);
        else if(ch == ')')
        {
            pop(&item);
            while(item != '(')
            {
                postStr[i++] = item;
                pop(&item);
            }
        }
        else
        {
            pop(&item);
            spr = indexPriority(sp, item);
            ipr = indexPriority(ip, ch);
            while(sp[spr][1] >= ip[ipr][1])
            {
                postStr[i++] = item;
                pop(&item);
                spr = indexPriority(sp, item);
            }
        }
    }
}
