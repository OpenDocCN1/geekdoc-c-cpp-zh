# Program S19:
Write a program for changing an integer to its equivalent name.

```
/* numname.c */
#include<stdio.h>
void main()
{
    long num, a[8], i, j, k, c = 0;
    printf("Enter a number : ");
    scanf("%d", &num);
    while(num > 0)
    {
        j = num % 10;
        a[c] = j;
        num = num / 10;
        c++;
    }
    a[c] = '\0';
    for(k = c - 1 ; k >= 0 ; k--)
    switch(a[k])
    {
        case 1 :      printf("ONE ");
                      break;
        case 2 :      printf("TWO ");
                      break;
        case 3 :      printf("THREE ");
                      break;
        case 4 :      printf("FOUR ");
                      break;
        case 5 :      printf("FIVE ");
                      break;
        case 6 :      printf("SIX ");
                      break;
        case 7 :      printf("SEVEN ");
                      break;
        case 8 :      printf("EIGHT ");
                      break;
    }
```
