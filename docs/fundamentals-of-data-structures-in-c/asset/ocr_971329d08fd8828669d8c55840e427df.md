binary_search(int list[], int find, int beg, int end)
{
    int mid;
    if(beg > end)
        return (-1);
    mid = (beg + end)/2;
    if(find == list[mid]);
        return(mid);
    else if(find < list[mid])
        return binary_search(list, find, beg, mid-1);
    else
        return binary_search(list, find, mid+1, end);
}
