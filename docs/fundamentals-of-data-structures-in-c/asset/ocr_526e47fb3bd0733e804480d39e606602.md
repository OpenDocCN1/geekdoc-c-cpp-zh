tree *findParNode(tree *btree, int data, tree **par)
{
    if(btree == NULL)
        return NULL;
    else if(data < btree->data)
    {
        *par = btree;
        return findParNode(btree->left,data, par) ;
    }
    else if(data > btree->data)
    {
        *par = btree;
        return findParNode(btree->right, data, par) ;
    }
    else if(data == btree->data)
        return btree;
}
