poly *createPoly()
{
    poly *p, *head = NULL;
    do
    {
        p = getNode();
        readNode(p);
        head = insertNode(head, p);
    }while(p -> exp != 0);
    return head;
}
poly *getNode()
{
    return (poly *)malloc(sizeof(poly));
}
void readNode(poly* newnode)
{
    int exp;
    float coef;
    printf("\nEnter the Coefficient : ");
    scanf("%f", &coef);
    printf("Enter the Exponent : ");
    scanf("%d", &exp);
    newnode -> coef = coef;
    newnode -> exp = exp;
    newnode -> next = NULL;
}
