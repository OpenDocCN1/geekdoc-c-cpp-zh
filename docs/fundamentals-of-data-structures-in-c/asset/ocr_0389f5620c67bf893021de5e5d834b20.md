int checkDir()
{
 int r, c;
 for(r = 0; r < nVertex; r++)
 for(c = 0; c < nVertex; c++)
 if(adjMat[r][c] != adjMat[c][r])
 return 1;
 return 0;
}

void createGraph()
{
 int r, c;
 printf("\n Enter the no. of Vertices : ");
 scanf("%d", &nVertex);
 for(r = 0; r < nVertex; r++)
 for(c = 0; c < nVertex; c++)
 {
 adjMat[r][c] = 0;
 if(r == c)
 insertEdge(r+1, c+1);
 }
}
