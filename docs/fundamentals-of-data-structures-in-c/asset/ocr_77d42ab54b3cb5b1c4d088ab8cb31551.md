/* Program to sort the numbers using bubble sort. */
/* bubble.c */
#include<stdio.h>
void bubble(int a[], int n);
int i, j, n, temp, a[25];
void main()
{
    printf("Enter the limit : ");
    scanf("%d",&n);
    printf("Enter the elements : ");
    for(i = 0 ; i < n ; i++)
    scanf("%d", &a[i]);
    bubble(a, n); /* Calling the function bubble() */
    printf("The sorted list is : ");
    for(i = 0 ; i < n ; i++)
    printf("%d ", a[i]);
}
void bubble(int a[], int n)
{
    for(i = 0 ; i <= n-1 ; i++)
    for(j = i+1 ; j <= n-1 ; j++)
    if(a[i] > a[j])
    {
    temp = a[i];
    a[i] = a[j];
    a[j] = temp;
    }
}
