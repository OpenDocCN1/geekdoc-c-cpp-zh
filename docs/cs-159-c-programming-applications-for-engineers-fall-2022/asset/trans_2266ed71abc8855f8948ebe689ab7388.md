```c
int main(void)
{
    int x = 3752281;
    int result = 0;

    do
    {
        result += makeCalculation(x);

        x = x - 100;
        x = x / 10;
    } while(x > result);

    printf("x: %d\n", x);
    printf("result: %d\n", result);

    return(0);
}

int makeCalculation(int n)
{
    int status;

    status = n % 100 / 50;

    switch(status)
    {
        case 2:  status = n % 1000;
                 break;
        case 1:  status = n % 100;
                 break;
        case 0:  status = n % 10;
        default: status = 0;
    }

    return(status);
}
```
