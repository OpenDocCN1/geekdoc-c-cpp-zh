# Use the program below for problems #17 - #19

```
#include<stdio.h>
void changeData(int, int, int, int);
int main(void)
{
   int x = 11;
   int y = 21;
   int z = 3;

   changeData(x, y, z, 1);
   changeData(y, z, x, 2);
   changeData(z, x, y, 3);

   return(0);
}

void changeData(int a, int b, int c, int num)
{
   a++;
   ++b;

   c = a % b + 1;

   printf("Result %d: %d\n", num, c);
}
```
