# Use the code segment below for problems 32 – 33

```
#define SIZE 10

int main(void)
{
    int result = 0;
    int *data;
    data = (int*)malloc(sizeof(int) * SIZE);
    result = populateArray(data);

    printf("result 1 = %d\n", result);
    printf("result 2 = %d\n", *data + SIZE);

    return(0);
}

int populateArray(int *array)
{
    int i = 0;
    int result = 0;

    while(i < SIZE)
    {
        *(array + i) = 2 * i - 1;
        result += *(array + i);
        i++;
    }

    return(result);
}
```
