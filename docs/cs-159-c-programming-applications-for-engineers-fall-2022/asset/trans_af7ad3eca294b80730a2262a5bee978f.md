```c
#include <stdio.h>

char getGrade(int);

int main()
{
  char grade;

  grade = getGrade(84);
  printf("Grade #1: %c - ", grade);

  grade = getGrade(68);
  printf("Grade #2: %c\n", grade);

  return(0);
}

char getGrade(int score)
{
  int pass = -1;

  switch((score + 7) / 10)
  {
  case 10: pass = 0;
  case 9: pass = 1;
  break;
  case 8: pass = 2;
  break;
  case 7: pass = 3;
  }

  return(('A' - 1) + pass);
}
```

## 以下程序用于第 26–27 题

```c
#include <stdio.h>
#define SIZE 7

int main(void)
{
    int x[SIZE] = {93, 90, 84, 72, 63, 72, 38};
    int y = 0;
    int z = 0;
    int i;

    for (i = 0; i < SIZE; i++)
    {
        y = x[i] < x[y] ? i : y;
        z = x[SIZE - i - 1] > z ? x[SIZE - i - 1] : z;
    }

    printf("y: %d\n", y);
    printf("z: %d\n", z);

    return (0);
}
```
