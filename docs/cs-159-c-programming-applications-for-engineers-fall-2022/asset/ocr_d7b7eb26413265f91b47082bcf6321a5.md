# The short-circuit method:

# Example (similar to) Program 5-1 Form text:

```
int a = 1;
int b = -1;
int c = 0;
printf("%d%d %d %d = %d\n", a, b, a && b);
printf("%d%d %d %d = %d\n", a, c, a && c);
printf("%d%d || %d = %d\n", a, b, a || b);
printf("%d%d || %d = %d\n", a, c, a || c);
printf("%d%d %d = %d\n", !a, !c, !a || !c);
printf("%d%d %d = %d\n", !a, !c, !a && !c);
printf("%d%d %d = %d\n", !a, !c, !(a && !c));
printf("b++ || ++c = %d\n", b++ || ++c);
printf("a = %d b = %d c = %d\n", a, b, c);
```

# Output:

+   1.
+   2.
+   3.
+   4.
+   5.
+   6.
+   7.
+   8.
+   9.
