```c
int main(void)
{
   int list[10] = {10, 2, 4, 3, 6, 8, 7, 9, 5, 1};
   sortArray(list, 10);
   return(0);
}

void sortArray(int list[], int last)
{
   int counter;
   int temp;
   int located;
   int current;

   for(current = 1; current < last; current++)
   {
      located = 0;
      temp = list[current];
      counter = current - 1;

      while(counter >= 0 && !located)
      {
         if(temp < list[counter])
         {
            list[counter + 1] = list[counter];
            counter--;
         }
         else
         {
            located = 1;
         }
         list[counter + 1] = temp;
      }
   }
}
```
