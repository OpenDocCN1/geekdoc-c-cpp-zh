int main(void)
{
   int x = 17;
   int y = 22;
   int z = 14;

   x = makeCalc(x, y, z);
   y = makeCalc(y, z, x);
   z = otherCalc(z, 2);

   printf("x: %d\n", x);
   printf("y: %d\n", y);
   printf("z: %d\n", z);
   return(0);
}

int makeCalc(int a, int b, int c)
{
   a += otherCalc(b, c);
   b += otherCalc(c, a);
   c += otherCalc(a, b);

   return(a + b + c);
}

int otherCalc(int a, int b)
{
   a -= b;
   b += a;
   a = b - a;

}
