```c
int main()
{
  char grade;

  grade = getGrade(84);
  printf("Grade #1: %c - ", grade);

  grade = getGrade(68);
  printf("Grade #2: %c\n", grade);

  return(0);
}

char getGrade(int score)
{
  int pass = -1;

  switch((score + 7) / 10)
  {
  case 10: pass = 0;
  case 9: pass = 1;
  break;
  case 8: pass = 2;
  break;
  case 7: pass = 3;
  }

  return(('A' - 1) + pass);
}
```

```c
{
```
