int total = 0;
int n = 351;

do
{
    if(n % 3 == 0)
    {
        total += n % 10;
        n /= 10;
    }
    else if(n % 2)
    {
        total += 1;
        n -- = 1;
    }
    else
    {
        total += 2;
        n -- = 2;
    }
}while(n > 0);

printf("total: %d\n", total);
printf("n: %d\n", n);
