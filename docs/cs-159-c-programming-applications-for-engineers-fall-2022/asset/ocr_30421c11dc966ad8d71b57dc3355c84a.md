```
int moreCalculations(int x, int n)
{
    int result = 0;

    printf("n: %d\n", n);

    if(x % n)
    {
        if(n < sqrt(x))
        {
            n += n % 2 + 1;
            result += moreCalculations(x, n);
        }
        else
        {
            result = 1;
        }
    }

    return(result);
}
```
