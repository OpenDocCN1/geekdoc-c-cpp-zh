## 使用下面的程序解决问题 9 – 10

```c
#include <stdio.h>
int makeCalculation(int, int);
int main(void)
{
    int n = 53642;
    int total = 0;

    do
    {
        total += makeCalculation(n, n % 2);
        n /= 10;
    } while(n > 0);

    printf("total: %d\n", total);

    return(0);
}

int makeCalculation(int n, int start)
{
    int result;

    result = start;

    while(n % 10 > 0)
    {
        switch(start)
        {
            case 0: result += n % 10;
            break;
            case 1: result *= n % 10;
        }
        n--;
    }

    return(result);
}
```
