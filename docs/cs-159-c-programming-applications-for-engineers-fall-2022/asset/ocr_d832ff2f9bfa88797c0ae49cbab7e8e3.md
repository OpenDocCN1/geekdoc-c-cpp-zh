# Use the program below for problems 26 – 27

```
#include <stdio.h>
#define SIZE 7

int main(void)
{
    int x[SIZE] = {93, 90, 84, 72, 63, 72, 38};
    int y = 0;
    int z = 0;
    int i;

    for (i = 0; i < SIZE; i++)
    {
        y = x[i] < x[y] ? i : y;
        z = x[SIZE - i - 1] > z ? x[SIZE - i - 1] : z;
    }

    printf("y: %d\n", y);
    printf("z: %d\n", z);

    return (0);
}
```
