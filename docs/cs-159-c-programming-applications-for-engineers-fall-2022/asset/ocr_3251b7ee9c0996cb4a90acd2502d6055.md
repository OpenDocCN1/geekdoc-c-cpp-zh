int ctZero(int n)
{
    int ct = 0;
    while(n > 0)
    {
        if(n % 10 == 0)
        {
            ct++;
        }
    }
    n = n / 10;
    return(ct);
}

int findMax(int n, int *maxLoc)
{
    int max = 0;
    int loc = 0;
    while(n > 0)
    {
        if(n % 10 > max)
        {
            max = n % 10;
            *maxLoc = loc;
        }
    }
    n = n / 10;
    loc++;
    }
    return(max);
}
