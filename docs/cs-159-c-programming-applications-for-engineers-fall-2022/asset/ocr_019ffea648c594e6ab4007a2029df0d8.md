int main(void)
{
    int index;
    int data[10] = {13, 3, 4, 9, 11, 10, 12, 1, 15, 10};
    randomArray(data);
    index = data[8] % 10;
    printf("result 1 = %d\n", data[index]);
    index = data[5] % 10;
    printf("result 2 = %d\n", *(data + index));
    return(0);
}
void randomArray(int *data)
{
    int i;
    for(i = 0; i < 9; i++)
    {
        *(data + i) = data[i] % 2 ? data[i] / 2 : data[i] * 2;
    }
}
