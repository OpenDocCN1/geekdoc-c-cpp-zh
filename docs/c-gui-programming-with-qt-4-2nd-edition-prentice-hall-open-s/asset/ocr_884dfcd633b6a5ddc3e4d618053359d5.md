The ThreadDialog class declares two variables of type Thread and some buttons to provide a user interface.

```
ThreadDialog::ThreadDialog(QWidget *parent)
 : QDialog(parent)
{
    threadA.setMessage("A");
    threadB.setMessage("B");

    threadAButton = new QPushButton(tr("Start A"));
    threadBButton = new QPushButton(tr("Start B"));
    quitButton = new QPushButton(tr("Quit"));
    quitButton->setDefault(true);

    connect(threadAButton, SIGNAL(clicked()),
           this, SLOT(startOrStopThreadA()));
    connect(threadBButton, SIGNAL(clicked()),
           this, SLOT(startOrStopThreadB()));
    ...
```

In the constructor, we call setMessage() to make the first thread repeatedly print 'A's and second thread 'B'.
