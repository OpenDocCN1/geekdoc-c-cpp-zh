void Tetrahedron::mouseMove(QMouseEvent event) {
    GLfloat dx = GLfloat(event->x() - lastPos.x()) width();
    GLfloat dy = GLfloat(event->y() - lastPos.y()) height();
    if (event->buttons() & Qt::LeftButton) {
        rotationX += 180 dy;
        rotationY += 180 dx;
        updateGL();
    } else if (event->buttons() & Qt::RightButton) {
        rotationX += 180 dy;
        rotationZ += 180 dx;
        updateGL();
    }
    lastPos = event->pos();
}
