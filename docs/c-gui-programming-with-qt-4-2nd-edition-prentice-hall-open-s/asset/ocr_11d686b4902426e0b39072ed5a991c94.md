we redo the example, but this time we will subclass QMImageData to postpone or avoid the (potentially expensive) conversions between QTableItems and QByteArray. Here's the definition of our subclass:

```
class TableMimeData : public QMImageData
{
    Q_OBJECT
public:
    TableMimeData(const QTable tableWidget,
                  const QTreeWidgetItemSelectionRange &range);

    const QTable table() const { return myTableWidget; }
    QTreeWidgetItemSelectionRange range() const { return myRange; }
    QStringList formats() const;

protected:
    QVariant retrieveData(const QString &format,
                          QVariant::Type preferredType) const;

private:
    static QString totoHtml(const QString &plainText);
    static QString toCsv(const QString &plainText);

    QString text(int row, int column) const;
    QString rangeAsPlainText() const;

    const QTable *myTableWidget;
    QTreeWidgetItemSelectionRange myRange;
    QStringList myFormats;
};
```

Instead of storing actual data, we store a QTableSelectionRange that specifies which cells are dragged and keep a pointer to the QTable. The formats() and retrieveData() functions are reimplemented from QMImageData.

```
TableMimeData::TableMimeData(const QTable *tableWidget,
              const QTableSelectionRange &range)
{
    myTableWidget = tableWidget;
    myRange = range;
    myFormats << "text/csv" << "text/html" << "text/plain";
}
```
