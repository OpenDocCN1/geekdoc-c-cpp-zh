Often we want to show a QProgressDialog while a long-running operation is taking place. QProgressDialog has a bar that keeps the user informed about the application's progress. QProgressDialog also provides a Cancel button that allows the user to abort the operation. Here's the code for saving a Spreadsheet file using this approach:

Code View:

```
bool Spreadsheet::writeFile(const QString &fileName)
{
  QFile file(fileName);
  ..
  QProgressDialog progress(this);
  progress.setLabelText(tr("Saving %1").arg(fileName));
  progress.setRange(0, RowCount);
  progress.setModal(true);

  for (int row = 0; row < RowCount; ++row) {
    progress.setValue(row);
    qApp->processEvents();
    if (progress.wasCancelled()) {
      file.remove();
      return false;
    }
    for (int column = 0; column < ColumnCount; ++column) {
      QString str = formula(row, column);
      if (!str.isEmpty())
        out << quint16(row) << quint16(column) << str;
    }
  }
  return true;
}
```
