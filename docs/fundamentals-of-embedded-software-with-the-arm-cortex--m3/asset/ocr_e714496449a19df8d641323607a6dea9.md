The output of the program illustrates the details of the 32-bit single-precision representation by printing out the binary contents of several floating-point numbers.

```c
#include <stdio.h>
int main()
{
    static float f[] =
    {
        -2.0, -1.0, -0.75, -0.5,
        0.0, 0.5, 0.75, 1.0, 2.0
    };
    int i, j;
    for (i = 8; i >= 0; i--)
    {
        void *p = &f[i];
        long bits = (*(long *) p);
        int zero = (bits == 0L);
        printf("%9.2f*:", f[i]);
        for (j = 0; j < 32; j++)
        {
            if ((j == 1) || j == 9) printf(" ");
            if (j == 0) printf("%c(%0d) ", '!', zero);
            printf("%d", bits < 0);
            bits <<= 1;
        }
        printf("\n");
    }
    return 0;
}
