int sizex = 350;

void smooth(short IN[][], short OUT[][]) {
    short[][] K = {{1, 2, 1}, {2, 4, 2}, {1, 2, 1}};
    for (int j = 0; j < sizey-2; j++) {
        for (int i = 0; i < sizex-2; i++) {
            int sum = 0;
            for (int r = 0; r < 3; r++)
                for (int c = 0; c < 3; c++)
                    sum += IN[i+c][j+r]*K[r][c];
            sum = sum / 16;
            OUT[j+1][i+1] = (short) sum;
        }
    }
}
