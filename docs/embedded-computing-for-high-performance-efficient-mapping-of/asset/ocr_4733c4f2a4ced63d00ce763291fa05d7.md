Boolean__ crc32file(char *name, DWORD *crc, long *charcnt) {
    timer_begin("crc32file");
    oldccrc2 = 0xFFFFFFFF; *charcnt = 0;
    if ((fin = fopen(name, "rb")) == NULL) {
        perror(name);
        timer_end("crc32file");
        return Error_;
    }
    while ((c=getc(fin))!=EOF) {
        ++charcnt;
        oldccrc2 = UPDC32(c, oldccrc2);
    }
    if (ferror(fin)) {
        perror(name);
        *charcnt = -1;
    }
    fclose(fin);
    *crc = ~oldccrc2;
    timer_end("crc32file");
    return Success_;
}
DWORD crc32buf(char *buf, size_t len) {
    timer_begin("crc32buf");
    oldccrc2 = 0xFFFFFFFF;
    oldccrc2 = UPDC32(*buf, oldccrc2);

        oldccrc2 = UPDC32(*buf, oldccrc2);
    }
    timer_end("crc32buf");

}

The timer_begin, timer_end and timer_report primitives are part of a specialized API developed to monitor the execution time and number of times a function is invoked and to provide a timing report. While our profiling strategy is relatively simple, it can be challenging to instrument the code manually if the code is very large and spans through many source files. Our strategy requires identifying functions that have loops and function calls and also all exit points (e.g., return statements) to insert the timer_end primitive.
