// Default constructor
public Player( )
{
}

// Constructor accepts any size
// pointsScored array.
public Player (string ln, string fn, string iden, int [ ] s, int numGames)
{
    numberOfGames = numGames;
    FillPointsScoredArray(s);
    lname = ln;
    fname = fn;
    id = iden;
}

public string FName
{
    get
    {
        return fname;
    }
    set
    {
        fname = value;
    }
}

public string LName
{
    get
    {
        return lname;
    }
    set
    {
        lname = value;
    }
}

public string ID
{
    get
    {
        return id;
    }
    set
    {
        id = value;
    }
}
