f₁(x₁,x₂,x₃) ≡ x₃ − x₁x₂x₃ = 0,
f₂(x₁,x₂,x₃) ≡ x₁ + x₂ + x₃ = 0,
f₃(x₁,x₂,x₃) ≡ x₁x₂ + x₂x₃ + x₃x₁ = 0.

# Newton-Raphson method for systems of non-linear equations
from math import *
from roots import *

def Func(f, x, n):                                   # zero: (0, 0, 0)
    f[1] = x[3] - x[1]*x[2]*x[3]
    f[2] = x[1] + x[2] + x[3]
    f[3] = x[1]*x[2] + x[2]*x[3] + x[3]*x[1]

n = 3
