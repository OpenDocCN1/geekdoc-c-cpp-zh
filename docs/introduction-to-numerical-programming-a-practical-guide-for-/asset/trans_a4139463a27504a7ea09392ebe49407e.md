//==============================================================================
double Exp(double x)
//------------------------------------------------------------------------------
// 由幂级数展开计算 exp(x)
// 对于 x < 0，避免因减法运算而可能产生的不稳定性
//------------------------------------------------------------------------------
{
    const double eps = 1e-14;                    // 相对精度
    double absx, f, t;
    int i;

    i = 0;
    f = t = 1e0;
    absx = fabs(x);
    while (fabs(t) > eps*fabs(f)) {
        i++;
        t *= absx/i;
        f += t;
    }
    return (x >= 0.0 ? f : 1e0/f);
}

//==============================================================================
double Sin(double x)
//------------------------------------------------------------------------------
// 由幂级数展开计算 sin(x)
//------------------------------------------------------------------------------
{
    const double eps = 1e-14;                    // 相对精度
    double f, t, x2;
    int i;

    i = 1;
    f = t = x;
    x2 = x*x;
    while (fabs(t) > eps*fabs(f)) {
        i += 2;
        t *= -x2/((i-1)*i);
        f += t;
    }
    return f;
}

//==============================================================================
double ArcSin(double x)
//------------------------------------------------------------------------------
// 由幂级数展开计算 arcsin(x)（|x| < 1）
//------------------------------------------------------------------------------
{
    const double eps = 1e-14;                    // 相对精度
    double f, t, x2;
    int i, i2;

    i = 1;
    f = t = x;
    x2 = x*x;
    while (fabs(t) > eps*fabs(f)) {
        i2 = i*i;
        i += 2;
        t *= i2*x2/((i-1)*i);
        f += t;
    }
    return f;
}

//==============================================================================
double Tan(double x)
//------------------------------------------------------------------------------
// 由连分数表示计算 tan(x)
//------------------------------------------------------------------------------
{
    const double eps = 1e-14;                     // 相对精度
    double a, b, p, pm1, pm2, q, qm1, qm2, r, rm1;

    a = 1e0; b = -x*x;
    pm1 = 0e0; p = x;
    qm1 = 1e0; q = 1e0;
    rm1 = 0e0; r = p/q;
    while (fabs(r-rm1) > eps*fabs(r)) {
        pm2 = pm1; pm1 = p;                           // 移动堆栈
        qm2 = qm1; qm1 = q;
        rm1 = r;
        a += 2e0;
        p = a*pm1 + b*pm2;
        q = a*qm1 + b*qm2;
        r = p/q;                                      // 新的收敛值
    }
    return r;
}

//==============================================================================
double Expl(double x)
//------------------------------------------------------------------------------
// 由连分数表示计算 exp(x)
//------------------------------------------------------------------------------
{
    const double eps = 1e-14;                     // 相对精度
    double a, b, p, pm1, pm2, q, qm1, qm2, r, rm1;
    int i;

    a = 1e0; b = x;
    pm1 = 1e0; p = 1e0 + x;
    qm1 = 1e0; q = 1e0;
    rm1 = 1e0; r = p/q;
    i = 1;
    while (fabs(r-rm1) > eps*fabs(r)) {
        i++;
        pm2 = pm1; pm1 = p;                           // 移动堆栈
        qm2 = qm1; qm1 = q;
        rm1 = r;
        a = (i%2 ? i : 2e0);
        b = -b;
        p = a*pm1 + b*pm2;
        q = a*qm1 + b*qm2;
        r = (q ? p/q : 9e99);                 // 新的收敛值
    }
    return r;
}

double Chebyshev(int n, double x, double d)
//---------------------------------------------------
// 计算 n 阶切比雪夫多项式及其在 x 处的导数 d
// 使用递推关系
//---------------------------------------------------
{
    double f, fm1, fm2, x2;
    int i;

    if (n == 0) {
        f = 1e0; d = 0e0;
    } else {
        f = x; fm1 = 1e0; x2 = 2*x;
        for (i=2; i<=n; i++) {
            fm2 = fm1; fm1 = f;
            f = x2*fm1 - fm2;
        }
        d = (x*x-1e0) ? n*(x*f-fm1)/(x*x-1e0) : n*n*f/x;
    }
    return f;
}

//================================================================================
double Legendre(int n, double x, double &d)
//---------------------------------------------------
// 计算 n 阶勒让德多项式及其在 x 处的导数 d
// 使用递推关系
//---------------------------------------------------
{
    double f, fm1, fm2;
    int i;

    if (n == 0) {
        f = 1e0; d = 0e0;
    } else {
        f = x; fm1 = 1e0;
        for (i=2; i<=n; i++) {
            fm2 = fm1; fm1 = f;
            f = ((2*i-1)*x*fm1 - (i-1)*fm2)/i;
        }
        d = (x*x-1e0) ? n*(x*f-fm1)/(x*x-1e0) : 0.5*n*(n+1)*f/x;
    }
    return f;
}

//================================================================================
double aLegendre(int l, int m, double x)
//--------------------------------------------------------------------------------
// 计算 l 阶、m >= 0 次的连带勒让德函数在 x 处的值
//--------------------------------------------------------------------------------
{
    double p, pm1, pm2, sqx;
    int i;

    if (l < m) return 0e0;

    p = 1e0; pm1 = 0e0;                     // 种子值：P(m,m,x), P(m-1,m,x)
    if (m) {
        sqx = -sqrt(1e0-x*x);
        for (i=1; i<=m; i++) p *= (2*i-1) * sqx;
    }

    for (i=m+1; i<=l; i++) {                 // 递推
        pm2 = pm1; pm1 = p;
        p = ((2*i-1)*x*pm1 - (i+m-1)*pm2)/(i-m);
    }
    return p;
}

//================================================================================
double Laguerre(int n, double x, double &d)
//--------------------------------------------------------------------------------
// 计算 n 阶拉盖尔多项式及其在 x 处的导数 d
// 使用递推关系
//--------------------------------------------------------------------------------
{
    double f, fm1, fm2;
    int i;

    if (n == 0) {
        f = 1e0; d = 0e0;
    } else {
        f = 1e0 - x; fm1 = 1e0;
        for (i=2; i<=n; i++) {
            fm2 = fm1; fm1 = f;
            f = ((2*i-1-x)*fm1 - (i-1)*fm2)/i;
        }
        d = x ? n*(f-fm1)/x : -n*f;
    }
    return f;
}

//================================================================================
double aLaguerre(int n, int k, double x)
//--------------------------------------------------------------------------------
// 计算 n 阶 k 次连带拉盖尔多项式在 x 处的值
// 使用递推关系
//---------------------------------------------------------------------------
{
    double f, fm1, fm2;
    int i;

    if (n == 0) {
        f = 1e0;
    } else {
        f = 1e0 + k - x; fm1 = 1e0;
        for (i=2; i<=n; i++) {
            fm2 = fm1; fm1 = f;
            f = ((2*i+k-1-x)*fm1 - (i+k-1)*fm2)/i;
        }
    }
    return f;
}

//===========================================================================
double Hermite(int n, double x, double &d)
//---------------------------------------------------------------------------
// 计算 n 阶埃尔米特多项式及其在 x 处的导数 d
// 使用递推关系
//---------------------------------------------------------------------------
{
    double f, fm1, fm2, x2;
    int i;

    if (n == 0) {
        f = 1e0; d = 0e0;
    } else {
        f = 2*x; fm1 = 1e0; x2 = 2*x;
        for (i=2; i<=n; i++) {
            fm2 = fm1; fm1 = f;
            f = x2*fm1 - 2*(i-1)*fm2;
        }
        d = 2*n*fm1;
    }
    return f;
}
