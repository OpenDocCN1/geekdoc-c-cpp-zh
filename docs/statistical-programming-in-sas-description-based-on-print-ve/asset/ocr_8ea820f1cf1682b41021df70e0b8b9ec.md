;
run;
data m2c;
    input ID var1;
    datalines;
1 11

;
run;

proc sql;
    select *
        from mlc,m2c;

    select *
        from mlc,m2c
        where mlc.id=m2c.id;

    select mlc.ID,v1,v2,var1,var2,var3
        from mlc,m2c
        where mlc.id=m2c.id;

    select *
        from mlc inner join m2c
        on mlc.id=m2c.id;

    select * from mlc;
    select * from m2c;
