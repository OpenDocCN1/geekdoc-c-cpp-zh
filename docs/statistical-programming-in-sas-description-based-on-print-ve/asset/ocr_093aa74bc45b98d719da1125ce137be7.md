# Display 10.11 Macro illustrating %GOTO

```
%macro ncheck(npts);
%* npts = needs to be greater than or equal to 1;
%if %sysevalf(&npts<1) %then %do;
  %put ERROR:  '&npts' must exceed 1;
  %put ERROR:  value of '&npts' = &npts;
  %goto badend;
%end;
%put Value of '&npts' = &npts;
%badend: ;
%mend ncheck;

%ncheck(1)
%ncheck(-2)
%ncheck(0.5)
```
