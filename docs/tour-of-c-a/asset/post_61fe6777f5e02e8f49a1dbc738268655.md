```cpp
int count_x(const char* p, char x)

    // count the occurrences of x in p
    // p is assumed to point to a zero-terminated array (or nothing)
{
    if (p==nullptr)
        return 0;
    int count = 0;
    for (;p!=0; ++p)
        if (*p==x)
            ++count;
    return count;
}
```

```cpp
int count_x(const char* p, char x)
  // count the occurrences of x in p
  // p is assumed to point to a zero-terminated array (or nothing)
{
  if (p==nullptr)
    return 0;
  int count = 0;
  while (*p) {
    if (*p==x)
      ++count;
    ++p;
  }
  return count;
}
```
