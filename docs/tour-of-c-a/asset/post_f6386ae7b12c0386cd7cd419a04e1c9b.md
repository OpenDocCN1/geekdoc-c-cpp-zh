template<typename Sequence, typename Value>
Value sum(const Sequence&s, Value v)
{
    for (auto x : s)
        v+=x;
    return v;
}
```

```cpp
void user(Vector<int>& vi, list<double>& ld, vector<complex<double>>& vc)
{
    int x = sum(vi,0);                       // the sum of a vector of ints (add ints)
    double d = sum(vi,0.0);                 // the sum of a vector of ints (add doubles)
    double dd = sum(ld,0.0);                // the sum of a list of doubles
    auto z = sum(vc,complex{0.0,0.0});      // the sum of a vector of complex<double>

```cpp
template<typename T>
class Less_than {
    T val; // value to compare against
public:
    Less_than(const T& v) :val{v} {}
    bool operator()(const T& x) const { return x<val; } // call operator
};

Less_than<int> lti {42}; // lti(i) will compare i to 42
Less_than l {"string"}; // lts(s) will compare s to "string"
```

```cpp
void f(int n, const string& s)
{
    bool b1 = lti(n);      // true if n
    bool b2 = lts(s);      // true if s
}
```

```cpp
template<typename C, typename P>
int count(C& c, P pred) // assume that C is a container and P is a predicate on its elements
{
    int cnt = 0;
    for (const auto& x : c) {
        if (pred(x))
            ++cnt;
    }
    return cnt;
}
```

```cpp
void f(const Vector<int>& vec, const list<string>& lst, int x, const string& s)
{
    cout << "number of values less than " << x << ": " << count(vec,Less_than{x}) << '\n';
    cout << "number of values less than " << s << ": " << count(lst,Less_than{s}) << '\n';
}

```cpp
template<typename C, typename Oper>
void for_each(C& c, Oper op) // assume that C is a container of pointers (see §8.2.1)
{
    for (auto& x : c) // pass op a reference to each pointed pointed
        op(x);
}
```

enum Init_mode { zero, seq, cpy, part }; // initializer alternatives

void user(Init_mode m, int n, vector<int>& arg, Iterator p, Iterator q)
{
    vector<int> v;

    // messy initialization code:
    switch (m) {
    case zero:
        v = vector<int>(n); // n elements initialized to 0
        break;
    case cpy:
        v = arg;
        break;
    }

    // ...

    if (m == seq)
        v.assign(p,q); // copy from sequence [p:q]
    // ...
}
```

```cpp
void user(Init_mode m, int n, vector<int>& arg, Iterator p, Iterator q)
{
    vector<int> v = [&] {
        switch (m) {
            case zero:  return vector<int>(n); // n elements initialized to 0
            case seq:   return vector<int>{p,q}; // copy from sequence [p:q]
            case cpy:   return arg;
        }
    }();
    // ...
}

```cpp
template <class F>
struct Final_action {
    explicit Final_action(F f) :act(f) {}
    ~Final_action() { act(); }
    F act;
};
