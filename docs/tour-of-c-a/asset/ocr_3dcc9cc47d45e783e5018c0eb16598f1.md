```
template<forward_iterator Iter>
void advance(Iter p, int n)
{
    while (n--)
        ++p;
}
// move p n elements forward
// a forward iterator has ++, but not + or +==
```

```
template<random_access_iterator Iter>
void advance(Iter p, int n)
{
    p+=n;
}
// move p n elements forward
// a random-access iterator has +=
```
