    vector<double>& v; // Il function object: do something with v

    void operator()(); // Il application operator: §7.3.2

  // ...

```cpp
void consumer() {
    while(true) {
        unique_lock lck (mmutex); // acquire mutex
        mcond.wait(lck, [] { return mqueue.empty(); }); // release mutex and wait; // re-acquire mutex upon wakeup

        auto m = mqueue.front(); // don't wake up unless mqueue is non-empty
        mqueue.pop(); // get the message
        lck.unlock(); // release mutex
        // ... process m ...
    }
}
```

void producer()
{
    while(true) {
        Message m;
        // ... fill the message ...
        scoped_lock lck{mutex}; // protect operations
        mqueue.push(m);
        mcond.notify_one();
    }
}
