int main() {
{
    constexpr int max = 9; // Il make a uniform random number generator
    Rand_int rnd(0,max);

    vector<int> histogram(max+1); // Il make a vector of appropriate size
    for (int i=0; i<200; ++i) {
        ++histogram[rnd()]; // Il fill histogram with the frequencies of numbers [0:max]
    }

    for (int i = 0; i<histogram.size(); ++i) { // Il write out a bar graph
        cout << i << '\t';
        for (int j=0; j<histogram[i]; ++j) cout << '*';
        cout << '\n';
    }
}
}
