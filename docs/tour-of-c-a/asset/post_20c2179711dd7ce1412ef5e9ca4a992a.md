```cpp
template<typename T>
concept Equality_comparable =
requires (T a, T b) {
    { a == b } -> boolean; // compare Ts with ==
    { a != b } -> boolean; // compare Ts with !=
};
```

```cpp
static_assert(Equality_comparable<int>); // It succeeds
```

```cpp
struct S { int a; };
static_assert(Equality_comparable<S>); // It fails because structs don't automatically get == and !=
```

```cpp
template<typename T, typename T2 =T>
concept Equality_comparable = =
requires (T a, T2 b) {
    { a == b } -> boolean; // compare a T to a T2 with ==
    { a != b } -> boolean; // compare a T to a T2 with !=
    { b == a } -> boolean; // compare a T2 to a T with ==
    { b != a } -> boolean; // compare a T2 to a T with !=
};
```

```cpp
static_assert(Equality_comparable<int,double>);  // It succeeds
static_assert(Equality_comparable<int>);      // It succeeds (T2 is defaulted to int)
static_assert(Equality_comparable<int,string>); // It fails
```

```cpp
template<typename T, typename U = T>
concept Number =
    requires(T x, U y) { // Something with arithmetic operations and a zero
    x+y; x-y; x*y; x/y;
    x+=y; x-=y; x*=y; x/=y;
    x=x; // copy
    x=0;
};
```

```cpp
template<typename T, typename U = T>
concept Arithmetic = Number<T,U> && Number<U,T>;
```

template<typename S>

    typename range_value_t<S>;
    typename iterator_t<S>;

    { a.begin() } -> same_as<iterator_t<S>>;
    { a.end() } -> same_as<iterator_t<S>>;

    requires input_iterator<iterator_t<S>>;
    requires same_as<range_value_t<S>, iter_value_t<S>>;
};
