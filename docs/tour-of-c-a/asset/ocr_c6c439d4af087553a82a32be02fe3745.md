A suitable definition of a copy constructor for Vector allocates the space for the required number of elements and copies the elements into it so that after each copy Vector has its own copy of the elements:

```
Vector::Vector(const Vector& a) // Il copy constructor
  :elem(new double[a.sz]), // Il allocate space for elements
    sz(a.sz)
  for (int i=0; i<sz; ++i) // Il copy elements
    elem[i] = a.elem[i];
```

```
Vector&operator=(const Vector& a) // Il copy constructor
{
  double* p = new double[a.sz];
  for (int i=0; i<a.sz; ++i)
    p[i] = a.elem[i];
  delete elem; // Il delete old elements
  elem = p;
  sz = a.sz;
  return *this;
}
```
