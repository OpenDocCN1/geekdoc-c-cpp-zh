template<typename C, typename V>
vector<C>::iterator find_all(const C& c, V v)
{
    vector<C>::iterator res;
    for (auto p = c.begin();p!=c.end(); ++p)
        if (*p==v)
            res.push_back(p);
    return res;
}

template<typename C, typename V>
auto find_all(const C& c, V v)
{
    range_value_t<C>* res;
    for (auto& x : c)
        if (x==v)
            res.push_back(&x);
    return res;
}
