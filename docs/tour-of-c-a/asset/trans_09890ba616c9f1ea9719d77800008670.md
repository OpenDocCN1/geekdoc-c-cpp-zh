```cpp
constexpr double nth(double x, int n) // 假设 0<=n
{
    double res = 1;
    int i = 0;
    while (i<n) { // while 循环：当条件为真时执行 (§1.7.1)
        res *= x;
        ++i;
    }
    return res;
```

```cpp
    // 计算 x 在 p 中出现的次数
    // 假设 p 指向一个零终止数组（或为空）
{
    if (p==nullptr)
        return 0;
    int count = 0;
    for (;p!=0; ++p)
        if (*p==x)
            ++count;
    return count;
}
```

```cpp
int count_x(const char* p, char x)
  // 计算 x 在 p 中出现的次数
  // 假设 p 指向一个零终止数组（或为空）
{
  if (p==nullptr)
    return 0;
  int count = 0;
  while (*p) {
    if (*p==x)
      ++count;
    ++p;
  }
  return count;
}
```
