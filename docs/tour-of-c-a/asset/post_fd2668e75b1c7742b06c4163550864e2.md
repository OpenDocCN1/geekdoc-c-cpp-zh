```cpp
template<typename C>
class Checked_iter {
public:
    using value_type = typename C::value_type;
    using difference_type = int;

    Checked_iter() { throw(); // forward_iterator requires a default constructor
    Checked_iter(const C& cc) : pc(&cc), p(cc.begin()) {}
    Checked_iter(const Checked_iter& cc) : pc(&cc), p(cc) {}

    Checked_iter& operator++() { check_end(); ++p; return *this; }
    Checked_iter& operator++(int) { check_end(); auto t = *this; ++p; return t; }
    value_type& operator+=(const&) { check_end(); return *p; }

    bool operator==(const Checked_iter& a) const { return p==a; }
    bool operator!=(const Checked_iter& a) const { return p!=a; }
private:
    void check_end() const { if (p == pc->end()) throw(); }
    C* pc; // default initialize to nullptr
    typename C::iterator p = pc->begin();
};
```

{
    vector<typename C::value_type*> res;
    for (auto& x : c) {
        if (x==v)
            res.push_back(&x);
