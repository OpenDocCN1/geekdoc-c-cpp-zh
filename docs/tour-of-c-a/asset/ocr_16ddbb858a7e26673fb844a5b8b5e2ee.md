int main()
{
    string from, to;
    cin >> from >> to;

    if (ifstream is{from}) {
        istream_iterator<string> ii{is};
        istream_iterator<string> eos {};

        if (ofstream os{to}) {
            ostream_iterator<string> oo {os, "\n"};

            vector<string> b {ii,eos};
            sort(b);

            unique_copy(b, oo);

            return !is.eof() || !os;
        }
    }
    return 1; // return error state
}
