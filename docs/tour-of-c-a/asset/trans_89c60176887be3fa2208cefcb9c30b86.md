```cpp
{
    vector<double> some_vec {1, 2, 3, 4, 5, 6, 7, 8, 9};
```

```cpp
void user(vector<double>& vec1, vector<double> vec2, vector<double> vec3)

    double res1;
    double res2;
    double res3;

    thread t1 (f,std::cref(vec1),&res1);
    thread t2 (F(vec2,&res2));
    thread t3 [&](){ res3 = g(vec3); };

    t1.join(); // Il 在使用结果前连接
    t2.join();
    t3.join();

    cout << res1 << " " << res2 << " " << res3 << '\n';
}
```
