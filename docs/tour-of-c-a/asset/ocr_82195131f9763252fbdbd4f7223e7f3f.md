void print_directory(path p) // print the names of all files in p
try
{
    if (is_directory(p)) {
        cout << p << ":\n";
        for (const directory_entry& x : directory_iterator{p}) {
            cout << " " << x.path() << '\\n';
        }
    }
}
catch (const filesystem_error& ex) {
    cerr << ex.what() << '\\n';
}
