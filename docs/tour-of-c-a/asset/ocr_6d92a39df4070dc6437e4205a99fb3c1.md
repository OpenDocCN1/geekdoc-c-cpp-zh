```
struct Record {
    string name;
    int product_code;
    // ...
};
```

```
struct Rhash { // a hash function for Record
    size_t operator()(const Record& r) const
    {
        return hash<string>()(r.name) ^ hash<int>()(r.product_code);
    }
};
```

unordered_set<Record,Rhash> my_set; // set of Records using Rhash for lookup
