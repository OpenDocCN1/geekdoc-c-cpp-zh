namespace System
{
    public class Object
    {
        public Object();
        ~Object();

        public virtual bool Equals(Object? obj);
        public virtual int GetHashCode();
        public Type GetType();
        public virtual string? ToString();
        protected Object MemberwiseClone();
    }
}
