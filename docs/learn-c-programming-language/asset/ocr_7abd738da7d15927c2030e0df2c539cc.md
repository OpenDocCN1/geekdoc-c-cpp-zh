// overloaded postfix ++ operator
Time operator++( int )
{
    // save the original value
    Time T(hours, minutes);
    // increment this object
    ++minutes;
    if(minutes >= 60)
    {
        ++hours;
        minutes -= 60;
    }
    // return old original value
    return T;
}
};
int main()
{
    Time T1(11, 59), T2(10,40);

    ++T1;                  // increment T1
    T1.displayTime();      // display T1
    ++T1;                  // increment T1 again
    T1.displayTime();      // display T1

    T2++;                  // increment T2
