movaps 16(%rdx), %mm8
movaps 48(%rdx), %mm9
movaps 80(%rdx), %mm10
movaps 112(%rdx), %mm11
#mult 4x14
movaps (%rdi), %mm2
movaps (%rsi), %mm0

movaps %mm3, %mm12
mulpd %mm0, %mm3
addp %mm2, %mm4
addp %mm3, %mm8
movaps 16(%rdi), %mm1
mulpd %mm1, %mm12
movaps (%rdi), %mm2
mulpd %mm1, %mm2

movaps (%rdx), %mm3
shufd $1, %mm3, %mm3
movaps %mm3, %mm2

mulpd %mm0, %mm2
addp %mm3, %mm7
addp %mm2, %mm5
movaps 16(%rdi), %mm2
shufd $1, %mm2, %mm2
movaps %mm2, %mm2
mulpd %mm0, %mm2
mulpd %mm1, %mm3
