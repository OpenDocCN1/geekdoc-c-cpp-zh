of an open file, satisfied.

The definition of read_mpi() closely parallels that of write_mpi().

```
void read_mpi(void *data, long len,
    char *fname, long disp{
    MPI_File ifile;
    MPI_File_open(MPI_COMM_WORLD,
    fname,
    MPI_MODE_RDONLY,
    MPI_INFO_NULL,
    &ifile);
    char datarep[200];
    printf(datarep, "native");
    MPI_File_set_view(ifile,
    disp,
    MPI_BYTE,
    MPI_BYTE,
    datarep,
    MPI_INFO_NULL);
    long offset = 0;
    int maxcount = 2000*1000*1000;
    while (len > 0){
        int count = (len<maxcount)?len:maxcount;
        MPI_File_read_at(ifile,
        offset,
        data,
        count,
        MPI_BYTE,
        MPI_STATUS_IGNORE);
        offset += count;
        data = ((char*)data+count);
        len -= count;
    }
    MPI_File_close(&ifile);
```
