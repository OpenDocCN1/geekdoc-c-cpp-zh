{
    phrase = str;
    count++;
}

//------------------------------------------------------------
// Returns this slogan as a string.
//------------------------------------------------------------
public String toString()
{
    return phrase;
}

//------------------------------------------------------------
// Returns the number of instances of this class that have been
// created.
//------------------------------------------------------------

{
    return count;
}
}
