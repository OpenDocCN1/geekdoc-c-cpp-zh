struct Iterator {
    handle_type coro;
    Iterator(auto p) : coro(p) { }
    void getNext() {
        if (coro) {
            coro.resume();
            if (coro.done()) {
                coro = nullptr;
            }
        }
    }
    T operator*() const {
        return coro.promise().coroValue;
    }
    Iterator operator++() {
        getNext();
        return *this;
    }
    bool operator==(const Iterator& i) const = default;
};

Iterator begin() const {
    if (!coro || coro.done()) {
        return Iterator(nullptr);
    }
    Iterator itor{coro};
    itor.getNext();
    return itor;
}
