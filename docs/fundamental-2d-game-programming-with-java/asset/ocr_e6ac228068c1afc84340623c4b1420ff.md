private void drawPolygon( Graphics g, Vector2f[] polygon ) {
Vector2f P;
Vector2f S = polygon[ polygon.length - 1 ];
for (int i = 0; i < polygon.length; ++i ) {
P = polygon[i];
g.drawLine( (int)S.x, (int)S.y, (int)P.x, (int)P.y );
S = P;
}
