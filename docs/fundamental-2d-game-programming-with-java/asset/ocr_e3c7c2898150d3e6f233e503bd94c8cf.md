@Overrride
public void loop( int count ) {
    lock.lock();
    try {
        clip.flush();
        clip.setFramePosition( 0 );
        clip.loop( count );
        while( !started ) {
            cond.await();
        }
        System.out.println( "It's Started" );
    } catch( InterruptedException e ) {
        e.printStackTrace();
    } finally {
        lock.unlock();
    }
}

@Override
public void restart() {
    restart = true;
    stop();
    restart = false;
    start();
}

@Override
protected void fireTaskFinished() {
    if( !restart ) {
        super.fireTaskFinished();
    }
}
