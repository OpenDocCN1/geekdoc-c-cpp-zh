private Vector2f calculateReflection(
Vector2f V, Vector2f P0, Vector2f P1 ) {
Vector2f Pv = P0.sub( P1 );
Vector2f n = Pv.perp().norm();
Vector2f Vn = n.mul( V.dot( n ) );
Vector2f Vp = V.sub( Vn );
return Vp.sub( Vn );
}
