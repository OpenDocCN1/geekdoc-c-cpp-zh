private Vector2f[] transform( Vector2f[] poly, Matrix3x3f mat ) {
	Vector2f[] copy = new Vector2f[ poly.length ];
	for( int i = 0; i < poly.length; ++i ) {
		copy[i] = mat.mul( poly[i] );
	}
	return copy;
}
