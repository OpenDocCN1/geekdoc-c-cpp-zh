public void run() {
    running = true;
    initializing();
    long curTime = System.nanoTime();
    long lastTime = curTime;
    double nsPerFrame;
    while (running) {
        curTime = System.nanoTime();
        nsPerFrame = curTime - lastTime;
        gameLoop(nsPerFrame / 1.0E9);
        lastTime = curTime;
    }
}
