protected void shutDown() {
    if ( Thread.currentThread() != gameThread ) {
        try {
            running = false;
            gameThread.join();
            onShutDown();
        } catch ( InterruptedException e ) {
            e.printStackTrace();
        }
    } else {
        System.exit( 0 );
    }
    SwingUtilities.invokeLater( new Runnable() {
        public void run() {
            shutDown();
        }
    });
}
