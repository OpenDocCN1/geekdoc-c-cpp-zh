private void resetHeldKey() {
    last = System.nanoTime();
    heldDown = -STEP;
}
private boolean updateHeldKey() {
    long current = System.nanoTime();
    heldDown += current - last;
    last = current;
    if ( heldDown > STEP ) {
        heldDown -= STEP;
        return true;
    }
    return false;
}
