Vector2f largestMin = Float.MAX_VALUE;
float smallestMax = Float.MAX_VALUE;

float f = n.dot( d );

swap = t0;
t0 = t1;
t1 = swap;
}
largestMin = Math.max( largestMin, t0 );
smallestMax = Math.min( smallestMax, t1 );

}

}
