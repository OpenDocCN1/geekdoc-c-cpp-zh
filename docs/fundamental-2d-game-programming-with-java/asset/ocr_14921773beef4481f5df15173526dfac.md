public synchronized void poll() {
    if ( isRelative() ) {
        mousePos = new Point( dx, dy );
    } else {
        mousePos = new Point( currentPos );
    }
    dx = dy = 0;
    polledNotches = notches;
    notches = 0;
    for ( int i = 0; i < mouse.length; ++i ) {
        if ( mouse[i] ) {
            polled[i]++;
        } else {
            polled[i] = 0;
        }
    }
}
