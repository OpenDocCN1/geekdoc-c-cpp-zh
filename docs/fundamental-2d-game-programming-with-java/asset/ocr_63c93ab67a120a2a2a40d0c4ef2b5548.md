@Override
protected void initialize() {
    super.initialize();

    radius = 0.125f;
    center = new Vector2f();

    p0 = new Vector2f( -0.65f, -0.2f );
    p1 = new Vector2f( 0.50f, 0.2f );
}

@Override
protected void processInput( float delta ) {
    super.processInput( delta );
    center = getWorldMousePosition();
}

@Override
protected void updateObjects( float delta ) {
    super.updateObjects( delta );

    Vector2f d = p1.sub( p0 );
    float len = d.len();
    d = d.norm();

    plane0 = plane1 = null;
    segment0 = segment1 = null;
    t0 = t1 = null;

    float[] intersections =
        lineCircleIntersection( p0, d, center, radius );
    if( intersections != null ) {
        t0 = intersections[0];
        plane0 = p0.add( d.mul( t0 ) );
        if( t0 > 0.0f && t0 < len ) {
            segment0 = plane0;
        }
    }

    t1 = intersections[1];
    plane1 = p0.add( d.mul( t1 ) );
    if( t1 > 0.0f && t1 < len ) {
        segment1 = plane1;
    }
}
}
