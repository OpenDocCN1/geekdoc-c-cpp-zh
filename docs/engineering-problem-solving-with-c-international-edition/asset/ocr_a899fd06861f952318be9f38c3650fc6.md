/*
* This function determines the number of positive,
* negative, and zero values in an array.
*/
void signs(int x[], int npts, int *npos,
           int *nzero, int *nneg)
{
    /* Declare variables. */
    int k;

    /* Set sums to zero. */
    *npos = *nzero = *nneg = 0;

    /* Update corresponding sums. */
    for (k=0; k<=npts-1; k++)
    {
        if (x[k] < 0)
            *nneg += 1;
        else
            if (x[k] > 0)
                *npos += 1;
            else
                *nzero += 1;
    }

    /* Void return. */
    return;
}
