/* This function computes the average power in a specified */

double power_w(double *ptr, int n)
{
    /* Declare and initialize variables. */
    int k;
    double xsquare=0;

    /* Compute sum of values squared in the array x. */
    for (k=0; k<=n-1; k++)
        xsquare += *(ptr+k)*(*(ptr+k));

    /* Return the average squared value. */
    return xsquare/n;
}
