Additional functions

ave_power(x, npts):
  set sum to zero
  set k to zero
  while k < npts - 1
    add x[k]^2 to sum
    increment k by 1
  return sum/npts

ave_magn(x, npts):
  set sum to zero
  set k to zero
  while k < npts - 1
    add |x[k]| to sum
    increment k by 1
  return sum/npts

crossings(x, npts):
  set count to zero
  set k to zero
  while k < npts - 2
    if x[k] * x[k + 1] < 0
      increment count by 1
    increment k by 1
  return count
