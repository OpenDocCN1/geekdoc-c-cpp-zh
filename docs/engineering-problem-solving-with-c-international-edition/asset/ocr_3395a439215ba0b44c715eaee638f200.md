```
back_substitute(a,n,soln):
    set soln[n - 1] to   -a[n-1][n]
                           -----------
                           a[n-1][n-1]
    set row to n - 2
    while row >= 0
        set col to n - 1
        while col >= row + 1
            subtract soln[col] * a[row][col]
            from a[row][n]
        subtract 1 from col
        set soln[row] to   a[row][n]
                           -----------
                           a[row][row]
        subtract 1 from row
```
