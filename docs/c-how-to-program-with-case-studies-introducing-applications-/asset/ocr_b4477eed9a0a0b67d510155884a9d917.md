    printf("The result is %d\n", mystery(x, y));
}

// Parameter b must be a positive integer
// to prevent infinite recursion
int mystery(int a, int b) {
    // base case
    if (1 == b) {
        return a;
    }
    else { // recursive step
        return a + mystery(a, b - 1);
    }
}
