const int *find(const int arr[])
{
    int i, j, cnt, max, pos;

    max = 0;
    for(i = 0; i < SIZE; i++)
{
        cnt = 0;
        for(j = i+1; j < SIZE; j++)
        {
            if(arr[i] == arr[j])
                cnt++;
        }
        if(cnt > max)
        {
            max = cnt;
            pos = i;
        }
    }
    if(max == 0)
        return nullptr;
    else
        return arr+pos;
}
