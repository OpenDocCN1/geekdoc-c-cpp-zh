    if(!fin) {
        fin.close();
        cout << "Error: read() failed\n";
        exit(EXIT_FAILURE);
    }

    grd_arr = new float[grd_num]; /* Allocate memory to store the grades. */
    fin.read((char*)grd_arr, grd_num * sizeof(float)); /* Read all grades and check if they are read successfully. */
    if(fin) {
        cout << "Enter grade: ";
        cin >> grd;
        for(i = 0; i < grd_num; i++)
            if(grd_arr[i] > grd)
                cout << grd_arr[i] << '\n';
    }
    else {
        cout << "Error: read(0) failed to read grades\n";
    }

    delete[] grd_arr;
    fin.close();
    return 0;
}
