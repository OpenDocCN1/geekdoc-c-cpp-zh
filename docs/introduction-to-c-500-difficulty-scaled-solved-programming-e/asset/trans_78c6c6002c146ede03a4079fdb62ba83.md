- 1. 在列表末尾插入一个学生。程序应读取学生数据并存储在节点中，该节点将成为列表的尾。
- 2. 在指定位置插入学生。程序应读取学生代码，找到与该代码对应的节点，并在之后创建一个新节点以插入新学生的数据。
- 3. 显示已存储学生的数据。
- 4. 查找学生。程序应读取学生代码，如果已注册，则显示其数据。
- 5. 修改学生的成绩。程序应读取学生代码和新成绩，并修改现有成绩。
- 6. 删除学生。程序应读取学生代码，如果已注册，则移除与该学生对应的节点。
- 7. 程序终止。

void add_list_end(const Node *p) {
    Node *new_node, *ptr;

    new_node = new Node(*p);
    new_node->next = nullptr;

    if (head == nullptr) {
        head = tail = new_node;
    } else {
        ptr = head;
        while (ptr->next != nullptr) {
            ptr = ptr->next;
        }
        ptr->next = new_node;
        tail = new_node;
    }
}

void add_list(const Node *p, int code) {
    Node *new_node, *ptr;

    new_node = new Node(*p);
    new_node->code = code;
    new_node->next = nullptr;

    if (head == nullptr) {
        head = tail = new_node;
    } else {
        ptr = head;
        while (ptr->next != nullptr && ptr->next->code != code) {
            ptr = ptr->next;
        }
        new_node->next = ptr->next;
        ptr->next = new_node;
        if (new_node->next == nullptr) {
            tail = new_node;
        }
    }
}

void show_list() {
    Node *p;
    p = head;
    while (p != nullptr) {
        cout << "Code: " << p->code << " Name: " << p->name << " Grade: " << p->grd << "\n";
        p = p->next;
    }
}

Node *find_node(int code) {
    Node *p;
    p = head;
    while (p != nullptr) {
        if (p->code == code)
            return p;
        p = p->next;
    }
    return nullptr;
}

int del_node(int code) {
    Node *p, *prev_node;
    p = head;
    while (p != nullptr) {
        if (p->code == code) {
            if (p == head) {
                head = p->next;
                if (head == nullptr) {
                    tail = nullptr;
                }
            } else {
                prev_node->next = p->next;
                if (p->next == nullptr) {
                    tail = prev_node;
                }
            }
            delete p;
            return 0;
        }
        prev_node = p;
        p = p->next;
    }
    return -1;
}

void free_list() {
    Node *p, *next_node;
    p = head;
    while (p != nullptr) {
        next_node = p->next;
        delete p;
        p = next_node;
    }
    head = tail = nullptr;
}
