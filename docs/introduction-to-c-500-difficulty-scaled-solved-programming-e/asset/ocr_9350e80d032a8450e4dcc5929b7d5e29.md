C.14.5 Write a function that takes two arrays of doubles (e.g., a1 and a2) of the same size and their number of elements, allocates memory to store the elements of a1 that are not contained in a2. The function should return a pointer to memory. Write a program that reads pairs of doubles and stores them into two arrays of 100 elements (e.g., p1 and p2). If either input is -1, the insertion of numbers should end. The program should use the function to display the elements of p1 that are not contained in p2.

```
#include <iostream>
#include <string>
#include <stdlib>

using std::cout;
using std::cin;

double *find_diff(const double a1[],const double a2[],int size, int *items);
/* items indicates how many elements are stored in the memory. A pointer is passed, so that the function may change its value. */

int main()
{
    const int SIZE = 100;
    int i, elems;
    double *p3, j, k, p1[SIZE], p2[SIZE];

    for(i = 0; i < SIZE; i++)
    {
        cout << "Enter numbers: ";
        cin >> j >> k;
        if(j == -1) || (k == -1))
            break;
        p1[i] = j;
        p2[i] = k;
    }
    elems = 0;
    p3 = find_diff(p1, p2, &elems);
    if(elems == 0)
        cout << "\n*** No different elements *****\n";
    else
    {
        for(i = 0; i < elems; i++)
            cout << p3[i] << '\n';
    }
    delete[] p3;
    return 0;
}
```
