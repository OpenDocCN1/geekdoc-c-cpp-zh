template <typename T> void Sort_Arr<T>::bubble_sort()
{
  bool reorder;
  int i, j;
  T tmp;

  for(i = 1; i < size; i++)
  {
    reorder = 0;
    for(j = size-1; j >= i; j--)
    {
      if(p[j] < p[j-1])
      {
        // Swap p[j]
        tmp = p[j];
        p[j] = p[j-1];
        p[j-1] = tmp;
        reorder = 1;
      }
    }
    if(reorder == 0)
      return;
  }
}
