```
    p1->show(); /* It should display the code of the Product object, that is,

    p2->show(); /* It should display the code and the title of the Book
```

Answer:

```
#include <iostream>
#include <string>
using std::cout;
using std::string;
```

```
class Product
{
protected:
    int code;
public:
    Product(int c);
    virtual -Product() {}; // Not needed, just to get used.
    virtual void show() const;
}
```

```
class Book : public Product
{
private:
    string title;
public:
    Book(string t, int c);
    virtual void show() const override;
}
```

```
void f(const Product *p1, const Product *p2);
```

```
Product::Product(int c)
{
    code = c;
}
```

```
void Product::show() const
{
    cout << "C:" << code << '\\n';
}
```

```
Book::Book(string t, int c) : Product(t)
{
    title = t;
}
```

```
void Book::show() const
{
    cout << "C:" << code << "T:" << title << '\\n'; /* Since the code is a
protected member we can access it. */
}
```
