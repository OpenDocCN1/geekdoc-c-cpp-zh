C.11.7 Write a function that takes an integer (e.g., n) and returns the value of: $1^3 + 2^3 + 3^3 + ... + n^3$. Write a program that reads a positive integer (suppose that the user enters a valid value) and uses the function to display the square root of that expression. Verify that the output value is equal to $n*(n+1)/2$. For the square root write another function using the Newton method to find the root of the argument (e.g., n). This method uses initial guess $r$ for the square root (e.g., n/2) and computes the sequence of values according to $r_{k+1}=0.5*(r_k + n/r_k)$. If the absolute difference between the new $r_k$ and the old $r$ is less than a tolerance (e.g., 0.0001), then the function should return the value of $r_k$, as the root. For the difference use the `fabs()` library function.

```
#include <iostream>
#include <cmath>

unsigned long int sum_cube(unsigned long int n);
double find_root(unsigned long int n);

int main() {
    int num;

    std::cout << "Enter number: ";
    std::cin >> num;

    std::cout << find_root(sum_cube(num)) << ' ' ' << num*(num+1)/2 << '\n';
    return 0;
}

unsigned long int sum_cube(unsigned long int n) {
    int i;
    unsigned long int sum;

    sum = 0;
    for (i = 1; i <= n; i++)
        sum += i*i*i;
    return sum;
}

double find_root(unsigned long int n) {
    double old_r, new_r;

    old_r = n/2.0;
    while (1) {
        new_r = 0.5*(old_r + n/old_r);
        if (fabs(new_r - old_r) < 0.0001)
            break;
        old_r = new_r;
    }
    return new_r;
}
```
