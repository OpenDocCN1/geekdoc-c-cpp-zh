```
Time& Time::inc_min()
{
    min++;
    if(min == 60) {
        min = 0;
        hr++;
    }
    return *this;
}
```

```
Time& Time::inc_sec()
{
    sec++;
    if(sec == 60) {
        sec = 0;
        inc_min();
    }
    return *this;
}
```

void Time::show() const
{
    std::cout << hr << ': ' << min << ': ' << sec << '\n';
}
