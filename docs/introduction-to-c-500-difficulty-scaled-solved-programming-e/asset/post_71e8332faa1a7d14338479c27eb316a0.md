void add_list_end(const Node *p) {
    Node *new_node, *ptr;

    new_node = new Node(*p);
    new_node->next = nullptr;

    if (head == nullptr)
        head = tail = new_node;
    else {
        ptr = head;
        while (ptr->next != nullptr) {
            ptr = ptr->next;
        }
        ptr->next = new_node;
        tail = new_node;
    }
}

void add_list(const Node *p, int code) {
    Node *new_node, *ptr;

    new_node = new Node(*p);
    new_node->code = code;
    new_node->next = head;

    if (head == nullptr)
        head = tail = new_node;
    else {
        ptr = head;
        while (ptr->next != nullptr && ptr->next->code != code) {
            ptr = ptr->next;
        }
        new_node->next = ptr->next;
        ptr->next = new_node;
        if (new_node->next == nullptr) {
            tail = new_node;
        }
    }
}

Node *find_node(int code) {
    Node *p;
    p = head;
    while (p != nullptr) {
        if (p->code == code)
            return p;
        p = p->next;
    }
    return nullptr;
}
