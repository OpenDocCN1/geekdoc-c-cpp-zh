void change(string& s, int& cnt)
{
    int i = 0;
    for(i = 0; i < s.length(); i++)
    {
        if(s[i] >= 'a' && s[i] <= 'z')
            s[i] -= 32;
        else if(s[i] >= 'A' && s[i] <= 'Z')
            s[i] += 32;
        else if(s[i] >= '0' && s[i] <= '9')
            cnt++;
    }
}
