void Account::deposit(double amt) {
    if (amt < 0)
        throw Err_Rpt("Error: Not acceptable amount to deposit\n");
    bal += amt;
}

void Account::withdraw(double amt) {
    if (amt < 0 || amt > bal)
        throw Err_Rpt("Error: Not acceptable amount to withdraw\n");
    bal -= amt;
}
