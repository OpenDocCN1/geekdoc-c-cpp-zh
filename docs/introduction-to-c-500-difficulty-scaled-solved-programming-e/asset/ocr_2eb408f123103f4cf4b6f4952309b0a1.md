7.16 Write a program that reads the number of students and the number of courses and creates a twimensional vector object. Then, the program should read the grades of the students and store them in the object. The program should display the average, the best, and worst grade of each student. It should display the positions in the object which holds the best and worst average grades. If more than one student has the same best or worst average grade, the program should display the first position. The program should force the user to enter grades within [0, 10] and positive values for the number of students and the courses.

```
#include <iostream>
#include <vector>
using std::cout;
using std::cin;
using std::vector;

int main() {
    int studs, courses, i, min_pos, max_pos;
    float sum, min_grd, max_grd, avg_grd, min_avg_avg_grd;
    do {
        cout << "Enter number of students: ";
        cin >> studs;
        cout << "Enter number of courses: ";
        cin >> courses;
    } while (studs <= 0 || courses <= 0);

    vector<vector<float>> grd(studs, vector<float>(courses));
    max_avg_grd = -1;
    min_avg_grd = 11;
    for(i = 0; i < studs; i++) {
        sum = 0;
        max_grd = -1;
        min_grd = 11;
        for(j = 0; j < courses; j++) {
            cout << "Enter grade of student " << i+1 << " for lesson " << j+1 << ": ";
            cin >> grd[i][j];
            while (grd[i][j] < 0 || grd[i][j] > 10);
            sum += grd[i][j];
            if(grd[i][j] > max_grd)
                max_grd = grd[i][j];
            if(grd[i][j] < min_grd)
                min_grd = grd[i][j];
        }
        avg_grd = sum/courses;
        if(avg_grd > min_avg_avg_grd) {
            max_avg_grd = avg_grd;
            max_pos = i;
        }
    }
```
