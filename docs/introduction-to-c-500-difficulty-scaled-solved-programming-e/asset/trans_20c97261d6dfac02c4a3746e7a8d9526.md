```cpp
template <typename T> void sort(vector<T>& v)
{
    int i, j, size;
    T tmp; // 每个类的构造函数都被调用；

    size = v.size();
    for(i = 0; i < size; i++)
    {
        for(j = i+1; j < size; j++)
        {
            if(v[i] > v[j]) /* 当比较元素时，每个类重载的 > 运算符函数会被调用 */
            {
                tmp = v[i];
                v[i] = v[j];
                v[j] = tmp;
            }
        }
    }
}
```
