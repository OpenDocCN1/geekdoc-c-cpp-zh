C.14.3 编写一个与 `memcmp()` 类似的函数。该程序应该读取两个小于 100 个字符的字符串、要比较的字符数，并使用该函数显示比较结果。

```cpp
#include <iostream>
#include <string>
using std::cout;
using std::cin;

int mem_cmp(const void *ptr1, const void *ptr2, size_t size);

int main()
{
    char str1[100], str2[100];
    int num;

    cout << "Enter first text: ";
    cin.getline(str1, sizeof(str1));

    cout << "Enter second text: ";
    cin.getline(str2, sizeof(str2));

    cout << "Enter the characters to compare: ";
    cin >> num;

    cout << mem_cmp(str1, str2, num) << '\n';
    return 0;
}

int mem_cmp(const void *ptr1, const void *ptr2, size_t size)
{
    char *p1, *p2;
    p1 = (char*)ptr1;
    p2 = (char*)ptr2;
    while (size != 0)
    {
        if(*p1 != *p2)
            return *p1 - *p2;
        p1++;
        p2++;
        size--;
    }
    return 0;
}
```

注释：循环比较 `p1` 和 `p2` 指向的字符。由于比较的是字符，我们将 `void*` 类型强制转换为 `char*`。如果所有字符都相同，`mem_cmp()` 返回 0，否则返回两个不匹配字符的差值。

C.14.4 编写一个函数，接受三个字符串指针作为参数，并将后两个字符串存储到第一个字符串中。第一个字符串的内存应该是动态分配的。编写一个程序读取两个小于 100 个字符的字符串，并使用该函数将它们存储在动态分配的内存中。

```cpp
#include <iostream>
#include <cstring>
using std::cout;
using std::cin;

void str_cat(char *fin, const char *str1, const char *str2);

int main() {
    char *fin, str1[100], str2[100];
    int len1, len2;

    cout << "Enter first text: ";
    cin.getline(str1, sizeof(str1));
    len1 = strlen(str1);

    cout << "Enter second text: ";
    cin.getline(str2, sizeof(str2));
    len2 = strlen(str2);
    // 分配内存存储两个字符串和空字符。
    fin = new char[len1+len2+1];
    str_cat(fin, str1, str2);
    cout << "Merged text: " << fin << '\n';
    delete[] fin;
    return 0;
}

void str_cat(char *fin, const char *str1, const char *str2) {
    while(*str1 != '\0') // 或者使用，while(*str1)
        *fin++ = *str1++;
    // 在第一个字符串之后拷贝第二个字符串。
    while(*str2 != '\0') // 或者使用，while(*str2)
        *fin++ = *str2++;
    *fin = '\0'; // 添加空字符。
}
```

注释：第一个循环将 `str1` 指向字符串的字符拷贝到 `fin` 指向的内存中。同样，下一个循环添加第二个字符串的字符。现在我们不再使用 `while` 循环，而是使用 `for` 循环。例如：

```cpp
void str_cat(char *fin, const char *str1, const char *str2) {
    for(;*str1;)
        *fin++ = *str1++;
    for(;*str2;)
        *fin++ = *str2++;
    *fin = '\0';
}
```

C.14.5 编写一个函数，接受两个相同 `double` 数组（例如 `a1` 和 `a2`）以及它们的元素数量，分配内存存储 `a1` 中不包含在 `a2` 中的元素。该函数应返回指向内存的指针。编写一个程序读取 `double` 对并将它们存储在两个包含 100 个元素的数组中（例如 `p1` 和 `p2`）。如果任一个输入为 -1，则停止插入数字。程序应使用该函数显示 `p1` 中不包含在 `p2` 中的元素。

```cpp
#include <iostream>
#include <string>
#include <stdlib>

using std::cout;
using std::cin;

double *find_diff(const double a1[], const double a2[], int size, int *items);
/* items 表示内存中存储了多少个元素。传递一个指针，以便函数可以更改其值。 */

int main()
{
    const int SIZE = 100;
    int i, elems;
    double *p3, j, k;
    double p1[SIZE], p2[SIZE];

    for(i = 0; i < SIZE; i++)
    {
        cout << "Enter numbers: ";
        cin >> j >> k;
        if(j == -1 || k == -1)
            break;
        p1[i] = j;
        p2[i] = k;
    }
    elems = 0;
    p3 = find_diff(p1, p2, SIZE, &elems);
    if(elems == 0)
        cout << "No different elements\n";
    else
    {
        for(i = 0; i < elems; i++)
            cout << p3[i] << '\n';
    }
    delete[] p3;
    return 0;
}

double *find_diff(const double a1[], const double a2[], int size, int *items)
{
    int i, j, found;
    double *mem = nullptr;

    for(i = 0; i < size; i++)
    {
        found = 0;
        for(j = 0; j < size; j++)
        {
            if(a1[i] == a2[j])
            {
                found = 1;
                break;
            }
        }
        if(!found)
        {
            if(mem == nullptr)
            {
                mem = new double[size];
            }
            mem[*items] = a1[i];
            (*items)++;
        }
    }
    return mem;
}

    {
        cout << "Enter text: ";
        cin.getline(str, sizeof(str));
        len = strlen(str);
        // 分配内存。
        ptr[i] = new char[len+1];

        {
            pos = i;
            max_len = len;
        }
    }
    cout << "Longer string: " << ptr[pos] << '\n';

    return 0;
}
