#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Student {
private:
    string name;
    float grd;
public:
    Student(const string& n = "", float g = 0);
    friend void sort(vector<Student>& v);
    void show() const;
};

```cpp
Student::Student(const string& n, float g) {
    name = n;
    grd = g;
}

void Student::show() {
    cout << "N:" << name << " G:" << grd << '\n';
}

void sort(vector<Student>& v) {
    int i, j, size;
    Student tmp;

    size = v.size();
    for (i = 0; i < size-1; i++) {
        for (j = i+1; j < size; j++) {
            if (v[i].grd > v[j].grd) {
                tmp = v[i];
                v[i] = v[j];
                v[j] = tmp;
            }
        }
    }
}
```

```cpp
int main() {
    int i;
    float grd;
    string name;
    vector<Student> st;

    while(1)
    {
        cout << "\nName: ";

        if(grd == -1)
            break;

        Student tmp(name, grd);

    }
    sort(st);
    for(i = 0; i < st.size(); i++)
        st[i].show();
    return 0;
}
```

**注释：** 我们需要一个构造函数来创建对象。由于对三个函数的限制，`sort` 函数必须声明为友元（friend）以便访问私有成员。

答案：在 `network.h` 头文件中放置类的声明。

```cpp
#ifndef NETWORK_H
#define NETWORK_H

#include <vector>
using std::vector;

const int MAX_NODES = 10;

class Node {
private:
    int id;
    int x, y;
    vector<int> nei_id; // 存储邻居的标识符
    vector<double> nei_dist; // 存储邻居的距离
public:
    Node(int id, int x, int y);
    void get_coord(int& x, int& y) const;
    void set_nei(int n_id, double dist);
    void show_nei() const;
};

#endif
```

```cpp
Network::Network()
{
    srand(time(NULL));
    for (int i = 0; i < MAX_NODES; i++)
        node.push_back(Node(i, rand() % 1000, rand() % 1000));
}
```

```cpp
void Network::find_nei()
{
    int i, j, x1, x2, y1, y2;
    double dist;

    for (i = 0; i < MAX_NODES; i++) {
        node[i].get_coord(x1, y1);
        for (j = i+1; j < MAX_NODES; j++) {
            node[j].get_coord(x2, y2);
            dist = sqrt(pow(x1-x2, 2) + pow(y1-y2, 2)); /* 应用勾股定理计算距离 */
            if(dist < 200) { // 存储邻居
                node[i].set_nei(j, dist);
                node[j].set_nei(i, dist);
            }
        }
    }
}
```

```cpp
void Network::show_topology() const
{
    for(int i = 0; i < MAX_NODES; i++)
        node[i].show_nei();
}
```

int main() {

    net = new Network();
    net->find_nei();
    net->show_topology();
    delete net;
    return 0;
}
