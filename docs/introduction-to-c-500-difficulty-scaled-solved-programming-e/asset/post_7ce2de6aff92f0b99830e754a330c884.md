```cpp
void add_stack(const Node *p)
{
    Node *new_node;
    new_node = new Node; // Allocate memory to create a new node.
    *new_node = *p; // Copy the student's data into the new node.
    new_node->next = head; // The new node is inserted at the beginning
of the stack. For example, when the first node is inserted the value of
new_node->next becomes equal to the initial value of head, which is
nullptr.
    head = new_node; // head points to the new node, therefore the new
node becomes the new head of the stack.
}
```

```cpp
void show_stack()
{
    Node *p;

    p = head;
    cout << "\n**** Student Data *****\n";
    while(p != nullptr)
    {
        cout << "C:" << p->code << " N:" << p->name << " G:" << p->grd
        << "\n";
        p = p->next; /* In each iteration, p points to the next node.
Once it becomes nullptr it means that there is no other node left and the
loop terminates. */
    }
}
```

```cpp
void pop()
{
    Node *p;

    p = head->next; /* p points to the next node after the head. This node
will be the new head of the stack. */
    cout << "\nStudent with code " << p->code << " is deleted\n";
    delete head; /* Release the allocated memory. The information about
the next node is not lost, because it was saved in p. */
    head = p; // head points to the new head of the stack.
}
```

```cpp
int size_stack()
{
    Node *p;
    int num;

    num = 0;
    p = head;
    while(p != nullptr)
    {
        p = p->next;
        num++; // This variable counts the nodes.
    }
    return num;
}
```

```cpp
void free_stack()
{
    Node *p, *next_node;

    p = head;
    while (p != nullptr) {
        next_node = p->next; /* next node points to the node
        after the one to be deleted. */
        delete p; /* Release the allocated memory. The information about
the next node is not lost, because it was saved in p. */
        p = next_node; // p points to the next node.
    }
}
```

Comments:
a. In add_stack(), we pass a pointer to the node.
b. To display the queue, we use a pointer to the head.
c. If the variable was declared locally, it becomes the tail.
