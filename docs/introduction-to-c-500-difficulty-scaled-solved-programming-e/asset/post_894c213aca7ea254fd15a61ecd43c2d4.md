int Prtcl::handle_up_msg(Msg *p) /* Handle the message coming from the upper layer. */
{
    int index;
    if(p->code == SETUP) {
        Call c(p->CRV); // Create an object to manage the call.
        calls.push_back(c); // Store the object
        index = calls.size()-1;
        map_CRV(p->CRV) = index; /* Map the CRV to the position of the object in the vector. */
        calls[index].fsm(); /* Forward the message to the object that manages the call. */
    } else if (p->code == RELEASE) {
        index = p->call_index;
        calls[index].fsm(p);
    } else {
        cout << "Upper message " << p->code << " not supported\n";
        return -1;
    }
    return 0;
}

int Prtcl::handle_low_msg(int data[]) /* Handle the message coming from the lower layer. */
{
    int index;
    Msg msg; /* The information coming from the lower layer is stored in the Msg members. */
    index = find_call(data[0]); /* We find the call, that is, the object which has the same CRV with the CRV contained in the message. */
    if(index == -1) {
        cout << "Non existing call\n";
        return -1;
    }
    msg.code = data[1];
    switch(msg.code) {
        case CONNECT:
            calls[index].fsm(&msg); /* Forward the message to the object that manages the call. */
            break;
        case RELEASE_COMPLETE:
            ret = calls[index].fsm(&msg); /* Here is an example to
            check the return value. The respective object is deleted. */
            if(ret == CLEAR_CALL)
            {
                calls.erase(calls.begin()+index);
                map_CRV.erase(data[0]);
            }
            break;
        default:
            cout << "Received message_" << msg.code << " not supported\n";
            return -1;
    }
    return 0;
}

int Prtcl::find_call(int CRV) const /* To find the object we use map_CRV
which maps the CRV with its position in the vector. */
{
    auto it = map_CRV.find(CRV);
    if(it != map_CRV.end())
        return it->second;
    return -1;
}

The handle_up_msg() is called when a message from the upper layer is received. In our example, this level connects to the application menu which provides some operations to perform. If the user selects the call option, a new object is created to manage the call and the message is supposed to be sent to the called user. In a real application, this message is created when the caller initiates communication with the called user (e.g., dials the number), the network card it to the physical medium (e.g., optical fiber), and the message traverses the ISDN network to the destination.

The handle_low_msg() is called when a message from the lower layer is received. For example, a real application, the message is received from the network card, it is passed to the driver of the application, and the driver forwards it to the application. In our example, the message is contained in the data array. In a real application, the handle_low_msg() and handle_up_msg() are called asynchronously and access the resources, I put them in threads and use semaphores for multi-threading. Thinking of this as a separate program. In this book we don't discuss about threads, I just give something similar in the future. Next files are fsm.cpp and fsm.h. These files implement the state machine of the protocol. For simplicity, when the message is received in the IDLE state, the state changes to CALL_INIT. When the call is in CALL_INIT state and the CONNECT message is received, the state changes to ACTIVE, which means that the call has been established. To release the call, the state changes from ACTIVE to RELEASE_INIT and the RELEASE message is received, the call is released and the respective object is deleted from the vector. For simplicity, using the switch statement. However, typical choice is to use an array of pointers to functions, in the example of Chapter 8, in order to improve performance.

int Call::fsm(Msg *p)
{
    switch (state)
    {
        case IDLE:
            if (p->code == SETUP)
            {
                state = CALL_INIT;
                send_msg(p);
            }
            else {
                cout << "Unexpected message_" << p->code << " in state_" << state << '\n';
                return UNKNOWN_MSG;
            }
            break;
        case CALL_INIT:
            if (p->code == CONNECT)
            {
                cout << "\nCall with CRV " << CRV << " is established\n";
                state = ACTIVE;
            }
            break;
        case ACTIVE:
            if (p->code == RELEASE)
            {
                state = RELEASE_INIT;
                send_msg(p);
            }
            break;
        case RELEASE_INIT:
            if (p->code == RELEASE_COMPLETE)
            {
                cout << "\nCall with CRV " << CRV << " is released\n";
                return CLEAR_CALL;
            }
            break;
        default:
            cout << "Not supported state\n";
            return UNKNOWN_STATE;
    }
    return 0;
}
