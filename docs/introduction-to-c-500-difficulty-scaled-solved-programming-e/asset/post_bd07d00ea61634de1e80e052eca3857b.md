**U.11.23** What is the output of the following program?

```cpp
#include <iostream>

void test(int val, int *tmp);

int main() {
    int *ptr, i, arr[] = {5, 10, 15};
    ptr = arr;
    for(i = 0; i < 2; i++) {
        test(*ptr, ptr);
        ptr++;
    } while (ptr >= arr) {
        std::cout << *ptr << ' ' << ptr << ' ';
        ptr--;
    }
    return 0;
}

void test(int val, int *tmp) {
    std::cout << ++val << ' ' << (tmp)++;
}
```
