## C.11.20 编写一个函数，以数组作为参数并检查它是否包含重复值。如果是，该函数返回出现次数最多的元素的指针，否则返回 `nullptr`。编写一个程序读取 100 个整数，将其存储在数组中并使用该函数显示出现次数最多的元素。注意：如果有多个元素出现的次数相同，该函数应返回找到的第一个。

```cpp
#include <iostream>

const int *find(const int arr[]);
const int SIZE = 100;
```

```cpp
int main()
{
    const int *ptr;
    int i, arr[SIZE];

    for(i = 0; i < SIZE; i++)
    {
        std::cout << "Enter number: ";
        std::cin >> arr[i];
    }
    ptr = find(arr);
    if(ptr == nullptr)
        std::cout << "No duplicated value is found\n";
    else
        std::cout << "The number " << *ptr << " appears the times\n";
    return 0;
}
```

```cpp
const int *find(const int arr[])
{
    int i, j, cnt, max, pos;

    max = 0;
    for(i = 0; i < SIZE; i++)
    {
        cnt = 0;
        for(j = i+1; j < SIZE; j++)
        {
            if(arr[i] == arr[j])
                cnt++;
        }
        if(cnt > max)
        {
            max = cnt;
            pos = i;
        }
    }
    if(max == 0)
        return nullptr;
    else
        return arr+pos;
}
```

```cpp
vector<string> find_same(const vector<string>& v1, const vector<string>& v2)
{
    int i, j, num;
    vector<string> v;

    num = v1.size();
    for(i = 0; i < num; i++) {
        for(j = 0; j < num; j++) {
            if(v1[i] == v2[j]) {
                v.push_back(v1[i]);
                break; /* Terminate the internal for next element.*/
            }
        }
    }
    return v;
}
```
