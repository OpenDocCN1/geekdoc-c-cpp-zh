C.18.3 Define Time class with private members the hours (e.g., hrs), the minutes (e.g., mins), and the seconds (e.g., secs). Add appropriate functions so that the following program works.

```cpp
int main()
{
    Time t1(23, 59, 59); /* There is a restriction that the class has
just one constructor. The members of t1 should be initialized with 0. Also,
    another restriction is that the constructor's parameters have the same names
    as the private members. */
    ++t2; /* The number of seconds increases by one. If the time is
23:59:59, in the case of t2, the new time becomes 0:0:0. */
    cout << (t1 == t2) << '\n'; /* Compare the two times and if they are
the same, the program should display 1, otherwise 0. For example, with these
values of t1 and t2 the program should display 1. */
    /* The time should be displayed in the h:m:s format. For
examples, 0:0:0. */
    return 0;
}
```

```cpp
class Time
{
private:
    int hrs;
    int mins;
    int secs;
public:
    Time(int hrs=0, int mins=0, int secs=0);
    bool operator+=(const Time& t);
    void operator+=();
    bool operator==(const Time& t);
    friend ostream& operator<<(ostream& out, const Time& t);
}
```

```cpp
Time::Time(int hrs, int mins, int secs)
{
    this->hrs = hrs; /* Due to the restriction about same names we use the
pointer. */
    this->mins = mins;
    this->secs = secs;
}

bool Time::operator==(const Time& t)
{
  if(hrs == t.hrs) && (mins == t.mins) && (secs == t.secs)
    return true;
  else
    return false;
}

void Time::operator++()
{
  secs++;
  if(secs == 60)
    {
      secs = 0;
      mins++;
      if (mins == 60)
        {
          mins = 0;
          hrs++;
          if(hrs == 24)
            hrs = 0;
        }
    }
}
ostream& operator<<(ostream& out, const Time& t)
{
  out << t.hrs << ' : ' << t.mins << ' : ' << t.secs << ' ';
  return out;
}
```

```cpp
  T t1(str); // The string should be stored in the s member.

  cout << "Enter text: ";
  cin.getline(str, sizeof(str));

```

Answer:

```cpp
#include <iostream>
#include <cstring>
using std::cout;
using std::cin;

class T {
private:
    char *s;
public:
    T(const char* str) {
        s = new char[strlen(str) + 1];
        strcpy(s, str);
    }
    ~T() {
        delete[] s;
    }
    bool operator==(const T& t) const {
        return strcmp(s, t.s) == 0;
    }
};

int main()
{
    char str[100];
    cout << "Enter text: ";
    cin.getline(str, sizeof(str));
    T t1(str);

    cout << "Enter text: ";
    cin.getline(str, sizeof(str));
    T t2(str);

    cout << (t1 == t2) << '\n';
    return 0;
}
```
