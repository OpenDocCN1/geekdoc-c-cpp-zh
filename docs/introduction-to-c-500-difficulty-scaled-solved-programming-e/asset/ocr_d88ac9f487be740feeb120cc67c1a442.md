# U.20.4 What is the output of the following program?

```
#include <iostream>
using std::cout;

class A {
public:
    int a;
    A(int i) { a = i; }
    virtual void f() const { cout << "A.f"; }
    void g() const { cout << "A.g"; }
};

class B : public A {
public:
    B(int i) : A(i) {}
};

class C : public A {
public:
    void f() const { cout << "C.f"; }
    void g() const { cout << "C.g"; }
};

void t(A* p) {
    p->f();
    p->g();
}

int main() {
    B b(10);
    C c;
    cout << b.a + c.a << '\n';
    t(&b);
    return 0;
}
```
