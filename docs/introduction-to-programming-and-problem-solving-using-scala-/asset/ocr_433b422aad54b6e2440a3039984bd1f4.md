```
val xmin = -1.5
val xmax = 0.5
val xstep = 0.01
val ymin = -1.0
val ymax = 1
val ystep = 0.01
points = for (x <- xmin to xmax by xstep)
      y <- ymin to ymax by ystep yield (x,y)
```
