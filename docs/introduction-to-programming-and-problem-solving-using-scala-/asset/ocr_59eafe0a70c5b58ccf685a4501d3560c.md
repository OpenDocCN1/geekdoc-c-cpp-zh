def insertionSortByEfficiency(a:Array[OperatorEfficiencyFactor]):Unit = {
for (j <- 1 until a.length) {
var i = j-1
var tmp = a(j)
while (i>=0 && a(i).factor>tmp.factor) {
a(i+1) = a(i)
i = i-1
}
a(i+1) = tmp
}
}
