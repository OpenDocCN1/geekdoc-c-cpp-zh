We can reduce the length of this code by utilizing bindings. The following code produces the same GUI, and it functions in the same way. However, instead of having handlers in the TextField that set the values in our recipe objects, in this version the recipe objects store their values using the StringProperty type, and bind the TextField properties to their values when they are selected.

import scalax.scene.input.KeyCombination

import java.io.PrintWriter

case Ingredient(name:StringProperty, amount:StringProperty)\case Recipe(name:StringProperty, ingredients:List[Ingredient], directions:StringProperty)

val app = new JFXApp {
  stage = new JFXApp.PrimaryStage(
    title = "Recipes"
    scene = new Scene(600,400) {
      // Menu
      val menuBar = new Menu()
      val fileMenu = new Menu("File")
      val openItem = new MenuItem("Open")
      openItem.accelerator = new KeyCombination(KeyCode.O, KeyCombination.Control)
      openItem.setOnAction={(e:ActionEvent) => openFile}
      val saveItem = new MenuItem("Save")
      saveItem.accelerator = new KeyCombination(KeyCode.S, KeyCombination.Control)
      saveItem.setOnAction={(e:ActionEvent) => saveFile}
      val exitItem = new MenuItem("Exit")
      exitItem.accelerator = new KeyCombination(KeyCode.X, KeyCombination.Control)
      exitItem.setOnAction={(e:ActionEvent) => System.exit(0)}
      fileMenu.items = List(openItem, saveItem, new SeparatorMenuItem, exitItem)
    }}
