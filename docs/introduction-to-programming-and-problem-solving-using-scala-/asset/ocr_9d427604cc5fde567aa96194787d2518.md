  if (index<a.length) Some(a(index)) else None
}

scala lst.find(_ == 5)
scala lst.find(_ == 9)
scala def recursiveWhile(cond:=>Boolean)(body:=>Unit) {
  if (cond) {
    body
    recursiveWhile(cond)(body)
  }
}
scala var i=0
scala recursiveWhile(i<10) {
  println(i)
  i += 1
}
