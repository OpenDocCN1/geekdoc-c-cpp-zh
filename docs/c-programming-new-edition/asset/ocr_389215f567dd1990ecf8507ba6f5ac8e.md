```c
#include <stdio.h>

// Function to find the k-th element in the merged array
int findKthElement(int arr1[], int N, int arr2[], int M, int k)
{
    int i = 0, j = 0, count = 0, result;
    while (i < N && j < M)
    {
        if (arr1[i] < arr2[j])
        {
            result = arr1[i];
            i++;
        }
        else
        {
            result = arr2[j];
            j++;
        }
        count++;
        if (count == k)
        {
            return result;
