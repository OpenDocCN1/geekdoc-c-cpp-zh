// Function to reverse a binary string
void reverseBinaryString(char *binary)
{
    int len = strlen(binary);
    for (int i = 0; i < len / 2; i++)
    {
        char temp = binary[i];
        binary[i] = binary[len - i - 1];
        binary[len - i - 1] = temp;
    }
}
