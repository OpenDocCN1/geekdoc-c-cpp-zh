             105 'NET VALUE',

             150 'TAX AMOUNT'.

    ENDAT.

                                      VBELN = wa_vbrp-vbeln.

      WRITE AT : 30 wa_vbrk-vbeln  LEFT-JUSTIFIED COLOR 2.
