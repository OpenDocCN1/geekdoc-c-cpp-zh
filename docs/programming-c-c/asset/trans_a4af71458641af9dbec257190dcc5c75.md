# 37. 编写一个统计字符串中元音、辅音和数字的 C 程序。

```c
#include<stdio.h>
int main(){
  char line[100];
  int vowels, consonant, digit, space, i;

  vowels = 0; consonant = 0; digit = 0; space = 0;

  printf("输入字符串: ");
  fgets(line,sizeof(line),stdin);

  for(i=0; line[i]!='\n'; i) {
    if(line[i]=='a' || line[i]=='e' || line[i]=='i' || line[i]=='o' || line[i]=='u') {
      vowels++;
    } else if(line[i]>='a' && line[i]<=='z') {
      consonant++;
    } else if(line[i]>='0' && line[i]<=='9') {
      digit++;
    } else if(line[i]==' ') {
      space++;
    }
  }

  printf("元音: %d", vowels);
  printf("辅音: %d", consonant);
  printf("数字: %d", digit);
  printf("空格: %d", space);
  return 0;
}
```
Output
输入字符串: 
元音: 5
辅音: 9
数字: 0
空格: 0

# 41. 编写一个使用指针交换两个数字的 C 程序。

```c
#include <stdio.h>
// 函数：使用指针交换两个数字
void swap(int *a, int *b)
{
    int t;
    t = *a;
    *a = *b;
    *b = t;
}

int main()
{
    int num1,num2;

    printf("输入 num1: ");
    scanf("%d", &num1);
    printf("输入 num2: ");
    scanf("%d", &num2);

    // 打印交换前的值
    printf("交换前: num1=%d,num2=%d\n", num1, num2);

    // 通过传递 num1 和 num2 的地址来调用函数
    swap(&num1,&num2);

    // 打印交换后的值
    printf("交换后: num1=%d,num2=%d\n", num1,num2);
    return0;
}
```

Output
输入 num1: 10
输入 num2: 20
交换前: num1=10,num2=20
交换后: num1=20,num2=10
