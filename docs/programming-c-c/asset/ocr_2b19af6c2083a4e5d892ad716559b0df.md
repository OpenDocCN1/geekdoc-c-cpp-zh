+   20. Write a C Program to check whether a number is palindrome or not.
#include<stdio.h>
int main() {
    int n, temp =0, remainder, num;
    printf("Enter an integer: ");
    scanf("%d",&n);
    num= n;

    // reversed integer is stored in temp
    while(n!=0){
        remainder= n%10;
        temp= temp *10+ remainder;
        n/=10;
    }

    // palindrome if originalN and reversedN are equal
    if(num==temp)
        printf("%d is a palindrome.",num);
    else
        printf("%d is not a palindrome.",num);

    return 0;
}

Output
Enter an integer: 343
343 is a palindrome.
