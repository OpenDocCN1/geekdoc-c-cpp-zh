
## 46. Write a C program to declare and initialize a union.

```c
#include <stdio.h>

// union declaration
union pack{
    char a;
    int b;
    double c;
};

int main() {
    union pack p; // object/variable declaration
    printf("\nOccupied size by union: %d\n", sizeof(p));

    // assign value to each member one by one other it will replace last value
    p.a='A';
    printf("\nValue of a:%c\n", p.a);

    p.b=10;
    printf("\nValue of b:%d\n", p.b);

    p.c=12345.6790;
    printf("\nValue of c:%f\n", p.c);

    // see, what will happen? if u will assign values together
    p.a='A';
    p.b=10;
    p.c=12345.6790;

    // here the last value of p c will be accessed by all members
    printf("\nValue of a:%c, b:%d, c:%f\n", p.a,p.b,p.c);

    return 0;
}
```

Output
Occupied size by union pack: 8
Value of a:A
Value of b:10
Value of c:12345.67900
Value of a:,-377957122, c:12345.67900

## 47. Write a C++ program to implement function overloading.

```cpp++
#include<iostream>
using namespace std;

int absolute(int);
float absolute(float);

int main() {
    int a = -5;
    float b = 5.5;

    cout << "Absolute value of " << a << " = " << absolute(a) << endl;
    cout << "Absolute value of " << b << " = " << absolute(b) << endl;
    return 0;
}

int absolute(int var) {
    if (var<0) {
        var = -var;
        return var;
    }
    return var;
}

float absolute(float var){
    if (var<0.0) {
        var = -var;
        return var;
    }
    return var;
}
```

Output
Absolute value of -5 = 5
Absolute value of 5.5 = 5.5

## 48. Write a C++ program to calculate an area of rectangle using encapsulation.

```cpp
#include<iostream>

using namespace std;

class Rectangle
{
    int length;
    int breadth;
public:
    void setDimension(int l, int b)
    {
        length = l;
        breadth = b;
    }
    int getArea()
    {
        return length * breadth;
    }
};

int main()
{
    int x,y;
    Rectangle rt;
    cout<<"Enter the length of the rectangle:";
    cin>>x;
    cout<<"Enter the breadth of the rectangle:";
    cin>>y;
    rt.setDimension(7, 4);
    cout<<rt.getArea() <<endl;
    return 0;
}
```

Output
Enter the length of the rectangle:30
Enter the breadth of the rectangle:20
600

## 49. Write a C++ program to add two numbers using data abstraction.

