#include <stdio.h>

# #error

The #error preprocessor directive indicates error. The compiler gives fatal error if #error directive is found and skips further compilation process.

# C #error Example:

```
#include <stdio.h>
#ifndef __MATH_H
#error First include then compile
#else
void main() {
float a;
a=sqrt(7);
printf("%%f",a);
}
#endif
```
