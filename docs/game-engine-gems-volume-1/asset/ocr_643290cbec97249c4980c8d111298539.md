void VectoringMode(int32 x, int32 y, int32 a, int32
{
    for (int i = 0; i < NUMBER_OF_ITERATIONS; ++i)
    {
        int32 d = S (y);   // (y >= 0) ? 0 : -1;
        int32 xi = x;
        int32 yi = y;
        x = x + CONDITIONAL_NEG(yi >> i, d);
        y = y - CONDITIONAL_NEG(xi >> i, d);
        a += CONDITIONAL_NEG(angles[i], d);
    }
    *rl = x;
    *ra = a;
}
