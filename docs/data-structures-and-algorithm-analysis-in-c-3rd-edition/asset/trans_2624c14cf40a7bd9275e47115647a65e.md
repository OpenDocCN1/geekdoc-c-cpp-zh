// Linked queue implementation
template <typename E> class LQueue: public Queue<E> {
private:
    Link<E>* front;        // Pointer to front queue node
    Link<E>* rear;         // Pointer to rear queue node
    int size;              // Number of elements in queue

public:
    LQueue(int sz = defaultSize) // Constructor
    { front = rear = new Link<E>(); size = 0; }

    ~LQueue() { clear(); delete front; }        // Destructor

    void clear() {              // Clear queue
        while(front->next != NULL) { // Delete each link node
            rear = front->next;
            front->next = rear->next;
            delete rear;
        }
        rear = front;
        size = 0;
    }

    void enqueue(const E& it) { // Put element on rear
        rear->next = new Link<E>(it, NULL);
        rear = rear->next;
        size++;
    }

    E dequeue() {               // Remove element from front
        Assert(size != 0, "Queue is empty");
        E it = front->next->element;  // Store dequeued value
        Link<E>* ltemp = front->next; // Hold dequeued link
        front->next = ltemp->next;    // Advance front
        if (rear == ltemp) rear = front; // Dequeue last element
        delete ltemp;                  // Delete link
        size--;
        return it;                     // Return element value
    }

    const E& frontValue() const { // Get front element
        Assert(size != 0, "Queue is empty");
        return front->next->element;
    }

    virtual int length() const { return size; }
};

图 4.28 展示了一个简单抽象字典类的定义。insert 和 find 方法是该类的核心。insert 方法接收一条记录并将其插入字典。find 方法接收一个键值，并返回字典中某个键与所提供键匹配的记录。如果字典中存在多条具有该键值的记录，则不要求具体返回哪一条。

// The Dictionary abstract class.
template <typename Key, typename E>
class Dictionary {
private:
    void operator =(const Dictionary&) {}
    Dictionary(const Dictionary&) {}
public:
    Dictionary() {} // Default constructor
    virtual ~Dictionary() {} // Base destructor
    // Reinitialize dictionary
    virtual void clear() = 0;
    // Insert a record
    // k: The key for the record being inserted.
    // e: The record being inserted.
    virtual void insert(const Key& k, const E& e) = 0;
    // Remove and return a record.
    // k: The key of the record to be removed.
    // Return: A matching record. If multiple records match
    // "k", remove an arbitrary one. Return NULL if no record
    // with key "k" exists.
    virtual E remove(const Key& k) = 0;
    // Remove and return an arbitrary record from dictionary.
    // Return: The record removed, or NULL if none exists.
    virtual E removeAny() = 0;
    // Return: A record matching "k" (NULL if none exists).
    // If multiple records match, return an arbitrary one.
    // k: The key of the record to find
    virtual E find(const Key& k) const = 0;
    // Return the number of records in the dictionary.
    virtual int size() = 0;
};

图 4.30 字典搜索示例。这里，工资单记录存储在两个字典中，一个按 ID 组织，另一个按姓名组织。两个字典都使用基于无序数组的列表实现。

字典的基本操作是查找与给定键匹配的记录。这就引出了如何从记录中提取键的问题。我们希望任何给定的字典实现都能支持任意的记录类型，因此我们需要某种足够通用的键提取机制。一种方法是要求所有记录类型都支持某个特定的返回键值的方法。例如，在 Java 中可以使用 Comparable 接口来实现这一效果。遗憾的是，当同一记录类型需要存储到多个字典中，且每个字典以记录的不同字段作为键时，这种方法就行不通了。这在数据库应用中很常见。另一种更通用的方法是提供一个专门负责从记录中提取键的类。遗憾的是，这种解决方案也并非在所有情况下都有效，因为有些记录类型根本无法编写键提取方法。²

根本问题在于，一条记录的键值并不是记录所属类的固有属性，也不是类中任何字段的固有属性。记录的键实际上是记录使用环境的一个属性。

一种真正通用的替代方案是，将给定记录关联的键作为字典中的一个单独字段显式存储。也就是说，字典中的每一项都将包含一条记录及其关联的键。这样的条目被称为“键值对”。显式存储键通常会重复记录中的某个字段。然而，键往往比记录小得多，所以这种额外的空间开销不会很大。图 4.31 展示了一个表示键值对的简单类。字典类的 insert 方法支持键值对实现，因为它接收两个参数：一条记录和该记录在该字典中的关联键。

> ²这种情形的一个例子是：我们有一组描述图书馆藏书的记录。这种记录的一个字段可能是主题关键词列表，通常每条记录存储几个关键词。我们的字典可能实现为一个按关键词排序的记录列表。如果一本书包含三个关键词，它会在列表中出现三次，每个关键词对应一次。然而，给定这条记录，我们无法简单地确定是关键词列表中的哪个关键词导致了该记录的出现。因此，我们无法编写一个从这种记录中提取键的函数。
