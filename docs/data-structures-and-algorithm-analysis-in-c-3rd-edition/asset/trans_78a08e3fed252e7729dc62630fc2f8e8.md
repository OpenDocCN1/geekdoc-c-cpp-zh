
```cpp
class Graphl : public Graph {
private:
    List<Edge>** vertex;        // List headers
    int numVertex, numEdge;     // Number of vertices, edges
    int *mark;                  // Pointer to mark array
public:
    Graphl(int numVert)
    { Init(numVert); }

    ~Graphl() {                 // Destructor
        delete [] mark; // Return dynamically allocated memory
        for (int i=0; i<numVertex; i++) delete vertex[i];
        delete [] vertex;
    }

    void Init(int n) {
        int i;
        numVertex = n;
        numEdge = 0;
        mark = new int[n];      // Initialize mark array
        for (i=0; i<numVertex; i++) mark[i] = UNVISITED;
        // Create and initialize adjacency lists
        vertex = (List<Edge>**) new List<Edge>*[numVertex];
        for (i=0; i<numVertex; i++)
            vertex[i] = new LList<Edge>();
    }

    int n() { return numVertex; } // Number of vertices
    int e() { return numEdge; }   // Number of edges

    int first(int v) { // Return first neighbor of "v"
        if (vertex[v]->length() == 0)
            return numVertex;       // No neighbor
        vertex[v]->moveToStart();
        Edge it = vertex[v]->getValue();
        return it.vertex();
    }

    // Get v's next neighbor after w
    int next(int v, int w) {
        Edge it;
        if (isEdge(v, w)) {
            if ((vertex[v]->currPos()+1) < vertex[v]->length()) {
                vertex[v]->next();
                it = vertex[v]->getValue();
                return it.vertex();
            }
        }
        return n(); // No neighbor
    }

    // Set edge (i, j) to "weight"
    void setEdge(int i, int j, int weight) {
        Assert(weight>0, "May not set weight to 0");
        Edge currEdge(j, weight);
        if (isEdge(i, j)) { // Edge already exists in graph
            vertex[i]->remove();
            vertex[i]->insert(currEdge);
        }
        else { // Keep neighbors sorted by vertex index
            numEdge++;
            for (vertex[i]->moveToStart();
                 vertex[i]->currPos() < vertex[i]->length();
                 vertex[i]->next()) {
                Edge temp = vertex[i]->getValue();
                if (temp.vertex() > j) break;
            }
            vertex[i]->insert(currEdge);
        }
    }

    void delEdge(int i, int j) {  // Delete edge (i, j)
        if (isEdge(i, j)) {
            vertex[i]->remove();
            numEdge--;
        }
    }

    bool isEdge(int i, int j) { // Is (i,j) an edge?
        Edge it;
        for (vertex[i]->moveToStart();
             vertex[i]->currPos() < vertex[i]->length();
             vertex[i]->next()) {      // Check whole list
            Edge temp = vertex[i]->getValue();
            if (temp.vertex() == j) return true;
        }
        return false;
    }

    int weight(int i, int j) { // Return weight of (i, j)
        Edge curr;
        if (isEdge(i, j)) {
            curr = vertex[i]->getValue();
            return curr.weight();
        }
        else return 0;
    }

    int getMark(int v) { return mark[v]; }
    void setMark(int v, int val) { mark[v] = val; }
};
```

图遍历算法可以通过为图中的每个顶点维护一个标记位来解决这两个问题。在算法开始时，所有顶点的标记位都被清除。当顶点在遍历过程中首次被访问时，其标记位被设置。如果在遍历过程中遇到已标记的顶点，则不会再次访问它。这可以防止程序在遇到环时进入无限循环。

遍历算法完成后，我们可以通过检查标记位数组来检查是否所有顶点都已处理。如果不是所有顶点都被标记，我们可以从另一个未标记的顶点继续遍历。注意，这个过程无论图是有向还是无向都适用。为了确保访问所有顶点，可以按如下方式对图 G 调用 graphTraverse：

```cpp
void graphTraverse(Graph* G) {
    int v;
    for (v=0; v<G->n(); v++)
        G->setMark(v, UNVISITED);    // Initialize mark bits
    for (v=0; v<G->n(); v++)
        if (G->getMark(v) == UNVISITED)
            doTraverse(G, v);
}
```

### 11.3.1 深度优先搜索

组织的图遍历的第一种方法称为深度优先搜索（DFS）。当搜索过程中访问顶点 V 时，DFS 会递归地访问 V 的所有未访问邻居。等价地说，DFS 会将从 v 出发的所有边添加到栈中。下一个要访问的顶点由弹出栈并跟随该边来确定。其效果是沿着图的一条分支跟随到其结论，然后回溯并跟随另一条分支，依此类推。DFS 过程可用于定义深度优先搜索树。该树由遍历过程中为访问任何新（未访问）顶点而跟随的边组成，并省略了通向已访问顶点的边。DFS 可应用于有向图或无向图。以下是 DFS 算法的实现：

```cpp
void DFS(Graph* G, int v) { // Depth first search
    PreVisit(G, v);                // Take appropriate action
    G->setMark(v, VISITED);
    for (int w=G->first(v); w<G->n(); w = G->next(v,w))
        if (G->getMark(w) == UNVISITED)
            DFS(G, w);
    PostVisit(G, v);               // Take appropriate action
}
```

此实现包含对 PreVisit 和 PostVisit 函数的调用。这些函数指定搜索过程中应采取的活动。就像先序树遍历需要在访问子树之前采取行动一样，一些图遍历要求在 DFS 中进一步的顶点之前处理顶点。或者，一些应用要求在处理完剩余顶点后采取行动；因此调用 PostVisit 函数。这将是使用第 1.3.2 节中描述的访问者设计模式的自然机会。

### 11.3.2 广度优先搜索

我们的第二个图遍历算法称为广度优先搜索（BFS）。BFS 在访问更远的顶点之前先检查与起始顶点相连的所有顶点。BFS 的实现与 DFS 类似，只是递归栈被队列取代。注意，如果图是一棵树且起始顶点在根，BFS 等价于自上而下逐层访问顶点。图 11.10 提供了 BFS 算法的实现。图 11.11 显示了一个图及其对应的广度优先搜索树。图 11.12 说明了图 11.11(a) 的 BFS 过程。

```cpp
void BFS(Graph* G, int start, Queue<int>* Q) {
    int v, w;
    Q->enqueue(start);        // Initialize Q
    G->setMark(start, VISITED);
    while (Q->length() != 0) { // Process all vertices on Q
        v = Q->dequeue();
        PreVisit(G, v);        // Take appropriate action
        for (w=G->first(v); w<G->n(); w = G->next(v,w))
            if (G->getMark(w) == UNVISITED) {
                G->setMark(w, VISITED);
                Q->enqueue(w);
            }
    }
}
```

### 11.3.3 拓扑排序

假设我们需要安排一系列任务，例如课程或建筑工程，其中一个任务必须在其先决条件完成后才能开始。我们希望将任务组织成一个线性顺序，以便一次完成一个任务而不违反任何先决条件。我们可以使用 DAG 对问题建模。图是有向的，因为一个任务是另一个任务的先决条件——顶点之间存在有向关系。它是无环的，因为环会表示一组相互冲突的先决条件，无法在不违反至少一个先决条件的情况下完成。将 DAG 的顶点按线性顺序排列以满足先决条件规则的过程称为拓扑排序。图 11.14 说明了该问题。此示例的一个可接受的拓扑排序是 J1, J2, J3, J4, J5, J6, J7。

可以通过对图执行 DFS 来找到拓扑排序。当访问顶点时，不采取任何行动（即函数 PreVisit 什么都不做）。当递归返回到该顶点时，函数 PostVisit 打印该顶点。这会以逆序产生拓扑排序。排序的开始位置并不重要，只要最终访问所有顶点即可。图 11.13 显示了基于 DFS 的算法的实现。
