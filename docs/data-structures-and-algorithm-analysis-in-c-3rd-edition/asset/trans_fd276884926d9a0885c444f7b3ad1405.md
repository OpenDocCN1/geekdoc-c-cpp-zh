removehelp(BSTNode<Key, E>* rt, const Key& k) {
    if (rt == NULL) return NULL;      // k is not in tree
    else if (k < rt->key())
        rt->setLeft(removehelp(rt->left(), k));
    else if (k > rt->key())
        rt->setRight(removehelp(rt->right(), k));
    else {                          // Found: remove it
        BSTNode<Key, E>* temp = rt;
        if (rt->left() == NULL) {    // Only a right child
            rt = rt->right();        // so point to right
            delete temp;
        }
        else if (rt->right() == NULL) { // Only a left child
            rt = rt->left();         // so point to left
            delete temp;
        }
        else {                      // Both children are non-empty
            BSTNode<Key, E>* temp = getmin(rt->right());
            rt->setElement(temp->element());
            rt->setKey(temp->key());
            rt->setRight(deletemin(rt->right()));
            delete temp;
        }
    }
    return rt;
}

本节介绍堆⁴数据结构。堆由两个性质定义。首先，它是一个完全二叉树，因此堆几乎总是使用 5.3.3 节介绍的完全二叉树数组表示来实现。其次，存储在堆中的值是部分有序的。这意味着任意节点存储的值与其子节点的值之间存在某种关系。根据这种关系的定义不同，堆有两种变体。

最大堆的性质是：每个节点存储的值都大于或等于其任一子节点的值。因为根节点的值大于或等于其子节点的值，而子节点的值又大于或等于它们各自子节点的值，所以根节点存储着树中所有值的最大值。

最小堆的性质是：每个节点存储的值都小于或等于其子节点的值。因为根节点的值小于或等于其子节点的值，而子节点的值又小于或等于它们各自子节点的值，所以根节点存储着树中所有值的最小值。

注意，无论是最小堆还是最大堆，一个节点的值与其兄弟节点的值之间都没有必然关系。例如，根节点左子树中所有节点的值可能大于右子树中每个节点的值。我们可以通过其有序关系的强弱来对比 BST 和堆。BST 在其节点上定义了一个全序：给定树中任意两个节点的位置，“左”侧的节点（等价于中序遍历中较早出现的节点）的键值小于“右”侧的节点。相比之下，堆实现的是偏序。给定两个节点在堆中的位置，只有当其中一个是另一个的后代时，我们才能确定它们键值的相对顺序。

最小堆和最大堆各有其用途。例如，7.6 节的堆排序使用最大堆，而 8.5.2 节的置换选择算法使用最小堆。本节其余示例将使用最大堆。

注意不要将堆的逻辑表示与其基于数组的完全二叉树物理实现混淆。两者并非同义词，因为堆的逻辑视图实际上是一种树结构，而典型的物理实现使用数组。

```
// Heap class
template <typename E, typename Comp> class heap {
private:
    E* Heap;                // Pointer to the heap array
    int maxsize;            // Maximum size of the heap
    int n;                  // Number of elements now in the heap

    // Helper function to put element in its correct place
    void siftdown(int pos) {
        while (!isLeaf(pos)) { // Stop if pos is a leaf
            int j = leftchild(pos); int rc = rightchild(pos);
            if ((rc < n) && Comp::prior(Heap[rc], Heap[j]))
                j = rc;         // Set j to greater child's value
            if (Comp::prior(Heap[pos], Heap[j])) return; // Done
            swap(Heap, pos, j);
            pos = j;            // Move down
        }
    }

public:
    heap(E* h, int num, int max)      // Constructor
        { Heap = h; n = num; maxsize = max; buildHeap(); }
    int size() const          // Return current heap size
        { return n; }
    bool isLeaf(int pos) const // True if pos is a leaf
        { return (pos >= n/2) && (pos < n); }
    int leftchild(int pos) const
        { return 2*pos + 1; }    // Return leftchild position
    int rightchild(int pos) const
        { return 2*pos + 2; }    // Return rightchild position
    int parent(int pos) const   // Return parent position
        { return (pos-1)/2; }
    void buildHeap()           // Heapify contents of Heap
        { for (int i=n/2-1; i>=0; i--) siftdown(i); }

    // Insert "it" into the heap
    void insert(const E& it) {
        Assert(n < maxsize, "Heap is full");
        int curr = n++;
        Heap[curr] = it;        // Start at end of heap
        // Now sift up until curr's parent > curr
        while ((curr!=0) &&
            (Comp::prior(Heap[curr], Heap[parent(curr)]))) {
            swap(Heap, curr, parent(curr));
            curr = parent(curr);
        }
    }
}
```
