20. What is the output of the below piece of code? (L4)

```
```cpp
#include<iostream>
using namespace std;
class Constructor_Array_Overloading
{
    int x;
public:
    Constructor_Array_Overloading()
    {
        x=0;
    }
    Constructor_Array_Overloading(int n)
    {
        x=n;
    }
    int get_value()
    {
        return x;
    }
};
int main()
{
    Constructor_Array_Overloading obj1[2];
    Constructor_Array_Overloading obj2[2]={1};
    int i;
    for(i=0;i<3;i++)
    {
        cout << "obj1[" << i << "]: " << obj1[i].get_value() << endl;
        cout << "obj2[" << i << "]: " << obj2[i].get_value() << endl;
    }
    return 0;
}
```
```

+   - (a) obj1[0] : 0  
+     obj2[0] : 1  
+     obj1[1] : 0  
+     obj2[1] : 0
+   - (b) Compilation error
+   - (c) Segmentation fault

+   - (d) obj1[0] : 0  
+     obj2[0] : 1  
+     obj1[1] : 0  
+     obj2[1] : 1

Answer: (a) Because, if one element of the array is initialized, rest all will be zero.
