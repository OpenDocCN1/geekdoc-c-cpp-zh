# wait_ms.asm : wait_ms.asm

```
; Millisecond delay subroutine - delay time set by MS_TIME_DEL
; F_CPU must be set to the frequency that the AVR is operating at

.if (F_CPU > 20000000)
.error "F_CPU too high for wait_ms subroutine. Maximum 20MHz"
.elif (F_CPU < 210000)
.error "F_CPU too low for wait_ms subroutine. Minimum 210kHz"
.elif (MS_TIME_DEL < 1)
.error "MS_TIME_DEL too low for wait_ms subroutine. Minimum 1ms"
.elif (((((MS_TIME_DEL * 1000) - 10) / 10) > 0xFFFFFFF)
// Maximum MS_TIME_DEL is 2,147,483s @ 20MHz and 204,522,525 @ 210kHz
// Maximum MS_TIME_DEL calculated by: (10,000 * 2^32) / F_CPU
.error "MS_TIME_DEL too big for wait_ms subroutine."
.endif

.equ WAIT_MS_CYCLES == ((((F_CPU * MS_TIME_DEL) / 1000) - 10) / 10)

wait_ms:
ldi R16, low (WAIT_MS_CYCLES)
ldi R17, byte2(WAIT_MS_CYCLES)
ldi R18, byte3(WAIT_MS_CYCLES)
ldi R19, byte4(WAIT_MS_CYCLES)

wait_ms_loop:
subi R16, 1
sbci R17, 0
sbci R18, 0
sbci R19, 0
mov R20, R16
or R20, R17
or R20, R18
or R20, R19
brbc SREG_Z, wait_ms_loop
nop
nop
ret
```

In the above code, tests for high and low limits are done using the .if directive, followed by a number of .elif directives. A .elif directive means “else if”. From the top of the code, the test are 1) if F_CPU is greater than 20MHz, flag an error, 2) else if F_CPU is less than 210kHz, flag an error, 3) else if MS_TIME_DEL is less than 1ms, flag an error, 4) else if the WAIT_MS_CYCLES calculation results in a too big value, flag an error. Because the .elif directive is used in this way, the first error that is found will stop the build process. After the error is fixed, the programmer can build the project again. If any other error is present, it will be caught after the second build of the project, and so on until all the limit errors have been corrected.
