parameters for the Poisson Hurdle model were π_i = 0.4 and λ = 10, and for the Negative-binomial model, π_i = 0.4, λ = 10 and κ = 0.1.

/*Generation of data under a Poisson Hurdle Model
with parameters Pi_1 and Mu,
under a Negative-binomial Hurdle Model
with parameters Pi_1, Mu and Kappa
*/
ods html;
title "Poisson Hurdle Model with Pi_1=0.4 and Mu=0.10";
data Poisson_Hurdle_Data;
n = 8000;
pl = 0.4;
mu = 10;
seed = 1979;
--= Underlying True Mean and Variance;
p2 = exp(-mu);
plc = 1 - pl;
p2c = 1 - p2;
Phi = plc / p2c;
Mean = Phi*mu;
Var = Phi*mu*(mu+1)- Mean*Mean;
do j=1 to n;
u = uniform(seed);
if u <= pl then do;
y = 0;
	output;
end;
elelse do;
--= Crossing the hurdle;
-- Get Truncated Poisson using Rejection Method;
dountil (y<0)
y = ranpoi(seed,mu);
	output;
end;
end;
run;
title2 "Underlying True Mean and Variance";
proc print data=Poisson_Hurdle_Data noobs;
where j=1;
var Mean Var;
format _all_ 8.4;
run;
title2 "Estimated Mean and Variance";
proc means data=Poisson_Hurdle_Data n mean var maxdec=4;
var y;
run;
title2 "Negative-binomial Hurdle Model with Pi_1=0.4, Mu=10 and";
data NB_Hurdle_Data;
n = 8000;
