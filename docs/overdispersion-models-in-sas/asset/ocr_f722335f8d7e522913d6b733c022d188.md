```
/* Module: MLERCM(m, T)                                               /
/* This Module computes the MLE from a Random-clumped Multinomial */
/* Distribution using the Fisher Information                 /
/* Matrix and the Fisher Scoring Algorithm                             /
/* m - cluster size                                                  /\n/* T - nxk matrix of counts (t1,t2,...,tk) where ti is a          /
/*      nxl vector of observed counts in category i                    /
/****************************************************************/
```

```
Start MLERCM(m, T);
print "**Fisher Scoring Method -- Random-clumped Multinomial Distribution ***";
pi0 = QLEpi(m,T);
rho0 = QLRho(m,T,pi0);
k = ncol(T);
n = nrow(T);
theta;
theta[1:(k-1)] = j(k,1,0); pi0;
theta[k] = rho0;

eps = 1E-8;
conv = 1;
iter = 0;
Do Until(conv<eps);
  iter = iter + 1;
  pi = theta[1:(k-1)];
  rho = theta[k];
  INFIM = Inv(n*FIRCM(m, pi, rho));
  Old = 0;
  Scores = j(k,1,0);
  Do j = 1 to n;
    tj = T[j,1:(k-1)];
    Scores = Scores + gradRCM(m, pi, rho, tj);
    Old = Old + log(ProbRCM(m, pi, rho, tj));
  End;

s = 0;
Lnew = Old - 1;
Do Until(Lnew > Old);
  thetaNew = theta + (0.5**si)*INFIM*Scores;
  thetaNew[1:(k-1)] = ;
  rho = thetaNew[k];
  Lnew = 0;
  Do j = 1 to n;
    tj = T[j,1:(k-1)];
    Lnew = Lnew + log(ProbRCM(m, pi, rho, tj));
  End;
  si = si + 1;
  if si > 20 then;
    print "NOTE: Step size has halved 20 times, convergence assumed";
```
