Note that (4.37) is a full model on both the betas and the alphas. This model permits us to estimate instantaneously the probability of ossification and the overdispersion parameter within treatment groups. The programming statements are presented in Program 4.5.

# Program 4.5: Fitting of a Full Model on both "Betas" and the "Alphas" under the Beta-binomial and Random-clumped Binomial Distributions

```
ods html;

title "Beta-binomial, Full Model on the Betas, Full Model on the Alphas";
proc nlmixed data=ossification cov;
parms b0=0, b1=0, b2=0, a0=0, a1=0, a2=0, a3=0;
if (tx='Control') then lin = a0;
else if (tx='TCPO') then lin = a0+a1;
else if (tx='PHT') then lin = a0+a2;
else if (tx='PHT+TCPO') then lin = a0+a2+a3;
rho = 1/(1-exp(-lin));
c = / rho * rho1;
if (tx='Control') then lin = b0;
else if (tx='TCPO') then lin = b0+b1;
else if (tx='PHT') then lin = b0+b2;
else if (tx='PHT+TCPO') then lin = b0+b1+b2+b3;
pi = 1/(1-exp(-lin));
pic = 1 - pi;
z = gamma(m) - lgamma(m-1) - lgamma(m-t*pic);
gamma(mc) = lgamma(c*pi) - lgamma(c*pic);
model t ~ general(1);
estimate ' 1/(exp(-b0));
estimate PI 'TCPO' 1/(exp(-b0-b1));
estimate PI 'PHT' 1/(exp(-b0-b2));
estimate 'PHT+TCPO' 1/(exp(-b1-b2-b3));
estimate 'Logarithm Odds-Ratio PHT vs Control when TCPO Absent ' b2;
estimate 'Rho Control' 1/(exp(-a0))/(1+exp(-a0));
estimate 'Rho PHT' 1/(exp(-a0-a2))/(1+exp(-a0-a2));
estimate 1/(exp(-a2));
estimate 'Rho PHT+TCPO' 1/(exp(-a0-a2-a3))/(1+exp(-a0-a2-a3));
run;
title "Random-clumped Binomial, Full Model on the Betas, Full Model on the Alphas";
proc nlmixed data=ossification cov;
parms b0=0, b1=0, b2=0, a0=0, a1=0, a2=0, a3=0;
if (tx='Control') then lin = a0;
else if (tx='TCPO') then lin = a0+a1;
else if (tx='PHT') then lin = a0+a2;
else if (tx='PHT+TCPO') then lin = a0+a2+a3;
rho = 1/(1-exp(-lin));
if (tx='Control') then lin = b0;
else if (tx='TCPO') then lin = b0+b1;
else if (tx='PHT') then lin = b0+b2;
else if (tx='PHT+TCPO') then lin = b0+b1+b2+b3;
pi = 1/(1-exp(-lin));
pic = 1 - pi;
pic = 1 - rho * pi + rho;
p2 = 1 - rho;
p2c = 1 - b2;
z = gamma(m) - lgamma(m-1) - lgamma(m-t);
z = z + log(pi*p1**t*pic**m + pic*p2**t*p2c**m);
model t ~ general(1);
```
