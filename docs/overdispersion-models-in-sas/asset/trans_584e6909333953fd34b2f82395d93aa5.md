```sas
ods html; ntitle "Binomial Distribution, Full Model on the Betas",
proc mixed ossification;
parms b0=0, b1=0, b2=0;
if (tx='Control')
then linp = b0;
else if (tx='TCPO')
then linp = b0+b1;
else if (tx='PHT')
then linp = b0+b2;
else if (tx='PHT+TCCO')
then linp = b0+b1+b2+b3;
pi = 1/(1+exp(linp));
```
