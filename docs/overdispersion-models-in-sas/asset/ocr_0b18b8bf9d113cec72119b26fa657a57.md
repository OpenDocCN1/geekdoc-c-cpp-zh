```
title "Beta-binomial Distribution, Full Model on the Betas, Common Rho";
proc nmlimed data=assignment;
  params b0=0, b1=0, b2=0, b3=0, a0=0;
  lin = 1/(1+exp(-lr));
  rho = 1/(1/lr - 1);
  c = if ('Control') then lr = b0;
      else if ('TCPO') then lr = b0+b1;
      else if ('PHT') then lr = b0+b2;
      else if ('PHT+TCPO') then lr = b0+b1+b2+b3;
  pi = 1/(1+exp(-ln));
  pi = 1 - pi;
  z = lgamma(m-1) + lgamma(t+1) - lgamma(m+t);
  z = lgamma(c) + lgamma(t+c*pi) + lgamma(m+c*pi) - lgamma(c*pi);
  model = general(1);
  estimate 'Pi Control' 1/(1+exp(-b0));
  estimate 'Pi TCPO' 1/(1+exp(-b1));
  estimate 'Pi PHT' 1/(1+exp(-b2));
  estimate 'Pi PHT+TCPO' 1/(1+exp(-b0-b1-b2));
  estimate 'Logarithm Odds-Ratio PHT When TCPO Absent ' b2;
  estimate 'Logarithm Odds-Ratio PHT When TCPO Present ' b2+b3;
  estimate 'Common Rho*Rho' 1/(1+exp(-a0))/(1+exp(-a0));
```

run;

```
title "Rho-Clumped Binomial Distribution, Full Model on the Betas, Common Rho";
proc nmlimed data=assignment;
  params b0=0, b1=0, b2=0, b3=0, a0=0;
  wh = 1/(1+exp(-lr));
  rho = 1/(1/lr - 1);
  if ('Control') then lr = b0;
      else if ('TCPO') then lr = b0+b1;
      else if ('PHT') then lr = b0+b2;
      else if ('PHT+TCPO') then lr = b0+b1+b2+b3;
  pi = 1/(1+exp(-ln));
  pi = 1 - pi * rho;
  pi = 1 - rho;
  z = lgamma(m-1) + lgamma(t+1) - lgamma(m+t);
  z = lgamma(pi) + lgamma(t+c*pi) + pi*p2*t+p2c*m_t;
  model = general(1);
  estimate 'Pi Control' 1/(1+exp(-b0));
  estimate 'Pi TCPO' 1/(1+exp(-b1));
  estimate 'Pi PHT' 1/(1+exp(-b2));
  estimate 'Pi PHT+TCPO' 1/(1+exp(-b0-b1-b2));
  estimate 'Logarithm Odds-Ratio PHT When TCPO Absent ' b2;
  estimate 'Logarithm Odds-Ratio PHT When TCPO Present ' b2+b3;
  estimate 'Common Rho*Rho' 1/(1+exp(-a0))/(1+exp(-a0));
```
