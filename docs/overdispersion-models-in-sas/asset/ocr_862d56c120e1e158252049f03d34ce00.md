# Program 5.2: Computation of Dean (1979) GOF Score Statistic for the Toxoplasmosis Example of Table 3.2

```
/*/*
Score Test Statistic, Dean (1992)
H0:Binomial Distribution Versus
H1:Beta-binomial or similar distribution such as
Random-clumped Binomial
  as far as Bernoulli outcomes are positively correlated

Toxoplasmosis in 34 cities of El Salvador
Example from Efron (1978, 1986)
*/
ods select none;
proc glimmix data=toxoplasmosis;
 model t/m = z*z z*z / link=logit dist=bin s;
  output out=data pred(noboplink pi) = pi;
run;
ods select all;
data pdata;
set pdata;
 pic  = 1 - pi;
 pipic = pi * pic;
 mpi   = m * pi;
 t_mpi = t - mpi;
 pit_mpi = pi * t_mpi;
 tpic  = t * pic;
 mm_l  = m *(t-1);
 aux   = ( t*t*mpi + pit_mpi - tpic ) / picic;
keep aux mm_l;
run;
proc means data=pdata sum noprint;
 var aux mm_l;
 output out=new=sum=aux mm_l;
run;
data new;
set new;
 label Z Z = "GOF Test";
 label Pval = "P-Value";
 format z = aux / sqrt(mm_l) .6f;
 pval = 1 - probnorm(z);
run;
ods title;
title title="Score Test Statistic, Dean (1992)";
proc print data=new noobs label;
 var z pval;
run;
ods html close;
```
