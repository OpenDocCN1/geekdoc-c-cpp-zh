## Program 10.5: Generation and Analysis of Data under a Beta-binomial GLOMM

```
/*
Beta-binomial Model with Logit Link and Random Effects
Random effects due to Cross-over Design

t given u ~ Beta-binomial(pi,rho,n)
logit(pi) ~ beta0 + betal * X + u
u ~ Normal(0,sigma=sigma)
*/
data GLOMM
n = 200;
pi_1 = 0.12;
pi_2 = 0.08;
rho_2 = 0.15; --- Overdispersion Parameter;
sigma = 1.2; --- Seed;
seed = 1983;
beta0 = log(pi_1/(1-pi_1));
betal = log(pi_2/(1-pi_2)) - beta0;
rho = sqrt(rho_2);
--- Sequence AB;
do Subj_id=1 to n;
Sequence = 'AB';
u = sigma * normal(seed);
--- Period 1;-
Period = 1;
Product = 1;
X = 1;
pi = 1 / (1 + exp(- beta0 - u));
yy = ranbin(seed,1,pi);
mm = ranbin(seed,m,rho);
t = yy * mm;
if m > mm then t = t + ranbin(seed,m,pi);
output;
--- Period 2;
Period = 2;
Product = 2;
X = 0;
pi = 1 / (1 + exp(- betal - u));
yy = ranbin(seed,1,pi);
mm = ranbin(seed,m,rho);
t = yy * mm;
if m > mm then t = t + ranbin(seed,m,pi);
output;
---
end;
--- Sequence BA;
do Subj_id=n+1 to n+n;
Sequence = 'BA';
u = sigma * normal(seed);
--- Period 1;-
Period = 1;
Product = 0;
pi = 1 / ( 1 + exp( beta0 - betal - u ) );
c = c * pi;
b = c - a;
y1 = rangam(seed, a);
y2 = rangam(seed, b);
p = y1 / ( y1 + y2);
t = ranbin(seed,m,p);
output:
-- Period 2;
Period = 2;
Product = 1;
pi = 1 / ( 1 + exp( beta0 - u ) );
c = c - a;
b = 
y1 = rangam(seed, a);
y2 = rangam(seed, b);
p = y1 / ( y1 + y2);
t = ranbin(seed,m,p);
output:
--;
end;
--;
```

```
proc glimmix
class Subj_id Sequence;

proc nixeded data=Glomm_RCB;
beparmms beta0=2.9,-2.8,-2.7,betal=0.2,0.3,0.4, a0=-0.5 0, sigma=1;
lnr a0;
rho = 1/(1+exp(-lnr));
np = beta0+betal*x + u;
pi = 1/(exp(-lnp));
pic = 1;
pic = (1 - rho )pi + rho;
p2 = 1 - pi;
p2c = p2;
z = lgamma(m+1) - lgamma(m-1);
ll = z + log(pi**p2c*(t - m) + pic**p2**p2c*(t - m));
t model = general ( l );
random u = normal (0,sigma*sigma) subject=Subj_id;
estimate 'P1' 1/(exp(-beta0-betal));
estimate 'P2' 1/(exp(-beta0));
estimate 'OR' exp(betal);
estimate 'Rho*Rho' 1/(exp(-a0))/(1+exp(-a0));
estimate 'Sigma*Sigma' Sigma*Sigma;
run;
```
