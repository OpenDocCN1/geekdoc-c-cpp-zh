ods html;
title "Zero-Inflated estimates of Beta0, Kappa and Sigma**2 PROC GLIMMIX";
proc glimmix data=zib_glomm method=quad order=data;
  class id group;
  model y = group / link=log dist=negbin solution;
  random int / subject=id;
run;

proc nlixed data=zib_glomm;
  parms b0=0.49 b1=0.3 kappa=0.8 w=0.1 0.2 0.3 sigma=1.13;
  bounds kappa > 0 and kappa < 1;
  mu = exp(b0+b1*group);
  a = 1/kappa;
  p = lgamma(kappa*a);
  z = lgamma(y+a) - lgamma(y+1);
  if y<=0 then
    ll = log((1-w)*p**a);
  else ll = log(w) + a * log(p) + y * log(1-p);
  model y = general(ll);
  random u-normal(0,sigma*sigma) subject=id;
  estimate "Expected Counts Group 1" (1-w)*exp(b0+b1);
  estimate "Expected Counts Group 0" (1-w)*exp(b0);
  estimate "Ratio Expected Counts Group 1 Va 0" exp(b1);
  estimate "Sigma*2" sigma*sigma;
run;\ods html close;
