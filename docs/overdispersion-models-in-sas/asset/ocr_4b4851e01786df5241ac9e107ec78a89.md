# Chapter 8 A Two-stage Maximum Likelihood Estimation Procedure 277

```
****************************************************************\n/* Module: Stage2DM(m,T)                                     */
/* * This Module performs the Two-Stage Procedure                    */
/* under the Dirichlet-Multinomial Distribution                 */
/* * m cluster size                                                  */
/* T - nk matrix of counts (t1,t2,...,tk) where ti is a *          */
/* nki vector of observed counts in category i                     */
******************************************************************
StartStage2DM(m, T);
  /* Stage 1 /*
  k = ncol(T);
  n = row(T);
  theta = j(k,1);
  theta = AMLEDM(m, T);

  /* Stage 2 */
  pi = theta(1:(k-1));
  rho = theta[k];
  InvFDM = Inv(FIMDM(m, pi, rho));
  Lold = 0;
  Scores = j(1,0);
  Do j = 1 to n
    tj = T[j,1:(k-1)];
    Scores = Scores + gradDDM(m, pi, rho, tj);
    Lold = Lold + log(ProbDDM(m, pi, rho, tj));
  End;

  -- Updates Theta, Step size will be halved up to 20 times:

si = 0;
Lnew = Lold - 1;
Do Until (Lnew > Lold);
  thetaNew = theta + (0.5*si)*InvFDM*Scores;
  pi = thetaNew(1:(k-1));
  rho = thetaNew[k];
  Lnew = 0;
  Do j = 1 to n
    tj = T[j,1:(k-1)];
    Lnew = Lnew + log(ProbDDM(m, pi, rho, tj));
  End;
  si = si + 1;
  if si > 20 then
    print "NOTE: At Stage 2, Step size halved 20 times without\n    further improvement";
    Lnew = Lold + 0.000001;
  end;
End;
if si <= 20 then theta = thetaNew;

Return(theta);

Finish;
Store-Module=Stage2DM;
```
