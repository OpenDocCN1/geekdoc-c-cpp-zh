  if freq = then delete;

ods html;

%GOF_BB(inds=haseman_soares,t=t,m,title2=DataSet III -- Haseman and Soares (1976));

* Construct QQ Plots;
data Resid_BB_RCB;
  set Resid_BB Resid_RCB;
run;
proc sort data=Resid_BB_RCB;
  by Distribution;
run;
proc rank data=Resid_BB_RCB out=new_qqplots normal=blom ties=mean;
  by Distribution;
  var Resid;
  ranks Nquant;
run;
proc spanel data=new_qqplots autolegend;
  panel Distribution;
  title "Dataset III -- Haseman and Soares (1976)";
  title2 "QQ-Plots of Residuals based on Observed and Expected Frequencies";
  reg x=Resid y=Nquant="Normal Quantiles";
run;
ods graphics off;\ods html close;
