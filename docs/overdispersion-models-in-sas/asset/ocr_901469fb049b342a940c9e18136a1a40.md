# Program 9.1 Generating Continuous and Binary Data from a Random Effects Model

```
data hair;
  seed = 1999;
  beta = 1.386294; -- P is approximately 0.8;
  do subj_id=1 to 1000;
    sl = 2*normal(seed);
    s2 = sqrt(sl+s2);
    do week=1 to 8;
      score = round(sl + normal(seed), 0.01);
      p = 1/(1+exp(-beta - s2));
      y = 0;
      u = uniform(seed);
      if u < p then y = 1;
      output;
    end;
  end;
keep subj_id score y;
run;
ods html;
title "*** Linear Mixed Model with G-side Random Effects ***";
proc glimmix data=hair;
  class subj_id;
  model score = / s;
  random int / subject=subj_id;
run;
title "*** Linear Mixed Model with R-side Random Effects ***";
proc glimmix data=hair;
  class subj_id;
  model score = / s;
  random _residual_ / subject=subj_id type=cs;
run;
```
