Program 4.6: Generating Data under the Beta-binomial and Random-clumped Binomial Distributions

```
/*
Generation of data under the Beta-binomial and Random-clumped Binomial distributions with parameters Pi, Rho and m
*/
#macro bb_rcb_data(pi,rho,m,n,seed);
---
Generation of data under the Beta-binomial distribution;
data bb;;
  format Dist S3;
  Dist = "BB";
  seed = &seed;
  c = 1/rho - 1;
  a = c *pi;
  b = c - a;
  do cluster=1 to n;
    y1 = rangam(seed, a);
    y2 = rangam(seed, b);
    p = y1/(y1+y2);
    t = ranbin(seed,m,p);
    output;
  end;
keep Dist;
run;
---
Generation data under the Random-clumped Binomial distribution;
data rcb;
  Dist = "RCB";
  seed = &seed;
  do cluster=1 to n;
    y = ranbin(seed, 1,pi);
    nn = ranbin(seed,m,rho);
    t = y * nn; -- Random clumping of the data;
    if nn < m then t = t + ranbin(seed,nn,pi);
    output;
  end;
```
