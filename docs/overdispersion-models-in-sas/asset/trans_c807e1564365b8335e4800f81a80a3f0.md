给定一个概率向量 $p, p=p(1),p(2),...p(d)$
该模块计算矩阵 $C(p)$ 的逆矩阵，假设
$A(p)*C(p)' = \text{Diag}(p) - p*p'$
其中 $C(p)$ 是通过 Cholesky 分解得到的。

矩阵 $A(i,j)$ 给出：

```
A(i,j) = { q(i-1) / ( p(i) * q(i) ) * 0.5 if i=j
          = p(i) / ( q(i-1) * q(i) ) * 0.5 if i>j
          = 0 if i<j
```

参见：Tanabe and Sagae (1992)

```
start_inv_cp(p,a,d);
q = q(1) // {1 - cumsum(p));
 i=1 to d;
  do il=i+1;
    a[i,il] = sqrt(q[i]) / p[i] / q[il,1]);
    if d > 1 then do;
      do j=1 to i-1;
        a[i,j] = sqrt(p[i]) / q[i] / q[il,1]);
      end;
    end;
  end;
finish_inv_cp;
```

第 1 部分 计算 Phi；

```
Scale = 0;
Phi = j(d,1,0);
z1 = z1;

do j=1 to n;
  p = t(pred_probs[j]);
  t = t(obs_ts[j,1:d]);
  call inv_cp(p,a,d);
  z = a * (t - m[j,1] * p) / sqrt(m[j,1]);
  scale = scale + z * z;
  phi = phi + z # z;
  zl = zl || z[1,1];
```

--- 第 2 部分) 计算广义多项方差

uu = j(d+k,1,0);
h = j(d+k,d,0);
inv_diag = inv( diag ( phi ) );

do j=1 to n;
p = t(pred_probs[j]);
nt = t(obs_ts[j,1:d]);
flf = cumsum(p);
flf = flf # (1 - flf);
f2f = flf;

temp = diag( phi );
if d > 1 then;
oi = 1 to d-1;
il = i+1;
temp[il,i] = -flf[il,i];
f2f[il,1] = f2f[il,1] - flf[il,i];
end;
end;

der = temp || f2f @ x_matrix[j,];
call inv_cp(p,a,d);
aux = der* a* inv_diag phi * a;
u = u + aux * ( t - m[j,1] * p);
h = h + m[j,1] aux * der;
