# 250 SAS 中的过度离散模型

```sas
*** 费舍评分法 -- 狄利克特多项分布 ***;

pi0 = QLEpi(m,T);
rho0 = QLErho(m,T,pi0);
k = ncol(T);
n = nrow(T);
theta = j(k,1,0);
theta[1:k-1] = pi0;
theta[k] = rho0;

eps = 1E-8;
conv = 1;
iter = 0;
Do Until(conv<eps);

    iter = iter + 1;
    pi = theta[1:k-1];
    rho = theta[k];
    INVNFIM = Inv(n*FIMDM(m, pi, rho));
    Old = 0;
    Scores = j(k,1,0);
    Do j = 1 to n;
        tj = T[j,1:k-1];
        Scores = Scores + gradDM(m, pi, rho, tj);
        Old = Old + log(ProbDM(m, pi, rho, tj));
    End;

    si = 0;
    Lnew = Old - 1;
    Do Until (Lnew > Old);
    thetaNew = theta + {0.5*si}INVNFIM*Scores;
    pi = thetaNew[1:k-1];
    rho = thetaNew[k];
    Lnew = 0;
        Do j = 1 to n;
            tj = T[j,1:k-1];
            Lnew = Lnew + log(ProbDM(m, pi, rho, tj));
        End;

        si = si + 1;
        if si > 20 then;
            print "NOTE: 步长减了二十次，假设收敛";
            Lnew = Old + 0.000001;
        end;
    End;

    theta = thetaNew;
    conv = abs(Lnew)/(abs(Lnew)+1E-6);
    conv_0 = conv;

    if si > 20 then conv = eps - 0.000001;
    if iter > 100 then;
        print "NOTE: 迭代 100 次后，尚未满足收敛";
        conv = eps - 0.000001;
    end;

End;

if conv_0 < eps then;
    print "NOTE: 收敛标准 (EPSILON-1E-8) 已满足";
end;
```

# 第 7 章 过度离散多项响应的似然模型

/* Module: MLERCM(m, T)                                               *
/* 此模块使用费舍信息                 *
/* Matrix和费舍评分算法从随机聚类多项 *                             *
/* 分布中计算 MLE                             *
/* m - 聚类大小                                                  *
/* T - nxk 计数矩阵 (t1,t2,...,tk)，其中 ti 是 *                          *
/* nxl 向量，表示类别 i 中的观测计数                    /
/****************************************************
Start MLERCM(m, T);
print "**费舍评分法 -- 随机聚类多项分布 ***";
pi0 = QLEpi(m,T);
rho0 = QLErho(m,T,pi0);
k = ncol(T);
n = nrow(T);
theta[1:k-1] = j(k,1,0);
theta[k] = rho0;

eps = 1E-8;
conv = 1;
iter = 0;
Do Until(conv<eps);

  iter = iter + 1;
  pi = theta[1:k-1];
  rho = theta[k];
  INFIM = Inv(n*FIMCM(m, pi, rho));
  Old = 0;
  Scores = j(k,1,0);
  Do j = 1 to n;
    tj = T[j,1:k-1];
    Scores = Scores + gradRCM(m, pi, rho, tj);
    Old = Old + log(ProbRCM(m, pi, rho, tj));
  End;

  si = 0;
  Lnew = Old - 1;
  Do Until (Lnew > Old);
    thetaNew = theta + {0.5*si}INVNFIM*Scores;
    pi = thetaNew[1:k-1];
    rho = thetaNew[k];
    Lnew = 0;
        Do j = 1 to n;
            tj = T[j,1:k-1];
            Lnew = Lnew + log(ProbRCM(m, pi, rho, tj));
        End;

        si = si + 1;
        if si > 20 then;
            print "NOTE: 步长减了二十次，假设收敛";
            Lnew = Old + 0.000001;
        end;
    End;

    theta = thetaNew;
    conv = abs(Lnew)/(abs(Lnew)+1E-6);
    conv_0 = conv;

    if si > 20 then conv = eps - 0.000001;
    if iter > 100 then;
        print "NOTE: 迭代 100 次后，尚未满足收敛";
        conv = eps - 0.000001;
    end;

End;

if conv_0 < eps then;
    print "NOTE: 收敛标准 (EPSILON-1E-8) 已满足";
end;

Return(theta);

Finish;

# 252 SAS 中的过度离散模型

```sas
******************************************************
/* 费舍信息矩阵                                */
******************************************************

******************************************************
/* Module: FIMDM(m,pi)                                */
/* 此模块返回狄利克特 *                                
/* 多项分布的费舍信息矩阵                                */
/* m = 聚类大小                                                    
/* pi = 概率向量 [pi1, pi2, ..., pk]               */
/* rho - 相关系数                                                    */
******************************************************

Start FIMDM(m, pi);
k = nrow(pi);
c = (1-rho)/rho;

FIM = j(k, c, 0);
PrK = 0;
PrC = 0;
pik = 1-sum(pi);
Do s = 1 to m;
    PrK = PrK + ProbDM(m, pi, rho, s)/(c*pi[s]-s)**2;
    PrC = PrC + (c+s-1)**-2;
end;

Do i = 1 to (k-1);
    PrI = 0;
    Do s = 1 to m;
        PrI = PrI + ProbDM(m, pi[i], rho, s)/(c*pi[i]+s-1)**2;
    end;

    FIM[i,j] = (c**2)*(PrI + PrK);
    Do j = (i+1) to (k-1);
        FIM[j,i] = (c**2)*(PrI + PrK);
    End;

    FIM[i,k] = c*(pi*PrK - pi[i]*PrI) * (2/rho**3);
    FIM[k,i] = c*(pi*PrK - pi[i]*PrI) * (2/rho**3);
    End;

    FIM[k,k] = (FIM[k,k] + (pi**2)*PrK - PrC) * (4/rho**6);
End;

Return (FIM);

Finish;
Store Module=FIMDM;
```
