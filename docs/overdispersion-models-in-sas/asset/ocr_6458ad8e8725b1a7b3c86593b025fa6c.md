Run;

%Macro DirMult_RCMult(data, label);

title &label;

Proc IML;

use &data;
read all into T;

n = nrow(T);
k = ncol(T);
m = sum(T(1,1));

Parameter = shape((Pi(1)' 'Pi(2)' 'Pi(3)' 'Pi(4)' 'Pi(5'
'Pi(6)' 'Pi(7)' 'Pi(8)' 'Pi(9)',k-1)) // (/Rho) // (/Log Likelihood);
PValue = j(i,1,0);

mattrib Estimate format=12.5;
mattrib StErr format=12.5;
mattrib ZStat format=0.2;
mattrib PValue format=pvalue6.4;

/**************************************/
/* MLE from DM Distribution
/**************************************/
Load Module=MLEDM;
pimLE1 = j((k-1),1,0);

thetaMLE = MLEDM(m, T);
pimLE1 = thetaMLE[1];
rhoMLE = thetaMLE[k];

INVFIM = Inv*FIDIM(m, pimLE1, rhoMLE);
SEI = j(i,1,0);
Do i = 1 to k;
SEI[i] = sqrt(INVFIM[i,i]);

End;

l = 0;
Do j = 1 to n;
tj = T[j,1:(k-1)];
tll = ll + log(ProBM(m, pimLE1, rhoMLE, tj));

End;

Estimate = pimLE1 // rhoMLE;
StErr = SEI;
ZStat = Estimate /SEI;
Estimate = Estimate // ll;
PValue = 1 - probnorm(ZStat);

print "Maximum Likelihood Estimates - Dirichlet-Multinomial Distribution"
print Parameter Estimate StErr ZStat PValue;
