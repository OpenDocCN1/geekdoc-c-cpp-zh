* Categories with expected frequencies smaller than 5 are collapsed with prior category;
do false = 1 to 1; Necessary to GOTO statement:

```
class = k;
index = 1;
s = 1;
aux2 = j(k,1,0);
aux4 = j(k,1,0);

begin2;
aux2[index] = expect[s];
aux4[index] = upperb[s];

while (s < k) {
  if (aux[index] < 5) then
    s = s+1;
    class = class-1;
    aux2[index] = aux2[index] + expect[s];
    aux4[index] = upperb[s];
  end;
  else
    index = index + 1;
    s = s+1;
    goto begin2;
  end
end;

if (aux2[index] < 5) then
  class = class-1;
  aux2[index-1] = aux2[index] + aux2[index-1];
end;

* Compute GOF test;

```
obs = (r, 1, 0); * Observed counts;
ratio = r/n;
do s = 1 to r-1
obs[s] = sum(ratio > lower[s]) & (ratio <= upper[s]);
end;
obs[r] = n - sum(obs[1:r]);
lower[1] = 1.00000;
obs_sum = sum(obs);
exp_sum = sum(expected);
