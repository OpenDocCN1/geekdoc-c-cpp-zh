*****************************************************************
Starting IML procedure;
********************************************************************

proc iml;

+   * Print input information;
file print;
********************************************************************************

+   * Module: FISHERB: This module can be called to
Compute Fisher information under BB distribution;
********************************************************************

```
start FISHERB;
dojo ij = 1 to size m;
AH = j(2,2,0);

doi i = 1 to mj[ij]+1;
u = -1;
mct = exp(gamma(mj[ij]+1)-gamma(u+1)-gamma(mj[ij]-u+1));
part1 = exp(gamma(c)-gamma(mj[ij]+c));
part2 = exp(gamma(u*pi)-gamma(c*pi));
part3 = exp(gamma(mj[ij]-u+c*-pi)-gamma(c-1*pi));
```

```
pt = (mct*part2*part3);

aux3 = 0;
aux2 = 0;
aux1 = 0;
```

```
aux3 = sum(1/(c;c+mj[ij] = 1));
if (u > 0) then aux1 = sum(1/(c:pi + u -1));
if (u < mj[ij]) then aux2 = sum(1/(c:cpic+mj[ij] - u));
```

```
derbb1 = c*(aux1-aux2);
derbb2 = (pi*aux1+pi*aux2-aux3) b;
```

+   * Compute Fisher information for each different;

```
AH[1,1] = AH[1,1] + derbb1*pt;
AH[1,2] = AH[1,2] + derbb2*pt;
AH[2,2] = AH[2,2] + derbb2*pt;
* end = 1 to m do loop;
end;
```

```
AH[2,1]=AH[1,2];
FIMB = FIMB + fj[ij]*AH;
```
