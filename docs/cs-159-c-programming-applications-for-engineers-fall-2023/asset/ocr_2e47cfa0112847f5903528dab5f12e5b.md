# Use the program below for problems #19 – #21

```
int main(void)
{
   int x = 1;
   float y = 0;
   float z = 2.75;
   y = alter_data(x, z);
   printf("y: %.2f\n", y);
   x *= z;
   printf("Result: %d\n", alter_data(-z, x));
   return(0);
}
int alter_data(float a, float b)
{
   return(a + (int)b + a * b);
}
```
