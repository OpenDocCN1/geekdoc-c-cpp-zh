```
char determineGrade(int score)
{
   char grade;

   if(score >= MINA)
   {
      grade = 'A';
   }
   else if(score >= MINB)
   {
      grade = 'B';
   }
   else if(score >= MINC)
   {
      grade = 'C';
   }
   else if(score >= MIND)
   {
      grade = 'D';
   }
   else
   {
      grade = 'F';
   }

   return(grade);
}
```
