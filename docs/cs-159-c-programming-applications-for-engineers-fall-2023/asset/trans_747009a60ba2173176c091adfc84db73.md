## 以下程序用于问题 #12 – #13

```
int calcResult(float, float);

int main(void)
{
    int x = 3;
    int y = 2;
    float z = 0.5;
    int value;

    value = calcResult(x / y, y);
    printf("Result #1: %d\n", value);

    value = calcResult(z * 3 / 2, z / 2);
    printf("Result #2: %d\n", value);

    return(0);
}

int calcResult(float a, float b)
{
    int c;
    c = a * b + a / b;
```
