# Use the program below for problems #1–#2

```
#include<stdio.h>
int makeCalc(int, int);
int main()
{
  int n = 4;
  int a;
  int result = 1;
  while(n > 2)
  {
    a = n--;
    result = makeCalc(a, result);
  }
  printf("result: %d\n", result);
  printf("a: %d n: %d\n", a, n);
  return(0);
}
```

```
int makeCalc(int a, int result)
{
  do
  {
    result *= a--;
  }while(a > 1);
  return(result);
}
```
