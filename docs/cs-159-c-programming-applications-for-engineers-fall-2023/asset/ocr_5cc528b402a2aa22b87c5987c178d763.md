# Use the program below for problems #45 – #46

```
#include<stdio.h>
int main(void)
{
 int a = 10;
 int b = 7;
 int *x;
 int *y;
 x = &b;
 y = &a;
 *x = -a % 5;
 *y -= b;
 printf("a = %d b = %d\n", a, b);
 x = y;
 b /= *x;
 y = &b;
 printf("*x = %d *y = %d\n", *x, *y);
 return(0);
}
```
