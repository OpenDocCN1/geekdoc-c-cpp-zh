   for(i = SIZE - 1; i > 0; i--)
   {
      changeArray(data, i);
   }
   return(0);
}
void changeArray(int x[], int index)
{
   int i;
   int temp = -1;
   for(i = index; i >= 0; --i)
   {
      if(x[i] % 2 != 0)
      {
         temp = i;
      }
   }
   if(temp != -1)
   {
      i = x[temp];
      x[temp] = x[index];
      x[index] = i;
   }
}
