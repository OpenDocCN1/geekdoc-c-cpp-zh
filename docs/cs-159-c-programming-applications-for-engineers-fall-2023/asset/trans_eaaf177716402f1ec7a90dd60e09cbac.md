## 使用以下程序解决第 48-49 题

```
#include<stdio.h>
void changeArr(int*, int*);

int main(void)
{
    int *x;
    int y[6] = {5, -1, 9, 10, 8};
    x = &y[5];
    changeArr(y, x);
    printf("%d %d\n", *x, y[3]);
    return(0);
}

void changeArr(int *a, int *b)
{
    int i;
    int temp;
    for(i = 0; i < 3; i++)

        temp = *(b - i);
        *(b - i) = a[i];
        a[i] = temp;

}
```
