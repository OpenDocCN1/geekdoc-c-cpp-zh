// and.cpp -- using the logical AND operator
#include <iostream>
const int ArSize = 6;
int main()
{
    using namespace std;
    float naaq[ArSize];

    while (i < ArSize && temp >= 0) // 2 quitting criteria
    {

        ++i;
        if (i < ArSize)              // room left in the array

        }
    }
