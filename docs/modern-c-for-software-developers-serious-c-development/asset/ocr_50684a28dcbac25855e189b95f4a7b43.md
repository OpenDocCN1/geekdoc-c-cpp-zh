If a class constructor has two or more arguments, we will have to use the following initialization syntax:

```
#include <iostream>
using namespace std;

class Foo {
  int h;
  int i;
public:
  // constructor with 2 parameters
  Foo(int j, int k) { h=j; i=k; }
  int get_i() { return i; }
  int get_h() { return h; }
};

int main() {
  Foo obj[3] = {
    Foo(1, 2), // initializer list of constructors
    Foo(3, 4),
    Foo(5, 6)
  };

  int i;
  for (i = 0; i < 3; i++)
    cout << obj[i].get_h();
```
