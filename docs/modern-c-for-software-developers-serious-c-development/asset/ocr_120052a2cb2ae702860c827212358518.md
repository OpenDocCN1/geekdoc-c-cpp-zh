void f(Base& r) {

}

  Base b;
  Derived1 d1;
  Derived2 d2;
  f(b); // pass a base class reference to f()
  f(d1); // pass a derived1 class reference to f()
  f(d2); // pass a derived2 class reference to f()
  return 0;
}
