```
class Foo
{
  public:
    int id;
    const char* name;
    Foo(int i, const char* nm)
    {
      id = i;
      name = nm;
      std::cout << "Foo constructor name = " << name << std::endl;
    }
  ~Foo() { std::cout << "destroying " << name << std::endl; }
  void negative(Foo object) { object.id = -object.id; }
}
```

```
int main()
{
  Foo(1, "foo");
  Foo bar(100, "bar");
  foo.negative(bar);

  return 0;
}
```

```
// output
Foo constructor name = foo
Foo constructor name = bar
destroying bar
destroying bar
destroying foo
```

Notice the copy of bar was passed to foo is destroyed after the call to negative returns and the object bar created on the stack of main is destroyed.
