# The code for the Add/Modify Customer form

```
public partial class frmAddModifyCustomer : Form
{
    public bool addCustomer;
    public Customer customer;

    private void frmAddModifyCustomer_Load(object sender, EventArgs args)
    {
        this.LoadStateCombo();
        if (addCustomer)
        {
            this.Text = "Add Customer";
            cbStates.SelectedIndex = -1;
        }
        else
        {
            this.Text = "Modify Customer";
            this.DisplayCustomer();
        }
    }

    private void LoadStateCombo()
    {
        List<State> states = new List<State>();
        try
        {
            states = DB.GetStates();
            cbStates.DataSource = states;
            cbStates.DisplayMember = "StateName";
            cbStates.ValueMember = "StateCode";
        }
        catch (Exception ex)
        {
            MessageBox.Show(ex.Message, ex.GetType().ToString());
        }
    }

    private void DisplayCustomer()
    {
        txtName.Text = customer.Name;
        txtAddress.Text = customer.Address;
        txtCity.Text = customer.City;
        cbStates.SelectedValue = customer.State;
        txtZipCode.Text = customer.ZipCode;
    }
}
```
