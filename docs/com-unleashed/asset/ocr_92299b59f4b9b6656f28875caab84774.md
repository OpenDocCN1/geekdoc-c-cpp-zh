LPCSTR FindOneOf(LPCSTR p1, LPCSTR p2)
{
    while ((p1 != NULL && *p1 != NULL)
    {
        LPCSTR p = p2;
        while (p != NULL && *p != NULL)
        {
            if (*p1 == *p)
                return CharNext(p1);
            p = CharNext(p);
        }
        p1 = CharNext(p1);
    }
    return NULL;
}
