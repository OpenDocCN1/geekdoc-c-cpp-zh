We can also implement this algorithm iteratively:

```
1 public static int count2SI(int num) {
2 int count2fs = 0, digit = 0;
3 int j = num, seidigits=0,position=0, pow10_pos = 1;
4 /* maintaining this value instead of calling pow() is a 6x perf
5 * gain (48s -> 8s) pow10_posMinus1. maintaining this value
6 * instead of calling Numof2s is an 2x perf gain (8s -> 4s).
7 * overall > 10x speed /
8 while (j > 0) {
9 digit = j % 10;
10 pow10_posMinus1 = pow10_pos / 10;
11 count2fs += digit * position * pow10_posMinus1;
12 /* we do not do this if digit <, >, or = 2
13 * Digit < 2 implies there are no 2s contributed by
14 * digit.
15 * Digit == 2 implies there are 2 * numof2s contributed by
16 the previous position + num of 2 * */
17 if (digit == 2) {
18 count2fs += seidigits + 1;
19 }
20 /* Digit > 2 implies there are digit * num of 2s by the prev.
21 * position + 10^position */
22 else if(digit > 2) {
23 count2fs += pow10_pos;
24 }
25 seidigits = seidigits + pow10_pos * digit;
26 pow10_pos *= 10;
27 position++;
28 j = j / 10;
29 }
30 return(count2fs);
31 }
```
