struct node *delete_beg(struct node *start)
{
    struct node *ptr;
    ptr = start;
    start = start -> next;
    start -> prev = NULL;
    free(ptr);
    return start;
}
struct node *delete_end(struct node *start)
{
    struct node *ptr;
    ptr = start -> next;
    while(ptr -> next != NULL)
    ptr = ptr -> next;
    ptr -> prev -> next = NULL;
    free(ptr);
    return start;
}

    ptr = start;

    ptr = ptr -> next;

    if(temp == start)
    start = delete_beg(start);

    {
    ptr -> prev = temp -> prev;

    free(temp);

}
struct node *delete_list(struct node *start)
{
    while(start != NULL)
    start = delete_beg(start);
