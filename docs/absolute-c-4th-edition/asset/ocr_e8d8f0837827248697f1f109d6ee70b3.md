template<class T>
T Queue<T>::remove()
{
    if (isEmpty())
    {
        cout << "Error: Removing an item from an empty queue.\n";
        exit(1);
    }

    T result = front->getData();

    Node<T> *discard;
    discard = front;
    front = front->getLink();
    if (front == NULL) //if you removed the last node
        back = NULL;

    delete discard;

    return result;
}
