# 116. 编写一个程序读取一个单词并显示其中包含元音和辅音的数量。

```cpp
#include <iostream>
#include <string>
#include <cctype>

int main() {
    std::string word;
    int vowelCount = 0, consonantCount = 0;

    // 获取输入的单词
    std::cout << "Enter a word: ";
    std::cin >> word;

    // 遍历每个字符并统计元音和辅音
    for (char c : word) {
        // 将字符转换为小写以统一处理
        c = std::tolower(c);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
            vowelCount++;
        } else if (std::isalpha(c)) { // 确保字符是字母
            consonantCount++;
        }
    }

    std::cout << "Number of vowels: " << vowelCount << std::endl;
    std::cout << "Number of consonants: " << consonantCount << std::endl;

    return 0;
}

# 118. 实现一个方法，使用重复字符的计数进行基础字符串压缩。例如，字符串 `aabcccccaaa` 将变为 `a2b1c5a3`。

```cpp
#include <iostream>
#include <string>
#include <sstream>

int main() {
    std::string input;
    std::cout << "Enter a string: ";
    std::cin >> input;

    if (input.empty()) {
        std::cout << "Compressed string: " << input << std::endl;
        return 0;
    }

    std::stringstream compressed;
    char currentChar = input[0];
    int count = 1;

    for (int i = 1; i < input.length(); i++) {
        if (input[i] == currentChar) {
            count++;
        } else {
            compressed << currentChar << count;
            currentChar = input[i];
            count = 1;
        }
    }

    // 加上最后一个字符及其计数
    compressed << currentChar << count;

    std::string result = compressed.str();
    if (result.length() < input.length()) {
        std::cout << "Compressed string: " << result << std::endl;
    } else {
        std::cout << "Compressed string: " << input << std::endl;
    }
    return 0;
}

# 119. Rawwords：编写一个算法检查一个单词是否为“rawword”（原始单词）。
如果一个单词的字母值之和（'a' = 1, 'b' = 2, 等）是一个质数，则该单词被视为“质词”。

```cpp
#include <iostream>
#include <string>

int main() {
    std::string word;
    std::cout << "Enter a word: ";
    std::cin >> word;

    int sum = 0;
    for(char c : word) {
        // 确保字符是小写字母
        if(c >= 'a' && c <= 'z') {
            sum += (c - 'a' + 1); // 'a' = 1, 'b' = 2, 等.
        }
    }

    // 检查是否为质数。质数只能被1和它整除。
    bool isPrime = true;

    // 边界情况：0和1不是质数
    if(sum <= 1) {
        isPrime = false;
    }

    // 检查从2到根(sum)的整除性
    for(int i = 2; i*i <= sum; i++) {
        if(sum % i == 0) {
            isPrime = false;
        }
    }

    if(isPrime) {
        std::cout << word << " is a rawword." << std::endl;
    } else {
        std::cout << word << " is not a rawword." << std::endl;
    }

    return 0;
}
``` # 120. 给定字符串列表，写一个函数确定最公共前缀。

```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

int main() {
    // 示例字符串列表
    std::vector<std::string> strs = {"flower", "flow", "flight"};

    if (strs.empty()) {
        std::cout << "No common prefix" << std::endl;
        return 0;
    }

    // 取最短字符串作为候选前缀
    std::string prefix = *std::min_element(strs.begin(), strs.end(), [](const std::string &s1, const std::string &s2) {
        return s1.size() < s2.size();
    });

    for (const std::string &str : strs) {
        // 将当前前缀与字符串比较
        // 如果不匹配，缩小前缀长度并重试
        while (str.substr(0, prefix.size()) != prefix) {
            prefix = prefix.substr(0, prefix.size() - 1);
            if (prefix.empty()) {
                std::cout << "No common prefix" << std::endl;
                return 0;
            }
        }
    }

    std::cout << "Longest Prefix: " << prefix << std::endl;
    return 0;
}

    for (int i = 0; i < 4; i++) {

        std::cout << std::endl;
    }

    // 转置矩阵
    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 4; j++) {
            transposed[j][i] = matrix[i][j];
        }
    }

    // 显示转置矩阵
    for (int i = 0; i < 4; i++) {
