// Check for duplicates
for (int i = 0; i < n; i++) {
  for (int j = i + 1; j < n; j++) {
    if (arr[i] == arr[j]) {
      hasDuplicate = true;
      break;
    }
  }
  if (hasDuplicate) {
    break;
  }
}

if (hasDuplicate) {
  std::cout << "The array contains duplicates.\n";
} else {
  std::cout << "The array does not contain duplicates.\n";
}

return 0;
}
