int main() {
    std::string word;
    int count = 0;

    // Get the word input
    std::cout << "Enter a word: ";
    std::cin >> word;

    // Loop through each character and count vowels
    for (char c : word) {
        // Convert the character to lowercase for uniformity
        c = std::tolower(c);
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u') {
            count++;
        }
    }

    std::cout << "Number of vowels: " << count << std::endl;

    return 0;
}
