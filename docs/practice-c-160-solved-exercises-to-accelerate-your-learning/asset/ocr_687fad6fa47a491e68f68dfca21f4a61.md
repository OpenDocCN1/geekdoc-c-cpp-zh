std::cin >> n;

// Allocate memory for the array
int* arr = new int[n];

// Prompt the user for the array elements
std::cout << "Enter " << n << " integers:" << std::endl;
for (int i = 0; i < n; i++) {
    std::cin >> arr[i];
}

// Find the largest element
int max = arr[0];
for (int i = 1; i < n; i++) {
    if (arr[i] > max) {
        max = arr[i];
    }
}

// Display the largest element
std::cout << "The largest element is: " << max << std::endl;
