int n = sizeof(arr) / sizeof(arr[0]);

std::cout << "Given array: \n";
for (int i = 0; i < n; i++) {
    std::cout << arr[i] << " ";
    std::cout << "\n";
}

quickSort(arr, 0, n - 1);

std::cout << "Sorted array: \n";
for (int i = 0; i < n; i++) {
    std::cout << arr[i] << " ";
    std::cout << "\n";
}

return 0;
}
