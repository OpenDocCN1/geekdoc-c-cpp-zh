# 97. 给定一个大小为 n 的数组，找到一个出现次数超过 n/2 次的元素。（该元素被称为多数元素。）

```cpp
#include <iostream>

int main() {
    int n;

    // 获取数组大小
    std::cout << "Enter the number of elements:";
    std::cin >> n;

    int arr[n];

    // 输入元素
    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> arr[i];
    }

    // Boyer-Moore 投票算法
    int candidate = -1;
    int count = 0;

    for (int i = 0; i < n; i++) {
      if (count == 0) {
        candidate = arr[i];
      }
      if (arr[i] == candidate) {
        count++;
      } else {
        count--;
      }
    }

    // 验证候选元素确实是多数元素
    count = 0;
    for (int i = 0; i < n; i++) {
      if (arr[i] == candidate) {
        count++;
      }
    }

    if (count > n / 2) {
        std::cout << "The majority element is: " << candidate << std::endl;
    } else {
        std::cout << "There is no majority element.\n";
    }

    return 0;
}
```

# 100. 给出一个随机数组，在不改变顺序的情况下将所有零推到末尾。

```cpp
#include <iostream>
#include <vector>

int main() {
    int n;

    std::cout << "Enter the number of elements: ";
    std::cin >> n;

    std::vector<int> nums(n);

    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> nums[i];
    }

    int count = 0; // 非零元素计数
    // 遍历数组，如果元素非零，则将其放在下一个可用位置
    for (int i = 0; i < n; i++) {
      if (nums[i] != 0) {
        nums[count++] = nums[i];
      }
    }

    // 将剩余位置置为零
    while (count < n) {
      nums[count++] = 0;
    }

    std::cout << "Array after pushing zeros to the end:\n";
    for (int num : nums) {
      std::cout << num << " ";
    }
    std::cout << std::endl;
    return 0;
}
```

# 101. 给出一个无排序整数数组，找到最长元素序列的长度。

```cpp
#include <iostream>
#include <unordered_set>
#include <vector>

int main() {
    std::vector<int> nums;
    std::unordered_set<int> set;
    int n, input, longestStreak = 0;

    std::cout << "Enter the number of elements: ";
    std::cin >> n;

    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> input;
        nums.push_back(input);
        set.insert(input);
    }

    for (int num : nums) {
      // 检查此数字是否为序列的起点
      if (set.find(num - 1) == set.end()) {
        int currentNum = num;
        int currentStreak = 1;

        // 检查序列中的后续元素
        while (set.find(currentNum + 1) != set.end()) {
          currentNum += 1;
          currentStreak++;
        }

        // 更新最长长度
        if (currentStreak > longestStreak) {
          longestStreak = currentStreak;
        }
      }
    }

    std::cout << "The length of the longest consecutive sequence is: " << longestStreak << std::endl;
    return 0;
}
```
