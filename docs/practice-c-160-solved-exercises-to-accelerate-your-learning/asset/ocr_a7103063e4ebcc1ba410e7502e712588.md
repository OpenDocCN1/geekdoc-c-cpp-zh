int main() {
  double sum = 0.0;
  int count = 0;
  double num;

  std::cout << "Enter numbers (enter 0 to stop): "
  << std::endl;

  do {
    std::cin >> num;
    if (num != 0) {
      sum += num;
      count++;
    }
  } while (num != 0);

  if (count != 0) {
    double average = sum / count;
    std::cout << "Average of the numbers entered: "
    << average << std::endl;
  } else {
    std::cout << "No numbers were entered." <<
  }
