# 7. Make a program that asks for the user's name, and prints the name immediately after.

```cpp
#include <iostream>
#include <string>

int main() {
    std::string name;
    std::cout << "Enter your name: ";
    std::getline(std::cin, name);
    std::cout << "Hello, " << name << "!" << std::endl;
    return 0;
}
```

# 8. Make a program that reads the names of three people in sequence, followed by their respective ages, and prints data in table format.

```cpp
#include <iostream>
#include <iomanip>
#include <string>

int main() {
    std::string name1, name2, name3;
    int age1, age2, age3;

    // Details for the first person
    std::cout << "Enter the name of the first person: ";
    std::getline(std::cin, name1);
    std::cout << "Enter the age of " << name1 << ": ";
    std::cin >> age1;
    std::cin.ignore(); // Clear the newline left by the previous input

    // Details for the second person
    std::cout << "Enter the name of the second person: ";
    std::getline(std::cin, name2);
    std::cout << "Enter the age of " << name2 << ": ";
    std::cin >> age2;
    std::cin.ignore(); // Clear the newline left by the previous input

    // Details for the third person
    std::cout << "Enter the name of the third person: ";
    std::getline(std::cin, name3);
    std::cout << "Enter the age of " << name3 << ": ";
    std::cin >> age3;
    std::cin.ignore(); // Clear the newline left by the previous input

    // Output the details in table format
    std::cout << std::endl;
    std::cout << std::left << std::setw(20) << "Name" << std::setw(5) << "Age" << std::setw(10) << "Height(m)" << std::endl;
    std::cout << "--------------------------------" << std::endl;
    // (Note: Height was not requested in the prompt logic but exists in the table structure)
    std::cout << std::left << std::setw(20) << name1 << std::setw(5) << age1 << std::endl;
    std::cout << std::left << std::setw(20) << name2 << std::setw(5) << age2 << std::endl;
    std::cout << std::left << std::setw(20) << name3 << std::setw(5) << age3 << std::endl;

    return 0;
}
```
