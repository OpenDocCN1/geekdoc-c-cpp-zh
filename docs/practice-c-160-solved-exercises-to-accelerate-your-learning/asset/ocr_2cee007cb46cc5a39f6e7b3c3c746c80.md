int main() {
    std::string name;

    // Get the name input
    std::cout << "Enter a name: ";
    std::getline(std::cin, name);

    // Check if the first character of the name is 'A'
    if (name[0] == 'A' || name[0] == 'a') {
        std::cout << "The name starts with the letter 'A'.\n";
    } else {
        std::cout << "The name does not start with the letter 'A'.\n";
    }

    return 0;
}
