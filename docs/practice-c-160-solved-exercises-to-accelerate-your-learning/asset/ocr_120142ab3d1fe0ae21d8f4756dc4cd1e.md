std::cin >> B;
if (A > B) {
    std::cout << "A should be less than or equal to B.\n";
    return 1; // Exit with error code
}

// Display numbers between A and B
std::cout << "Numbers between " << A << " and " << B << " are:\n";

for (int i = A; i <= B; i++) {
    std::cout << i << " ";
}

std::cout << std::endl; // Newline after printing all numbers
return 0;
}
