```
// Sliding window approach
for (int i = k; i < n; i++) {
    sum += nums[i] - nums[i - k];
    if (sum < min_sum) {
        min_sum = sum;
        start_index = i - k + 1;
    }
}
```

```
std::cout << "The subarray of length " << k << "with the least average is:\n"
for (int i = start_index; i < start_index + k; i++) {
    std::cout << nums[i] << " ";
}
std::cout << std::endl;
```
