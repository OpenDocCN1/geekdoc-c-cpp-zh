> 列表 6.12: timestamp.cpp

```cpp
#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    int size; // 表格的行列数
    cout << "Please enter the table size: ";
    cin >> size;
    // 打印一个 size x size 的乘法表

    // 第一个标题：1 2 3 4 5 等
    cout << endl;
    // 打印列标题
    int column = 1;
    while (column <= size) {
        cout << setw(4) << column; // 打印该列的标题。
        column++;
    }

    // 打印分隔线：+---------------------
    cout << endl << "+";
    column = 1;
    while (column <= size) {
        cout << "----"; // 打印该列的线。
        column++;
    }
    cout << endl;

    // 打印表格内容
    int row = 1;
    while (row <= size) { // 表格有 size 行。
        cout << setw(2) << row << " ||"; // 行标题。
        column = 1; // 每行重置列。
        while (column <= size) { // 表格有 size 列。
            int product = row*column; // 计算积
            cout << setw(4) << product; // 元素
            column++;
        }
        row++; // 移动到下一行
        cout << endl;
    }
