### Listing 18.6: genericpoint.h
```cpp
#ifndef GENERICPOINT_H
#define GENERICPOINT_H_

template <typename T>
class Point {
public:
    T x;
    T y;
    Point(T x, T y) : x(x), y(y) {}
};

#endif
```

If a client declares a Point object as
`Point<int> pixel(10, 10);`

the compiler will instantiate a Point<int> class that looks like

```cpp
class Point {
public:
    int x;
    int y;
    Point(int x, int y) : x(x), y(y) {}
};
```

If a client declares a Point object as
`Point<float> pixel(10.0, 20.0);`

the compiler will instantiate a Point<double> class that looks like

```cpp
class Point {
public:
    double x;
    double y;
    Point(double x, double y) : x(x), y(y) {}
};
```

Neither of these instantiations in any source code, a programmer will not see them. The compiler substitutes actual types (int or double) for the parameterized type (T).

As a significant example, Listing 18.7 (genericcompar.h) contains the declaration of a Compar class that is the generic equivalent to Listing 17.18 (compar.h).

### Listing 18.7: genericcompar.h
```cpp
#ifndef GENERICCOMPAR_H
#define GENERICCOMPAR_H_

/*
 * Comparer objects manage the comparisons and element
 * interchanges on the selection sort function below.
 * */

template <typename T>
class Comparator {
    // The object's data is private, so it is inaccessible to
    // clients and derived classes

    // Keeps track of the number of comparisons
    // performed
    int compare_count;

    // Keeps track of the number of swaps performed
    int swap_count;

    // Function pointer directed to the function to
    // perform the comparison
    bool (*comp)(const T&, const T&);

protected:
    // Method that actually performs the comparison
    // Derived classes may customize this method
    virtual bool compare_impl(const T& m, const T& n) {
        return comp(m, n);
    }

    // Method that actually performs the swap
    // Derived classes may customize this method
    virtual void swap_impl(T& m, T& n) {
        T temp = m;
        m = n;
        n = temp;
    }

public:
    // The client must initialize a Comparator object with a
    // suitable comparison function.
    Comparator(bool (*f)(const T&, const T&)),
        compare_count(0), swap_count(0), comp(f) {}

    // Resets the counters to make ready for a new sort
    void reset() {
        swap_count = 0;
    }

    // Method that performs the comparison. It delegates
    // the actual work to the function pointed by comp.
    // this method logs each invocation.
    bool compare(const T& m, const T& n) {
        compare_count++;
        return compare_impl(m, n);
    }

    // Method that performs the swap. Interchange the values
    // of a and b which are passed by reference
    // this method logs each invocation.
    void swap(T& m, T& n) {
        swap_count++;
        swap_impl(m, n);
    }

    // Returns the number of comparisons this object has
    // since it was created.
    int comparisons() const {
        return compare_count;
    }

    // Returns the number of swaps this object has
    // since it was created.
    int swaps() const {
        return swap_count;
    }
};
```

```cpp
#include <iostream>
#include <vector>
#include "genericcompar.h"

using namespace std;

/*
* selection_sort(a, compare)
* Arranges the elements of vector a in an order
* determined by the compare object.
* compare is a function that compares two integers.
* The contents of a are physically rearranged.
*/
template <typename T>
void selection_sort(vector<T>& a, Comparer<T>& compare) {
    int n = a.size();
    for (int i = 0; i < n - 1; i++) {
        // Note: i, small, and j represent positions within
        // a[i], a[small], and a[j] represents the elements at
        // those positions. small is the position of the smallest
        // value seen so far, we use it to find the smallest
        // than a[i];
        int small = i;
        // See if a smaller value can be found in the array
        for (int j = i + 1; j < n; j++) {
            if (compare(a[j], a[small])) {
                small = j; // Found a smaller value
            }
        }
        // Swap a[i] and a[small]; if a smaller value was found
        if (i != small) {
            compare.swap(a[i], a[small]);
        }
    }
}

Recall the IntList linked list list class from Listings 16.6 and 16.7. It implemented a linked list of integer elements. Suppose we wanted to make a linked list of std::string objects. Without C++ templates, we would have to copy the IntList code and modify it so it works with string objects instead of integers. We could call this StringList. The classes would be almost identical; the only major difference would be the type of elements the list can hold. C++'s class template feature is ideal for designing generic container classes. Listing 18.9 defines a generic linked list in its LList class.

### Listing 18.9: list.h
```cpp
#include <iostream>

template <typename T>
class LList {
    /*
     * An object that holds an element in a linked list.
     * This type is local to the LList class and inaccessible outside this class.
     */
    struct Node {
        T data;         // A element of the list
        Node* next;     // The node that follows one in the list
        // Constructor
        Node(const T& d) : data(d), next(nullptr) {}
    };

    Node *head; // Points to the first element in the list

    /
     * Private helper methods
     */
    // insert(p, elem)
    // Inserts elem onto the back of the list pointed to p.
    // p points to a node in a list
    // elem is the element to insert.
    // Returns a pointer to the list that contains elem.
    Node *insert(Node *p, const T& elem) {
        if (p) // Is p non-null?
            p->next = insert(p->next, elem);
        else // p is null, make a new node
            p = new Node(elem);
        return p;
    }

    /*
     * print(p)
     * Prints the contents of the linked list
     * p points to a linked list of nodes.
     */
    void print(Node *p) const {
        while (p) {
            std::cout << p->data << " "; // While p is not null
                                         // node's data
            p = p->next; // Move to next node
        }
        std::cout << std::endl;
    }

    /*
     * Returns the length of the linked list pointed to p.
     */
    int length(Node *p) const {
        if (p)
            return 1 + length(p->next); // 1 + length of rest of list
        else
            return 0; // Empty list has length zero
    }

    /*
     * dispose(p)
     * Deallocate the memory held by the list pointed to p.
     */
    void dispose(Node *p) {
        if (p) {
            dispose(p->next); // Free up the rest of the list
            delete p; // Deallocate this node
        }
    }

public:
    /*
     * The constructor makes an initially empty list
     */
    LLList() : head(nullptr) {}
};
```
