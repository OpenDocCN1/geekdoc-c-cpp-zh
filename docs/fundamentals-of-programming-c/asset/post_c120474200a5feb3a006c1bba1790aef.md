All that is left is to add the row and column titles and the lines that bound the edges of the table. Listing 6.12 (timestamp.cpp) adds the necessary code.

> Listing 6.12: timestamp.cpp

```cpp
#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    int size; // The number of rows and columns in the table
    cout << "Please enter the table size: ";
    cin >> size;
    // Print a size x multiplication table

    // First heading: 1 2 3 4 5 etc.
    cout << endl;
    // Print column heading
    int column = 1;
    while (column <= size) {
        cout << setw(4) << column; // Print heading for this column.
        column++;
    }

    // Print line separator: +---------------------
    cout << endl << "+";
    column = 1;
    while (column <= size) {
        cout << "----"; // Print line for this column.
        column++;
    }
    cout << endl;

    // Print table contents
    int row = 1;
    while (row <= size) { // Table has size rows.
        cout << setw(2) << row << " ||"; // Print heading for this row.
        int column = 1; // Reset column for each row.
        while (column <= size) { // Table has size columns.
            int product = row*column; // Compute product
            cout << setw(4) << product; // Next element
            column++;
        }
        row++; // Next row
        cout << endl;
    } // Move cursor to next row
}
```
