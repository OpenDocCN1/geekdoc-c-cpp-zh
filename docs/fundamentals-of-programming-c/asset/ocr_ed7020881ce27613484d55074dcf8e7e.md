/*
 * dispose(p)
 * Deallocate the memory held by the list pointed to by p.
 */
void IntList::dispose(IntList::Node *p) {
    if (p) {
        dispose(p->next); // Free up the rest of the list
        delete p; // Deallocate this node
    }
}
