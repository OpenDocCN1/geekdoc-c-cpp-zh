# Listing 6.15: exitnested.cpp

```
#include <iostream>

using namespace std;

int main() {
    // Compute some products
    int op1 = 2;
    while (op1 < 100) {
        int op2 = 2;
        while (op2 < 100) {
            if (op1 + op2 == 3731) {
                goto end;
            }
            cout << "Product is " << (op1 + op2) << endl;
            op2++;
        }
        op1++;
    }
:
cout << "The end" << endl;
```
