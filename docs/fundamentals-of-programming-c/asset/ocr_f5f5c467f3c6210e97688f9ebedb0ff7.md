#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    int size; // The number of rows and columns in the table
    cout << "Please enter the table size: ";
    cin >> size;
    // Print a size x multiplication table

    // First heading: 1 2 3 4 5 etc.
    cout << endl;
    // Print column heading
    int column = 1;
    while (column <= size) {
        cout << setw(4) << column; // Print heading for this column.
        column++;
    }

    // Print line separator: +---------------------
    cout << endl << "+";
    column = 1;
    while (column <= size) {
        cout << "----"; // Print line for this column.
        column++;
    }
