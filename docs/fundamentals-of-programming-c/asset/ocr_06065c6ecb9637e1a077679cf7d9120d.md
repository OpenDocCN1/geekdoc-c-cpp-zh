Listing 17.19: compar.cpp
#include "compar.h"

// Method that actually performs the comparison
// Derived classes may customize this method
bool Compar::compare_impl(int m, int n) {
    return comp(m, n);
}

// Method that actually performs the swap
// Derived classes may customize this method
void Compar::swap_impl(int& m, int& n) {
    int temp = m;
    m = temp;
    n = temp;
}

// The client must initialize a Compar object with a
// suitable comparison function.
Compar::Compar(bool (*xf)(int, int)):
    compare_count(0), swap_count(0), comp(xf);

// Resets the counter to make ready for a new sort
void Compar::reset() {
    compare_count = 0;
}

// Method that performs the comparison. It delegates
// the actual work to the function pointed to by.
// This method logs each invocation.
bool Compar::compare(int m, int n) {
    compare_count++;
    return compare_impl(m, n);
}
