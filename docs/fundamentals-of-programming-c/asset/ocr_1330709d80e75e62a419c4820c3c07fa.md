#include <iostream>
#include <vector>

using namespace std;

// Print the contents of vector v, traversing with a
// caller supplied increment value (inc)
void print(const vector<int>& v, int inc) {
    for (auto p = begin(v); p != end(v); p += inc)
        cout << p << ' ';
    cout << endl;
}

// Print the contents of the vector v backwards
// traversing with a caller supplied decrement value (dec)
void print_reverse(const vector<int>& v, int dec) {
    auto p = end(v);
    while (p != begin(v)) {
        p -= dec;
        cout << p << ' ';
    }
    cout << endl;
}

int main() {
    vector<int> vec(20);
    for (int i = 0; i < 20; i++)
        vec[i] = i;

    print(vec, 1);
    print(vec, 2);
    print(vec, 4);
    print(vec, 5);
    print(vec, 10);
    cout << endl;

    print_reverse(vec, 1);
    print_reverse(vec, 2);
    print_reverse(vec, 4);
    print_reverse(vec, 5);
    print_reverse(vec, 10);
}
