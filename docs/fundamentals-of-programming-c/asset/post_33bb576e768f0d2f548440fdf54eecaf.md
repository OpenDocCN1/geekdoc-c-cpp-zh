In order to determine if any of the elements is less than A[i], we introduce a new variable named small. The purpose of small is to keep track of the smallest element found so far. We
will set small equal to i initially because we wish to locate any element less than the element at
position i.

```cpp
// n is A's length
for (int i = 0; i < n - 1; i++) {
    // small is the position of the smallest value we've seen
    // so far; we use it to find the smallest value less than A[i]
    int small = i;
    for (int j = i + 1; j < n; j++) {

            small = j; // Found a smaller element, update
    }
```

```cpp
/*
* selection_sort
* Arranges the elements of vector into ascending order.
* a is a vector that contains integers.
*/
void selection_sort(vector<int>& a) {
    int n = a.size();
    for (int i = 0; i < n - 1; i++) {
        // Note: i, small, and j represent positions within a
        // a[i], a[small], and a[j] represents the elements at
        // positions.
        int small = i;
        // If a smaller value can be found later in the vector
        // we've seen so far; we use it to find the smallest value less
        // than a[i]
        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[small])
                small = j; // Found a smaller value
        }
        // Swap a[i] and a[small], if a smaller value was found
        if (i != small)
            swap(a[i], a[small]);
    }
}
```
