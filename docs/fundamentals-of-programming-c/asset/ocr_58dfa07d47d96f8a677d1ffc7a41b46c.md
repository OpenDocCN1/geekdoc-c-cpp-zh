17. Consider the following C++ program that prints one line of text:

```

using namespace std;

    int input;

    if (input < 10) {
        if (input != 5)
            cout << "wow";
        else
            input++;
    } else {
        if (input == 17)
            input += 10;
        else
            cout << "whoa";
    }
    cout << input << endl;
}
```
