```
#include <iostream>

using namespace std;

/*
 * get_int_range(first, last)
 * Forces the user to enter an integer within a
 * specified range
 * first is either a minimum or maximum acceptable value
 * last is the corresponding other end of the range,
 * either a minimum or maximum = value
 * Acquires an acceptable value from the user
 */
int get_int_range(int first, int int, int last) {
    // If the larger number is provided first,
    // switch the parameters
    if (first > last) {
        int temp = first;
        first = last;
        temp = temp;
    }
    // Insist on values in the range first...last
    cout << "Please enter a value in the range " <<
         << first << "..." << last << ": ";
    int in_value; // User input value
    bool bad_entry;
    do {
        cin >> in_value;
        bad_entry = (in_value < first || in_value > last);
        if (bad_entry) {
            cout << in_value << " is not in the range " <<
                 << first << "..." << last << endl;
            cout << "Please try again: ";
        }
    } while (bad_entry);
    // in_value at this point is guaranteed to be within range
    return in_value;
}

/*
 * main
 * Tests the get_int_range function
 */
int main() {
    cout << "first: " << get_int_range(10, 20) << endl;
    cout << "second: " << get_int_range(20, 10) << endl;
    cout << "third: " << get_int_range(5, 5) << endl;
    cout << "fourth: " << get_int_range(-100, 100) << endl;
}
```

Listing 9.11 (betterinputfunc.cpp) forces the user to enter a value within a specified range. This functionality can be used in many programs.
