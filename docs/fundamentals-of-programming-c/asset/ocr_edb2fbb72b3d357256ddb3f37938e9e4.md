# 19.5 Exception Mechanics

An exception handler within a catch block may take a few steps to handle the exception and re-throw the exception or completely a different exception.

```
Listing 19.9: rerow. cpp
#include <iostream>
#include <fstream>
#include <vector>

using namespace std;

void filter(vector<int>& v, int i) {
  v.at(i)++;
}

void compute(vector<int>& a) {
  for (int i = 0; i < 6; i++) {
    try {
      filter(a);
    } catch (exceptions& ex) {
      cout << "****************************************************" << endl;
      cout << "* Loop terminated prematurely" << endl;
      cout << "* when i = " << i << endl;
      cout << "****************************************************" << endl;
      throw ex;
    }
  }
}

int main() {
  vector<int> list { 10, 20, 30, 40, 50 };
  try {
    compute(list);
  } catch (exception& e) {
    cout << "Caught an exception: " << e.what() << endl;
  }
  cout << "Program finished" << endl;
}}
```
