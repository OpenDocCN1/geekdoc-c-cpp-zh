```
string message; // Identifies the exception and filename
public:
// Constructor establishes the exception object's message
FileNotFoundException(const string filename):
    message("" + filename + "" not found") { }
// Reveal message to clients
const char *what() const {
    return message.c_str();
}
// Creates and returns a vector of integers from data stored
// in a text file.
// filename: name of the text file containing the data
// Returns a vector containing the data in the file, if possible
vector<int> load_vector(const string filename) { // Open the file for reading
    if (fin.open(filename)) // Did the file open successfully?
        int n;
        fin >> n; // Size of data set
        vector<int> result(n); // Allocate space for the vector
        int i = 0;
        while (fin >> value)
            result.at(i++) = value; // Insert it into the vector
        fin.close(); // Close the file
        return result;
    } // Return the populated vector
else // Could not open the text file
```
