IntList::Node *IntList::insert(Node *p, int n {
   // Check that we are not at the end of the list and
   // that the current list element is less than the new
   // item to add.
   if (p && p->data < n)
      p->next = insert(p->next, n); // Insert into rest of list
   else {
      Node *temp = new Node;
      temp->data = n;
      temp->next = p;
      p = temp;
   }

}
