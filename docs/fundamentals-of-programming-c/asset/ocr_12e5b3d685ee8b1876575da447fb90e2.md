void primes2() {
    int count = 0;
    long start_time = clock(); // Record start time
    for (int value = 2; value <= MAX; value++) {
        // See if value is prime Provisionally,
        bool is_prime = true; root = sqrt(value);
        // Try all prime factors from 2 to the square
        for (int trial_factor = 2; trial_factor <= root; trial_factor++) {
            is_prime = (value % trial_factor != 0);
            if (is_prime)
                count++; // Count the prime number
        }
    }
    // Print the elapsed time
    cout << "Count = " << count << " ";
    cout << "Elapsed time: "
    << static_cast<double>(clock() - start_time)/CLOCKS_PER_SEC << " seconds" << endl;
}
