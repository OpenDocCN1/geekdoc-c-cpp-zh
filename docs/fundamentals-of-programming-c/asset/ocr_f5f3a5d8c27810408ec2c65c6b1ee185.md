4. Consider the following C++ code:

```
#include <iostream>

using namespace std;

int max(int n) {
    return n;
}

int max(int m, int n) {
    return (m >= n) ? m : n;
}

int max(int m, int n, int r) {
    int x = m;
    if (n > x)
        x = n;
    if (r > x)
        x = r;
    return x;
}

int main() {
    cout << max(4) << endl;
    cout << max(4, 5) << endl;
    cout << max(5, 4) << endl;
    cout << max(1, 2, 3) << endl;
    cout << max(2, 1) << endl;
    cout << max(2, 1, 2) << endl;
}
```
