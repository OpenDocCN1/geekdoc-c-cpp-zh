def iterator = new Iterator[(K,V)] {
    val stack = new ListStack[AVLNode[K,V]]
    pushRunLeft(AVLNode.this)
    def hasNext:Boolean = !stack.isEmpty
    def next:(K,V) = {
        val n = stack.pop()
        pushRunLeft(n.right)
        n.key -> n.data
    }
    def pushRunLeft(n:ImmutableAVLTreeMap[K,V]) {
        n match {
            case e:AVLEmpty[K] =>
            case i:AVLNode[K,V] =>
                stack.push(i)
                pushRunLeft(i.left)
        }
    }
}
