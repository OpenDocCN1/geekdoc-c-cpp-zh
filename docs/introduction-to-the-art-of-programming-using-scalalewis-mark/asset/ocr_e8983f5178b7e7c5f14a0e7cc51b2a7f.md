private def forCollaborators[A](f:Collaborator=>A):Seq[A] = {
(for(c <- sketches.keys) yield try {
Some(f(c))
} catch {
case ex:RemoteException => None
}).toSeq.filter(!_.isEmpty).map(_.get)
}
