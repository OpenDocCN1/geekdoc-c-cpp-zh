/* output the main menu and collect input */
int menu(void)
{
    int ch;

    printf("Show, A)dd, R)emove, Q)uit: ");
    ch=getchar();
    /* delete excess input */
    while(getchar()!='\n')
        ;
    return(toupper(ch));
}
