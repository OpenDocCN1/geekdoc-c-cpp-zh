```
operator =
{ arithmetic-operator
| bitwise-operator
| parts-selection-operator
| dereferencing-operator
| location-operator
| logical-operator
| augmentation-operator
| relational-operator
| equality-operator
}

arithmetic-operator = { + | - | * | / | % }

bitwise-operator = { & | | | ^ | << | >> | ~ }

parts-selection-operator = { . | -> }

dereferencing-operator = *

location-operator = &

logical-operator = { && | || | ! }

augmentation-operator = { ++ | -- }

relational-operator = { < | > | <= | >= }

equality-operator = { == | != }
```
