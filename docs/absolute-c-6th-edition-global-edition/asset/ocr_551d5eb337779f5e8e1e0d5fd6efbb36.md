```
16     void setData(const T& theData) { data = theData; }

18   private:

20     Node<T> *link;
21   };
22 
23   template<class T>
24   class Queue
25   {
26   public:
27     Queue( );
28     //Initializes the object to an empty queue.
29     Queue(const Queue<T>& aQueue);
30     Queue<T>& operator =(const Queue<T>& rightSide);
31     virtual ~Queue( );
32     void add(T item);
33     //Postcondition: item has been added to the back of the queue.
34     T remove( );
35     //Precondition: The queue is not empty.
36     //Returns the item at the front of the queue

```

This is the same definition of the template class Node that we gave for the stack interface in Display 17.17. See the "Tip: A Comment on Namespaces" for a discussion of this duplication.

You might prefer to replace the parameter type T with const T&.
