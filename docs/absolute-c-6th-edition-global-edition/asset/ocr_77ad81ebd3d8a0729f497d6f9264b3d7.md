Money::Money(int theDollars, int theCents)
{
    if ((theDollars < 0 && theCents > 0) ||
        (theDollars > 0 && theCents < 0))
    {
        cout << "Inconsistent money data.\n";
        exit(1);
    }
    dollars = theDollars;
    cents = theCents;
}
