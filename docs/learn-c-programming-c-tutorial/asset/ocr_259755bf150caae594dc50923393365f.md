printf( "Array values using pointer\\n");
for ( i = 0; i < 5; i++ )
{
    printf("*(p + %d) : %f\\n", i, *(p + i) );
}

printf( "Array values using balance as address\\n");
for ( i = 0; i < 5; i++ )
{
    printf("*(balance + %d) : %f\\n", i, *(balance + i) );
}

return 0;
}
