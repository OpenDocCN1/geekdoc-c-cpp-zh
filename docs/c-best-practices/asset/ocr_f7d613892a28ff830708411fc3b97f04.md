The if constexpr construct in C++17 can simplify this code: if constexpr option for compile time behavior change.

```
    #include <stdexcept>
    #include <type_traits>
    #include <utility>

    template <typename Numerator, typename Denominator>
    auto divide(Numerator numerator, Denominator denominator) {
        if constexpr (std::is_integral_v<Numerator> &&
                      std::is_integral_v<Denominator>) {
            // is integral division
            if (denominator == 0) {
                throw std::runtime_error("divide by 0!");
            }
        }

        return numerator / denominator;
    }
```
