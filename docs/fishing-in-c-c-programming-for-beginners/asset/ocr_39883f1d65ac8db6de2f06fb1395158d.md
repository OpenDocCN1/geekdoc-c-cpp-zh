```
#include<iostream>
using namespace std;
namespace ns1
{
    int a=1;
    int b=2;
}
namespace ns2
{
    float a=3.33;
    double b=4.44;
}
int main()
{
    using namespace ns1;
    cout<<a<<endl;
    cout<<b;
}
```

1
2
