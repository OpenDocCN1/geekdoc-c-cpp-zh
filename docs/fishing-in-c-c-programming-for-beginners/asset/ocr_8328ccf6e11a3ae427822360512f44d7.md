using namespace std;
int main()
{
    fstream fin;
    fin.open("example.txt", ios::in);
    if(fin==NULL)
    {
        cout<<("Can't open the file");
        return -1;
    }
    char ch;
    while(!(fin.eof()))
    {
        fin>>ch;
        cout<<ch;
    }
    fin.close();
    return 0;
}
