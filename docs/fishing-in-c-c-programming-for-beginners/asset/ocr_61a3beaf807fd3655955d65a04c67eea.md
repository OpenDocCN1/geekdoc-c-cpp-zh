# Reading a string from a file

We can also read a string from file. For this purpose we use a function getline()

```
```cpp
#include<iostream>
#include<fstream>
using namespace std;
int main()
{
    fstream fin;
    fin.open("example.txt",ios::in);
    if(fin==NULL)
    {
        cout<<<"Can't open the file");
        return -1;
    }
    char ch[10];
    while(!fin.eof())
    {
        fin.getline(ch,10);
        cout<<ch;
    }
    fin.close();
    return 0;
}
