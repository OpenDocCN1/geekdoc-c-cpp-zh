```
using namespace std;

int main()
{
    fstream fin("example.txt",ios::out);
    if(!fin.is_open())
    {
        cout<<"Can't open file";
        return -1;
    }

    fin<<"Hello World";
    if(fin.good())
    {
        cout<<"File written successfully";
        fin.clear();
    }

    fin.close();
    return 0;
}
```

Output

File written successfully
