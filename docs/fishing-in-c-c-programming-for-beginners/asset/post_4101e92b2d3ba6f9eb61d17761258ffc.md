
length=p;
breadth=q;
return length*breadth;
}
class side_area
{
public:
int x,z;
int calculate_side_area(int,int);
};
int side_area::calculate_side_area(int a,int b)
{
x=a;
z=b;
return x*z;
}
class cuboid : public base_area, public side_area
{
public:
int volume();
};
int cuboid::volume()
{
return length*breadth*z;
}
class surface_area : public cuboid
{
    public:
        int calculate_total_surface_area();
};
int surface_area::calculate_total_surface_area()
{
    return 2*(length*breadth+length*z+breadth*z);
}

int main()
{
    class surface_area s1;
    cout<<"Base area of cuboid is: "<<s1.calculate_base_area(3,4)<<endl;
    cout<<"Area of side wall is: "<<s1.calculate_side_area(3,5)<<endl;
    cout<<"Volume of cuboid is: "<<s1.volume()<<endl;
    cout<<"Total surface area of cuboid is: "<<s1.calculate_total_surface_area()<<endl;
}

Volume of cuboid is: 60

## Operator overloading

You may think that the operators like +,-,*,/ etc works on pre-defined data types like int, float, etc. then can these work on user defined data types like class. Yes they can! The operators can be overloaded and you can use them in whatever way you want.

The syntax of overloading function for an operator whose symbol is say opr is:

```
return_type operator opr()
{
//statements
}
```

e.g. for overloading an operator say + an example is:

```
int operator +()
{
return value1+value2;
}
```

Normally, we cannot increase, decrease any object. E.g. Following program will generate an error:

```
#include<iostream>
using namespace std;
class counting
{
    int count;
    public:
    counting(){count=0;}
    int return_value(){return count;}
};
int main()
{
    counting c1;
    cout<<c1.return_value()<<endl;
    ++c1;
    cout<<c1.return_value();
}
```

This program will generate an error saying “[Error] no match for ‘operator++’ (operand type is ‘counting’)”.

### Overloading increment and decrement operator (++ & --)

Now, we shall overload the operator ++ so that we can increase or decrease the objects.

Consider the following program:

```
#include<iostream>
using namespace std;
class counting
{
int count;
public:
counting() {count=0;}
int return_value(){return count;}
void operator++(){++count;}
};
int main()
{
counting c1;
cout<<c1.return_value()<<endl;
++c1;
cout<<c1.return_value();
}
```

Output

0

1

Similarly, we can overload the decrement operators too.

### Using return types in operator overloading

However, we cannot store the value since it has not returned anything. So, if we want to store the value in an object then we can return the value so that we can store it in an object. Following program illustrates this fact:

```
#include<iostream>
using namespace std;
class counting
{

int count;

public:

counting(){ count=0;}
void set_value(int a){count=a;}
int return_value(){return count;}
counting operator++();

};

counting counting::operator++()
{

++count;

counting cx;

cx.set_value(count);
return cx;
}

int main()
{
    counting c1,c2;
    cout<<c1.return_value()<<endl;
    ++c1;
    cout<<c1.return_value()<<endl;
    c2=++c1;
    cout<<c2.return_value();
}
```

Output
0
1
2

### Using this pointer in operator overloading

Using this pointer can further be written as:

```
#include<iostream>

}

### Overloading post-increment operator

Until now, we have overloaded only prefix operator not postfix operator. What if we try to overload the postfix operator.

Consider the following program:

```
#include<iostream>
using namespace std;
class counting
{
    int count;
    public:
    counting() { count=0;}
    int return_value(){return count;}
    void operator++(){++count;}
};

int main()
{
    counting c1;
    ++c1;
    cout<<c1.return_value()<<endl;
    c1++;
    cout<<c1.return_value()<<endl;
}
```

This program will display an error “[Error] no ‘operator++(int)’ declared for postfix ‘++’ [-fpermissive]”.

To overload the postfix operator, we have to pass an argument of integer type, whose value is ignored but it is an indication of the fact that we want to overload the postfix operator.

The syntax for post-increment operator is

```
return_type operator++(int)
{
//Statements
}
```

Consider the following program:

```
#include<iostream>
using namespace std;
class counting
{
    int count;
public:
    counting(){ count=0;}
    void set_value(int a){count=a;}
    int return_value(){return count;}
    counting operator++(); //preincrement
    counting operator++(int); //postincrement
};
counting counting::operator++()
{
    count++;
    counting cx;
    cx.set_value(count);
    return cx;
}
counting counting::operator++(int)
{
    count++;
    set_value(count);
    return *this;
}
int main()
{
    counting c1,c2,c3;
    c2=c1++;
    c3=++c2;
    cout<<c2.return_value()<<endl;
    cout<<c3.return_value();
}
```

Output
2
2

### Overloading operator+

Consider the addition of two objects:

```
third=first+second;
```

Since + is not a unary operator, so the syntax of overloading operator+ is:

```
object_return_type operator+(object_return_type& rhs);
```

where rhs = Right most side object which is second in our example.

Consider the following program:

```
typedef unsigned short USHORT;
#include <iostream>
using namespace std;
class Counter
{
    public:
        Counter();
        Counter(USHORT initialValue);
        ~Counter(){}
        USHORT GetItsVal() const { return itsVal; }
        void SetItsVal(USHORT x) {itsVal = x; }
        Counter operator+ (const Counter &);

    private:
        USHORT itsVal;
};

Counter::Counter(USHORT initialValue):
    itsVal(initialValue)
{}

Counter::Counter():
    itsVal(0)
{}

Counter Counter::operator+ (const Counter & rhs)
{
    return Counter(itsVal + rhs.GetItsVal());
}

int main()
{
    Counter first(2), second(4), third;

    third = first + second;

    cout << "first: " << first.GetItsVal()<< endl;
    cout << "second: " << second.GetItsVal() << endl;
    cout << "third: " << third.GetItsVal() << endl;
    return 0;
}
```

Output
first: 2
second: 4
third: 6

Here, for the place of rmo, second is passed. In the same way as above minus operator can also be overloaded.

### Overloading assignment operator (=)

This is an amazing operator. Consider the object relation say second=first; C++ supports many of the overloading features of assignment operator.

Consider the following program:

```
typedef unsigned short USHORT;
#include <iostream>
using namespace std;
class Counter
{
public:
Counter();
Counter(USHORT initialValue);
~Counter(){}
USHORT GetItsVal() const { return itsVal; }
void SetItsVal(USHORT x) {itsVal = x; }
Counter operator+(const Counter &);
private:
USHORT itsVal;
};

Counter::Counter(USHORT initialValue):
itsVal(initialValue)
{}

Counter::Counter():
itsVal(0)
{}

Counter Counter::operator+ (const Counter & rmo)
{
    return Counter(itsVal + rmo.GetItsVal());
}

int main()
{
    Counter first(2);
    Counter second=first;
    cout << "second: " << second.GetItsVal() << endl;
    return 0;
}
```

Output

second: 2

Here, in this program you can see that C++ already has an overloaded assignment operator.

### Overloading equality operator (==)

Like other binary operators, equality operators can also be overloaded in the same way.

The syntax of overloading the equality operator is as follows:

```
object_return_type operator==(object_return_type &rhs);
```

Following program illustrates to overload the equality operator.

```
#include <iostream>
using namespace std;
class DOG
{
public:
DOG();
int age() const { return Age; }
void SetAge(int x) { Age = x; }
int operator==(const DOG &);
private:
int Age;
};
DOG::DOG()
{
Age = 5;
}
