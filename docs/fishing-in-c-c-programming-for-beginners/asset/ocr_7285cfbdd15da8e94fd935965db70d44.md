Consider the following program:

```
#include<iostream>
#include<fstream>
using namespace std;
int main()
{
    fstream fin("example.txt",ios::out);
    if(fin==NULL)
    {
        cout<<"Can't open file";
        return -1;
    }
    fin<<"Hello World";
    fin.close();
    fin.open("example.txt",ios::in);
    int a;
    fin>>a;
    if(fin.fail())
    {
        cout<<"Failed to extract an integer";
    }
    fin.close();
}
```

# Output

Failed to extract an integer
