Consider the following program:

```
#include<iostream>
#include<fstream>
using namespace std;
int main()
{
    fstream fin;
    fin.open("example.txt",ios::in | ios::out);
    if(fin==NULL)
    {
        cout<<("Can't open the file");
        return -1;
    }
    fin<<"Hacking: Exploiting computer security with Aman";
    fin.seekg(5,ios::beg);
    int i=fin.tellg();
    cout<<i;
    fin.close();
    return 0;
}
```

Output

5

This program places the cursor at fifth offset then it returns its position.

tellp() – This function returns the position of ‘put’ cursor in a file in terms of offset
