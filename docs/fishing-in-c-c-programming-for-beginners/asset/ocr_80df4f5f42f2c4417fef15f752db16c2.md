fstream fin;

{
    cout<<("Can't open the file");
    return -1;
}

fin<<"I am Aman";
fin.seekp(4,ios::beg);

char ch;

fin.seekg(0,ios::beg);

while(!fin.eof())

    ch=fin.get();
    cout<<ch;
}

return 0;
}

tellg() – This function returns the position of ‘get’ cursor in a file in terms of offset
