#include<iostream>
using namespace std;
class area
{
    int *x, *y;
    public:
    area (int a,int b)
    {
        x=new int;
        y=new int;
        *x=a;
        *y=b;
    }
    ~area()
    {
        delete x;
        delete y;
        cout<<"/nPointers deleted";
    }
    int calculate_area (void)
    {
    return *x**y;
    }
};
int main()
{
    class area parallelogram(4,2);
    cout<<"Area of parallelogram is "<<parallelogram.calculate_area();
    return 0;
}
