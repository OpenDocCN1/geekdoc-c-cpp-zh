#define IPEXPR "([0-9]{1,3})\\.([0-9]{1,3})\\.([0-9]{1,3})\\.([0-9]{1,3})"
int main(void)
{
    char line[1024];
    char *address = NULL;
    char delim[] = ".,;:\^+-_(){ }[]<>*%&%$#@!?~-/\\|=\\t\\r\\n";
    int retval = 0;
    regex_t re;
    setlocale(LC_ALL,"");

    if(regcomp(&re, IPEXPR, REG_EXTENDED) != 0)
        return 1;

    while((fgets(line, 1024, stdin)) != NULL)
    {
        if(strchr(line, '.') == NULL)
            continue;

        address = strtok(line, delim);
        while(address != NULL)
        {
            if(strlen(address) <= 15)
                if((retval = regexec(&re, address, 0, NULL, 0)) == 0)
                    printf("%s\n", address);

            address = strtok(NULL, delim);

        } /* while */
    } /* while */
    return 0;
}

## 166. 接受密码的 C 程序。

这是一个简单的 C 语言登录程序。接受密码时，它使用“*”符号遮盖每个字符，并在用户按下 Enter 键后在下一行显示密码。它还接受退格键并做出相应处理。

```c
#include<stdio.h>
#include<conio.h>
char pw[25],ch;
int i;
void main()
{
    clrscr();
    puts("Enter password");
    while(1)
    {
        if(i<0)
            i=0;
        ch=getch();
        if(ch==13)
            break; /*13 is ASCII value of ENTER*/
        if(ch==8) /*ASCII value of BACKSPACE*/
        {
            putch('\b');
            putch(' ');
            putch('\b');
            i--;
            continue;
        }
        pw[i++]=ch;
        ch='*';
        putch(ch);
    }
    pw[i]='\0';
    printf("\n\n%s",pw);
    getch();
}
```
