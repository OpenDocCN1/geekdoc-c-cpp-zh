# Algorithm No. 5: To print the series 1 to N using for loop.

+   1. Read N.
+   2. Repeat Step 3 for I:=1 to N:
+   3. Write I.
+      [End of Step 2 loop]
+   4. Exit.

# Algorithm No. 6: To print the series 1 to N using while loop.

+   1. Read N.
+   2. Set I:= 1
+   3. Repeat Step 4 and 5 While I<= N:
+   4. Write I.
+   5. Set I:=I+1.
+      [End of Step 3 loop]
+   6. Exit.

# Algorithm No. 7: To print the series 1 to N using do while loop.

+   1. Read N.
+   2. Set I:=1.
+   3. Write I.
+   4. Set I:=I + 1.
+   5. if I<=N, then: goto step3.
+   6. Exit.
