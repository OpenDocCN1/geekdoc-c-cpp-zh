```
class Person
{
  public string faculty_name;
  public string[] PhoneNumbers;
}
var persons = new[] {
  faculty_name = "Patrick Tones",
  PhoneNumbers = new[] { "750-055-0144", "750-055-0145" },
  new Person { faculty_name = "Gretchen Rivas",
  PhoneNumbers = new[] { "755-055-0163" } };
};
nXElement faculties = new XElement("faculties", f in persons
  select XElement("faculty",
    new XElement("name", f.faculty),
    from p in f.PhoneNumbers
    select new XElement("phone", p))));
Console.WriteLine(faculties);
```
