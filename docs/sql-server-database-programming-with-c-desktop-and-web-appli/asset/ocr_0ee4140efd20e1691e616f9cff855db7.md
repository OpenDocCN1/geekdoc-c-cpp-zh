# ASP.NET Web Services

| WebServiceDt Project | SQLDelete() |

```
[Method]
public void SQLDelete(string sName)
{
    string cmdString = "dbo.WebDeleteStudent";
    SqlConnection connection = new SqlConnection();
    SqlBase Result = new SqlBase();
    int intDelete;

    Result.SQLOK = true;
    connection.Open();
    if (connection == null)
    {
        Result.SQLOK = false;
        Result.Error = "Database connection is failed";
        ReportError(Result);
        return;
    }
    SqlCommand command = new SqlCommand(cmdString, connection);
    command.CommandType = CommandType.StoredProcedure;
    command.Parameters.Add("@sName", SqlDbType.VarChar).Value = sName;
    intDelete = command.ExecuteNonQuery();
    if (intDelete == 0)
    {
        Result.Error = "Data deleting failed";
        ReportError(Result);
    }
    connection.Close();
    command.Dispose();
    return;
}
```
