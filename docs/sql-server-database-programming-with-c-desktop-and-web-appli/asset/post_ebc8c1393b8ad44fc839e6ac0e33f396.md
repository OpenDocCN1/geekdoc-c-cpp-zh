```csharp
private void CourseList_SelectedIndexChanged(object sender, EventArgs e)
{
    string cmdString = "SELECT course_id, credit, classroom, schedule, enrollFrom FROM Course WHERE course_id = '" + CourseList.Text + "';
    SqlDataAdapter adapter = new SqlDataAdapter();
    SqlCommand command = new SqlCommand(cmdString);
    DataTable dataTable = new DataTable();
    adapter.SelectCommand = command;
    command.Connection = loginForm.Connection;
    command.CommandType = CommandType.Text;
    cmdString = "SELECT * FROM Course WHERE=@course_id";
    command.Parameters.Add("@course_id", SqlDbType.Char).Value = CourseList.Text;
    if (ComboBox.Text == "DataAdapter Method")
    {
        adapter.SelectCommand = command;
        adapter.Fill(dataTable);
        if (dataTable.Rows.Count > 0)
        {
            FillCourseTextBox(dataTable);
        }
        else
        {
            MessageBox.Show("No matched course information found!");
        }
        dataTable.Dispose();
        adapter.Dispose();
    }
    else if (ComboBox.Text == "DataReader Method")
    {
        SqlDataReader reader = command.ExecuteReader();
        if (reader.HasRows)
        {
            FillCourseTextBox(reader);
        }
        else
        {
            MessageBox.Show("No matched course information found!");
        }
        reader.Close();
        reader.Dispose();
    }
    else
    {
        //Invalid Method is selected
        MessageBox.Show("Invalid Method selected!");
    }
    command.Dispose();
}
```

- III. Otherwise, if the ExecuteReader() method fails, a warning message is displayed to remind and ask the users to handle it.
- IV. Some cleaning jobs are performed to close the DataReader object since we no longer need it.
- V. If no valid query method has been selected, a warning message is shown up.
- VI. Finally, the used SqlCommand object is closed and disposed.
