Let's have a closer look at this piece of code to see how it works.

- A. The namespace for the SQL Server Data Provider System.Data.SqlClient is added into this for us to use all data components in that namespace.
- B. A global SQL connection is declared here. The access mode for this object is Public, which means any object or method defined in this project can use this connection to access the SQL Server database CSE_DEPT.

The advantage of using this connection is that more parameters are included in this connection and make the connection more secured and safe. - E. A new instance of the SqlConnection is created and initialized with the connection string as the argument.
- F. A try-catch block is utilized to catch any exception caused by opening this connection. The advantage of using this kind of strategy is to avoid unnecessary system process and simplify this debug procedure. Multiple catch blocks are utilized to track and find any possible exception or error, including SqlException and InvalidOperationException.
- G. This step is used to confirm whether our database connection is successful or not. If not, an error message is displayed and the project is exited.

After a database connection is successful, we need to use this connection to access the SQL Server database to perform our data query job.

using System.Data;
using System.Data.SqlClient;
using System.Data.Linq;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Windows.Forms;

namespace RTOSelect_Project
{
    public partial class LogInform : Form
    {
        public SQLConnection sqlConnection = null;
        public LogInform()
        {
            InitializeComponent();
            // stringString = "Server=DESKTOP-24JPUFB\\SQL2019EXPRESS;" + 
            // "Initial Catalog=CSE_DEPT; + "Integrated Security=True";
            string sqlString = "Server=DESKTOP-24JPUFB\\SQL2019EXPRESS;" + 
            "Initial Catalog=CSE_DEPT" + "User ID=SMART;Password=Happy2022";
            sqlConnection = new SQLConnection(sqlString);
            try {
                sqlConnection.Open();
            } catch (Exception e) {
                MessageBox.Show("SQL Server Error");
                MessageBox.Show("Error Code = " + e.ErrorCode);
                MessageBox.Show("Error Message = " + e.Message);
            } catch (InvalidOperationException e) {
                MessageBox.Show("Invalid Message = " + e.Message);
                if (sqlConnection.State != ConnectionState.Open) {
                    MessageBox.Show("Database connection Failed");
                    Application.Exit();
                }
            }
        }
    }
}
