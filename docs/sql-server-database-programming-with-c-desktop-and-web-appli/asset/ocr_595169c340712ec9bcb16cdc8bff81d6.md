private[] getStudentImage()
{
    int length;
    string imgPath = Server.MapPath("/StudentImage/\");
    if (!System.IO.Directory.Exists(imgPath))
        System.IO.Directory.CreateDirectory(imgPath);
    FileUpload.Save(imgPath + System.IO.Path.GetFileName(FileUpload.FileName));
    PhotoBox.ImageUrl = "/StudentImage/" + FileUpload.FileName;
    byte[] img = new byte[FileUpload.PostedFile.ContentLength];
    length = FileUpload.PostedFile.InputStream.PostedFile.InputStream.Read(img, 0, length);
    return img;
}
