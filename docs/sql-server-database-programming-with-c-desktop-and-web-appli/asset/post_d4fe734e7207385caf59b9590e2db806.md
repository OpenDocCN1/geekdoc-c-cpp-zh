public SQLInsertRecord GetInsert(string fullName)
{
    string cmdString = "SELECT Course.course_id FROM Course Join Faculty = @name";
    SqlConnection connection = new SqlConnection(connectionString);
    SQLInsertRecord result = new SQLInsertRecord();
    SqlCommand command = new SqlCommand(cmdString, connection);
    SqlDataReader reader;

    result.SqlInsertError = true;
    connection = null;
    if (connection == null)
    {
        result.SqlInsertError = "Database connection is failed";
        ReportError(result);
        return null;
    }

    command.Connection = connection;
    command.CommandText = cmdString;
    command.Parameters.Add("@name", SqlDbType.VarChar).Value = fullName;
    reader = command.ExecuteReader();
    if (reader.HasRows)
    {
        FillCourseReader(ref result, reader);
    }
    else
    {
        result.SqlInsertError = "No match found";
        ReportError(result);
    }
    reader.Close();
    connection.Close();
    command.Dispose();
    return result;
}
