void ProcessObject(ref SQLSelect.SQLResult wsResult) {
    if (wsResult.SQLRequestOK) {
        FillFacilityObject(wsResult);
    } else {
        MessageBox.Show("Web service is wrong: " + wsResult.Message);
    }
}

void FillFacilityObject(DataSet wsData) {

}
