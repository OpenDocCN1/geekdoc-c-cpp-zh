private void ProcessObject(ref WS_SQLSelect.SQLSelectResult wsResult)
{
    if (wsResult.SQLRequestOK == true)
        FillFacultyObject(wsResult);
    else
        MessageBox.Show("Faculty information has not been retrieved: " + wsResult.SQLRequestError());
}

private void FillFacultyObject(WS_SQLSelect.SQLSelectResult result)
{
    txtFacultyID.Text = result.FacultyID;
    txtFacultyName.Text = ComboName.Text;
    txtOffice.Text = result.Office;
    txtPhone.Text = result.FacultyPhone;
    txtCollege.Text = result.FacultyCollege;
    txtTitle.Text = result.FacultyTitle;
    txtEmail.Text = result.FacultyEmail;
    byte[] bage = result.FacultyImage;
    PhotoObject.Image = Image.FromStream(new System.IO.MemoryStream(bage));
}
