protected void cmdSelect_Click(object sender, EventArgs args)
{
    WebClientSelectProject.WS_SQLSelect.WebWebServiceSelect wsqlSelect;
    wsqlSelect = new WebClientSelectProject.WS_SQLSelect.WebServiceSelect();
    WebClientSelectProject.WS_SQLSelect.WS_SQLResult wsqlResult;
    wsqlResult = new DataSet(new DataSet());
    string errMsg;
    if (Combo.Text == "Object Method")
{
        try {
            wsqlResult = wsqlSelect.GetSelect(Combo.Text);
        } catch (Exception err)
        {
            errMsg = "Service is wrong: " + err.Message;
            Response.Write("<script>alert('" + errMsg + "')</script>");
        }
        SourceObject(ref wsqlResult);
    } else if (Combo.Text == "Stored Procedure")
    {
        try {
            wsqlResult = wsqlSelect.GetSP(Combo.Text);
        } catch (Exception err)
        {
            errMsg = "Service is wrong: " + err.Message;
            Response.Write("<script>alert('" + errMsg + "')</script>");
        }
        SourceObject(ref wsqlResult);
    } else
    {
        try {
            (dataSet = wsqlSelect.GetSelectDataSet(Combo.Text));
        } catch (Exception)
        {
            errMsg = "Service is wrong: " + err.Message;
            Response.Write("<script>alert('" + errMsg + "')</script>");
        }
        FillFacultyDataSet(ref dataSet);
    }
}

    if (wsResult.SQLRequestOK == true)
        FillFacultyDataSet(ref wsResult);
    else
        Response.Write("<script>alert('" + errMsg + "')</script>");
}
