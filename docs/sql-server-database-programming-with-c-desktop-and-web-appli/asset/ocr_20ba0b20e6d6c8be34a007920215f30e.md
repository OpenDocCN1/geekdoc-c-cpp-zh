| private int CheckFacultyCollection() {
  int check = 0;
  foreach (KeyValuePair<string, string> strCheck in FacultyCollection) {
    if (strCheck.Key == string.Empty) {
      MessageBox.Show(strCheck.Key + " is empty!");
      check++;
    }
  }
  return check;
} | |
