private void cmdInsert_Click(object sender, EventArgs e)
{
    int check, intInsert;
    string cmdString="INSERT INTO faculty (faculty_id, faculty_name, title, office, phone, college, email, image) " +
        "VALUES (@faculty_id, @faculty_name, @title, @office, @phone, @college, @image)";
    SqlDataAdapter dataAdapter=new SqlDataAdapter();
    SqlCommand sqlCommand=new SqlCommand();
    InitFacultyInfo();
    if (check == 0)
    {
        sqlCommand.Connection=loginForm.Connection;
        sqlCommand.CommandType=CommandType.Text;
        sqlCommand.Text=cmdString;
        InsertParameters(sqlCommand);
        dataAdapter.SqlCommand=sqlCommand;
        intInsert=dataAdapter.ExecuteNonQuery();
        if (intInsert != 0)
        {
            MessageBox.Show("The data insertion failed");
        }
        else
        {
            PhotoBox.Image=null;
            CurrentFaculty();
            ClearFacultyInfo();
            cmdInsert.Enabled=false;
        }
    }
    MessageBox.Show("Fill All Faculty Information box, enter a NULL for blank column");
}
