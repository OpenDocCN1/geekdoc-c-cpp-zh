```
private void cmdSelect_Click(object sender, EventArgs e)
{
    string strCourse = "SELECT course_id FROM Course JOIN Faculty"
strCourse += " ON (Course.id = Faculty.id) AND (Faculty.faculty_name = @name)";
    SqlDataAdapter dataAdapter = new SqlDataAdapter(strCourse);
    SqlCommand cmdCourse = new SqlCommand(strCourse);
    DataTable courseTable = new DataTable();
    SqlDataReader dataReader = dataAdapter.ExecuteReader();
    cmdCourse.Connection = loginForm.Connection;
    cmdCourse.CommandType = CommandType.Text;
    cmdCourse.CommandText = strCourse;
    cmdCourse.Parameters.Add("@name", SqlDbType.Char).Value = ComboName.Text;
    if (ComboMethod.Text == "DataReader Method") {
        CourseDataAdapter.SelectCommand = cmdCourse;
        CourseDataAdapter.Fill(courseTable);
    } else {
        if (sqlCourseTable.Rows.Count > 0) {
            FillTable(courseTable);
        }
        else {
            MessageBox.Show("No matched course found!");
            courseTable.Dispose();
        }
    } else if (ComboMethod.Text == "DataReader Method") {
        SqlDataReader dataReader = cmdCourse.ExecuteReader();
        if (dataReader.HasRows) {
            FillTable(dataReader);
        } else {
            MessageBox.Show("No matched course found!");
            dataReader.Close();
        }
        dataReader.Dispose();
    } else {
        MessageBox.Show("Invalid Method Selected!");
        cmdCourse.Dispose();
        courseList.SelectedIndex = 0;
    }
}
```

FIGURE 5.82 The completed codes for the course click method.

+   B. All Data Provider-dependent objects, such as the CourseDataAdapter, Command, DataReader and DataTable, are generated here those objects will be used later.
C. The SqlCommand object sqlSqlCommand is initialized with the connection object obtained from the constructor of the LoginForm class, connection string, command text, command type and command parameter.
D. The parameter name must be identical to the dynamic name @name, which is defined in the query string and is exactly located in the ON clause. The parameter's value is the content of the Faculty Name comboBox, which should be selected by the user of the projects runs.
E. If the DataAdapter Method is used by the user, initialized SqlCommand is assigned to the SelectCommand property of the DataAdapter and the Fill() method is executed to fill the table.
F. The Rows.Count property of the filled table is greater than 0, which means at least one row has been matched and filled into the data table. A user-defined method FillTable(table) is called to fill course_id into the CourseList box.
G. Otherwise, the Rows.Count is less than or equal to 0, which means that the fill operation is failed. A warning message is displayed for that case.
