
## Example: Output the contents of a file to standard output.

```
// file4.c

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    FILE *fp;

    if ((fp = fopen("infile.txt", "r"))) {
        fprintf("File could not be opened.\n");
        exit(1);
    }

    while((c = getc(fp)) != EOF) {
        putchar(c);
    }
    fclose(fp);
    return 0;
}
```

## Example: Copy the contents of one file to a second file and output the contents of the second file.

```
// file6.c

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    FILE *fpIn, *fpOut;

    if (!(fpIn = fopen("infile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    if (!(fpOut = fopen("outfile.txt", "w"))) {
        printf("File could not be opened for output.\n");
        exit(1);
    }

    while((c = getc(fpIn)) != EOF) {
        putc(c, fpOut);
    }

    fclose(fpIn);
    fclose(fpOut);

    if (!(fpIn = fopen("outfile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    while((c = getc(fpIn)) != EOF) {
        putchar(c);
    }

    fclose(fpIn);
    return 0;
}
```

Similarly there are equivalents to the functions printf() and scanf() which read or write data to files. These are called fprintf() and fscanf(). The functions are used in the same way, except that the fprintf() and fscanf() take the file pointer as an additional first argument:

```
fprintf(fp, formatSpecifiers, arguments);
```

Performs the analogous operations of the printf() function on a file. It returns the number of bytes output. In the event of error, it returns EOF.

Function prototype:

```
int fprintf(FILE *fp,
formatSpecifiers, arguments);
```

```
fscanf(fp, format_specifiers, arguments);
```

Performs the analogous operations of the scanf() function on a file. It returns the number of arguments that are successfully assigned or the value of EOF if the end of file is reached.

Function Prototype:

```
int fscanf(FILE *fp, format_specifiers,
arguments);
```

Additionally, there are equivalent functions to gets() and puts() for file manipulation:

```
fgets(s, N, fp);
```

Characters are read from the file pointed to by fp, and placed into string s until either N-1 characters have been read or a newline character is read, whichever comes first. Unlike gets(s), if a newline character is read, it is placed in s. In both cases, s is terminated with a NULL character. The value of s (pointer to char) is returned.

Function prototype:

```
char *fgets(char *s, int N, FILE *fp);
```

```
fputs(s, fp);
```

The null-terminated string s is copied to the file pointed to by fp, except that the terminating NULL character itself is not copied. Unlike puts(s), a newline character is not appended.

Function prototype:

```
int fputs(char *s, FILE *fp);
```

Finally, we have a function for testing whether or not we are at the end of file (EOF).

```
feof(fp);
```

Tests for the end of file. Returns an integer value that is non-zero if all the data from the specified file has been read, and zero otherwise.

Function Prototype:

```
int feof(FILE *fp);
```

## Example: Output the contents of a file line by line

```
// file7.c

#include <stdio.h>
#include <stdlib.h>

#define MAX 81

int main(void) {
    char line[MAX];
    FILE *fp;

    if (! (fp = fopen("infile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    while (!feof(fp)) {
        fgets(line, MAX, fp);
        printf("%s", line);
    }
    fclose(fp);
    return 0;
}
```

## Example: Concatenate two files into a third file and output the third file.

```
// file9.c

#include <stdio.h>
#include <stdlib.h>

#define MAX 81

int main(void)
{
    char line[MAX];
    FILE *fpIn, *fpIn1, *fpOut;

    if ((fpIn1 = fopen("infile1.txt", "r")) == NULL) {
        fprintf("File could not be opened for input.\n");
        exit(1);
    }

    if ((fpIn2 = fopen("infile2.txt", "r")) == NULL) {
        fprintf("File could not be opened for input.\n");
        exit(1);
    }

    if ((fpOut = fopen("outfile.txt", "w")) == NULL) {
        fprintf("File could not be opened for output.\n");
        exit(1);
    }

    while (!feof(fpIn1)) {
        fgets(line, MAX, fpIn1);
        fputs(line, fpOut);
    }

    while (!feof(fpIn2)) {
        fgets(line, MAX, fpIn2);
        fputs(line, fpOut);
    }

    fclose(fpIn1);
    fclose(fpIn2);
    fclose(fpOut);

    if ((fpIn = fopen("outfile.txt", "r")) == NULL) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    while (!feof(fpIn)) {
        fgets(line, MAX, fpIn);
        printf("%s", line);
    }

    return 0;
}
```

## A Note on Microsoft C Compilers (Again)

If you are using a Microsoft compiler, the use of fopen() might produce a warning message during the build phase. Instead, Microsoft offers a more secure form of fopen() called fopen_s() which returns an error code of 0 if successful or non-zero if failure, rather than a file pointer.

Here's the usage:

```
errorCode = fopen_s(&fp, "infile.txt", "r");
if (errorCode != 0) {
    printf("File could not be opened.\n");
    exit(1);
}
```

Once again, we can combine the fopen_s() call, the assignment of the returned error code, and the test against the NULL pointer, into a single expression:

```
if ((fopen_s(&fp, "infile.txt", "r")) != 0) {
    printf("File could not be opened.\n");
    exit(1);
}
```

Or even more concisely:

```
if ((fopen_s(&fp, "infile.txt", "r"))) {
    printf("File could not be opened.\n");
    exit(1);
}
```

Note: If you want to strictly use ANSI C and not Microsoft C without getting the warning messages, add the following line of code before you include any header files:

```
#define _CRT_SECURE_NO_WARNINGS
```
