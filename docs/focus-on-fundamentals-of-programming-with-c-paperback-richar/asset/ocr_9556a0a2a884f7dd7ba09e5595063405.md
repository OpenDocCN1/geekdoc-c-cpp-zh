void outputMatrix(int mat[][4]) {
 int i, j;

 for (i = 0; i < 4; ++i) {
    for (j = 0; j < 4; ++j) {
       printf("%4d", mat[i][j]);
    }
    printf("\\n");
 }
}

int isSymmetric(int mat[][4]) {
 int i, j;

 for (i = 0; i < 4; ++i) {
    for (j = 0; j < 4; ++j) {
       if (mat[i][j] != mat[j][i]) {
          return 0;
       }
    }
 }

 return 1;
}
