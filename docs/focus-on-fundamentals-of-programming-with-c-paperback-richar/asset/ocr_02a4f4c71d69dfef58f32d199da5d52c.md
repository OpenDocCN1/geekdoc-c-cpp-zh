# Example: Copy the contents of one file to a second file and output the contents of the second file.

```
// file6.c

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int c;
    FILE *fpIn, *fpOut;

    if (!(fpIn = fopen("infile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    if (!(fpOut = fopen("outfile.txt", "w"))) {
        printf("File could not be opened for output.\n");
        exit(1);
    }

    while((c = getc(fpIn)) != EOF) {
        putc(c, fpOut);
    }

    fclose(fpIn);
    fclose(fpOut);

    if (!(fpIn = fopen("outfile.txt", "r"))) {
        printf("File could not be opened for input.\n");
        exit(1);
    }

    while((c = getc(fpIn)) != EOF) {
        putchar(c);
    }

    fclose(fpIn);
    return 0;
}
```
