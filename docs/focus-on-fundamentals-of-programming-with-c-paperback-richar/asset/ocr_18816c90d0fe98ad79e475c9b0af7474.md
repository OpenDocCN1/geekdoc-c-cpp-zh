```c
// doublep.c
#include <stdio.h>
void fool(char *p);
void foo2(char **q);
int main(void) {
    char *p = "Hello";
    char *q = "Bye";
    printf("Function: main() ---> p = %s\n", p);
    fool(p);
    printf("Function: main() ---> p = %s\n", p);
    printf("Function: main() ---> q = %s\n", q);
    foo2(&q);
    printf("Function: main() ---> q = %s\n", q);
    return 0;
}
void fool(char *pp) {
    pp++;
    printf("Function: fool() ---> pp = %s\n", pp);
}
void foo2(char **qq) {
    (*qq)++;
    printf("Function: foo2() ---> *qq = %s\n", *qq);
}
