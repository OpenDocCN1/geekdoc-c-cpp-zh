
线性链表 .c 文件中的包含语句如下：

```
// list.c

#include <stdio.h>
#include <stdlib.h>
#include "list.h"
```

请注意以下语句：

```
#include "list.h"
```

这条语句指示编译器读取 list.h 头文件中的语句。
此时，你应该能够将 list.c 文件与程序的其余部分分别编译。因此，当你准备在程序中使用链表代码时，会在项目中创建一个新的源代码文件，并包含链表函数的头文件：

```
#include "list.h"
```

你还需要通过将源代码文件添加到项目中的其他源代码文件中，来引入链表的 list.c 源代码。

## 线性链表代码

```
// list.h

#ifndef _LIST_H
#define _LIST_H

typedef struct node {
    int info;
    struct node *next;
} *NODEPTR;

NODEPTR getNode(void);
void freeNode(NODEPTR p);
void addFront(NODEPTR *list, int num);
int deleteFront(NODEPTR *list);
void addLast(NODEPTR *list, int num);
int deleteLast(NODEPTR *list);
void insertAfter(NODEPTR p, int num);
void deleteAfter(NODEPTR p);
void insert(NODEPTR *list, int num);
int numElements(NODEPTR list);
int sumElements(NODEPTR list);
void outputList(NODEPTR list);
int contains(NODEPTR list, int num);
int isEmpty(NODEPTR list);
void freeList(NODEPTR *list);
NODEPTR select(NODEPTR list, int num);
int remove(NODEPTR *list, int num);
NODEPTR copyList(NODEPTR list);
void concatList(NODEPTR *list1, NODEPTR list2);
NODEPTR intersect(NODEPTR list1, NODEPTR list2);

#endif
```

```
// list.c

#include <stdio.h>
#include <stdlib.h>
#include "list.h"

// returns the address of a newly created node
NODEPTR getNode(void) {
    NODEPTR p;

    p = (NODEPTR) malloc(sizeof(struct node));
    if (p == NULL) {
        error("List overflow.");
    }
    return(p);
}

// frees the node pointed to by p
void freeNode(NODEPTR p) {
    free(p);
}

// adds a new node to the front of the list
void addFront(NODEPTR *list, int num) {
    NODEPTR p;

    p = getNode();
    p->info = num;
    p->next = *list;
    *list = p;
}
```

// 删除列表前端的节点

```
int deleteFront(NODEPTR *list) {
    int temp;
    NODEPTR p;

    if (*list == NULL) {
        error("List underflow.");
    }
    p = *list;
    temp = p->info;
    *list = p->next;
    freeNode(p);
    return temp;
}
```

// 向列表末尾添加元素

```
void addLast(NODEPTR *list, int num) {
    NODEPTR p, q;

    p = getNode();
    p->info = num;
    p->next = NULL;
    if (*list == NULL) {
        *list = p;
    } else {
        q = *list;
        while (q->next) {
            q = q->next;
        }
        q->next = p;
    }
}
```

// 删除列表中的最后一个元素

```
int deleteLast(NODEPTR *list) {
    NODEPTR p, q;
    int temp = -1;

    if (*list) {
        p = *list;
        q = NULL;
        while (p->next) {
            q = p;
            p = p->next;
        }
        if (!q) {
            *list = NULL;
        } else {
            q->next = NULL;
        }
        temp = p->info;
        freeNode(p);
    }
    return temp;
}
```

// 在节点 p 指向的节点之后插入新节点

```
void insertAfter(NODEPTR p, int num) {
    NODEPTR q;

    if (p == NULL) {
        error("Void insertion.");
    }
    q = getNode();
    q->info = num;
    q->next = p->next;
    p->next = q;
}
```

// 删除节点 p 指向的节点之后的节点

```
int deleteAfter(NODEPTR p) {
    int temp;
    NODEPTR q;

    if (p == NULL || p->next == NULL) {
        error("Void deletion.");
    }
    q = p->next;
    temp = q->info;
    p->next = q->next;
    freeNode(q);
    return temp;
}
```

// 将节点插入到排序列表中的正确位置

```
void insert(NODEPTR *list, int num) {
    NODEPTR p, q;

    p = *list;
    q = NULL;
    while (p != NULL && p->info < num) {
        q = p;
        p = p->next;
    }
    if (q == NULL) {
        addFront(list, num);
    } else {
        insertAfter(q, num);
    }
}
```

// 返回列表中的节点数

```
int numElements(NODEPTR list) {
    int count=0;
    while (list) {
        count++;
        list = list->next;
    }
    return count;
}
```

// 返回列表中整数的和

```
int sumElements(NODEPTR list) {
    int sum=0;

    while (list) {
        sum += list->info;
        list = list->next;
    }

    return sum;
}
```

// 遍历列表并输出列表内容

```
void outputList(NODEPTR list) {
    while (list) {
        printf("%8d", list->info);
        list = list->next;
    }

    printf("\n\n");
}
```

// 如果在列表中找到该项，则返回 true

```
int contains(NODEPTR list, int num) {
    NODEPTR p;

    p = list;
    while (p != NULL && p->info != num) {
        p = p->next;
    }

    return p != NULL;
}
```

// 确定列表是否为空

```
int isEmpty(NODEPTR list) {
    return list == NULL;
}
```

// 释放列表中的所有节点

```
void freeList(NODEPTR *list) {
    NODEPTR p;
    while (*list) {
        p = *list;
        *list = p->next;
        freeNode(p);
    }
}
```

// 返回具有请求值的节点的指针；
// 否则返回 NULL

```
NODEPTR select(NODEPTR list, int num) {
    while (list) {
        if (list->info == num) {
            return list;
        }
        else {
            list = list->next;
        }
    }
    return list;
}
```

// 删除列表中第一次出现的项

```
int remove(NODEPTR *list, int num) {
    NODEPTR p, q;
    p = *list;
    q = NULL;
    while (p != NULL && p->info != num) {
        q = p;
        p = p->next;
    }
    if (p == NULL) {
        return -1;
    }
    else {
        return q == NULL ? deleteFront(list) : deleteAfter(q);
    }
}
```

// 制作列表的副本

```
NODEPTR copyList ( NODEPTR list ) {
    NODEPTR newlist, p, q;

    newlist = NULL;
    while ( list ) {
        p = getNode();
        p->info = list->info;
        p->next = NULL;
        if ( newlist == NULL ) {
            newlist = p;
        }
        else {
            q->next = p;
        }
        q = p;
        list = list->next;
    }
    return newlist;
}
```

// 连接两个列表

```
void concatList ( NODEPTR *list1, NODEPTR list2 ) {
    NODEPTR p;

    if ( *list1 == NULL ) {
        *list1 = list2;
    }
    else if ( list2 ) {
        p = *list1;
        while ( p->next ) {
            p = p->next;
        }
        p->next = list2;
    }
}
```

// 返回指向包含两个列表交集的新列表的指针

```
NODEPTR intersect(NODEPTR list1, NODEPTR list2) {
    NODEPTR p, q, r, s, newlist;
    int found;
    p = list1;
    newlist = NULL;
    r = NULL;
    while (p) {
        q = list2;
        found = 0;
        while (q && !found) {
            if (p->info == q->info) {
                found = 1;
            } else {
                q = q->next;
            }
        }
        if (found) {
            s = getNode();
            s->info = p->info;
            s->next = NULL;
            if (!r) {
                newlist = s;
            } else {
                r->next = s;
            }
            r = s;
        }
        p = p->next;
    }
    return newlist;
}
```
