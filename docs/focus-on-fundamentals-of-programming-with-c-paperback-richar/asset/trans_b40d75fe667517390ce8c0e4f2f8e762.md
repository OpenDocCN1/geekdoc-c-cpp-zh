```c
// insertionSort.c
void insertionSort(int list[], int max) {
   int i, j, temp;
   for (i = 1; i < max; i++) {
      temp = list[i];
      j = i - 1;
      while (j >= 0 && temp < list[j]) {
         list[j+1] = list[j];
         j = j - 1;
      }
      list[j+1] = temp;
   }
}
```

## 对数组排序：qsort() 库函数

标准 C 库包含一个名为 qsort() 的异常强大且通用的排序例程。它是名为快速排序的特定排序算法的实现。qsort() 例程有四个参数：

- 数据数组的基地址
- 数组中的项目数
- 每个项目的宽度（字节）
- 一个函数的地址，该函数可以比较数组中包含类型的两个数据项，并按如下方式返回有符号值：
  - 如果 p < q，则 < 0
  - 如果 p = q，则 0
  - 如果 p > q，则 > 0

如果您使用的是 C 的原始数据类型之一，通常可以用几行代码构建一个适当的比较函数。示例如下：

```c
// compare.c
int compare(const void *p, const void *q) {
    if (*(int *)p < *(int *)q) {
        return -1;
    }
    return *(int *)p > *(int *)q;
}
```

在 compare() 函数中：

- 如果 p < q，则返回 -1
- 如果 p = q，则返回 0
- 如果 p > q，则返回 1

下面的代码用整数随机填充一个 100 元素的数组。显示数组内容，调用 qsort()，然后再次显示数组内容，展示排序后的数据项。

```c
// qsort.c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

#define SIZE 100

int compare(const void *p, const void *q);
void output(int num[], int size);

int main()
{
   int i, num[SIZE];

   srand((unsigned)time(NULL));
   for (i = 0; i < SIZE; i++) {
      num[i] = rand() % 50000;
   }

   printf("%s\n%s\n", "Before sort:", "----------------");
   output(num, SIZE);
   qsort(num, SIZE, sizeof(int), compare);
   printf("%s\n%s\n", "After sort:", "----------------");
   output(num, SIZE);

   return(0);
}

int compare(const void *p, const void *q) {
   if (*(int *)p < *(int *)q) {
      return -1;
   }

   return *(int *)p > *(int *)q;
}

void output(int num[], int size) {
   int i;

   for (i = 0; i < size; ++i) {
      printf("%6d", num[i]);
      if (i % 10 == 9 || (i == size -1)) {
         printf("%s", "\n");
      }
      else {
         printf("%s", ' ');
      }
   }
}
```
