void reverse(vector<int>& a, int i, int j)
{
    while (i < j)
    {
        swap(a[i], a[j]); i++; j--;
    }
}

bool next_permutation(vector<int>& a)
{
    for (int i = a.size() - 1; i > 0; i--)
    {
        if (a[i - 1] < a[i])
        {
            int j = a.size() - 1;
            while (a[i - 1] > a[j]) { j--; }
            swap(a[i - 1], a[j]);
            reverse(a, i, a.size() - 1);
            return true;
        }
    }
    return false;
}

void print(const vector<int>& a)
{
    for (int i = 0; i < a.size(); i++)
    {
        cout << a[i] << " ";
    }
    cout << endl;
}

    {

    }
