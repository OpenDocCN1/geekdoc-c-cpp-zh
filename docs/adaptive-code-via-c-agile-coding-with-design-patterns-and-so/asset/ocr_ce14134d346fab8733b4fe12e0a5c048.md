```
public class Adaptee
{
    public void MethodB()
    {
    }
}
// ...
public class Adapter : Adaptee
{
    public void MethodA()
    {
        MethodB();
    }
}
// ...
class Program
{
    static Adapter dependency = new Adapter();
    static void Main(string[] args)
    {
        dependency.MethodA();
    }
}
```
