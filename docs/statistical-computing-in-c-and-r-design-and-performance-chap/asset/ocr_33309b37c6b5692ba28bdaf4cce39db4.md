The details required for implementing Algorithm 7.2 have been collected into another new method for class Matrix.

```
Matrix::cholesky(const Vector& RHS) const {
    double temp = 0;
    Matrix G(nRows, nCols);
    for(int j = 0; j < nRows; j++){//proceed by columns
        if(pA[j][j] == 0) {
            cout << "Singular system!" << endl;
            exit(1);
        }
        temp = pA[j][j];starting value for diagonal recursion
        if (j > 0) {
            for(int k = 0; k < j; k++)
                temp += G[j][k]*G[j][k];
        }
        G.pA[j][j] = sqrt(temp);
    }
    for(int i = 0; (j + 1); i < nRows; i++)//now the rest
        temp = pA[i];
        for(int k = 0; k < j; k++)
            temp -= G[i][k]*G[j][k];
        G.pA[i][j] = temp/G[j][j];
    }
}
cout << "Cholesky factor:" << endl;
nG.printMatrix();
Vector h = G.forward(RHS);
Vector b = G.trans().backward(h);
return b;
```

The code implements the scheme of Algorithm 7.2 to compute the lower-triangular Cholesky factor and store it in a Matrix object. In general there is no need to retain G, although the developments in Section 7.2.5 demonstrate this need not always be the case. Once G has obtained it provides the matrix that is used in forward-solving the transpose of G is used for back-solving.
