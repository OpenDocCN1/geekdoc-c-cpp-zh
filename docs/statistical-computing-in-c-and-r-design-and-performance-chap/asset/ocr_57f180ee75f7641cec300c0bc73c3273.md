```
//Overloaded multiplication operators
Matrix operator*(const Vector& v1, const Vector& v2) {
Matrix B(v1.getRows(), v2.getRows());
for(int i = 0; i < v1.getRows(); i++)
for(int j = 0; j < v2.getRows(); j++)
B.pA[i][j] = v1[i]*v2[j];
return B;
```
