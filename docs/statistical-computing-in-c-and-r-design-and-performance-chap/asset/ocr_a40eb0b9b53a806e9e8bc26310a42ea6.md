The one-dimensional array constructor with the pointer argument acts like the constructor that was used for class Vector in Section 3.9 in that it creates the Array1D object using a pointer (or array). The two-dimensional case involving a pointer is much more involved. A one-dimensional array of dimension m*n must be given to the constructor with the contents of the underlying array arranged in row major form.

In Listing 7.9 below of the contents of a matrix are read from a file and subject to various manipulations using tools from the TNT package. The scope resolution operator is employed to specify references to names in the TNT namespace.

Listing 7.9 tntEx1.cpp

```
//tntEx1.cpp
#include <iostream>
#include <fstream>
#include "tnt.h"
using std::cout; using std::endl;
int main(int argc, char** argv){
    int m = atoi(argv[2]); int n = atoi(argv[3]);
    double* pc = new double[n];
    double* pA = new double[m*n];
    std::ifstream fin;
    fin.open(argv[1]);
    for(int i = 0; i < m; i++){
        for(int j = 0; j < n; j++)
            fin >> pA[i*n + j];//row major form
        fin >> pc[i];
    }
    TNT::Array1D<double> c(n, pc);
    cout << "The vector c" << endl;
    cout << c << endl;
    TNT::Array2D<double> A(m, n, pA);
    cout << "The matrix A" << endl;
    cout << A << endl;
    TNT::Array2D<double> B(m, n);
    for(int i = 0; i < m; i++){
        for(int j = 0; j < n; j++)
            B[i][j] = (double)(i + 1)*(j + 1);
    }
    cout << "The matrix B" << endl;
    cout << B << endl;
    cout << "The sum of A and B" << endl;
    cout << A + B << endl;
    cout << "The product of A and B" << endl;
    cout << matmul(A, B) << endl;
    return 0;
}
```
