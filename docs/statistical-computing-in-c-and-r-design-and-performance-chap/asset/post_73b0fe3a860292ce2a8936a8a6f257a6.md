Vector& Vector::operator=(const Vector& v) {
    if(this == &v)//avoid self assignment
        return *this;
    // define/redefine object's members
    if(nRows != v.nRows) {
        if(pa != 0)
        delete [] pa;

        nRows = v.nRows;
        pa = new(std::nothrow) double[nRows];
        pointerCheck();
    }
    for(int i = 0; i < nRows; i++)
    pa[i] = v.pa[i];
    return *this;
