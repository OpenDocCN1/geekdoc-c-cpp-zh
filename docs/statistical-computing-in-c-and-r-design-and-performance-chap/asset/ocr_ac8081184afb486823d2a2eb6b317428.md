```
//vecEx1.cpp
#include <iostream>
#include <vector>

using std::; using std::endl; using std::vector;

voidbtnPrint(int size, int cap){
    cout << "The size is " << size << endl
      << " and the capacity is " << cap << endl;
}

int main() {
    vector<int> v(1);
    v[0] = 12;

    printSize(v.size(), v.capacity());
    v.push_back(1);
    printSize(v.size(), v.capacity());
    v.push_back(2);
    printSize(v.size(), v.capacity());
    for(int i = 0; i < v.size(); i++)
        cout << "[" << i << "] = " << v[i] << " ";
    cout << endl;

    v.pop_back();
    vector<int>::iterator iterator = v.insert(v.begin(), 1, 17);
    cout << "Dereferenced = " << *iter << endl;
    for(iter iter = v.begin(); iter != v.end(); iter++)
        cout << *iter << " ";
    cout << endl;
    v.clear();
    printSize(v.size(), v.capacity());

    return 0;
}
```

Listing 10.1 begins with a print utility program. Then the main function initializes a container for int variables, the integer 12 in its first slot and uses the push.back method to append two integers to the array. After each addition, the values returned from size are printed to standard output before examining the contents of the vector object. Next, the inserted element is removed with pop.back and the insert method is employed to insert the 17 between the first and second int in the container (i.e., in the slot with 1). The position for insertion is designated by incrementing the begin iterator for the vector object. A new iterator is defined to hold the one returned from insert and this new iterator is dereferenced to view the int
