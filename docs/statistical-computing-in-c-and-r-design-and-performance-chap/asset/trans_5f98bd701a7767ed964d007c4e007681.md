Vector pdBand::cholesky(const Vector & RHS) const {
  double temp = 0;
  int low, up;
  int rows = this->getRows();
  pdBand G(rows, bw);

  for(int i = 0; i < rows; i++) {
    if(pA[i][0] == 0){
        cout << "Singular system!" << endl;
        exit(1);
    }
    temp = pA[i][0];
    low = std::max(0, i - bw);
    for(int k = low; k < i; k++) {
        temp -= G[k][i]*G[k][i];
    }
    G.pA[i][0] = sqrt(temp);
    up = std::min(i + bw, rows - 1);
    for(int j = i + 1; j <= up; j++) {
        temp = pA[j][i];
        for(int k = low; k < i; k++) {
            temp -= G[k][j]*G[k][i];
        }
    }
  }
  return G;
}

Matrix pdBand::bandToFull() const {
  double** pTemp = new double*[getRows()];
  if(pTemp == NULL) {
    cout << "Memory allocation for pTemp" << endl;
    exit(1);
  }
  for(int i = 0; i < getRows(); i++) {
    pTemp[i] = new double[getRows()];
    for(int j = 0; j < getRows(); j++) {
      pTemp[i][j] = 0;
    }
  }
  for(int i = 0; i < getRows(); i++) {
    for(int j = 0; j < getRows(); j++) {
      if(abs(i-j) <= bw) {
        pTemp[i][j] = pA[i][j]; // 基于上下文的逻辑
      }
    }
  }
  // ... bandToFull 的剩余
  return Matrix(); // 逻辑占

        return *this;
    // 定义对象的成员

        nRows = v.nRows;
        pa = new(std::nothrow) double[nRows];
        pointerCheck();

    return *this;
