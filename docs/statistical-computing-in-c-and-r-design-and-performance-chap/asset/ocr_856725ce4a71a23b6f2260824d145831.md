Vector Matrix::bandChol(const Vector& RHS, Matrix& G,
int bWidth) const {
    double temp = 0;
    int low, up;

    for(int j = 0; j < nCols; j++){
        if(pA[j][j] == 0){
            cout << "Singular system!" << endl;
            exit(1);
        }
        temp = pA[j][j];
        low = std::max(0, j - bWidth);
        for(int k = low; k < j; k++) {
            temp -= G[j][k]*G[j][k];
        }
        G.pA[j][j] = sqrt(temp);
        up = std::min(j + bWidth, nRows - 1);
        for(int i = j + 1; i <= up; i++) {
            temp = pA[i][j];
            for(int k = low; k < j; k++) {
                temp -= G[i][k]*G[j][k];
            }
            G.pA[i][j] = temp/G.pA[j][j];
        }
    }

    Vector h = G.bandFor(RHS, bWidth);
    Vector b = G.trans().bandBack(h, bWidth);
    return b;
}
