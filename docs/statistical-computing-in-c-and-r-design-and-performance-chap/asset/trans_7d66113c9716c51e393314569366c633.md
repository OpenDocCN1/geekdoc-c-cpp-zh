        temp = 0;

        C.p[i] = temp;

}

Matrix::trans() const {
    Matrix B(nCols, nRows, 0.0);
    for(int i = 0; i < nCols; i++)
        for(int j = 0; j < nRows; j++)
            B.p[i][j] = pA[j][i];
    return B;
}

Vector::backward(const Vector& RHS) const {
    double temp; // 临时存储
    Vector b(nRows, 0.0);
    // 初始化递归
    if(pA[nRows-1][nRows-1] != 0)
        b[nRows-1] = RHS[nRows-1]/pA[nRows-1][nRows-1];
    else {
        cout << "奇异方程组!" << endl;
        exit(1);
    }
    // 现在处理剩余行
    for(int i = nRows-2; i >= 0; i--) {
        if(pA[i][i] != 0) {
            temp = RHS[i];
            for(int k = i+1; k < nRows; k++)
                temp -= b[k]*pA[i][k];
            b[i] = temp/pA[i][i];
        } else {
            cout << "奇异方程组!" << endl;
            exit(1);
        }
    }
    return b;
}

Vector Matrix::forward(const Vector& RHS) const {
    double temp; // 临时存储
    Vector b(nRows, 0); // 结果矩阵
    // 初始化递归
    if(pA[0][0] != 0)
        b[0] = RHS[0]/pA[0][0];
    else {
        cout << "奇异方程组!" << endl;
        exit(1);
    }
    // 现在处理剩余行
    for(int i=1; i<nRows; i++) {
        if(pA[i][i] != 0) {
            temp = RHS[i];
            for(int k = 0; k < i; k++)
                temp -= b[k]*pA[i][k];
            b[i] = temp/pA[i][i];
        } else {
            cout << "奇异方程组!" << endl;
            exit(1);
        }
    }
    return b;
}

Vector Matrix::gauss(const Vector& RHS) const {
    double multiplier = 0;
    Matrix G(nRows, nRows, pA);
    Vector h(nRows, RHS);

    for(int j=0; j < nCols; j++) { // 需要消除的列
        for(int jj = j; jj < nCols; jj++) { // 当前操作
            if(G[jj][j] == 0) {
                cout << "错误！除以零!" << endl;
                exit(1);
            }
            for(int i = j+1; i < nRows; i++) { // 对 RHS 执行相同操作
                multiplier = G[i][j]/G[jj][j];
                G[i][jj] -= multiplier*G[jj][jj];
                h[i] -= multiplier*h[j]; // 注意：OCR 文本中的逻辑此处似乎有些碎片
            }
        }
    }
    // 现在求解
    Vector b = G.backward(h);
    return b;
}

        b[nRows-1] = RHS[nRows-1]/pA[nRows-1][nRows-1];
    else {
        cout << "奇异方程组!" << endl;
        exit(1);
    }

    // 处理剩余行
    for(int i = nRows - 2; i >= 0; i--) {
        if(pA[i][i] != 0) {
            temp = RHS[i];
            up = std::min(nCols, i + bWidth);
            for(int k = i + 1; k < up; k++)
                temp -= b[k]*pA[i][k];

            cout << "奇异方程组!" << endl;
            exit(1);
        }
    }
    return b;
}
