```
import AppTheme (paperColor) ①
import CSS.Background (backgroundColor)
import CSS.Cursor (cursor, pointer)
import CSS.Display (display, flex)
import CSS.Flexbox (flexDirection, row, column, flexGrow)
import CSS.Geometry (minWidth, paddingRight)
import CSS.Size (rem)
import Entity.User (User(..))
import Halogen.HTML.Core (ClassName(..))

render :: State -> H.ComponentHTML Action Slots m
render { authorized, users } = ②
  if not authorized then HH.text "NOT AUTHORIZED" else ③
  HH.div [
    HC.style do
      display flex
      flexDirection row ④
      flexGrow 1.0
  ]
  [ HH.div [ ⑤
    HC.style do
      display flex
      flexDirection column
      minWidth (rem 20.0)
      paddingRight (rem 2.0)
  ]
  [ HH.ul_ ⑥
    (users <#> \(User { userName }) -> ⑦
      HH.li [
        HP.class_ $ ClassName "list-group-item" ⑧
      , HC.style do
          backgroundColor paperColor
          cursor pointer
      ]
      [ HH.text userName ])
  ]
  ]
```

① Added these imports except for Entity.User, which (..) was added.

② Destructure State.

③ We first make sure that the the user is authorized to view this page and that they didn’t just type in the URL to this location in the Address Bar or use a link they got from someone who does have authorization.
