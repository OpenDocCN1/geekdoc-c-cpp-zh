import Data.Either (Either, either)

import Control.Monad.Trans.Class (class MonadTrans, lift)

liftEither :: ∀ e m a. MonadError e m => Either e a -> m a
liftEither = either throwError pure

liftSuccess
  :: ∀ e a m t
   . MonadError e (t m)
   => Monad m
   => MonadTrans t
   => m (Either e a)
   -> t m a
liftSuccess ma = ma # lift >>= liftEither
