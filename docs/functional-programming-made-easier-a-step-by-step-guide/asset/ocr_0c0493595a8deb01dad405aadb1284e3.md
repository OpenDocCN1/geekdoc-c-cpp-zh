```
import Effect.Aff.Class (class MonadAff, LiftAff)

withAVar :: ∀ a b m. MonadAff m => AVar a → (a → m (Tuple a b)) → m b
withAVar aVar f = do
  value <- liftAff $ AVar.take aVar
  Tuple newValue result <- f value
  liftAff $ AVar.put newValue aVar
  pure result
```
