import ChalkStyles (Style, bold, red, dim, strikethrough) ① ②

test :: Effect Unit

  let colorful styles = chalk $ red : styles ④
  log $ colorful [ bold, strikethrough ] "Test" ⑤
    <> colorful [] "Test" <> colorful [ dim ] "Test" ⑥
