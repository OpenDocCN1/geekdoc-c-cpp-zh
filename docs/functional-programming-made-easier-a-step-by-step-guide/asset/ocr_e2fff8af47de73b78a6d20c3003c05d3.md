ugly :: ∀ e. ParserError e => Parser e (Array String) ❶
ugly = do
  p1 <- range' 1 4 digit
  constChar ','
  constChar ' '
  p2 <- some' (letter <|> constChar ' ')
  p3 <- many' digit
  pure [p1, p2, p3]
