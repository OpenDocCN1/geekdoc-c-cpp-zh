import Data.Either (isLeft)
import Node.Encoding (Encoding(..))
import Node.FS.Aff (exists, writeTextFile)

loadAccounts :: Aff (Array Account)
loadAccounts = do
  exists <- try $ exists accountsFile
  when (isLeft exists)
    $ writeTextFile ASCII accountsFile bootstrapAccount
  pure []
```
```

+   4. We don’t care why it doesn’t exist, just that it doesn’t. That’s why we’re using `isLeft`.
+   5. We WANT it to crash if we cannot write so there is no use of `try` here.
