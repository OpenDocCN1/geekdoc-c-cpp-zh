def sign(a, b):
    if (b >= 0.0):
        return abs(a)
    else:

def Forces(t, w, PE, PEorW):                     # Forces
    # invr2 = 0.
    r2cut = 9.                                   # Switch: PEorW = 1 for PE
    PE = 0.
    for i in range(0, Natom):
        fx[i][t] = fy[i][t] = 0.0
    for i in range(0, Natom-1):
        for j in range(i + 1, Natom):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            if (abs(dx) > 0.50*L):
                dx = dx - sign(L, dx)            # Interact with closer image
            if (abs(dy) > 0.50*L):
                dy = dy - sign(L, dy)
            r2 = dx*dx + dy*dy
            if (r2 < r2cut):
                if (r2 == 0.):                   # To avoid 0 denominator
                    r2 = 0.0001
                invr2 = 1.0/r2
                wij = 48.*(invr2**3 - 0.5) *invr2**3
                fijx = wij*invr2*dx
                fijy = wij*invr2*dy
                fx[i][t] = fx[i][t] + fijx
                fy[i][t] = fy[i][t] + fijy
                fx[j][t] = fx[j][t] - fijx
                fy[j][t] = fy[j][t] - fijy
                PE = PE + 4.*(invr2**3)*((invr2**3) - 1.)
                w = w + wij
    if (PEorW == 1):
        return PE
    else:
        return w
