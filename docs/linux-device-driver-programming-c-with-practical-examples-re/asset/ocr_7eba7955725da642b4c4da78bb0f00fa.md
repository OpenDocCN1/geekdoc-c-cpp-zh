#include <unordered_map>
std::unordered_map<int, void *> hash_table;

void hash_table_insert(int key, void *data) {
    hash_table[key] = data;
}

void *hash_table_lookup(int key) {
    auto it = hash_table.find(key);
    if (it != hash_table.end())
        return it->second;
    return NULL;
}
