float calculate(operator_t operator, float lh_f, float rh_f)
{
float result_f= 0.0F;
switch (operator)
{
case op_add:
result_f= lh_f + rh_f;
break;
case op_subtract:
result_f= lh_f - rh_f;
break;
case op_multiply:
result_f= lh_f * rh_f;
break;
case op_divide:
result_f= lh_f / rh_f;
break;
default:
/* Do nothing. */
break;
}
return result_f;
}
