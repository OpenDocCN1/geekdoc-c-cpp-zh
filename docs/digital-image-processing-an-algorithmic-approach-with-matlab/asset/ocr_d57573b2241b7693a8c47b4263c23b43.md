x=imread('sr71.jpg');
x = rgb2gray(x) ;
[r,c] = size(x);

% Gaussian Noise
xng = imnoise(x,'gaussian',0,0.1);
figure ; imshow(xng)
XNG = fft2(xng,1024,1024) ;
figure ; imshow(fftshift(abs(XNG))*0.00001)
G1 = XNG .* H1 ;
g1 = ifft2(G1) ;
g1 = g1(1:r,2:c) ;
figure ; imshow(g1*0.004)
G2 = XNG .* H2 ;
g2 = ifft2(G2) ;
g2 = g2(1:r,2:c) ;
figure ; imshow(g2*0.004)
