public static double CalculateDistance(double
                                     numberOfStepsPerMin,
                                     int totalMinutes)
{
    const double STRIDE = 2.5;
    const int FEET_PER_MILE = 5280;

    return numberOfStepsPerMin * STRIDE * totalMinutes /
           FEET_PER_MILE;
}
