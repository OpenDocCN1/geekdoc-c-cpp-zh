public class TestOfStack
{
    static void Main(string[] args)
    {
        Stack stack = new Stack(5);

        stack.Push("test");
        stack.Push(100);
        stack.Push(200);
        stack.Push(3.6);

        Console.WriteLine("Values in the Stack are: " +
            (double)stack.Pop() + ", " +
            (int)stack.Pop() + ", " +
            (int)stack.Pop() + ", " +
            (string)stack.Pop());
    }
}

As illustrated in Examples 11–25 and 11–26, without generics you could define the data type as an object because it can hold any type. To retrieve the items, the last four lines in Example 11–26 show that casts had to be performed to unbox or extract the values from the `object` type. In Example 11–26, because the last item placed on the Stack was a `string`, the first call to Pop( ) retrieved the string. The program would have terminated abnormally if the order of the casting was changed.
