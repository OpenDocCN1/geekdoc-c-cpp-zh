using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace FluentLearning
{
    class myClass
    {
        ArrayList myArray;

        myClass()
        {
            myArray = new ArrayList();

            for (Int32 x = 1; x <= 100; x++)
            {
                myArray.Add(x);
            }

            myArray[5] = 325;

            foreach (Int32 x in myArray)
            {
                Console.WriteLine(myArray[x]);
            }
        }
    }
}
