# 3. In the code-behind file, MainWindow.xaml.cs, add the following methods:

```
private void GenerateData()
{
  Car car = new Car() { Make = "Athlon", Model = "XYZ" };
  this.DataContext = car;
}

private void btnShowDataCellValue_Click(object sender, RoutedEventArgs e)
{
  Car dc = this.DataContext as Car;
  MessageBox.Show("Make: " + dc.Make + " \nModel: " + dc.Model");
}

private void btnChangeDataCellValue_Click(object sender, RoutedEventArgs e)
{
  Car dc = this.DataContext as Car;
  dc.Make = "Changed Make";
  dc.Model = "Changed Model";
}
```
