  // InputDecimal -- Return a positive decimal number
  // from the keyboard.
  public static decimal InputDecimal(string prompt)
  {
    while (true)
    {
      // Keep until the user gets it right.
      Console.Write("Enter " + prompt + \*");
      // Retrieve a decimal value from the keyboard.
      string input = Console.ReadLine();
      decimal value = Convert.ToDecimal(input);
      // Exit the loop if the value is correct.
      if (value >= 0)
      {
        // Return the decimal value entered by the user.
        return value;
      }
      // Otherwise, generate an error on incorrect input.
      Console.WriteLine(prompt + " *" cannot be negative!");
      Console.WriteLine("Try again!");
      Console.ReadLine();
    }
  }

  // OutputTable -- Given the principal and interest,
  // generate a value table for the number of periods
  // indicated duration. (Implements Section 3.).
  public static void OutputTable(decimal principal, decimal interest, decimal duration)
  {
    for (int i = 1; i <= duration; i++)
    {
      // Calculate the value of the principal plus interest.
      decimal interestPaid = principal * (interest / 100);
      // Now add the new principal by adding
      // the interest to the previous principal.
      principal = principal + interestPaid;
      // Round off the principal to the nearest cent.
      principal = decimal.Round(principal, 2);
      // Output the result:
      Console.WriteLine(i + " *-\* " + principal);
    }
  }
}
