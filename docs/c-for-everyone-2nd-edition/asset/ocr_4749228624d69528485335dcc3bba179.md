    padding = 4 - scanline_size % 4;

        int pos = stream.tellg(); // Go to the start of the pixel

        int green = stream.get();
        int red = stream.get();

        process(blue, green, red); // Process the pixel

        stream.seekg(pos); // Go back to the start of the pixel

        stream.put(green);
        stream.put(red);
