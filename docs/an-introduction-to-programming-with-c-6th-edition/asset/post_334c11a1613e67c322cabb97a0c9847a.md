//Lab11-3.cpp - finds the lowest and highest
// scores in an integer array
//Created/revised by <your name> on <current date>

#include <iostream>
using namespace std;

//function prototypes
void findMinMax(int scores[], int size, int &low, int &high);

int main() {
    //declare arrays and variables
    int scores[100];
    int size;
    int low, max;

    cout << "Enter the number of scores (1-100): ";
    cin >> size;

    for (int i = 0; i < size; i++) {
        cout << "Enter score " << i + 1 << ": ";
        cin >> scores[i];
    }

    findMinMax(scores, size, low, max);

    cout << "The lowest score is: " << low << endl;
    cout << "The highest score is: " << max << endl;

    system("pause");
    return 0;
}

void findMinMax(int scores[], int size, int &low, int &high) {
    low = scores[0];
    high = scores[0];
    for (int i = 1; i < size; i++) {
        if (scores[i] < low)
            low = scores[i];
        if (scores[i] > high)
            high = scores[i];
    }
}
