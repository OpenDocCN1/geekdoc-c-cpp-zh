#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    //declare array
    double sales[4] = {0.0}; //array declaration

    //fill array with data
    for (int sub = 0; sub < 4; sub += 1) {
        cout << "Enter the sales for Region ";
        cout << sub + 1 << " ";";
        cin >> sales[sub];
    }
} //end for
