#include <iostream>
using namespace std;

int main() {
    //declare variables
    int sales = 0;
    int totalSales = 0; //accumulator
    for (int region = 1; region < 3; region += 1) {
        //get current region's first amount
        cout << "First sales amount for Region " << region << ": ";
        cin >> sales;
        do {
            //add the sales amount to the total
            //for the region
            totalSales += sales;
            //get the next sales amount
            //current region
            cout << "Next sales amount for Region " << region << ": ";
            cin >> sales;
        } while (sales > 0);
        //display the current region's total
        cout << "****Region " << region << " sales: $" << totalSales << endl;
        //reset the accumulator
        totalSales = 0;
    } //end for

    system("pause");
    return 0;
}
