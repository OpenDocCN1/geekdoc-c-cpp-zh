//TryThis16.cpp - 向顺序访问
//访问文件写入 26 个字母
//Created/revised by <name> on <current date>

#include <iostream>
#include <fstream>
using namespace std;

int main() 
{
    char letter = ' ';

    //创建文件对象并打开文件
    ofstream outfile;
    outfile.open("TryThis16.txt");

    //判断文件是否已打开
    if (outfile.is_open())
        //获取字母表中的 26 个字母并
        //将每个写入文件
        for (int x = 1; x <= 26; x = x + 1) {
            cout << "Enter letter " << x << ":";
            cin >> letter;
            outfile << letter << endl;
        }
        //关闭文件
        outfile.close();
    else {
        cout << "The file could not be opened." << endl;
    }
    return 0;
} //main函数结束
```
