## The Random Numbers Program

```cpp
///Random Numbers.cpp - displays the highest
///random number stored in an array
//Created/revised by your <name> on <current date>

#include <iostream>
#include <ctime>
using namespace std;

// function prototypes
void display(int numbers[], int numElements);
int getHighest(int numbers[], int numElements);

int main() {
    // declare array
    int randNums[5] = {};

    // initialize random number generator
    srand(static_cast<int>(time(0)));
    // assign random integers from 1 through 100 to the array
    for (int sub = 0; sub < 5; sub++) {
        randNums[sub] = 1 + rand() % (100 - 1 + 1);
    }

    // display the array
    displayArray(randNums, 5);

    // display the highest number
    cout << endl << "Highest number: " << getHighest(randNums, 5) << endl;

    return 0;
}

void displayArray(int numbers[], int numElements) {
    // display the contents of the array
    for (int sub = 0; sub < numElements; sub++) {
        cout << numbers[sub] << endl;
    }
    // end for
}

int getHighest(int numbers[], int numElements) {
    int high = numbers[0];
    int x = 1;
    // search for highest number
    while (x < numElements) {
        if (numbers[x] > high) {
            high = numbers[x];
            // end if
        }
        x += 1;
    } // end while
    return high;
}
```
