#include <iostream>
using namespace std;

int main() {
    //declare array and variables
    int salaries[10] = {23000, 26000,
    34000, 21000,
    54000, 45000,
    36000, 38000,
    75000, 34000};
    int searchFor = 0;

    cin >> searchFor;

    while (searchFor >= 0) {

            if (salaries[sub] > searchFor) {
                numEarnOver += 1;
            }
