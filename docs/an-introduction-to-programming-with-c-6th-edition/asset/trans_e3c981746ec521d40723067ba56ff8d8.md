```cpp
//Lab14-3.cpp - 将记录保存到顺序访问
//文件并计算且显示文件中存储的销售额总额
//由 <你的名字> 在 <当前日期> 创建/修订
#include <iostream>
#include <string>
#include <fstream>
using namespace std;

// 函数原型
int getChoice();
void addRecords();
void displayRecords();
void displayTotal();
void displayAvg();

int main()
{
  int choice = 0;
  do
    // 获取用户菜单
    choice = getChoice();
    if (choice == 1)
      addRecords();
    else
      if (choice == 2)
        displayRecords();
      else
      if (choice == 3)
          displayTotal();
      else
        if (choice == 4)
          displayAvg();
    // 结束 if
} while (choice != 5);
system("pause");
return 0;
} // main 函数结束

int getChoice()
{
    // 显示菜单并返回选择
    int menuChoice = 0;
    cout << "Menu" << endl;
    cout << "1 Add Records" << endl;
    cout << "2 Display Records" << endl;
    cout << "3 Total Sales" << endl;
    cout << "4 Average Sales" << endl;
    cout << "5 Exit the program" << endl;
    cout << "Choice (1, 2, 3, 4, or 5)? ";
    cin >> menuChoice;
    cin.ignore(100, '\n');
    return menuChoice;
}

void addRecords()
{
    // 将记录保存到顺序访问文件
    string name = "";
    int sales = 0;
    ofstream outFile;
    // 以追加模式打开文件
    outfile.open("sales.txt", ios::app);
    // 如果打开成功，获取
    // 销售人员名称和销售额，
    // 然后将信息写入文件；
    // 否则，显示消息
    if (outFile.is_open()) {
        cout << "Salesperson's name (X to stop): ";
        getline(cin, name);
        while (name != "X") {
            cout << "Sales: ";
            cin >> sales;
            cin.ignore(100, '\n');
            outfile << name << " #" << sales << endl;
            cout << "Salesperson's name (X to stop): ";
            getline(cin, name);
        }
    }
}

void displayTotal()
{
    // 销售人员名称和销售额，
    // 将销售额累加累加器，
    // 否则，显示
    // 消息
    ifstream infile;
    int total = 0;
    string name;
    int sales;
    if (infile.is_open()) {
        getline(infile, name, '#');
        infile >> sales;
        infile.ignore();

        while (!infile.eof()) {
            total += sales;
            getline(infile, name, '#');
            inFile >> sales;
            inFile.ignore();
        }
        infile.close();
        cout << "Total sales $" << total << endl;
    } else {
        cout << "File not be opened." << endl;
    }
}

void displayAvg()
{
    // 计算并显示平均销售
    string name = "";
    int sales = 0;
    int totalSales = 0;
    int numSales = 0;
    double avgSales = 0.0;
    ifstream infile;
    // 打开文件
    infile.open("sales.txt");
    // 如果打开成功，读取
    // 销售人员名称和销售额，然后
    // 将销售额累加累加器并加 1
    // 到计数器；否则，显示消息
    if (infile.is_open()) {
        getline(infile, name, '#');
        infile >> sales;
        infile.ignore();

        while (!inFile.eof()) {
            totalSales += sales;
            numSales += 1;
            getline(inFile, name, '#');
            inFile >> sales;
            inFile.ignore();
        }
        inFile.close();

        // 计算并显示平均销售
        avgSales = static_cast<double>(totalSales) / numSales;
        cout << "Average sales $" << avgSales << endl;
    } else {
        cout << "File not be opened." << endl;
    }
}
```
