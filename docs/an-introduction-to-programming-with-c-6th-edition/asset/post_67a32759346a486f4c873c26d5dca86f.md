```cpp
//Modified Company.cpp
//Created by <your name>
//on <date>
#include <iostream>
using namespace std;

//Displays the contents of a two-dimensional array
int main() {
    int orders[4][3] = {};
//enter data into the array
    for (int region = 0; region < 4; region++) {
        for (int month = 0; month < 3; month++) {
            cout << "Region " << region + 1 << ", "
                 << month + 1 << ": ";
            cin >> orders[region][month];
        }
    } //end for

    //display the contents of the array
    displayArray(orders);

    system("pause");
    return 0;
} //end of main function

void displayArray(int nums[4][3]) {
    for (int region = 0; region < 4; region++) {
        cout << "Region " << region + 1 << ": " << endl;
        for (int month = 0; month < 3; month++) {
            cout << "Month " << month + 1 << ": " << nums[region][month] << endl;
        }
    } //end for
} //end of displayArray function
```
