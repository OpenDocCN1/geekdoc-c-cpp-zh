#include <iostream>
using namespace std;

int main() {
    //declare variables
    int multiplicand = 0;
    int product = 0;

    cout << "Multiplicand (negative number to end): ";
    cin >> multiplicand;

    do //begin loop
    {
        int multiplier = 1;
        do //begin loop
        {
            product = multiplicand * multiplier;
            cout << multiplicand << " * " << multiplier << " = " << product << endl;
            cout << " << = " << product << endl;
            multiplier += 1;
        } while (multiplier < 10);

        cout << endl;
        cout << "Multiplicand (negative number to end): ";
        cin >> multiplicand;
    } while (multiplicand >= 0);

    system("pause");
    return 0;
}
//end of main function
