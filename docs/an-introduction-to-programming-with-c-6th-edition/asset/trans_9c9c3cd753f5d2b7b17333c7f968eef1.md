# LAB 7-3 修改

程序显示正确的总分 (255) 和等级 (C)。程序显示教授输入了 7 个分数。

```cpp
//Lab7-3.cpp
#include <iostream>
using namespace std;
int main()
{
//declare variables
int score = 0;
int totalPoints = 0; //accumulator
char grade = ' ';
int numScores = 0; //counter

//get first score
cout << "Enter score (-1 to stop): ";
cin >> score;
while (score != -1)
{
//update accumulator and counter
totalPoints += score;
numScores++;
//get next score
cout << "Enter score (-1 to stop): ";
cin >> score;
}

//determine grade
if (numScores > 0)
{
    if (totalPoints >= 90)
        grade = 'A';
    else if (totalPoints >= 80)
        grade = 'B';
    else if (totalPoints >= 70)
        grade = 'C';
    else if (totalPoints >= 60)
        grade = 'D';
    else
        grade = 'F';

    cout << "Total points: " << totalPoints << endl;
    cout << "Grade: " << grade << endl;
}
else
{
    cout << "No scores were entered." << endl;
}

system("pause");
return 0;
}
```
