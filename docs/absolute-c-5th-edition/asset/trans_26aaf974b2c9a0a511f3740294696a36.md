int BankAccount::dollarsPart(double amount) {
    double doubleContents = amount * 100;
    int intCents = round(fabs(doubleContents)) % 100;
    // 在负数上可能会出现异常
    if (amount < 0) {
        intCents = -intCents;
    }
    return intCents;
}

# 第 7 章 构造函数和其他工具

## 类成员变量

类成员可以拥有一个类型为另一个类的成员变量。拥有一个类成员变量并没有特殊要求，但有一种特殊的符号允许在外部层类的构造函数内部调用成员变量的构造函数。显示 7.3 给出了一个示例。

## 显示 7.3 类成员变量 (3部分中的第 1 部分)

```cpp
#include <iostream>
#include <cstdlib>
using namespace std;

class DayOfYear {
public:
    DayOfYear(int month, int dayValue);
    DayOfYear(int month);
    DayOfYear();
    void output(); // DayOfYear 与显示 7.1 相同，但我们重复了你所需的细节。
    int getMonth();
    int getDay();
private:
    int month;
    int day;
    void testDate();
};

class Holiday {
public:
    Holiday(int month, int day, bool enforcement);
    void output();
private:
    DayOfYear date; // 类成员
    bool parkingEnforcement; // bool 类型
};

int main() {
    Holiday h(2, 1, true); // 类的实例化
    cout << "Testing the class.\n";
    h.output(); // 类的输出
    return 0;
}

DayOfYear::DayOfYear(int monthValue, int dayValue)
    : month(monthValue), day(dayValue)
{
    testDate();
}

DayOfYear::DayOfYear(int monthValue)
    : month(monthValue), day(1)
{
    testDate();
}

DayOfYear::DayOfYear()
    : month(1), day(1)
{
    testDate();
}

// 使用 iostream 和 cstdlib
void DayOfYear::testDate() {
    if ((month < 1) || (month > 12)) {
        cout << "Illegal month value!\n";
        exit(1);
    }
    if ((day < 1) || (day > 31)) {
        cout << "Illegal day value!\n";
        exit(1);
    }
}

// 使用 iostream
void DayOfYear::output() {
    switch (month)
        case 1:
            cout << "January "; break;
        case 2:
            cout << "February "; break;
        case 3:
            cout << "March "; break;
        case 4:
            cout << "April "; break;
        case 5:
            cout << "May "; break;
        case 6:
            cout << "June "; break;
        case 7:
            cout << "July "; break;
        case 8:
            cout << "August "; break;
        case 9:
            cout << "September "; break;
        case 10:
            cout << "October "; break;
        case 11:
            cout << "November "; break;
        case 12:
            cout << "December "; break;
        default:
            cout << "Error in DayOfYear::output.";

    cout << day;
}

int DayOfYear::getMonth() {
    return month;
}

int DayOfYear::getDay() {
    return day;
}

Holiday::Holiday(int month, int day, bool enforcement)
    : date(month, day), parkingEnforcement(enforcement)
{
}

Holiday::Holiday() : date(1, 1), parkingEnforcement(false)
{
}

void Holiday::output() {
    date.output();
    cout << endl;
    if (parkingEnforcement)
        cout << "Parking laws will be enforced.\n";
    else
        cout << "Parking laws will not be enforced.\n";
}
```
