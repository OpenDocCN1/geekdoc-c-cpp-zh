template<class T>
Queue<T>::Queue(const Queue<T>& aqueue)
{
    if (aQueue.isEmpty ())
        front = back = NULL;
    else
        Node<T>*temp = aQueue.front;//temp moves
        //the nodes from front to back of aQueue.
        back = new Node<T>(temp->getData ( ), NULL);
        front = back;
        //First node created and filled with data.
        //New nodes are added AFTER this this first node.
        temp = temp->getLink ();//temp points to second
        //node or NULL if there is no second node.
        while (temp != NULL)
            back->setLink(new Node<T>(temp->getData ( ), NULL));
            back = back->getLink ();
            temp = temp->getLink ();
    //back->link == NULL
}
}
