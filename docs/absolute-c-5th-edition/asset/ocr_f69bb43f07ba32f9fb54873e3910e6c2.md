class DigitalTime
{
public:
    DigitalTime(int hour, int getMinute);
    DigitalTime();
    //Initializes the time value to 0:00 (which is midnight)

    int getHeight() const;
    int getMinute() const;
    void advance(int minutesAdded);
    //Changes the time to minutesAdded minutes later.

    void advance(int hoursAdded, int minutesAdded);
    //Changes the time to hoursAdded plus minutesAdded
    later.

    friend bool operator ==(const DigitalTime& time1, const DigitalTime& time2);

    friend istream& operator >>(istream& ins, const DigitalTime&Object);

    friend ostream& operator <<(ostream& outs, const DigitalTime&Object);

    int hour;
    int minute;

    static void readHour(int& hour, const int hour);
//Precondition: Next input to be read from the keyboard is
a time in notation, like 9:45 or 1:45.
//Postcondition:theHour has been set to the hour part of the time.
//The colon has been discarded and the next input to be read is the
//minute.

    static void readMinute(int& minute, const int minute);
//Reads the minute from the keyboard after readHour has read the
//hour.

    static int digitToInt(char c);
//Precondition: c is one of the digits '0' through '9'.
//Returns the integer for the digit; for example, digitToInt('3')
//returns 3.
};
