    HashTable::HashTable()
    {
        for (int i = 0; i < SIZE; i++) {
            hashArray[i] = NULL;
        }
    }

    HashTable::~HashTable()
    {
        for (int i=0; i<SIZE; i++) {
            Node<string> *next = hashArray[i];
            while (next != NULL) {
                Node<string> *discard = next;
                next = next->getLink();
                delete discard;
            }
        }
    }

    int HashTable::computeHash(string s)
    {
        int hash = 0;
        for (int i = 0; i < s.length(); ); i++) {
            hash = hash + s[i];
        }
        return hash % SIZE;
    }
