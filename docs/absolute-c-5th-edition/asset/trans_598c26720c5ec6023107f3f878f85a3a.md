14. // 使用 cstddef:
```cpp
template<class T>
Stack<T>::Stack(const Stack<T>& aStack) {
    if (aStack.isEmpty()) {
        top = NULL;
    } else {
        Node<T> *temp = aStack.top; // 从 aStack 的顶部到部遍历节点。
        Node<T> *end; // 指向新栈的末。
        end = new Node<T>(temp->getData(), NULL);
        top = end;
        // 创建了第一个节点并填充了数据。
        // 新节点现在添加到此第一个节点之后。
        temp = temp->getLink(); // 将 temp 移动到第二个节点
        // 如果没有第二个节点，则为 NULL。
        while (temp != NULL) {
            end->setLink(new Node<T>(temp->getData(), NULL));
            temp = temp->getLink();
            end = end->getLink();
        }
    }
} // end->link == NULL;
```

```cpp
// 使用 cstddef;
template<class T>
void Queue<T>::add(T item)
{
    if (isEmpty()) {
        front = back = new Node<T>(item, NULL); // 将 front
        // 和 back 指向同一个节点。
    } else {
        back->setLink(new Node<T>(item, NULL));
        back = back->getLink();
    }
}

// 使用 cstdlib 和 iostream
template<class T>
T Queue<T>::remove()
{
    if (isEmpty()) {
        cout << "Error: 从空队列中移除项目。\n";
        exit(1);
    }
    T result = *front->getData();
    Node<T>*discard;
    discard = front;
    front = front->getLink();
    if (front == NULL) // 如果删除了最后一个节点
        back = NULL;

    delete discard;
    return result;
}
```
