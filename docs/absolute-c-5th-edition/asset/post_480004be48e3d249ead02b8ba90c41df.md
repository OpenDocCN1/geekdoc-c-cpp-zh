int BankAccount::dollarsPart(double amount) {
    double doubleContents = amount * 100;
    int intCents = round(fabs(doubleContents)) % 100;
    // can misbehave on negatives
    if (amount < 0) {
        intCents = -intCents;
    }
    return intCents;
}
// Uses math::

    return static_cast<int>(floor(number + 0.5));

# CHAPTER 7 Constructors and Other Tools

## Class Member Variables

A class member may have a member variable whose type is another class. There is nothing special that you need to have a class member, but there is special notation to allow the invocation of the member variable's constructor within the constructor of the outer class. An example is given in Display 7.3.

## Display 7.3 Class Member Variable (part 1 of 3)

```cpp
#include <iostream>
#include <cstdlib>
using namespace std;

class DayOfYear {
public:
    DayOfYear(int month, int dayValue);
    DayOfYear(int month);
    DayOfYear();
    void output(); // The DayOfYear is the same as in Display 7.1, but we have repeated the details you need for this.
    int getMonth();
    int getDay();
private:
    int month;
    int day;
    void testDate();
};

class Holiday {
public:
    Holiday(int month, int day, bool enforcement);
    void output();
private:
    DayOfYear date; // member variable of class type
    bool parkingEnforcement; // true enforced
};

int main() {
    Holiday h(2, 1, true);
    cout << "Testing the class.\n";
    h.output(); // Invocations of constructors from the class
    return 0;
}

Holiday::Holiday() : date(1, 1), parkingEnforcement(false)
/*Intentionally empty */

Holiday::Holiday(int month, int day, bool enforcement)
    : date(month, day), parkingEnforcement(enforcement);
```
