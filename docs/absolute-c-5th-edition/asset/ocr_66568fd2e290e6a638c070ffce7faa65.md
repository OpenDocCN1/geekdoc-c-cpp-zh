Display 19.12 Program Using the set Template Class (part 2 of 2)

```
int main( {
{
    set<char> s;

    s.insert('A');
    s.insert('D');
    s.insert('D');
    s.insert('C');
    s.insert('C');
    s.insert('B');

    cout << "The set contains:\n";
    for (auto p = s.begin(); p != s.end(); p++)
        cout << *p << " " << endl;

    cout << "Set contains 'C':" << endl;
    if (s.find('C')==s.end())
        cout << " no " << endl;
    else
        cout << " yes " << endl;

    cout << "Removing C.\n";
    s.erase('C');
    for (auto p = s.begin(); p != s.end(); p++)
        cout << *p << " " << endl;

    cout << "Set contains 'C':" << endl;
    if (s.find('C')==s.end())
        cout << " no " << endl;
    else
        cout << " yes " << endl;

    return 0;
}
```
