8. What is the output produced by the following program?

```
#include <iostream>
using std::cout;

void sampleFunction(double test) throw (int);

int main() 
{
    try {
        cout << "Trying.\n";
        sampleFunction(98.6);
        cout << "Trying after call.\n";
    }
    catch(int) {
        cout << "Catching.\n";
    }
    cout << "End program.\n";
    return 0;
}

void sampleFunction(double test) throw (int)
{
    cout << "StartingFunction.\n";
    if (test < 100)
        throw 42;
}
```
