// This is the implementation file set.cpp.

#include <iostream>
#include "listtools.h"
#include "set.h"
using std::cout;
using std::endl;
using LinkedListSavitch::Node;
using LinkedListSavitch::search;
using LinkedListSavitch::headInsert;

namespace SetSavitch
{
    template<class T>
    Set<T>::~Set() {
        Node<T>*ToDelete = head;
        while (head != NULL) {
            head = head->getLink();
            delete toDelete;
            ToDelete = head;
        }
    }

    template<class T>
    bool Set<T>::contains(T target) const {
        Node<T>* result = search(head, target);
        if (result == NULL)
            return false;
        else
            return true;
    }

    void Set<T>::output() {
        Node<T>* iterator = head;
        while (iterator != NULL) {
            cout << iterator->getData() << " ";
            iterator = iterator->getLink();
        }
    }
}
