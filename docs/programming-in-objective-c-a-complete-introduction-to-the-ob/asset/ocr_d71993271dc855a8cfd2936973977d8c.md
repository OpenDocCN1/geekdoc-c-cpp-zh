int main(int argc, char *argv[]) {
    NSAutoreleasePool *pool = [NSAutoreleasePool alloc init];
    unsigned int w1 0xA0A0A0A0, w2 0x00007777;
    NSLog("%%x %x %x, %x & %x", w1, w2, w1, w2);
    NSLog("%%x %x %x", -w1, -w2, -w3);
    NSLog("%%x %x %x, %x | %x | %x", w1, w1, w1, w2 | w3);
    NSLog("%%x %x", w1 | w2 & w3, -(-w1 & -w3));
    NSLog("%%x %x", -(w1 & -w2), -(-w1 | -w2));
    [pool drain];
    return 0;
}

# The Left Shift Operator

When a left shift operation is performed on an operation, the bits contained in the value are left shifted. The associated with this operation is the number of places (bits) that the value is to be shifted. Bits that are shifted out through the high-order bit of the data item are lost, and 0s are always shifted through the low-order bit of the value. So if w1 is w1 << 1;

which can also be expressed as

```
w1 << 1;
```
