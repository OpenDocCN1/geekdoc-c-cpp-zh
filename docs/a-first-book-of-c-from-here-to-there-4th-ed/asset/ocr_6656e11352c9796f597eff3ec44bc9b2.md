```
#include <iostream>
#include <string>
using namespace std;

bool isvalidInt(string); // function prototype (declaration)
int main()
{
    string value;
    int number;

    cout << "Enter an integer: ";
    getline(cin, value);

    if (!isvalidInt(value))
        cout << "The number you entered is not a valid integer.";
    else
    {
        number = atoi(value.c_str());
        cout << "The integer you entered is " << number;
    }

    return 0;
}
```
