template<typename T>
T interpolation_search_helper(T A[], std::size_t l,
    std::size_t r, T value)
{
    std::size_t found = 0, next;

    if (A[l] == value)
    {
        found = l;
    }
    else if ((l == r) || (A[l] == A[r]))
    {
        found = 0;
    }
    else
    {
        next = l + (value - A[l]) * (r - l) / (A[r] -
            A[l]);

        if (value < A[next])
        {
            found = interpolation_search_helper(A,
                l, next - 1, value);
        }
        else
        {
            found = interpolation_search_helper(A,
                next, r, value);
        }
    }
    return found;
}
