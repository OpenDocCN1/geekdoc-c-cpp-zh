#include <iostream>

struct A
{
    int m;

    A(int i) : m(i)
    {
        std::cout << "A("
                  << m << ")" << std::endl;
    }

    ~A()
    {
        std::cout << "~A()" << std::endl;
    }

    operator bool()
    {
        return m != 0;
    }
};

void func_simulate()
{
    int i = 10;

label:
    {
        A a = i;

        if (a)
        {
            i = 0;
            goto label;
        }
    }
}

int main()
{
    func_simulate();
}
