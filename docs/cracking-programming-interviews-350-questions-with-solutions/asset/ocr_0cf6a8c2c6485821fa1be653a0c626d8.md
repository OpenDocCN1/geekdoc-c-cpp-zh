# **Question 13 Understanding Point of Declaration**

What is the output of this program? How the template arguments are deduced for the class template B?

Program 8.16: Understanding Point of Declaration

```
~~~cpp
#include <iostream>

void test()
{
    int a1 = 10;
    {
        int a1 = a1;
        std::cout << "a1:_" << a1
                  << std::endl;
    }

    const int a2 = 20;
    {
        int a2[a2];
        std::cout << "size_of_array:"
                  << sizeof(a2)/sizeof(int)
                  << std::endl;
    }

    const int a3 = 30;
    {
        enum { a3 = a3 };
        std::cout << "enum_a3_is:_"
                  << a3 << std::endl;
    }
}

struct A
{
    enum { e = 40 };
    int arr[A::e];
};

typedef int T;

template<typename T = T, T n = 0>
struct B {};

int main()
{
    test();

    A a;
    std::cout << "length_of_array_arr_is:_"
              << sizeof(a.arr)/sizeof(int)
              << std::endl;
}
~~~
```
