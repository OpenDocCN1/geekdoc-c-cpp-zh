4  template <int m, int n>

8  template<int m, int n>
9  void spiral_print(TwoDimArray<m, n> & a)
10 {
11     int end_row_index = m;
12     int end_col_index = n;
13
14     int cur_index, start_row_index = 0,
15         start_col_index = 0;
16     while (start_row_index < end_row_index &&
17            start_col_index < end_col_index)
18     {
19         // Print the first row from the remaining
20         rows
21         for (cur_index = start_col_index; cur_index
22              < end_col_index; ++cur_index)
23         {
24             std::cout << a[start_row_index][
25                 cur_index] << " ";
26         }
27         start_row_index++;
28
29         // Print the last column from the remaining
30         columns
31         for (cur_index = start_row_index; cur_index
32              < end_row_index; ++cur_index)
33         {
34             std::cout << a[cur_index][end_col_index
35                 - 1] << " ";
36         }
37         end_col_index--;
38
39         // Print the last row from the remaining
40         rows
41         if ( start_row_index < end_row_index)
42         {
43             for (cur_index = end_col_index-1;
44                  cur_index >= start_col_index; --
45                  cur_index)
46             {
47                 std::cout << a[end_row_index-1][
48                     cur_index] << " ";
49             }
50         }
51     }
52 }
