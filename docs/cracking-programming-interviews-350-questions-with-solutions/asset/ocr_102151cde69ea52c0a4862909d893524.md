template <typename ForwardIterator>
ForwardIterator first_max_element(
    ForwardIterator first , ForwardIterator last )
{
    if (first == last) return last;
    ForwardIterator max_result = first;
    while (++first != last)
    {
        if (*max_result < *first)
            max_result = first;
    }
    return max_result;
}
