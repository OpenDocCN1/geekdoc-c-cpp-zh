 6:     l_i ← 0
 7:     while start ≠ end do
 8:         max_ending_here ← (max_ending_here + *start)

10:             max_ending_here ← 0
11:             l ← start + 1
12:         end if
13:         if max_so_far < max_ending_here then
14:             max_so_far ← max_ending_here
15:             l ← l_i
16:             r ← start
17:         end if
18:         start ← start + 1
19:     end while
20:     return <max_so_far, l, r>
