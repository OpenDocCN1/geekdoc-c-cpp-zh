TwoDimArray<m, n> & a, int x)
{
    size_t i = 0, j = n - 1;
    ListIndices list_indices;
    int current_col_index = j;

    while(j <= n - 1)
    {
        if(a[i][j] < x) i += 1;
        else if(a[i][j] > x) j -= 1;
        else // a[i][j] == x
        {
            current_col_index = j;
            while(current_col_index >=0 &&
                a[i][current_col_index] == x)
                list_indices.push_back(
                    PairIndices(i, current_col_index--));

            ++i;
        }

    }

    return list_indices;
}
