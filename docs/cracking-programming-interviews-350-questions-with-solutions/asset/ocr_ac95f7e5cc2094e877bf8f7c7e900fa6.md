int C; // Celebrity
i = 0;
while(i < Size)
{
    person_stack.push(i);
    i = i + 1;
}

int A = person_stack.top();
person_stack.pop();

int B = person_stack.top();
person_stack.pop();

while(person_stack.size() != 1)
{
    if( KnowCeleb[A][B] )
    {
        A = person_stack.top();
        person_stack.pop();
    }
    else
    {
        B = person_stack.top();
        person_stack.pop();
    }
}

// Potential celebrity candidate
C = person_stack.top();
person_stack.pop();

// Last candidate was not examined,
// it leads one excess comparison (optimize)
if (KnowCeleb[C][B])
    C = B;

if (KnowCeleb[C][A])
    C = A;

i = 0;
while(i < Size)
{
    if(C != i)
        person_stack.push(i);
    i = i + 1;
}

while(!person_stack.empty())
{
    i = person_stack.top();
    person_stack.pop();
}
