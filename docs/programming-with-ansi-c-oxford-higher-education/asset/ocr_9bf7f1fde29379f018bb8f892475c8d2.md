Right-click function. It takes the reference of the file the mouse is pointing to at the moment as an input

```
void MyApp OnRightClick(News &NewsItem, int Choice)
{
string Address;
switch(Choice)
// Choice represents what is chosen after right-click
{
case SENDNEWS:
cout << "Enter address";
cin >> Address;
NewsItem.SendNews(Address);
break;
case STORENEWS:
NewsItem.StoreNews();
break;
// The following is not acceptable
/* case PRINTNEWS:

}
}
```
