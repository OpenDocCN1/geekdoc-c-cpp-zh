void dll::hsort()
{
    hpfy(); //we begin with heapifying the entire heap
    dnode *ptr=h2; // Now we are pointing from head h2 to the head h1
    int tcnt=ncnt;
    while((tcnt>=2))
    {
        int t=ptr->data;
        ptr->data=h1->data;
        h1->data=t;
        swapdown(1,--tcnt);
        ptr=ptr->blink;
    }
}
