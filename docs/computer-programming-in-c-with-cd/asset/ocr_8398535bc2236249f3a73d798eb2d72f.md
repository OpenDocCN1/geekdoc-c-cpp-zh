                                      // matrix multiplication
{
    int i,j,k,row,col,temp[10][10];
    for(i=0;i<r;i++)

                temp[i][j]=num[i][k]*m.num[k][j]; //Formula for matrix
                                                  // multiplication

    col=c;
    return matrix(temp,row,col);
}
