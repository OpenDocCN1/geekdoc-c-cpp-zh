# Q5. What is the output of the following program?

```
#include <iostream.h>

class CharClass {
    char ch;
public:
    CharClass(char c) {
        ch = c;
    }
    char display_ch() {
        return ch;
    }
};

int main()
{
    CharClass ob[10] = {
        CharClass('A'),
        CharClass('B'),
        CharClass('C'),
        CharClass('D'),
        CharClass('E'),
        CharClass('F'),
        CharClass('G'),
        CharClass('H'),
        CharClass('I'),
        CharClass('J')
    };
    int i;

    for(i = 0; i <10; i++)
        cout << ob[i].display_ch() << ' ';
    cout << endl;
    return 0;
}
```
