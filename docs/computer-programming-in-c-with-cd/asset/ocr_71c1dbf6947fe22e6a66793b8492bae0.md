#include <iostream.h>
#include<stdio.h>

struct date
{
    int day;
    int month;
    int year;
};
struct employee
{
    char name[30];
    int age;
    float salary;
    date join_date ;
};
int main()
{

    employee emp;
    cout<<"Enter name : ";
    gets(emp.name);
    cout<<"Enter age : ";
    cin>>emp.age;
    cout<<"Enter salary : ";
    cin>>emp.salary;
    cout<<"Enter date of joining : "<<endl;
    cout<<"Enter day :";
    cin>>emp.join_date.day;
    cout<<"Enter month : ";
    cin>>emp.join_date.month;
    cout<<"Enter year : ";
    cin>>emp.join_date.year;
    cout<<"\nThe description of employee are -->"<<endl;
    cout<<"--------------------------------------"<<endl;
    cout<<"Name :"<<emp.name<<endl;
    cout<<"Age :"<<emp.age<<endl;
    cout<<"Salary :"<<emp.salary<<endl;
    cout<<"Date of joining :"<<emp.join_date.day<<"/"<<emp.join_date.month<<"/"<<emp.join_date.year;
    return 0;
}
