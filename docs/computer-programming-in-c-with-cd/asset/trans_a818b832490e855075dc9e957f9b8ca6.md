    {
        node * Ptr=Head;

        while (Ptr->link!=NULL)
        {
            if(Ptr->data > Ptr->link->data)
            {
                int Temp= Ptr->data;
                Ptr->data = Ptr->link->data;
                Ptr->link->data=Temp;
