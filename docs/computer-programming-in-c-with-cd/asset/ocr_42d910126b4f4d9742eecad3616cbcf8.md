```cpp
#include <iostream.h>
void swap(int &a, int &b)
{
    int temp;
    temp = a;
    a = b;
    b = temp;
}
int gcd(int x, int y)
{
    if(x<y)
        swap(x,y);
    while(y>0)
    {
        int re = x % y;
        x = y;
        y = re;
    }
    return x;
}
int lcm(int x, int y)
{
    int l=0;
    l = x * y/gcd(x,y);
    return l;
}
void main()
{
    int val1, val2;
    cout<<"Enter first number ";
    cin>>val1;
    cout<<"Enter second number ";
    cin>>val2;
    cout<<"Greatest common divisor of "<<val1<<" and "<<val2<<" is "
        <<gcd(val1,val2)<<endl;
    cout<<"Lowest common multiple of "<<val1<<" and "<<val2<<" is "
        <<lcm(val1,val2)<<endl;
}
