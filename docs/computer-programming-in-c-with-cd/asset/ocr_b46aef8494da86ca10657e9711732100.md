{
    char name[10];
    int rollno,m1,m2,m3,t;
    node *ptr=head;
    node *p=new node;
    cout<<"\nEnter roll no,name,marks of 3 subjects:\n";
    cin>>rollno>>name>>m1>>m2>>m3;
    t=m1+m2+m3;
    strcpy(p->data.name,name);
    p->data.rollno=rollno;
    p->data.m1=m1;
    p->data.m2=m2;
    p->data.m3=m3;
    p->data.t=t;

    node *back=NULL;
    while((rollno>ptr->data.rollno)&&(ptr!=NULL))
    {
        back=ptr;
        ptr=ptr->link;
    }
    p->link=ptr;
    if(back==NULL)
        head=p;
    else
        back->link=p;
    display();
}
