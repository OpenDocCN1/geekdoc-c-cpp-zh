```cpp
float SUMFUN(float X, int N)
{

    float sum=0;
    float Term=X;
    int K=2;
    for (int I=1; I<=N; I++)
    {
        sum=sum + Term;
        Term=Term*X*X/(K*(K+1))*(-1);
        K=K+2;
    }

    return sum;
}
```
