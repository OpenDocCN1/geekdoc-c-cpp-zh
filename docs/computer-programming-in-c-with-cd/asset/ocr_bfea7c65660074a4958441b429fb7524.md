# ▶ Listing 4: Calling a Base Class Function by using a Derived Class Object

```cpp
#include<iostream.h>

class Base
{
public:
  void MyFuncB()
  {
    cout<<"Function of base class"<<endl;
  }
};
class Derived:public Base
{
  public:
  void MyFuncD()
  {
    cout<<"Function of derived class"<<endl;
  }
};
int main()
{
  Derived d;
  cout<<"Main Program started"<<endl;
  d.MyFuncB();
  d.MyFuncD();
  cout<<"Main Program ended"<<endl;
  return 0;
}
