#include<iostream.h>

class FloydT
{
        public:
        void printFloydT();
};

void FloydT :: printFloydT()
{
        int i,j,k=1;

        for(i=1;k<=91;i++)
        {
                cout<<endl;

                for(j=1;j<=i;j++)
                {
                        cout << k ;
                        ++k;
                }
        }
}
void main()
{
        FloydT ft;
        ft.printFloydT();
}
