    float sum = 0;
    for(int i=1;i<=n;i++)

                       // in the array
{
    float mn;
    sort();
    if(n%2==0)
        mn=(a[n/2]+a[n/2+1])/2.0;
    else
        mn=a[n/2+1];
    return(mn);
}

float stat::stdvtn() /*Function for finding the standard variation of all
                        the elements in the array as compared to the average value*/
{
    int sum, std;
    float av=average();
    for(int i=1;i<=n;i++)
        sum+=(a[i]-av)*(a[i]-av);
    std=sqrt(sum);
    return(std);
}
