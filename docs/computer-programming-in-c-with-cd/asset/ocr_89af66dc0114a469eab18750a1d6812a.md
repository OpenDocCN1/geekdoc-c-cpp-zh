#include <string.h>
class StringClass {
    char *p;
public:
    StringClass(char *s);
    StringClass(StringClass &o);
    ~StringClass() {
        delete [] p;
    }
    char *get() {
        return p;
    }
};
StringClass::StringClass(char *s)
{
    int l;
    l = strlen(s)+1;
    p = new char [l];
    strcpy(p, s);
}
StringClass::StringClass(StringClass &o)
{
    int l;
    l = strlen(o.p)+1;
    p = new char [l];
    strcpy(p, o.p);
}
void show(StringClass x)
{
    char *s;
    s = x.get();
    cout << s << endl;
}
int main()
{
    StringClass a("www.kogent.com"), b("www.dreamtech.com");
    a=b;
    StringClass c=b;
    show(a);
    show(b);
    show(c);
    return 0;
}
