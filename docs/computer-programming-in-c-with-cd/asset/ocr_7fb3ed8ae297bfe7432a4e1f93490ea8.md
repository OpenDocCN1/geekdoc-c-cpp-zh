#include <iostream.h>

class Stack {
    int stck[3],size,top;
public:
    Stack();
    ~Stack();
    void push(int i);
    int pop();
};

Stack::Stack(){
    top = 0;
    size=3;
    cout << "Stack Initialized\n";
}

Stack::~Stack(){
    cout << "Stack Destroyed\n";
}
void Stack::push(int i){
    if( top == size ) {
        cout << "Stack is full.\n";
        return;
    }
    stck[ top ] = i;
    top++;
}
int Stack::pop() {
    if( top == 0 ) {
        cout << "Stack underflow.\n";
        return 0;
    }
    top--;
    return stck[ top];
}
int main()
{
