Listing 32 shows how to create an array of pointer objects of a class (you can find the ArrayPointerObjects.cpp file on the CD in the Code\Chapter7 folder):

# Listing 32: Creating an Array of Pointer Objects of a Class

```
```cpp
#include<iostream.h>
class ABC
{
int num1;
//private data member
public:
int num2;
//public data member
void setValue(int);
void displayValue();
};
void ABC::setValue(int p)
{
num1=p;
}
void ABC::displayValue()
{
cout<<"num1 = "<<num1<<endl;
cout<<"num2 = "<<num2<<endl;
}
void main()
{
ABC obj[4];
//Declaring an array of objects
for(int i=0;i<4;i++)
{
obj[i].setValue(i+1);
obj[i].num2=(i+1)*5;
}
ABC *ptr[4];
for(int k=0;k<4;k++)
ptr[k]=&obj[k];
for(int j=0;j<4;j++)
{
cout<<"Object no. "<<j+1<<endl;
ptr[j]->displayValue();
cout<<endl;
}
}
```
