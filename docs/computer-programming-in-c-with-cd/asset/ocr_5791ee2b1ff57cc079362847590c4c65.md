void display(int array[], int n)
{
        int i;

        {
                cout<<array[i]<<endl;
        }

void sort(int array[], int n)
{
        int temp = 0;
        for (int i=0; i<n; i++)
        {
                for(int j=0; j<=i; j++)
                {
                        if(array[i] < array[j])
                        {
                                temp = array[i];
                                array[i] = array[j];
                                array[j] = temp;
                        }
                }
        }
        display(array, n);
}
