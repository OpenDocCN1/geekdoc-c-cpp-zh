## Listing 5: Creating Three-dimensional Array

```cpp
#include<iostream.h>
int main()
{
  int num=0;
  int Arr[2][4][3]={{{1,2,3},{4,5,6},{7,8,9}},{{3,0,0},{8,9,4},{4,4,4},{2,0,9}}};
  for(int i=0;i<2;i++)
  {
      for(int j=0;j<4;j++)
      {
          for(int k=0;k<3;k++)
          {
              if(Arr[i][j][k]==4)
                  ++num;
          }
      }
  }
  cout<<"The number of 4 in this array is "<<num;
  return(0);
}
```
