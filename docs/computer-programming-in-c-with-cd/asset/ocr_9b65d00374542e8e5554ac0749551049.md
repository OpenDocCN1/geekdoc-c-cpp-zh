class distance
{
    private:
    float meter;
    public:
    void init()
    {
        cout<<"Enter The Distance";
        cin>>meter;
    }
    void add(distance d1,distance d2)
    {
        meter =d1.meter+d2.meter;
    }
    void subtract(distance d1,distance d2)
    {
        meter=d1.meter-d2.meter;
    }
    void display()
    {
        cout<<"distance ="<<meter;
    }
};

{
