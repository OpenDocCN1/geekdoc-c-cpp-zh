#include <iostream.h>

class ThisPtr {
    int a, b;
public:
    ThisPtr(int n, int m) {
        this->a = n;
        this->b = m;
    }
    int add() {
        return this->a + this->b;
    }
    void show();
};

void ThisPtr::show()
{
    int t;

    t = this->add();
    cout << t << endl;
}

int main()
{
    ThisPtr ob(10, 14);
    ob.show();
    return 0;
}
