{

    foreach(char busySymbol in Utility.BusySymbols())
    {
        if(task.IsCompleted)
        {
            Console.Write('\b');
            break;
        }
        Console.Write(busySymbol);
    }

    Console.WriteLine();

    Console.WriteLine(task.Result);
    if (!task.IsCompleted)
    {
        throw new Exception("Task Should Be Completed");
