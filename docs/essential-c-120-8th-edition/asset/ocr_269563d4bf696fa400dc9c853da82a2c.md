```
public static bool GreaterThan(int first, int second)
{
    return first > second;
}
```

```
public static bool AlphabeticalGreaterThan(
    int first, int second)
{
    int comparison;
    comparison = (first.ToString()).CompareTo(second.ToString());
    return comparison > 0;
}
```
