```
#include <ctype.h>
/* Beginning of main function */
main()
{
	/* Declare variable */
	char c_character;

	/* Input Data */
	printf("Enter a character: ");
	scanf("%c",&c_character);
/* Check for entered character and determine its type */

	/* switch structure begins */
	switch(c_character)
	{
	default:
		if(isdigit(c_character))
			printf("\nCharacter %c is number.\n");
		else if (isalpha(c_character))
			printf("\nCharacter %c is consonant.\n");
		else
			printf("\nCharacter %c is special character.\n");
		break;
	case 'A':
	case 'a':
	case 'E':
	case 'e':
	case 'I':
	case 'i':
	case 'O':
	case 'o':
	case 'U':
	case 'u':
		printf("\nCharacter %c is vowel\n",c_character);
	}	/* switch structure ends */
	return;
}	/* End of main function */
```
