int sumofarray(int i_tempmat[][3])
{
int i_row,i_col, i_sum=0;
/* Addition of elements of an array*/
for (i_row=0;i_row<=2;i_row++)
{
for (i_col=0;i_col<=2;i_col++)
{
i_sum=i_sum + i_tempmat[i_row] [i_col];
}
}
return i_sum;
}
