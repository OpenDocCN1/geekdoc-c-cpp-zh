    /* Trip-i begins */
    for (j = 1; j <= n-i ; j++)
    {
        if (a[j] <= a[j+1])
        { /* Interchanging values */
            t = a[j];
            a[j] = a[j+1];
            a[j+1] = t;
        }
        else
            continue ;
    }
} /* sorting ends */
/* calculation of median */
if ( n % 2 == 0)
    median = (a[n/2] + a[n/2+1])/2.0 ;
else
    median = a[n/2 + 1];
