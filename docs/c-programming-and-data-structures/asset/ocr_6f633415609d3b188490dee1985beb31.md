# EXAMPLE A.20 Write a program to determine whether or not a given number is an Armstrong.

```
/*Program for demonstrating the use of while statement*/
#include <stdio.h>
#include <conio.h>
#include <math.h>

void main()
{
    int num, temp, sum=0, i;
    clrscr();
    printf("\nEnter a number : ");
    scanf("%d", &num);/*Reading num*/
    temp=num;
    while(temp>0)/*Computing Armstrong value*/
    {
        i=temp%10;
        sum=sum+i*i*i;
        temp=temp/10;
    }
    if(sum==num)/*Checking whether num is Armstrong or not*/
    {
        printf("%d is an Armstrong number",num);
    }
    else
    {
        printf("%d is not an Armstrong number",num);
    }
    getch();
}
```
