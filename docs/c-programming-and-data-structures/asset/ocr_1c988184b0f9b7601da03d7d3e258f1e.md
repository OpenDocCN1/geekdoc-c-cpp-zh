# EXAMPLE A.52 Write a program to compute the two's compliment.

# Program

```
/*Program for demonstrating the use of global variables*/
#include <stdio.h>
#include <conio.h>
#include <string.h>

char a[16]; /*Declaring a character array as a global variable*/

void main()
{
    void twos(void); /*Declaring the function prototype for computing two's compliment*/
    clrscr();
    printf("Enter a binary number: ");
    gets(a);/*Reading a binary string*/

    twos(); /*Calling the function*/

    printf("\n2's compliment = %s",a); /*Printing the result stored in a by the twos function*/
    getch();
}

void twos(void)
{
    int i,j,k,len;
    len=strlen(a);
    for(k=0;a[k]!='\0'; k++)
    {
        if (a[k]!='0' && a[k]!='1')
        {
            printf("\nincorrect binary number format...the program will quit");
            getch();
            exit(0);
        }
    }
    /*Computing the two's compliment*/
    for(i=len-1;a[i]=='1'; i--)
    ;
    for(j=i-1;j>=0;j--)
    {
        if(a[j]=='1')
            a[j]='0';
        else
            a[j]='1';
    }
}
```
