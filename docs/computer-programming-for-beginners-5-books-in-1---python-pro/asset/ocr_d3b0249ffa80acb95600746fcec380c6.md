            int x = 15;
            /* check a boolean condition via if statement */
            if (x < 10) {
                /* to be printed of the condition is true */
                Console.WriteLine("x is less than 10");
            }
            Console.WriteLine("The value of x is : {0}", x);
            Console.ReadLine();
        }
