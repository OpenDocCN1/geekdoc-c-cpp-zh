function countBy(items, groupName) {
  let counts = [];
  for (let item of items) {
    let name = groupName(item);
    let known = counts.findIndex(c => c.number == name);
    if (known == -1) {
      counts.push({number, count: 0});
    } else {
      counts[known].count++;
    }
  }
  return counts;
}
