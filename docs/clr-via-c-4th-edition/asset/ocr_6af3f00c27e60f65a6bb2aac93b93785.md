public Int32 FillArray([WriteOnlyArray] /* [Out] implied */ Int32[] data) {
    // NOTE: Original array contents MAY not be marshaled in;
    // write to the array before reading from it
    for (Int32 n = 0; n < data.Length; n++) data[n] = n;
    return data.Length;
}
