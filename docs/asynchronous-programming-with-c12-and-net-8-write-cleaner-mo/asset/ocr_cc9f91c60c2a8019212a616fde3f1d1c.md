            try
            {

                foreach (var post in posts)
                {
                    Console.WriteLine($"PostId: {post.Id}, Title:
{post.Title}");
                }
            }
            catch (Exception ex)

                Console.WriteLine("Error fetching posts: " + ex.Message);
            }
        }
    }
}
