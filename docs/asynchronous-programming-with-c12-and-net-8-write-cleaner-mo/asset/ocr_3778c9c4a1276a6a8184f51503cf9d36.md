```
private readonly AppDbContext _context;
public ProductsController(AppDbContext context)
{
    _context = context;
}
[HttpGet]
public async Task<IActionResult> GetProductsAsync()
{
    try
    {
        var products = await
_context.Products.ToListAsync();
        return Ok(products);
    }
    catch (Exception ex)
    {
        return StatusCode(500, "Internal server error");
    }
}
}
```

In this example, the `GetProductsAsync` action in the `ProductsController` asynchronously fetches products from the database using EF Core's `ToListAsync` method. This ensures that the web server remains responsive while querying the database.
