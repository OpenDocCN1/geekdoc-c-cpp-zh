Example 10 Conversions between Reference Types
This example shows the cases of implicit and explicit reference conversions listed in section 5.3.2. The explicit conversions are performed using type cast expressions (section 12.20).

```
interface I1 { }
interface I2 : I1 { }
interface J { }
class B : I2 { }
class C : B, J { }
delegate void D(String s);

Object obj = new B();      // Implicit B-->I2
I2 b2 = new B();           // Implicit B-->I2
B cl = new C();             // Implicit C-->B
I1 b3 = b2;               // Implicit I2-->I1
I1[] a1 = new I2[];     // Implicit I2[]->I1[]
Array array = new int[5];  // Implicit int[]->Array
delegate di = new D(Print);  // Implicit D-->Delegate\Cn = null;               // Implicit null type-->C
B b4 = (B)b4;              // Explicit Object-->B
 c2 = (C)c1;               // Explicit B-->C
J b5 = (J)c1;               // Explicit C-->J
B b6 = (B)b2;               // Explicit I2-->B
I1 i2 = (I1)i2;             // Explicit I2-->I1
I2[] a2 = (I2[]){a1};       // Explicit I1[]->I2[]
int[] int2 = (int[])intial; // Explicit Array->int[]
d dy = (O)dy;                // Explicit Delegate-->D
```

# Example 11 Boxing and Unboxing Conversions

```
interface I { void Print(); };
struct S : I {
public int i;
public S(int i) { this.i = i; }
public void Print() { Console.WriteLine(i); }
}

int i = 7;
Object obj = i;               // Implicit boxing-->Object
int j = 5 *(int)obj;          // Explicit unboxing-->int
S s1 = new S(s1);
s s2 = 22;                    // Implicit boxing S-->I
s1.i = s2;
s1.Print();
s s3 = (S)s2;                 // Explicit unboxing I-->S
s3.Print();                  // 11
```
