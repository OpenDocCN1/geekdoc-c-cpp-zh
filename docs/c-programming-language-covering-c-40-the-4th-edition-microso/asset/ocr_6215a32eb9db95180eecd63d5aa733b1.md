```csharp
using System;

class Test
{
    static void Main() {

    }

    public static int F(string s) {
        Console.WriteLine(s);
        return 1;
    }
}

class A
{
    public static int X = Test.F("Init A");
}

class B
{
    public static int Y = Test.F("Init B");
}
```
```

```
```text
Init A
Init B
1 1
```
```

or this output:

```
```text

Init A
1 1
```
