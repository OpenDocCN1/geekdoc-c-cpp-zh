function CursorTrail(image, length, state)
{
  var w = GetWindowWidth()
  var h = GetWindowHeight()
  var c = 'CT_'

  if (!state) return clearInterval(CT_IID)

  if (!O('TT_0'))
{
    for (var j = 0; j < 10; ++j)
    {
      var newming = document.createElement('img')
      newming.setAttribute('id', c + j)
      document.body.appendChild(newming)
      Position(newming, ABS)
      Opacity(newming, j + 1) * 9
      newming.src = image
      O(c + j).X = -9999
      O(c + j).Y = -9999
    }
  }

  else Show(c + j)

  if (j > 0)
  {
    O(c + j - 1).X = O(c + j).X
    O(c + j - 1).Y = O(c + j).Y
  }

  O(c + 9).X = MOUSE_X < (w - 12) ? MOUSE_X : -9999
  O(c + 9).Y = MOUSE_Y < (h - 20) ? MOUSE_Y : -9999
}
