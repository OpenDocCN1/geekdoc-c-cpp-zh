# The Plug-ins

```
function X(id) {
  var obj = O(id)
  var offset = obj.offsetLeft

  if (obj.offsetParent) {
    while (obj = obj.offsetParent)
      offset += obj.offsetLeft

  return offset
}
function Y(id) {
  var obj = O(id)
  var offset = obj.offsetTop

  if (obj.offsetParent) {
    while (obj = obj.offsetParent)
      offset += obj.offsetTop

  return offset
}
```
