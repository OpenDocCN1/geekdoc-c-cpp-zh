function FlyIn(id, x, y, msecs)
{
  if (id instanceof Array)
{
    for (var j = 0; j < id.length; ++j)
      FlyIn(id[j], x, y, msecs)
    return
  }

  if (O(id).Fi_Flag) return
  else O(id).Fi_Flag = true

  var tox = X(id)
  var toy = Y(id)
  var fromx = tox + x
  var fromy = toy + y
  var xstep = x / (msecs / INTERVAL)
  var ystep = y / (msecs / INTERVAL)
  var count = 0

function DoFlyIn()
{
  GoTo(id, fromx, xstep = count, fromy, ystep = count)

  if (count++ >= msecs / INTERVAL)
{
    O(id).Fi_Flag = false
    GoTo(id, tox, toy)
    clearInterval(id)
  }
}

# # About the Plug-in

This plug-in performs a wave or ripple effect from the start to end of text contained within arguments. It requires the following arguments:

+   - **id** An object, ID, or an array of objects and/or IDs
- **number** The number of times to repeat the ripple—infinite, if number is 0
- **msecs** The number of milliseconds the ripple should take
