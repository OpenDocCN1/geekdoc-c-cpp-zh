# The Plug-in

```
function EmbedYouTube(video, width, height, hq, full, auto)
{
  if (hq) hq = '&amp;ap=%2526fmt%3D18'
  else hq = ''

  if (width && height) height = width * 0.7500
  if (!width && height) width = height * 1.3333
  if (!video) width = 425
  if (!height) height = 324

  var temp = InsVars("<object width='#1' height='#2'>\n" +
    "<param name='movie' value='http://www.youtube.com/v/\n" +
    "#3&amp;fs=1&autoplay=#4#5', width, height, video,\n    auto, hq)>")

  temp += InsVars("<param name='allowFullScreen' "\n" +
    "value='#1'</param><param name='allowscriptaccess' "\n" +
    "value='allows&amp;quot;</param>, full)")

  temp += InsVars("<embed src='http://www.youtube.com/\n" +
    "/v/#1&amp;fs=1&autoplay=#2#3' type='application/\n" +
    "x-shockwave-flash;allowallowsciptaccess='always' "\n" +
    "allowfullscreen='true', video, auto, hq>")

  temp += InsVars("width='#1' height='#2'></embed></object>")

  return temp
}
```
