Finally, we return an Either:

```
public class Try<T>
{
    public Either<Exception, T> Result { get; set; }
    public Try(Func<T> @try)
    {
        try
        {
            Result = new Right<Exception, T>(@try());
        }
        catch (Exception e)
        {
            Result = new Left<Exception, T>(e);
        }
    }
}
```
