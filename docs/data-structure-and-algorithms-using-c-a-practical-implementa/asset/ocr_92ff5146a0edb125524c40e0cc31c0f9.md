```
DELETE(QUEUE[SIZE], FRONT, REAR)
STEP 1 : IF (FRONT = -1) THEN :
    WRITE : "UNDERFLOW"
    RETURN
[END OF IF]
STEP 2 : WRITE: QUEUE[FRONT]
```

# ALGORITHM FOR TRAVERSE OPERATION

```
TRAVERSE(QUEUE[SIZE], FRONT, REAR)
STEP 1 : IF (FRONT = -1) THEN :
    WRITE : " QUEUE IS EMPTY "
    RETURN
[END OF IF]
STEP 2: SET I:=0
STEP 3 : REPEAT FOR I = FRONT TO REAR
    WRITE : QUEUE[I]
[END OF LOOP]
STEP 4: RETURN
```
