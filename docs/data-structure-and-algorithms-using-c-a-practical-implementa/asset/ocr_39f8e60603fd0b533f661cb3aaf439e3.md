# ALGORITHM

+   Step-1 Repeat through step-3 while I < n
+   Step-2 Repeat through step-3 while k= I+1 to n
+   Step-3 If arr[i] >arr[k]
+   Temp = arr[k]
+   arr[k] = arr[i]
+   arr[i] = temp
+   Step-4 exit

# Program for selection sort

 int n,*arr;

 arr = new int[n];
 for(i = 0 ; i < n ; i++)

 cout<<endl<<"Enter a number";
 cin>>arr[i];
 }
 cout<<endl<<"LIST BEFORE SORTING :\n";
 for(i=0;i<n;++i)
 cout<<setw(5)<<arr[i];
 for(i=0; i<n-1 ; i++)
 for(k = i+1; k<n;k++)
 {
 if(arr[i] > arr[k])
 {
 temp = arr[k];
 arr[k] = arr[i];
 arr[i]=temp ;
 }
 }
 cout<<endl<<"LIST AFTER SORTING :\n";
 for(i=0;i<n;++i)
 cout<<setw(5)<<arr[i];
}
