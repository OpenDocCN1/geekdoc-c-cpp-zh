//heapify() method
void heapify(int n) {
int k,i,j,item;
for (k=1;k<=n;k++) {
item = a[k];
i = k;
j = (i-1)/2;
while ((i>0)&&(item>a[j])) {
a[i] = a[j];
i = j;
j = (i-1)/2;
}
}
}
