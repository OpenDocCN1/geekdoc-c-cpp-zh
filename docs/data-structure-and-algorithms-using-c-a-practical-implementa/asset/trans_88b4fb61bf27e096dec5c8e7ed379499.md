#include <iostream>
using namespace std;
#define SIZE 50
class dequeue {
 int a[50], f, r;
 public:
 dequeue();
 void insert_at_beg(int);
 void insert_at_end(int);
 void delete_fr_front();
 void delete_fr_rear();
 void show();
};
dequeue::dequeue() {
 f = -1;
 r = -1;
}
void dequeue::insert_at_end(int i) {
 if (r >= SIZE - 1) {
 cout << "\n insertion is not possible, overflow!!!!";
 } else {
 if (f == -1) {
 f++;
 r++;
 } else {
 r++;
 }
 a[r] = i;
 cout << "\nInserted item is " << a[r];
 }
}

 a[--f] = i;

 }
}
void dequeue::delete_fr_front() {
 if (f == -1) {
 cout << "deletion is not possible: deque is empty";
 return;
 } else {
 cout << "the deleted element is: " << a[f];
 if (f == r) {
 f = r = -1;
 return;
 } else
 f = f + 1;
 }
}
void dequeue::delete_fr_rear() {
 if (f == -1) {
 cout << "deletion is not possible: deque is empty";
 return;
 } else {
 cout << "the deleted element is: " << a[r];
 if (f == r) {
 f = r = -1;
 } else
 r = r - 1;
 }
}
void dequeue::show() {
 if (f == -1)
 cout << "Deque is empty";
 else {
 for (int i = f; i < r; i++)
 cout << a[i] << " ";
 }
}

## 链表

链表是表示数据结构的一种方式，该结构可以是线性的也可以是非线性的。链表中的元素将在内存中随机分配，并彼此之间存在关系。链表中的元素称为节点。

由于内存使用得当，链表比数组更好。

### 7.1 为什么使用链表？

数组始终要求内存按顺序排列，而链表只需要一次内存分配，这足以存储数据。在数组的情况下，即使内存空间大于所需空间，也可能无法分配内存，因为该空间可能不是按顺序排列的。

### 7.2 链表的类型

链表有四种类型，例如：

- 单链表
- 双链表
- 循环链表
- 头链表

### 链表节点的结构

链表节点具有存储数据及其下一节点地址的能力，数据可能因用户需求而异，因此最好将节点的数据类型选择为结构体，它能够存储不同类型的元素。

节点的一般格式为

```
struct tagname
{
Data type member1;
Data type member2;
............
............
Data type membern;
struct tagname *var;
};
```

示例：

```
struct link
{
int info;
struct link *next;
};
```

这种结构也称为自引用结构。
