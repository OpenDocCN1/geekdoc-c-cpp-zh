# Example

# WAP to find out the GCD of two numbers

```
#include<iostream.h>
class GCD
{
private :
int a,b;
public :
void input();
void find();
}
inline void GCD :: input()
{
cout<<endl<<"Enter two numbers";
cin>>a>>b;
}
inline void GCD :: find()
{
int r;
if(a>b) //Makes a as greater than b
{
r=a;
a=b;
b=r;
}
r = a%b;
while(r!=0)
{
a = b;
b = r;
r = a%b;
}
cout<<"GCD"<<b;
}
void main()
{
GCD obj;
obj.input();
obj.find();
}
```
