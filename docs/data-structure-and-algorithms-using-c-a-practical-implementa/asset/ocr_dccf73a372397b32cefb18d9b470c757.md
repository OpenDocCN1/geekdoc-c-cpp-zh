void adjust(int n) {
    int i,j,item;
    j = 0;
    item = a[j];
    i = 2*j+1;
    while(i<=n-1) {
        if(i+1 < n-1)
            if(a[i] < a[i+1])
                i++;
        if(item<a[i]) {
            a[j] = a[i];
            j = i;
            i = 2*j+1;
            else
                break;
        }
        a[j] = item;
    }
}

int partition(int low, int high)
{
    long int pivot = a[high];     //assign the pivot element
    int i = (low - 1);
    for (int j = low; j <= high-1; j++)
    {
        if (a[j] <= pivot)
        {
            i++;     // increment index of smaller element
            swap(&a[i], &a[j]);
        }
    }
    swap(&a[i + 1], &a[high]);
    return (i + 1);
}
