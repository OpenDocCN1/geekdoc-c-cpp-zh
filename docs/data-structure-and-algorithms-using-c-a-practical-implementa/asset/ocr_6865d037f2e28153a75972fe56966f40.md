```
{

};
```

# ALGORITHM FOR CREATION OF SINGLE LINK LIST

```
struct link
{
int info;
struct link *next;
};
CREATE(START,NODE)
[START IS THE STRUCTURE TYPE OF VARIABLE]
[NODE IS THE STRUCTURE TYPE OF POINTER]
```
