## 弗洛伊德-沃肖尔算法程序

```c
#include<stdio.h>
int i, j, k, n, x, y, dist[10][10];
void FloydWarshall()
{
    for (k = 0; k < n; k++)
    {
        printf("\n For K = %d", k);
        for (i = 0; i < n; i++)
        {
            for (j = 0; j < n; j++)
            {
                if (dist[i][k] + dist[k][j] < dist[i][j])
                {
                    printf("\nFor i=%d, j = %d, dist[%d][%d] + dist[%d][%d] < dist[%d][%d], dist[%d][%d] = %d",
                           i, j, i, k, k, j, i, j, i, j, dist[i][k] + dist[k][j]);
                }
            }
        }
    }
}
```
