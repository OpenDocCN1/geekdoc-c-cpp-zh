private List<Integer> sort(List<Integer> list) {
    if (list.size() <= 1)
        return list;
    else if (list.size() == 2){
        int first = list.get(0);
        int second = list.get(1);
        if (first > second)
            return asList(second, first);
        else
            return asList(first, second);
    }
    else {
        return list;
    }
}
