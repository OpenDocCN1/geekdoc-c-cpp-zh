// T: 12:50 Stack.java
public class Stack {
  private int size = 0;
  private int element;

  public void push(int element) {
    size++;
    this.element = element;
  }

  public int pop() {
    if (size == 0)
      throw new Underflow();
    --size;
    return element;
  }
}
