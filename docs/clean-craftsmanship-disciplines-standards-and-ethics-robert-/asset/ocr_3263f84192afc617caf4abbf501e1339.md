private List<Integer> factorsOf(int n) {
    ArrayList<Integer> factors = new ArrayList<>();
    int divisor = 2;
    while (n > 1) {

private List<Integer> factorsOf(int n) {
    ArrayList<Integer> factors = new ArrayList<>();

        for (; n % divisor == 0; n /= divisor)
