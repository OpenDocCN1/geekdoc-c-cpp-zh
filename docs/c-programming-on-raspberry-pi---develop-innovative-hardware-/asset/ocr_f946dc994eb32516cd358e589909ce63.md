void Display(unsigned int No, unsigned int L)
{
        unsigned int j, i, m, p, r;
        m = L - 1;
        for(i = 0; i < L; i++)
        {
                j = 1;
                for(p = 0; p < m; p++) j = 2*j;          // Power of 2
                r = No & j;
                if(r > 0) r = 1;
                digitalWrite(LEDs[i], r);
                m--;
        }
}
