```c
/*-----------------------------------------------------------
                         LCD秒计数器
                         ==================

本程序将LCD以4位模式连接到树莓派。
程序每秒计数一次并显示在LCD上

作者：Dogan Ibrahim
文件：LCDSeconds.c
日期：2020年12月
------------------------------------------------------------*/
#include <wiringPi.h>
#include <lcd.h>

//
// 树莓派与LCD之间的连接
//
#define D4 2
#define D5 3
#define D6 17
#define D7 27
#define RS 22
#define E 10

//
// 主程序开始
//
int main(void)
{
    int lcd, count = 0;

    wiringPiSetupGpio();
    lcd = lcdInit(2, 16, 4, RS, E, D4, D5, D6, D7, 0, 0, 0, 0);
    lcdClear(lcd);
    lcdPosition(lcd, 0, 0);
    lcdPuts(lcd, "SECONDS:");
    while(1)
    {
        lcdPosition(lcd, 0, 1);
        count++;
        delay(1000);
        lcdPrintf(lcd, "%d", count);
    }
}
```
