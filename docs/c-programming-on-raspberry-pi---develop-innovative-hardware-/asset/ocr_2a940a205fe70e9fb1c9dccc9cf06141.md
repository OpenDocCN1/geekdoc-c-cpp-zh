#include <pigpio.h>
#include <stdio.h>

#define Button 2

int count = 0;
int flag;

//
// This is the ISR which increment count
//
void Trigger()
{
    count++;
    flag = 1;
}

//
// Start of MAIN program
//
int main(void)
{
    gpioInitialise();
    gpioSetMode(Button, PI_INPUT);
    flag = 0;

    gpioSetISRFunc(Button, FALLING_EDGE,0,Trigger);

    while(1)
    {
        if(flag == 1)
        {
            flag = 0;
            printf("Count = %d\n", count);
            while(gpioRead(Button) == 0);
            time_sleep(0.01);
        }
    }
}
