/********************************************
      TABLES OF SQUARES
      ******************

Table of squares from 1 to 10

Author: Dogan Ibrahim
File   : squares.c
Date   : December 2020
********************************************/
#include <stdio.h>

int main(void)
{
    int k;

    printf("\nTABLE OF SQUARES");
    printf("\n=================\n");

    for(k = 1; k <= 10; k++)
    {
        printf("%4d\t%d\n", k, k*k);
    }

    return 0;
}
