// Configure LED ports as outputs,turn OFF LEDs. Also configure the
// Button port as input
//
void Configure()
{
    char k;
    for(k = 0; k < 8; k++)
    {
        gpioSetMode(LEDs[k], PI_OUTPUT);
        gpioWrite(LEDs[k], OFF);
    }
    gpioSetMode(Button, PI_INPUT);
}
