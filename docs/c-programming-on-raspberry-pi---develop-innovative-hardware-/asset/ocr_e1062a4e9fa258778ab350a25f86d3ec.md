// This function positions cursor at col,row. Top left is 0,0
//
void lcd_goto(int col, int row)
{
        char address, c;
        c = col + 1;
        if(row == 0)address = 0;
        if(row == 1)address = 0x40;
        address += c - 1;
        lcd_write(0x80 | address, 0);
}

//
// This function initializes the LCD
//
void lcd_init(void)
{
        char i;
        delay(120);
        for(i = 0; i < 8; i++)digitalWrite(PCF+i, 0);
        delay(50);
        digitalWrite(D4, 1); digitalWrite(D5, 1);
        lcd_strobe();
        delay(10);
        lcd_strobe();
        delay(10);
        lcd_strobe();
        delay(10);
        digitalWrite(D4, 0);
        lcd_strobe();
        delay(5);
        lcd_write(0x28, 0);                         //28
        delay(1);
        lcd_write(0x08,0);                          //06
        delay(1);
        lcd_write(0x01,0);
        delay(10);
        lcd_write(0x06,0);
        delay(5);
        lcd_write(0x0C,0);
        delay(1);
        digitalWrite(LED, 1);
}
