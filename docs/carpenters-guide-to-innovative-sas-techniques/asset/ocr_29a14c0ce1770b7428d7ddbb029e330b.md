%macro BldRpt;
ods pdf file="&path\\results\\ell_4_3.pdf"; ⑤
        style=journal;

proc sql noprint; ⑥
   select distinct edu
      into :edu1 - :edu99
      from advprdt.demog (keep=edu);
   %let educnt=&sqlobs;
   quit;

%do i = 1 %to &educnt; ⑦
   ods pdf anchor="&edu&i"; ⑧
   ods proclabel "Symptom Summary"; ⑨
   title3 "&edu&i Years of Education";
   proc report data=advprdt.demog
               (where=(edu=&edu&i ⑦))
               contents="&edu&i Years" ⑩
               nowd;
      columns symp_sex,wt;
      define symp / group;
      define sex   / across 'Gender';
      define wt    / analysis mean;
      run;
%end;
ods pdf close;
%mend bldrpt;
%bldrpt
