```
/* example of & and * operators */
ptr = &num1; /* 'ptr' now contains the
address of 'a'*/
printf("Value of num1 is %d\n", num1);
printf("*ptr is %d.\n", *ptr);

/* example of ternary operator */
num1 = 10;
num2 = (num1 == 1)? 20: 30;
printf( "Value of num2 is %d\n", num2 );
num2 = (num1 == 10)? 20: 30;
printf( "Value of num2 is %d\n", num2 );
return 0;
}
```

+   - **Output:**
+     - Value of num1 is 4.
+     - *ptr is 4.
+     - Value of num2 is 30.
+     - Value of num2 is 20.
