void blockIP(const std::string& ip) {

    std::string cmd = "iptables -A INPUT -s " + ip + " -j DROP";

    system(cmd.c_str());

}

int main() {

    blockIP("192.168.1.100"); // replace with the IP you want to block

    return 0;

}
