# List 6-9 Approximate LINQ implementations of the operators in the HashSet

```
public static IEumerable<T> SymmetricExcept<T>(
    this IEnumerable<T> first,
    IEnumerable<T> second = null)
{
    if (first == null)
    {
        throw new ArgumentNullException("first");
    }

    if (second == null)
    {
        throw new ArgumentNullException("second");
    }

    return first.Except(second)
        .Concat(
            second.Except(first));
        );
}

public static bool Overlaps<T>(
    IEnumerable<T> first,
    IEnumerable<T> second)
{
    if (first == null)
    {
        throw new ArgumentNullException("first");
    }

    if (second == null)
    {
        throw new ArgumentNullException("second");
    }

    return second.Intersect(first).Distinct().Any();
}
```
