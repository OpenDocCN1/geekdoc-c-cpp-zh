## Listing 7-17 The Segment operator, which segments the contents of the source sequence into equal-sized groups

```csharp
public static IEnumerable<Grouping<int, T>> Segment<T>(
    this IEnumerable<T> source,
    int segments)
{
    if (source == null)
        throw new ArgumentNullException("source");

    if (segments <= 0)
        throw new ArgumentOutOfRangeException("segments");

    return new SegmentIterator<T>(source, segments);
}
```

```csharp
private static IEnumerable<IGroup<int, T>>
SegmentIterator<T>(IEnumerable<T> source,
    int segments)
{
    // calculate the number of elements per segment
    int count = source.Count();
    int perSegment = (int)Math.Ceiling(
        (decimal)count / segments);

    // build the empty groups
    Grouping<int, T[]> groups =
        new Grouping<int, T>([segments]);

    for (int i = 0; i < segments; i++)
    {
        Grouping<int, T> g =
            new Grouping<int, T>(perSegment);

        g.key = i + 1;
        groups[i] = g;
    }

    // fill the groups and yield results
    // when each group is full.
    int index = 0;
    int segment = 1;
    Grouping<int, T> group = groups[0];
    using (Enumerator<T> e = source.GetEnumerator())
    {
        while (e.MoveNext())
        {
            group.Add(e.Current);
            index++;

            // yield return when we have filled each group
            if ((segment < segments) &&
                (index == perSegment))
            {
                yield return group;
                index = 0;
                segment++;
                group = groups[segment - 1];
            }
        }
    }
```
