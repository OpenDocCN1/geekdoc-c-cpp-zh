void DList :: Traverse()
{
    DLL_Node *Curr;
    Curr = Head;
    if(Curr == Null)
        cout << "The list is empty \n";
    else
    {
        while(Curr != Null)
        {
            cout << Curr->Data << "\t";
            Curr = Curr->Next;
        }
        cout << endl;
    }
}
