void btree :: remove_key (btnode *curr, int pos)
{
    int p;
    for(p = pos + 1; p <= curr->count; p++)
    {
        curr->data[p - 1] = curr->data[p];
        curr->child[p - 1] = curr->child[p];
    }
    curr->count--;
}

void btree :: successor (btnode *curr, int pos)
{
    btnode *leaf;
    leaf = curr->child[pos];
    while(leaf->child[0])
    {
        leaf = leaf->child[0];
        curr->data[pos] = leaf->data[0];
    }
}
