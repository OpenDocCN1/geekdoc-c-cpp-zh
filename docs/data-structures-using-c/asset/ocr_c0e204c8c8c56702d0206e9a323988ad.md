// rotation: Left
AVLNode *AVLNode :: Left (AVLNode *A)
//function is called with unbalanced node as a parameter
{
    AVLNode *B;
    B = A->right;
    A->right = B->left;
    B->left = A;
    A->height = height (A);
    B->height = height (B);
    return B; // Set new root to B
}

// Rotation : Right
AVLNode *AVLNode ::Right (AVLNode *A)
{
    AVLNode *B;
    B = A->left;
    A->left = B->right;
    B->right = A;
    A->height = height (A);
    B->height = height (B);
