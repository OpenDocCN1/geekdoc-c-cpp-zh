Void Heap_Sort (int a[], int n)
{
    // create heap
    int i, temp;
    for(i = (n - 1)/2, i >= 0; i--)
    reheapdown(a, n, i);
    // delete first value and swap it with last
    while(n > 0)
    {
        //swap first and last element
        temp = a[0];
        a[0] = a[n - 1];
        a[n - 1] = temp;
        n--; // decrement count
        reheapdown(a, n, 0);
    }
}
