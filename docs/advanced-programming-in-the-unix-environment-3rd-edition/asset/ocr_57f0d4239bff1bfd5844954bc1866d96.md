   int          err;
   struct termios  buf;

   if (ttystate != RESET) {
     errno = EINVAL;
     return(-1);

    * Echo off, canonical mode off, extended input
    * processing off, signal chars off.

    * No SIGINT on BREAK, CR-to-NL off, input parity
    * check off, don't strip 8th bit on input, output
    * flow control off.
