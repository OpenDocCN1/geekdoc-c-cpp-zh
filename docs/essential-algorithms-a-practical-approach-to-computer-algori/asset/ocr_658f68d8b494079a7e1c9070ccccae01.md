```
// Draw the gasket
ierpinskiGasket(Integer: depth,
Point:point1, Point:point2, Point:point3)
// If this is depth 0, fill the remaining triangle.
If (depth == 0) Then
 Point:points[] = {point1, point2, point3 }
 FillPolygon(points)
Else
 // Find points on the left, right, and bottom of the triangle.
 lpoint = new Point(
(point1.X + point2.X) / 2,
(point1.Y + point2.Y) / 2)
 bpoint = new Point(
(point2.X + point3.X) / 2,
(point2.Y + point3.Y) / 2)
 rpoint = new Point(
(point3.X + point1.X) / 2,
(point3.Y + point1.Y) / 2)
 // Draw the triangles at the corners.
ierpinskiGasket(depth - 1, gr, lpoint, rpoint),
ierpinskiGasket(depth - 1, gr, lpoint, point2, bpoint)
sierpinskiGasket(depth - 1, gr, rpoint, bpoint)
End If
ierpinskiGasket
```
