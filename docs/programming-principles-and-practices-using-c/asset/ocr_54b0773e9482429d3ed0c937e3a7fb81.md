double term()
{
	double left = primary();
	Token t = ts.get(); 		// get the next Token from the Token stream

	while (true) {
		switch (t.kind) {
		case '*':
			left *= primary();
			t = ts.get();
			break;
		case '/':
		{
			double d = primary();
			if (d == 0) error("divide by zero");
			left /= d;
			t = ts.get();
			break;
		}
		default:
			ts.putback(t); 		// put t back into the Token stream
			return left;
		}
	}
}
