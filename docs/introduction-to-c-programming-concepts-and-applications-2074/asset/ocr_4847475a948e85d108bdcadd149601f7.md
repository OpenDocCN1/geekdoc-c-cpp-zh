4 What would be the output of the following code?

```
#include <iostream>
using namespace std;

class pointcounter {
  private:
  int points=0;
  int bonus=500;

  public:
  void increasepoints(int p) {
    points += p;
  }
  void givebonus() {
    points += bonus;
  }
  int number() {
    return points;
  }
};

int main()
{
  pointcounter gamescore;
  cout << "Score is now: " << gamescore.number() << endl;
  gamescore.increasepoints(10);
  cout << "Score is now: " << gamescore.number() << endl;
  gamescore.givebonus();
  cout << "Score is now: " << gamescore.number() << endl;
}
```
