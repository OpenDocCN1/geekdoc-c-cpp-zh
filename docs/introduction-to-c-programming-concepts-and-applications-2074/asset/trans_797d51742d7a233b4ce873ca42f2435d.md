class bad_value_exception {

    bool entertainment_package;

        if (entertainment_package) {
            cout << " with the entertainment package";
            total_price += 1200.0;
        }
        if (safety_enhancements) {
            cout << " with safety enhancements";
            total_price += 2100.0;
        }
        cout << " costs " << total_price << endl;
    }
};

int main() {
    car x;
    x.print_info();
