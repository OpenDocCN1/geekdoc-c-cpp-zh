3 What would be the output of the following program?

```
#include<iostream>
using namespace std;

void A(int x) {
    x++;
    cout << x << endl;
}

void B(int& x) {
    x++;
    cout << x << endl;
}

int main() {
    int x = 2;
    A(x);
    cout << x << endl;
    B(x);
    cout << x << endl;
}
```
