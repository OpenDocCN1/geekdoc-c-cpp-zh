This will use arrays. You'll use one array for storing the product IDs and one for storing the rating. You will compute the average rating as you read the data. Then, you will go through the list and print the ID for the products with a rating that is greater than the average:

+   c cout << x[0] << endl;
d int x[] = {10, 20, 30, 40, 50};

Note that you could have written x[5] instead of x[i] and had the same result.

+   e float x[4][6];

```
#include <iostream>
using namespace std;
int main() {
    int ID[100];
    float rating[100];
    int totalnumber = 0;
    float sumofratings = 0.0;
    int i;
    cout << "Enter an ID and a rating, entering an ID of 0 to stop" << endl;
    int id;
    float r;
    while(id > 0) { // Repeat until ID is in read
        ID[id] = r; // Store the product ID
        totalnumber] = r; // Store the product rating
        totalnumber++; // Increment count by 1
        sumofratings += r; // Sum up ratings in next product
    }
    float averaging = sumofratings/totalnumber;
    for(i=0;i<totalnumber;i++) {
        if (rating[i] > averaging) {
            cout << ID[i] << endl;
        }
    }
}
```

Although there are several ways this could be done, the most straightforward is to read the numbers into an array of 5 integers and loop through the array in reverse order:

```
#include <iostream>
using namespace std;
int main() {
    int nums[5];
    int i;
    cout << "Enter 5 integers:" << endl;
    for(i=0;i<5;i++) {
        cin >> nums[i];
    }
    cout << "The numbers in reverse order:"
    for(i=4;i>=0;i--) {
        cout << nums[i] << endl;
    }
}
```
