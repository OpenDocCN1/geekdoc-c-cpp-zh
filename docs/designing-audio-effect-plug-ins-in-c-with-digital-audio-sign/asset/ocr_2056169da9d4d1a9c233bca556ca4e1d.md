    if(m_uTimeConstant == digital)

        m_RightDetector.init((float)m_nSampleRate, m_fAttack_mSec,

    }
    else
    {
        m_LeftDetector.init((float)m_nSampleRate, m_fAttack_mSec, m_fRelease_mSec,

        m_RightDetector.init((float)m_nSampleRate, m_fAttack_mSec,
                             m_fRelease_mSec, true, DETECT_MODE_RMS, false);

}
