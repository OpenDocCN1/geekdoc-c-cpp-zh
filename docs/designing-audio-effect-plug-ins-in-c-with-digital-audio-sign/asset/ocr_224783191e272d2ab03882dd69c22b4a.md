bool __stdcall CDDLModule::prepareForPlay()
{
    // setup our delay 1ine
    m_nBufferSize = 2*m_nSampleRate; // 2 seconds delay @ fs

    // delete it if it exists
    if(m_pBuffer)
        delete [] m_pBuffer;

    // create the new buffer
    m_pBuffer = new float[m_nBufferSize];

    return true;
}
