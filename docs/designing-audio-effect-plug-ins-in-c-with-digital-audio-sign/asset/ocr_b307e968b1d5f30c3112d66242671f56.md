	switch(m_uOscType)

	case sine:

	case saw:

					m_SawtoothArray[nReadIndexNext],
					fFrac);

				m_SawtoothArray_BL5[nReadIndex],
				m_SawtoothArray_BL5[nReadIndexNext],
