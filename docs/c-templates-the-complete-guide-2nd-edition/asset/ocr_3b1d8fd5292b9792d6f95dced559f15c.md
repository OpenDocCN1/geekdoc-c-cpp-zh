```
auto a = nullptr;
auto b = nullptr;
is_same_v<decltype(a),decltype(b)>      // yields true

using A = int;
is_same_v<A,int>                        // yields true

auto x = [] (int) {};
auto y = x;
auto z = [] (int) {};
is_same_v<decltype(x),decltype(y)>      // yields true
is_same_v<decltype(x),decltype(z)>      // yields false
```
