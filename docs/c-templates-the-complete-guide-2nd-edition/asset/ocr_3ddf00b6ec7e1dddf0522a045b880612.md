int f1(int);      // #1
int f1(double);   // #2
f1(4);            // calls #1: perfect match (#2 requires a standard conversion)

int f2(int);      // #3
int f2(char);     // #4
f2(true);         // calls #3: match with promotion

class X {
    public:
        X(int);
};
int f3(X);        // #5
int f3(...);      // #6
f3(7);            // calls #5: match with user-defined conversion
//                 (#6 requires a match with ellipsis)
