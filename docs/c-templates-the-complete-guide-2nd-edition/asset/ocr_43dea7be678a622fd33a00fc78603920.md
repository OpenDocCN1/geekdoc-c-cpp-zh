```
#include <iterator>

template<typename Iter>
auto accum (Iter start, Iter end)
{
using VT = typename std::iterator_traits<Iter>::value_type;

VT total{}; // assume this actually creates a zero value
while (start != end) {
total += *start;
++start;
}
return total;
}

#endif //ACCUM_HPP
```

Because a partial specialization for pointers exists, these traits are conveniently used with any ordinary pointer types.
