```
template<typename> struct PMClassT;
template<typename C, typename M> struct PMClassT<M C::*> {
using Type = C;
};
template<typename PM> using PMClass = typename PMClassT<PM>::Type;

template<auto PMD> struct CounterHandle {
PMClass<decltype(PMD)>& c;
CounterHandle(PMClass<decltype(PMD)>& c): c(c) {
}
void incr() {
++(c.*PMD);
}
};

struct S {
int i;
};

int main() {
S s{41};
CounterHandle<&S::i> h(s);
h.incr();// increases s.i
}
```

Here we used a helper class template PMClassT to retrieve from a pointer-to-member type its “parent” class type, using class template partial specialization¹⁰ (described in Section 16.4 on page 347). With an auto template parameter, we only have to specify the pointer-to-member constant &S::i as a template argument.
