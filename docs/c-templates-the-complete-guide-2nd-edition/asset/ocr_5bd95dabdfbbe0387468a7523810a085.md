template<typename T>
class B {
    public:
        enum E { e1 = 6, e2 = 28, e3 = 496 };
        virtual void zero(E e = e1);
        virtual void one(E&);
};

template<typename T>
class D : public B<T> {
    public:
        void f() {
            typename D<T>::E e;    // this->E would not be valid syntax
            this->zero();          // D<T>::zero() would inhibit virtuality
            one(e);                // one is dependent because its argument
                                   // is dependent
        }
};
