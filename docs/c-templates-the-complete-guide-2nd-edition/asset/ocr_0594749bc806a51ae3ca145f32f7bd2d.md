{
    T tmp(*a);
    *a = *b;
    *b = tmp;
}

template<typename T>
void quickExchange(Array<T>* a, Array<T>* b)  // #2

    a->exchangeWith(b);
}

void demo(Array<int>* p1, Array<int>* p2)

    quickExchange(&x, &y);                 // uses #1
