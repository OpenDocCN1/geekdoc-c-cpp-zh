long fibi(int n) { // 迭代斐波纳契生成器
  // fibi(46) 是 long 类型能容纳的最大值
  Assert(n > 0) && (n < 47); // 输入范围超出范围
  long past, prev, curr; // 存储临时值
  past = prev = curr = 1; // 初始化
