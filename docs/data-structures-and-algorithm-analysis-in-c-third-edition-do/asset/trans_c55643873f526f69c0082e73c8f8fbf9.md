```cpp
// 使用二分查找查找
E find(const Key& k) const {
    int l = 1;
    int r = list->length();
    while (l+1 <= r) { // 当 l 和 r 相同时停止
        int i = (l+r)/2; // 检查剩余子数组的中间
        list->moveToPos(i);
        KVpair<Key, E> temp = list->getValue();
        if (k < temp.key()) r = i; // 在左侧
        if (k == temp.key()) return temp.value(); // 找到
        if (k > temp.key()) l = i; // 在右侧
    }
    return NULL; // "k" 不在字典中
}
```
