# 11.2 Graph Implementations

```
// Implementation for the adjacency matrix representation
class Graph : public Graph {
private:
    int numVert, numEdge; // Store number of vertices, edges
    int **matrix; // Pointer to adjacency matrix
    int *mark; // Pointer to mark array
public:
    Graph(int numVert) // Constructor
    Init(numVert); //

    Graph() { // Destructor
    delete [mark]; // Return dynamically allocated memory
    for (int i=0; i<numVert; i++)
    delete [matrix[i]];
    delete [matrix];
}

    void Init(int n) { // Initialize the graph
    int i;
    numVert = n;
    numEdge = 0;
    mark = new int[n]; // Initialize mark array
    for (i=0; i<numVert; i++)
    mark[i] = UNVISITED;
    matrix = (int **)new int[numVert]; // Make matrix
    for (i=0; i<numVert; i++)
    matrix[i] = new int[numVert];
    for (int i=0; i<numVert; i++) // Initialize to 0 weights
    for (int j=0; j<numVert; j++)
    matrix[i][j] = 0;
}

int n() { return numVert; } // Number of vertices
int e() { return numEdge; } // Number of edges

// Return first neighbor of "v"
int first(int v) {
    for (int i=0; i<numVert; i++)
    if (matrix[v][i] != 0) return i;
    return numVert; // Return n if none
}

// Return v's next neighbor after w
int next(int v, int w) {
    for (int i=w+1; i<numVert; i++)
    if (matrix[v][i] != 0)
    return i;
    return numVert; // Return n if none
}
```
