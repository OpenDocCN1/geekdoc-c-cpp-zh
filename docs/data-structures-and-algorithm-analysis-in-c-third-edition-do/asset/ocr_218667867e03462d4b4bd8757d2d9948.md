    ~UADict() { delete list; } // Destructor
    void clear() { list->clear(); } // Reinitialize

```
// Use sequential search to find the element to remove
void remove(const Key& k) {
    E temp = find(k); // "find" will set list position
    if(temp != NULL) list->remove();
    return temp;
}
```

```
void removeAny() { // Remove the last element
    Assert(size() != 0, "Dictionary is empty");
    list->moveToEnd();
    list->prev();
    KeyValuePair<Key, E> e = list->remove();
    return e.value();
}
```

```
// "k" using sequential search
E find(const Key& k) const {
    for(list->moveToStart(); list->currPos() < list->length(); list->next()) {
        KeyValuePair<Key, E> temp = list->getValue();
        if (k == temp.key()) {
            return temp.value();
        }
    }
    return NULL; // "k" does not appear in dictionary
}
```
