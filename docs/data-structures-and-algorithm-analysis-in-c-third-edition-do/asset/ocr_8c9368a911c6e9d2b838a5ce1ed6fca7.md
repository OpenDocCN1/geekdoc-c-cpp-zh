// Dijkstra's shortest paths algorithm with priority queue
void Dijkstra(Graph* G, int* D, int s) {
    int i, v;
    // v is current vertex
    DijkElem temp;
    DijkElem E[G->get(v)]; // Heap array with lots of space
    temp.distance = 0; temp.vertex = s;
    E[0] = temp; // Initialize heap array
    DijkElem[DComp] = 1, G->get(E()); // Create heap array
    for (i=0; i<G->get(n); i++) {\ // Now, get distances
        do {
            if (H->size() == 0) return; // Nothing to remove
            temp = H->removesFirst();
            while (temp.v == VISITED) {
                v = temp.vertex;
                G->setMark(v, VISITED);
                if (D[v] == INFINITY) // Unreachable vertices
                for (w=G->first(v); w; w=G->next(v,w)) { // Update D
                    if (D[w] > D[v] + G->weight(v, w)) { // Update D
                        D[w] = D[v] + G->weight(v, w);
                        temp.vertex = w;
                        H->insert(temp); // Insert new distance in heap
                    }
                }
            }
        } while (0);
    }
}

If a MST had a cycle, a cheaper MST could be found by removing one of the edges in the cycle. Then, the MST is a free tree with |V|-1 edges. The name "minimum-cost spanning tree" comes from the fact that the required set of edges forms a tree, it spans the vertices (i.e., it connects them together), and has this minimum cost.
