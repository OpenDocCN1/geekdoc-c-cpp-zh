// Array-based queue implementation
template <typename E> class AQueue: public Queue<E> {
private:
    int maxSize;               // Maximum size of queue
    int front;                // Index of front element
    int rear;                  // Index of rear element
    E *listArray;              // Array holding queue elements

public:
    AQueue(int size =defaultSize) { // Constructor
        // Make list array one position larger for empty slot
        maxSize = size+1;
        rear = 0; front = 1;
        listArray = new E[maxSize];
    }

    ~AQueue() { delete [] listArray; } // Destructor

    void clear() { rear = 0; front = 1; } // Reinitialize

    void enqueue(const E it) { // Put "it" in queue
        Assert((rear+2) % maxSize) != front, "Queue is full");
        rear = (rear+1) % maxSize; // Circular increment
        listArray[rear] = it;
    }

    E dequeue() { // Take element out
        Assert(length() != 0, "Queue is empty");
        E it = listArray[front];
        front = (front+1) % maxSize; // Circular increment
        return it;
    }

    const E& front() const { // Get front value
        Assert(length() != 0, "Queue is empty");
        return listArray[front];
    }

    virtual int length() const { // Return length
        return ((rear+maxSize) - front + 1) % maxSize;
    };

### 4.3.2 Linked Queues

The linked queue is a straightforward adaptation of the linked list. Figure 4.27 shows the linked queue class declaration. Methods **front** and **rear** pointers to the front and rear queue elements, respectively. We will use a header node, which allows for a simpler implementation of the enqueue operation by avoiding any special cases when the queue is empty. In initialization, the **front** and **rear** pointers will point to the header node, and front will always point to the header node while rear points to the true last node in the queue. Method **enqueue** places the new element in a link node at the end of the linked list (i.e., the node that **rear** advances **rear** to point to the new link node. Method **dequeue** returns the first element of the list.

## 4.4 Dictionaries

In this section we describe a simple interface for a collection, called a dictionary. // Linked queue implementation
template <typename E> class LQueue: public Queue<E> {
    private:
    Link<E>* front;               // Pointer to front queue node
    Link<E>* rear;                // Pointer to rear queue node
    int size;                      // Number of elements in queue

public:
    LQueue(int sz =defaultSize) // Constructor
    { front = new Link<E>(); size = 0; }

    ~LQueue() { clear(); }        // Destructor

    void clear() {
        // Clear queue
        while(front->next != NULL) { // Delete each link node
            rear = front;
            delete rear;
        }
        rear = front;
        size = 0;
    }

    void enqueue(const E& it) { // Put element on rear
        rear->next = new Link<E>(it, NULL);
        rear = rear->next;
        size++;
    }

    E dequeue() { // Remove element from front
        Assert(size != 0, "Queue is empty");
        E it = front->next->element; // Store dequeued value
        Link<E>* ltmp = front->next; // Hold dequeued link
        front->next = ltmp->next;      // Advance front
        if (rear == ltmp) rear = front; // Delete last element
        delete ltmp;
        size--;
        return it;
    }

    const E& front() const { // Get front element
        Assert(size != 0, "Queue is empty");
        return front->next->element;
    }

    virtual int length() const { return size; }
};
