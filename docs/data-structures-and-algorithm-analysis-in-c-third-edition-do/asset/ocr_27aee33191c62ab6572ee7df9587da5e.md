Chap. 5 Binary Trees

```
// Huffman tree node abstract base class
template <typename E> class HuffmanNode {
public:
    virtual HuffmanNode() // Base destructor
    virtual int weight() = 0; // Return frequency
    virtual bool isLeaf() = 0; // Determine type
};

template <typename E> // Leaf node subclass
class LeafNode : public HuffmanNode<E> {
private:
    E it; // Value
    int wgt; // Weight
public:
    LeafNode(const E& val, int freq) // Constructor
        { it = val; wgt = freq; }
    int weight() { return wgt; }
    bool isLeaf() { return true; }
};

template <typename E> // Internal node subclass
class IntNode : public HuffmanNode<E> {
private:
    HuffmanNode<E>* lc; // Left child
    HuffmanNode<E>* rc; // Right child
    int wgt; // Subtree weight
public:
    IntNode(HuffmanNode<E>* l, HuffmanNode<E>* r) // Subtree weight
        { wgt = l->weight() + r->weight() + 1; lc = l; rc = r; }
    int weight() { return wgt; }
    bool isLeaf() { return false; }
    void setLeft(HuffmanNode<E>* b) 
        { lc = b; }
    HuffmanNode<E>* getLeft() const { return lc; }
    void setRight(HuffmanNode<E>* b) 
        { rc = b; }
    HuffmanNode<E>* getRight() const { return rc; }
};
```

**Figure 5.27** Implementation for Huffman tree nodes. Internal nodes and leaf nodes are represented by separate classes, each derived from an abstract base class.
