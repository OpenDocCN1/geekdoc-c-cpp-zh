```cpp
        for (int i=0; i<numVert; i++)

        delete [matrix];
    }

    void Init(int n) { // Initialize the graph
        int i;
        numVert = n;
        numEdge = 0;
        mark = new int[n]; // Initialize mark array
        for (i=0; i<numVert; i++)
            mark[i] = UNVISITED;
        matrix = (int **)new int[numVert]; // Make matrix
        for (i=0; i<numVert; i++)
            matrix[i] = new int[numVert];
        for (int i=0; i<numVert; i++) // Initialize to 0 weights
        for (int j=0; j<numVert; j++)
            matrix[i][j] = 0;
    }

    int n() { return numVert; } // Number of vertices
    int e() { return numEdge; } // Number of edges

    // Return first neighbor of "v"
    int first(int v) {
        for (int i=0; i<numVert; i++)
            if (matrix[v][i] != 0) return i;
        return numVert; // Return n if none
    }

    // Return v's next neighbor after w
    int next(int v, int w) {
        for (int i=w+1; i<numVert; i++)
            if (matrix[v][i] != 0)
            return i;
        return numVert; // Return n if none
    }
```

## 11.3 Graph Traversals

Often it is useful to visit the vertices of a graph in some specific order based on the graph's topology. This is known as a graph traversal and is similar in concept to a tree traversal. Recall that tree traversals visit every node exactly once, in some order such as preorder, inorder, or postorder. Multiple traversals exist because various applications require the nodes to be visited in a particular order. For example, visiting a BST's nodes in ascending order requires an inorder traversal as opposed to some other. Standard graph traversals also exist. Each is appropriate for solving certain problems. For example, many problems in artificial intelligence are modeled using graphs. The problem domain may consist of a collection of states, with connections between pairs of states. The problem may require getting from a specified start to a specified goal by moving through the connections. To solve this problem, the vertices of the graph must be searched in some organized manner. traversal algorithms typically begin with a start vertex and attempt to visit all remaining vertices from there. Graph traversals deal with troublesome cases: First, it may not be possible. Second, the graph may contain cycles, and we must make sure that cycles do not cause the algorithm to go into an infinite loop.

Graph traversal algorithms can solve both of these problems by maintaining a mark bit for each vertex on the graph. At the beginning of the algorithm, the mark bit for all vertices is cleared. The mark bit for a vertex is first visited during the traversal. If a marked vertex is encountered during traversal, it is not visited a second time. This keeps the program from going into an infinite loop when it encounters a cycle.

Once the traversal algorithm completes, we can check to see if all vertices have been processed by checking the mark array. If not all vertices are marked, we can continue the traversal from another unmarked vertex. Note that this process works regardless of whether the graph is directed or undirected. To ensure visiting vertices, graphTraverse could be called as follows on a graph G:

```cpp
void graphTraverse(Graph* G) {
  int v;
  for (v=0; v<G->n(); v++) {
    G->setMark(v, UNVISITED);
  }
  for (v=0; v<G->n(); v++) {
    if (G->getMark(v) == UNVISITED)
      depthTraverse(G, v);
  }
}
```

### 11.3.1 Depth-First Search

```cpp
class Graph : public Graph {
private:
    List<Edge>** vertex; // List headers
    int numVertex, numEdge; // Number of vertices, edges
    int *mark; // Pointer to mark array
public:
    Graph(int numVertex);
    Init(numVertex);

    ~Graph() {
        delete [mark]; // Return dynamically allocated memory
        for (int i=0; i<numVertex; i++)
            delete [vertex][i];
        delete [vertex];
    }

    void Init(int n) {
        numVertex = n;
        numEdge = 0;
        mark = new int[n]; // Initialize mark array
        for (i=0; i<numVertex; i++)
            mark[i] = UNVISITED;
        // Create and initialize adjacency lists
        vertex = (List<Edge>**)[numVertex];
        for (i=0; i<numVertex; i++)
            vertex[i] = new List<Edge>();
    }

    int n() { return numVertex; } // Number of vertices
    int e() { return numEdge; } // Number of edges

    int first(int v) { // Return first neighbor of "v"
        if (vertex[v]->length() == 0) // No neighbor
            return numVertex;
        vertex[v]->moveToStart();
        Edge e = vertex[v]->getValue();
        return e.vertex();
    }

    int next(int v, int w) { // Get v's next neighbor after w
        int it = 0;
        if (isEdge(v, w)) {
            if (vertex[v]->currPos() + 1 < vertex[v]->length()) {
                vertex[v]->next();
            }
            it = vertex[v]->getValue();
            return it.vertex();
        }
        return numVertex(); // No neighbor
    }

    // Set edge(i, j) to "weight"
    void setEdge(int i, int j, int weight) {
        Assert(weight>0, "May not set weight to 0");
        Edge currEdge(j, weight);
        if (isEdge(i, j)) { // Edge exists in graph
            vertex[i]->remove();
            vertex[i]->insert(currEdge);
        } else { // Keep neighbors sorted by vertex index
            numEdge++;
            for (vertex[i]->currPos() < vertex[i]->length()) {
                Edge temp = vertex[i]->getValue();
                if (temp.vertex() > j) break;
            }
            vertex[i]->insert(currEdge);
        }
    }

    void deleteEdge(int i, int j) { // Delete edge (i, j)
        if (isEdge(i, j)) {
            vertex[i]->remove();
            numEdge--;
        }
    }

    bool isEdge(int i, int j) { // Is (i, j) an edge?
        Edge it;
        for (vertex[i]->currPos() < vertex[i]->length()) {
            vertex[i]->next(); // Check whole list
            Edge temp = vertex[i]->getValue();
            if (temp.vertex() == j) return true;
        }
        return false;
    }

    int weight(int i, int j) { // Return weight of (i, j)
        Edge curr;
        if (isEdge(i, j)) {
            curr = vertex[i]->getValue();
            return curr.weight();
        } else return 0;
    }

    int getMark(int v) { return mark[v]; }
    void setMark(int v, int val) { mark[v] = val; }
};
```
