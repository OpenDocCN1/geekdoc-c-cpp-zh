// Sorted array-based list
// Inherit from AList as a protected base class
template <typename Key, typename E>
class SList : protected AList<KeyValuePair<Key, E> > {
public:
    SList(int size=defaultSize) :
        AList<KeyValuePair<Key, E> >(size) {}

    ~SList() // Destructor

    // Redefine insert function to keep values sorted
void insert(KeyValuePair<Key, E>& it) { // Insert at right
    KeyValuePair<Key, E> curr=it;
    for (moveToStart(); currPos() < length(); next(curr)) {
        curr = getValue();
        if(curr > it.key())
            break;
    }
    AList<KeyValuePair<Key, E> >::insert(it); // Do AList insert
};

Class SList (SAL stands for "sorted array-based list") inherits from class AList; however it does so using AList as a protected base class. This means that SAL has available for its use any member functions of AList, but those member functions are not necessarily available to the user of SList.
