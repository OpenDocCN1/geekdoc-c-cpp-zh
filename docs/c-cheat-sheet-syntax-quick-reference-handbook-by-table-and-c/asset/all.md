#define syntax:

#define identifier value

// "#define" preprocessor is used to define a constant.

// Constant name usually is uppercase.

e.g.

#include <iostream>

using namespace std;

#define LENGTH 100 // define a constant

#define WIDTH  50     // define a constant

int main(){

int rectangle;  

rectangle = LENGTH * WIDTH;

cout << rectangle;

return 0;

}

// Output:   50000

-------------------------------------------------------------------------------

#define syntax:

#define macrosName(parameters) expression

// "#define  macrosName(parameters)  expression" is a macros definition, which is used to substitute "expression" with "macrosName(parameters)".

e.g.

#include <iostream>

using namespace std;

#define   Multiply(x,y)   x*y    // macro definition

int main( ){

double area = Multiply(10, 20);  // Namely area=x*y;

cout << area << endl;

}

// Output:   200

-------------------------------------------------------------------------------

#variable syntax:

#define macrosName( variable ) #variable

// #variable can convert the macrosName(variable) to a string.

e.g.

#include <iostream>

using namespace std;

#define macros( var ) #var   

// #var converts macros( var ) to a string

int main (){

cout << macros( Hello World ) << endl;

return 0;

}

// Output:    Hello World

// "macros(Hello World)"  has been converted to "Hello World"

-------------------------------------------------------------------------------

#if…#else…#endif syntax:

#if expression

compile code1       // if true, compile this.

#else

compile code2       // if false, compile this.

#endif

// Part of the C++ program will be compiled if the condition test is true, it will not be compiled if the condition test is false.

e.g.

#include<iostream>

using namespace std;

#define ID 100

int main( ){

#if ID > 0

cout<< "ID>0" << endl;

#else

cout<< "ID<0" << endl;

#endif

return 0;

}

// Output:   ID>0

-------------------------------------------------------------------------------

#ifdef…#else…#endif syntax:

#ifdef  identifier

compile code1      // if defined, compile this.

#else

compile code2      // if not defined, compile this.

#endif

// Part of the C++ program will be compiled if the string has been defined previously, it will not be compiled if the string has not been defined.

e.g.

#include<iostream>

using namespace std;

#define VARIABLE  "This is a string."

int main( ){

#ifdef VARIABLE

cout<< "VARIABLE has been defined" <<endl;

#else

cout<< "VARIABLE has not been defined" << endl;

#endif

return 0;

}

// Output:   VARIABLE has been defined.

## syntax:

token1 ## token2

// The ## operator is used to join two tokens.

e.g.

#include <iostream>

using namespace std;

#define concat(x, y) x ## y

// convert "concat(x, y)" to "xy"

int main(){

int mn = 200;

cout << concat(m, n);

return 0;

}

// Output:   200

// "concat(m, n)" has been converted to "mn".

*pointer syntax:

dataType  *pointer;      // declare a pointer

*pointer = &variable; // *pointer accesses the variable's value

pointer = &variable;     // pointer points to the variable's address

// "dataType *pointer" declares a pointer.

// Pointer name is preceded by a " * ".

// & operator gets the memory address of a variable.  

e.g.

#include <string>

#include <iostream>

using namespace std;

int main( ) {

int num = 20;

int *pt = &num;      // *pt accesses the num's value

cout << "The value of num is: " << *pt << endl;

pt = &num;     // pt points to the num's address

cout << "The address of num is: " << pt << endl;

return 0;

}

// Output:  The value of num is: 20

The address of num is: 0x7fff1bbfbaa8

&variable // & operator gets the memory address of a variable.

*pointer = &variable; // *pointer accesses the variable's value

pointer = &variable;     // pointer points to the variable's address

&argument syntax:

type functionName ( &a, &b ) {  }

// (&a, &b) not only can receive the parameter from the function caller, and but also pass the parameter to the function caller.

e.g.

void swap ( int &a, int &b) {    // reference arguments

int temp;

temp = a, a = b, b = temp;   //  reference exchange

}

int main( ) {

int x = 10, y = 20;

cout << x << " " << y << endl;

swap ( x, y );   // call the function, and pass the parameters

cout << x << " " << y << endl;

return 0;

}

// Output:

10   20

20   10

? : operator syntax:

(test-expression) ? (if-true-do-this) : (if-false-do-this);

// When test-expression returns true, then run "if-true-do-this".

// When test-expression returns false, then run "if-false-do-this".

e.g.

int main( ) {

int a = 100; int b =200;

string result = (a < b) ? "apple" : "banana";

// (test-expression) ? (if-true-do-this) : (if-false-do-this);

cout <<  result << endl;

return 0;

}

// Output:  apple

~ destructor() syntax:

~ destructor();

// A destructor is used to release the memory occupation, clean up the constructor data.

// Destructor name is the same as class name, but it is preceded by a ~ symbol.

// The destructor has no any arguments, and has no return value.

e.g.

class Flower {    // declare a class

public:

string c1, c2;

Flower() {  c1= "yellow";  c2 = "purple"; } ;  // constructor

~ Flower();     // define a destructor.

};

// The declaration of destructor must follow the constructor.

abstract class syntax:

virtual type functionName( ) = 0;

// "virtual type functionName( ) = 0;" defines a "pure virtual method".  The "pure virtual method" is always inside an "abstract class".

Any class that contains a pure virtual function is named "abstract class". The abstract class as a base class includes one or more "pure virtual function", which must always be overridden by the derived class.

e.g.

#include<iostream>

using namespace std;

class Ball{     // define an abstract class

public:

virtual void play( ) = 0;    // define a pure virtual function

};

class Volleyball: public Ball{

public:

void play( ){

cout << "Play volleyball." << endl;

}

};

class Football: public Ball{

public:

void play( ){

cout << "Play football." << endl;

}

};

void sport(Ball &b){     // Ball’s alias is b

b.play( );     // call play( ) in derived class

}

int main( ){

Volleyball v;    // creates an object

Football f;     // creates an object

sport(v);

sport(f);

return 0;

}

// Output:

Play volleyball.

Play football.

acos() syntax:

acos(num)

// Return the arccosine of a number in radian.

e.g.

#include <iostream>

#include <cmath>

using namespace std;

int main(){

double num = 0.5;

volatile syntax:

int volatile variable;       // declare a variable

volatile int variable;       // declare a variable

int volatile *pointer;       // declare a pointer

volatile int *pointer;       // declare a pointer

// "volatile" tells the compiler that the variable value in multithread may unexpectedly change at any time; the compiler should read the variable from memory rather than from register.

// "volatile" tells the compiler not to optimizes this variable value.

// "volatile" tells the compiler not to cache variables in register.

// "volatile" tells the compiler not to change variable assignment order.

e.g.

int main(){

volatile int number;

number++;

}

int main(){

int number;

number++;

}

// Explanation:

// "volatile int number": variable cannot be copied from memory to registers.

// "int number": variable can be copied from memory to registers.

-------------------------------------------------------------------------------

write() syntax:

cout.write(stringArray, stringLength);

// Output a string

// "stringArray" defines the string as an array.

// "stringLength" defines the string size.

e.g.

using namespace std;

int main() {

char ch[ ] = "This is a test";   // define a string array

cout.write(ch, strlen(ch)) << endl;    // output a string

return 0;

}    

// Output:  This is a test

write a file syntax:

ofstream fileObject ("myFile.txt");     // create an output object

fileObject << myString << endl;     // write myString to file

// "ofstream fileObject ("myFile.txt");" creates an output-file object "fileObject".

// "ofstream" class is used to write data to a file.

#include <iostream>

#include <fstream>

#include <string>

}
