// Single initial block
module blockingNonblocking();
  // declare temp registers
  reg a,b,c,d,e,f,g,h,i,j,k;
  // initialize reg variables
  initial
    begin
    a = 0; b = 0; c = 0; d = 0; e = 0; f = 0;
    g = 0; h = 0; i = 0; j = 0; k = 0; l = 0;
    end
  initial
    begin
    // delay on right hand side
    // blocking
    a = #10 1;
    b = #2 1;
    c = #4 1;
    // non-blocking
    d = #40 1;
    e <= #2 1;
    f <= #4 1;
    // blocking left hand side
    // blocking
    #10 g = 1;
    #2 h = 1;
    #4 i = 1;
    // non-blocking
    #10 j = 1;
    #2 k <= 1;
    #4 l <= 1;
    end
  initial
    begin
    $display("Time, a, b, c, d, td, e, tf, g, th, i, tj, k, tk");
    $monitor($time, " %b %b %b %b %b %b %b %b %b %b %b %b",

    end
