#include <stdio.h>
#define max(a,b) ((a) > (b) ? (a) : (b))

void main(void)
{
    int i = 0;
    int j = 0;
    int a = 3;
    int b = 4;
    int c = 6;
    int d = 5;

    i = max (a, b);
    j = max (c, d);
    printf ("i = %s, j = %s\\n", i, j);

    i = max(a++, b++);
    printf ("a = %s, b = %s\\n", a, b);
    j = max(c++, d++);
    printf ("c = %s, d = %s\\n", c, d);

    return;
}
