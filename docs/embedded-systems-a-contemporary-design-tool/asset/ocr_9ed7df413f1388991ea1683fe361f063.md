int computeArea(Point p0, Point pt1)
{
    int area = (pt1.x - p0.x) * (pt1.y - p0.y);
    return area;
}

// Rectangle perimeter function
int computePerimeter(Point p0, Point pt1)
{
    int perimeter = 2*(pt1.x - p0.x) + 2*(pt1.y - p0.y);
    return perimeter;
}
