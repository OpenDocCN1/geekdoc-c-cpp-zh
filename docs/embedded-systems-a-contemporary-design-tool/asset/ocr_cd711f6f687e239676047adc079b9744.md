# A.4.3.4.5 SEQUENTIAL LOGIC

The three code modules in Figure A.29 evolve the behavioral implementations of the gated SR latch, the master-slave SR flip-flop, and the 2-bit binary counter designed earlier at the gate and dataflow levels.

```
// Behavioral Level Model
// Gated SR Latch
module gated_sr_latch(q, qnot, s, r, c, r, enab);
reg q, qnot;
always @( -clr or enab)
begin
if(-clr)
begin
q = 1'b0;
qnot = 1'b1;
end
end
else
begin
if(s & ~r)
begin
q <= s;
qnot <= r;
end
else if(~s & r)
begin
q <= s;
qnot <= r;
end
end
endmodule

// Use two SR Latches in a master slave
// configuration to build a flip-flop
module srmffs(q, qnot, s, r, clk, clr);
module srmffs(q, qnot, s, r, clk, clr);
input q, s, clk;
output q, qnot;
gSrLatch master(qm, qmnot, qnot, s, r, clk, clr);
gSrLatch slave(q, qnot, qm, qmnot, clr, -clk);
endmodule

// Build a synchronous two bit binary up counter
// using master slave SR flip-flops
module TwoBitCounter(qA, qB, clk, clr);
input clk, clr;
output qA, qB;
reg sA, sR;
wire qA, qnotB, qB;
always @(posedge clk)
begin
qA = qA & qnot & qB;
sR = ~qA & qB;
end
end
end
srmff FFB(qB, qBnot, qBnot, qB, clk, clr);
srmff FFA(qA, qAnot, sA, sR, clk, clr);
endmodule
```
