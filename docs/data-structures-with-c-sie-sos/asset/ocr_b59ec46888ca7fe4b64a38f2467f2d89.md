int PAR(char S1[][80],int N1,int NUM1)
{
int K;
int i;
char BLANK[6]=(" ");
char TEMP[80];

K=0;
while(K<N1-1)
{
i=0;

while(S1[K][i]!='\\0')
{
TEMP[i]=S1[K][i];
i=i+1;
}

if(strcmp(SUBSTR(TEMP,1,5),BLANK)==0)
NUM1=NUM1+1;
K=K+1;
}
return(NUM1);
}
