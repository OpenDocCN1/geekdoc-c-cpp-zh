```
int INDEX(char *STR1,char *STR2)
{
 int m,n;
 int index,flag;
 for(m=0;m<strlen(STR1);m++)
 {
 index=m;
 flag=1;
 for(n=0;n<strlen(STR2);n++)
 {
 if(STR1[m+n]==STR2[n])
 ;
 else
 {
 flag=0;
 break;
 }
 }
 if(flag==0)
 continue;
 else
 return(index);
 }
 if(m==strlen(STR1))
 return(-1);
}
```

INDEX(T,'THEN') = -1

Remark: Since 0 is a valid index location in C, we have used -1 to denote instances where pattern does not match the text.
