//
// Receives timer messages
//
void CMainView::OnTimer(UINT nIDEvent)
{
    ++m_nHeartbeat;

    if (m_nHeartbeat > 1)
        ((CStatic *)GetDlgItem(IDC_ICON_STATUS))->SetIcon(m_hYellow);

    if (m_nHeartbeat > 2)
        ((CStatic *)GetDlgItem(IDC_ICON_STATUS))->SetIcon(m_hRed);

    CFormView::OnTimer(nIDEvent);
}
