Fig. 7.9: The derivation relations between class Matrix and class Linequ.

Matrix and Linequ are defined in linequ.h. The member functions of the two classes are implemented in linequ.cpp. The main() is in lemain.cpp. An object of Linequ is defined in main(), which we can use to solve the linear equations with four variables.

```
//linequ.h file 1, defining classes
#include <iostream>
#include <cmath>
using namespace std;
class Matrix //define base class Matrix
public:
    Matrix(int dims=2); //constructor
    ~Matrix(); //destructor
    void setMatrix(double* rmat); //set initial value for matrix
    void printM(); //display matrix
protected: //protected data members
    int index; //dimension of the matrix
    double* MatrixA; //starting address of the matrix in memory
};
class Linequ: public Matrix //define publicly derived class Linequ
public: //external interface
    Linequ(int dims=2); //constructor
    ~Linequ(); //destructor
```
