```
//implementation of the function member of Class Linequ
//Public Function member No.1 Constructor
Linequ::Linequ(int dims):Matrix(dims)    //constructor of derived class Linequ
{
    //call base-class constructor, pass
    //arguments
    //dynamic memory allocation
    sums=new double[dims];
    solu=new double[dims];
}
```

```
//Public Function member No.2 Destruction function
Linequ::~Linequ()    //destructor of derived class Linequ
{
    //call destructor of base class
    //free memory
    delete[] sums;
    delete[] solu;
}
```

```
//Public Function member No.3 Set Line Equation function
void Linequ::setLinequ(double *a,double *b) //set the linear equations
{
    setMatrix(a);    //call base-class functions
    for(int i=0;i<index;i++)
        sums[i]=b[i];
}
```
