void InsertionSort(T A[], int n)
{
    int i, j;
    T temp;

    //Insert the elements with index 1-n-1 to the appropriate position
    //in the sorted sequence
    for (i = 1; i < n; i++)
    {
        //Scan elements from A[i-1] to A[0], find the appropriate
        //position to insert A[i]
        j = i;
        temp = A[i];
        while (j > 0 && temp < A[j-1])
            //Compare them one by one until temp=A[j-1], j is the
            //position to insert
            //If j==0, 0 is the appropriate position
            A[j] = A[j-1]; //Move elements forward one by one, make it
            //possible to insert immediately when the
            //appropriate is found
            j--;
        }
        //The position is found, insert immediately
        A[j] = temp;
    }
}
