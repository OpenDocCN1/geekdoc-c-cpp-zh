    int js,l,k,i,j,is,p,q;

        d=0.0;
        for (i=k;i<=index-1;i++) {
            for (j=k;j<=index-1;j++) {
                t=fabs(MatrixA[i*index+j]);
                if (t>d)
                    d=t;js[k]=j;is=i;
            }
        }
        if (d+1.0==1.0) l=0;

            if (js[k]!=k)
                for (i=0;i<=index-1;i++) {

                if (!is!)

                        MatrixA[p]; MatrixA[p]=MatrixA[q]; MatrixA[q]=t;
