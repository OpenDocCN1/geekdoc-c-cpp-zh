```cpp
void Calculator::Enter(int num) //Push operand num into the stack
{ S.Push(num); }

//Pop 2 operands out of stack, store them into opnd1 and opnd2
//If the stack does not contain enough operands, return False and
//output the message
Boolean Calculator::GetTwoOperands(int& opnd1, int& opnd2)
{
    if (S.IsEmpty()) //Check whether the stack is empty
    {
        cerr << "Missing operand!" << endl;
        return False;
    }
    opnd1 = S.Pop(); //Pop the right operand
    if (S.IsEmpty()) //Check whether the stack is empty
    {
        cerr << "Missing operand!" << endl;
        return False;
    }
    opnd2 = S.Pop(); //Pop the right operand
    return True;
}
```

```cpp
void Calculator::Compute(char op) //Commit calculation
{
  Boolean result;
  int operand1, operand2;
  result = GetTwoOperands(operand1, operand2); //Pop 2 operands
  if (result == True) //If succeed, commit the calculation
    //and push the result to the stack

    switch (op) {
      case '+': S.Push(operand2+operand1);
        break;
      case '-': S.Push(operand2-operand1);
        break;
      case '*': S.Push(operand2*operand1);
        break;
      case '/': if (operand1 == 0) //Check whether the divisor is 0
        {
          cerr << "Divide by 0!" << endl;
          S.ClearStack();
          //Clear the stack if the divisor is 0
        } else
          S.Push(operand2/operand1);
          break;
      case '^': S.Push(pow(operand2,operand1));
        break;
    }
    cout << S.Peek() << ' '; //Output the result of calculation
  } else
    S.ClearStack();
}

  char c[20];

        //Judge '-' is a minus or a subtraction

        //convert it to an integer and push

        break;
```

# 9.2.5 Queues

Unlike stacks, queues are also linear collections with restricted access pattern. We can only add elements on one side (head), and only remove elements on the other side (tail). It is also possible to implement queues in both arrays and linked lists. The definition of status of a queue is similar to one for stacks. A queue should provide a head pointer, and tail pointer as data members, and it also needs to provide interfaces of these operations:

- Initialization
- Enqueue
- Clear
- Access head

Figure 17 shows a trivial implementation in static arrays. We can see all elements moving towards the head when dequeuing an element, which is very inefficient. A way to design a queue is to make it circular one. The idea is shown in Figure 18. The queue maintains two pointers pointing to the head and the tail respectively. The queue changes these pointers when enqueueing and dequeuing elements. The important part is to use modular operation to wrap the queue, ensuring the pointers are reset to zero when cross the boundary. The queue also needs count elements. Example 9.10 is an implementation using this method.

```cpp
//Definition of class
template <class T>
class Queue
{
private:
    int front, rear, count; //Pointer of queue head and tail,
    //number of elements
    T qlist[MaxSize]; //Array of elements of queue

public:
    Queue(void); //Constructor, initialize queue head,
    //tail and number of elements
    void QInsert(const T& item); //Enqueue new element
    T QDequeue(void); //Dequeue element
    void ClearQueue(void); //Clear the queue

    T QFront(); //Get the element of queue head

    //Test the status of queue
    int QLength(void) const; //Get the length of the queue
                             //(number of elements)
    int QEmpty(void) const; //Check whether the queue is empty
    int QFull(void) const; //Check whether the queue is full
};

//Constructor, initialize queue head, tail and number of elements
template<class T>
Queue<T>::Queue(void) : front(0), rear(0), count(0)
{}
```

```cpp
template<class T>
void Queue<T>::QInsert(const T& item)
{//Insert element to the queue tail(enqueue)
{
    if (count == MaxSize)//Terminate the program if queue is full
    {
        std::cerr << "Queue overflow!" << std::endl;
        exit(1);
    }
    count++; //Increase the number of elements
    qlist[rear] = item; //Insert the element to queue tail
    rear = (rear+1) % MaxSize; //Increase the pointer of queue tail
                 //by 1, implement circular queue by
                 //reminder
}
```

```cpp
template<class T>
T Queue<T>::Delete(void) //Delete the queue head, return its
//value(Dequeue)
{
    T temp;
    if (count == 0) //Terminate the program if queue is empty
    {
        std::cerr << "Deleting from an empty queue!" << endl;
        exit(1);
    }
    temp = qlist[front]; //Save the value of queue head
    count--; //Decrease the number of elements
    front = (front+1) % MaxSize; //Increase the pointer of queue head
                               //by 1, implement circular queue by
                               //reminder
    return temp; //Return the value of queue head
}
```

```cpp
template<class T>
T Queue<T>::Front(void) const //Return the value of queue head
{ return qlist[front]; }
```

```cpp
template<class T>
```
