{
    if (S.IsEmpty()) //Check whether the stack is empty
    {
        cerr << "Missing operand!" << endl;
        return False;
    }
    opnd1 = S.Pop(); //Pop the right operand
    if (S.IsEmpty()) //Check whether the stack is empty
    {
        cerr << "Missing operand!" << endl;
        return False;
    }
    opnd2 = S.Pop(); //Pop the right operand
    return True;
}
