**Example 5.11:** Personnel information management program.
The whole program is divided into three files: class definition header employee.h, class implementation file employee.cpp, and main function file 5_11.cpp. We should link 5_11.cpp and employee.cpp together after compiling. If we use the VC++ environment, we should put 5_11.cpp and employee.cpp in the same project.

```
//employee.h
class employee
{
protected:
    int individualEmpNo;        //individual number
    int grade;                  //grade
    float accumPay();           //monthly pay
    static int employeeNo;     //currently maximum employee number
public:
    employee();                  //constructor
    ~employee();                 //destructor
    void promote(int);          //promotion function
    void SetAccumPay(float pa); //setting monthly pay function
    int GetIndividualEmpNo();   //getting number function
    int GetGrade();             //getting grade function
    float GetAccumPay();        //getting monthly pay function
};
```

```
//employee.cpp
#include<iostream>
#include "employee.h"
using namespace std;
int employee::employeeNo=1000; //basic value of employee number is

employee::employee() {
    individualEmpNo=employeeNo++; //the newly added employee number is
    grade=1;                      //the currently maximum employee number plus 1
    accumPay=0.0;                 //initial value of grade is 1
                              }//initial value of monthly pay is 0
void employee::promote(int increment) //promotion, the level promoted is
    grade+=increment;             //indicated by the increment
}
```
