```csharp
            get { return messages; }

            messages = new HashSet<Message>();
        }

        public void AddMessage(Message msgToAdd)
        {
            messages.Add(msgToAdd);
        }

        public void Display(IDisplayControl db)
        {
            db.LoadMessages(messages);
            db.Run();
        }

        public void DailyPurge(ClientBook clients)
        {
            // code omitted here
        }

        private bool ToBeDeleted(Message m)
        {
            return m.HasExpired();
        }
    }
}
```

The code above shows the creation of a typed collection of 'Message' called Messages and methods to add and display messages.

The method to display messages requires an object of type DisplayBoardControl to be passed as a parameter. Initially a DummyBoard object will be provided however when a real board is purchased then this object will replace the DummyBoard object. Case Study

For each message: If the message has expired or if the client has run out of credit Set the DaysRemaining for that message to zero. Remove all expired messages. See code below for this...

```csharp
void DailyPurge(ClientBook clients)
{
    Client client;

    // loop through current messages and decrement days
    foreach (Message m in messages)
    {
        m.DecrementDays(); // deduct 1 from days remaining
        try {
            // decrease client credit for this message
            client = clients.GetClient(m.ClientID);
            client.DecreaseCredit(m.Cost);
        }
        catch (UnknownClientException uce) {
            MessageBox.Show("INTERNAL ERROR IN MessageSet.\r\nDetails: " + uce.Source + " " + \r\nMessage details " + uce.Message + ": \r\n");
        }
    }
    // loop through current messages and expire those whose credit <=0
    foreach (Message m in messages)
    {
        try {
            client = clients.GetClient(m.ClientID);
            if (client.Credit < 0) {
                m.DaysRemaining = 0;
            }
        }
        catch (UnknownClientException) {
            // Do nothing as unknown clients have been reported to error log already
        }
    }
    // Remove all expired messages
    messages.RemoveWhere(ToBeDeleted);
}
```

# 11.9 Programming the Client Classes

The system needs a Client class, with methods to decrease the client's credit (when a message has been displayed for them). Ultimately it will also need a method to allow a client to pay for and increase their credit but we will not add this functionality in the first version of this system. This class is simple and is very similar to programming the Message class and is therefore not shown here.

Programming the ClientBook class is also similar to programming MessageSet, however there are a few significant :

- * All clients have a ClientID so ClientBook uses a sorted dictionary instead of a Set.
- * The method GetClient() would fail if no client exists with the specified ClientID. We need to build in protection in case a client cannot be found.
- * The method AddClient() could also fail if a client already exists with the specified ID.
