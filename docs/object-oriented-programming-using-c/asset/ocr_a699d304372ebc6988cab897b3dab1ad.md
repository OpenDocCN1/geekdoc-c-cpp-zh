public void SaleType (ISaleItem PSI)
{
    if (pSI is Publication)
{
        Console.WriteLine("This is a Publication");
    }
    else if (pSI is Ticket)
{
        Console.WriteLine("This is a Ticket");
    }
    else
{
        Console.WriteLine("This is an an unknown sale type");
    }
}
