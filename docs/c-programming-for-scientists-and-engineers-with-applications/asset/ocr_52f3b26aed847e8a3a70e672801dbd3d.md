void functoptubble(float array1[ ], int n)
{
    int i = 1, j,flag = 0;
    float temp;

    while(!flag && i <= n)
    {
        flag = 1;      /* * assume already sorted */, j--)
        for(j = n - 1; j >= i; j--)
        {
            if(array1[j] > array1[j - 1])
            {
                temp = array1[j];
                array1[j] = array1[j - 1];
                array1[j - 1] = temp;
                flag = 0;      /* * indicate not already sorted */
            }
        }
        i++;
    }
    return;
