/* funcscalemat: Set Up the Scaling Matrix */
/*                                                                */
**********************************************************************************
/* local variables:                                */
/*   sx - scaling factor for x coordinates      */
/*   sy - scaling factor for y coordinates        */
/*                                                                */
/* Output Variables:                               */
/*   scalemat1[ ][IMAX] - scaling matrix          */
/*                                                                */
**********************************************************************************
```

```
void funcscalemat(float scalemat1[ ][IMAX])
{
    float sx, sy;

    printf("Input x and y coordinate scaling factors:\n");
    scanf("%f %f",&sx, &sy);

    scalemat1[0][0] = sx;
    scalemat1[1][1] = sy;
    scalemat1[2][2] = 1.0;

    return;
