void funcsort(char name1[][STRLNNG])
{
    int i, j;
    char temp[STRLNNG];

    for(i = 0; i < NUMSTR - 1; i++)
    {
        for(j = i + 1; j < NUMSTR; j++)
        {
            if(strcmp(name1[i], name1[j]) > 0)
            {
                strcpy(temp, name1[i]);
                strcpy(name1[i], name1[j]);
                strcpy(name1[j], temp);
            }
        }
    }
    return;
}
