    scanf("%f %f",&radius,&angle);
    printf("\nRadius = %5.2f Angle = %5.2f\n", radius, angle);

    theta = PI * angle / 180.0;
    x = radius * cos(theta);
    y = radius * sin(theta);

    printf("x = %5.2f y = %5.2f\n", x, y);

    return 0;
}
