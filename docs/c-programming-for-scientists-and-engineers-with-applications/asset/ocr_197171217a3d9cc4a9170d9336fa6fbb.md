/*
 * Output Variables:
 */
 * count - number of above average days
 */
 * sum - sum of the numbers
 */
 * average - average of numbers
 */
/*
 * function called:
 * funcmonth - return number of days in the month
 */
 ************************************************************************************

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int funcmonth(char []);

int main()
{
    int i, ndays, count;
    char month[10];
    float sum, average;
    float *avgtemp;

    scanf("%s", month);

    ndays = funcmonth(month);

    avgtemp = (float*)calloc(ndays, sizeof(float));

    sum = 0.0;
    for(i = 0; i < ndays; i++)
    {
        scanf("%f", avgtemp);
        sum += *avgtemp;
    }

    avgtemp = avgtemp - ndays;
    average = sum / ndays;
    printf("average temperature for %s is %.2f\n", month, average);

    count = 0;
    for(i = 0; i < ndays; i++)
    {
        if(*avgtemp > average)
            count++;
        avgtemp++;
    }
