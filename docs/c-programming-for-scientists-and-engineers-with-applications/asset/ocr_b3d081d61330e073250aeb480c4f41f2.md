```
void funcinput(char name[])STRLNG)
{
    FILE *inptr;
    int i;

    inptr = fopen("strinfile.dat", "r");

    for(i = 0; i < NUM; i++)
        fgets(name[i], STRLNG, inptr);

    fclose(inptr);
    return;
}
```

```
void funcoutput(char name[])STRLNG)
{
    FILE *outptr;
    int i;

    outptr = fopen("stroutfile.dat", "w");

    for(i = 0; i < NUM; i++)
        fputs(name[i], outptr);

    fclose(outptr);
    return;
}
```
