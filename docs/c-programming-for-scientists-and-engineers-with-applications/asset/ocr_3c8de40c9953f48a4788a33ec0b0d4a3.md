int main ()
{
    const float PI = 3.141593;
    const float DENSITY = 441.48;

    int numrod;
    float diameter,length, cost, area, weight, volume,
    total_cost;

    printf("Input number of rods, diameter, length, and cost:\n");
    scanf("%d %f %f %f", &numrod, &diameter, &length, &cost);

    printf("\nNumber of rods=%5d\n", numrod);\n    printf("Diameter of rods=%6.2f\n", diameter);
    printf("Length of rods=%6.2f\n", length);\n    printf("Cost per pound of steel=%6.2f\n\n", cost);

    diameter = diameter / 12.0;
    area = PI * diameter * diameter / 4.0;
    volume = numrod * area * length;
    weight = volume * DENSITY;
    total_cost = weight * cost;

    printf("Total weight of steel=%10.2f pounds\n", weight);\n    printf("Total cost = $%8.2f\n", total_cost);

    return 0;
}
