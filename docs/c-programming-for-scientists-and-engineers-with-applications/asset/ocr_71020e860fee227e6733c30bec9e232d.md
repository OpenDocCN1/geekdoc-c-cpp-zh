void funcompute(CUBICLE *room1)
{
    int i = 0;
    for(i = 0; i < ENTRIES; i++)
    {
        room1[i].areawall = 2 * ( room1[i].length * room1[i].height
        + room1[i].width * room1[i].height);
        room1[i].paintcost = room1[i].areawall * COSTQFT;
    }
    return;
