********************************************************************
/* *
*   funcpressure: Compute the Pressure in Pounds *
*      per Square Inch *
/* *
********************************************************************
/* *
*   Input Parameters: pressure[] [][JMAX] - pressure of water in *
*      pounds per square inch *
/* *
*   Computed Variables:
*      pressure[][JMAX] - pressure of water in *
*      pounds per square inch *
/* *
*   Output Parameters:
*      pressure[][JMAX] - pressure of water in *
*      pounds per square inch *
/* *
********************************************************************
void funcpressure(float depth1[][JMAX], float pressure1[][JMAX])
{
    int i, j;

    for(i = 0; i < IMAX; i++)
        for(j = 0; j < JMAX; j++)
            pressure1[i][j] = 0.4333 * depth1[i][j];

    return;
}
