# 11.2 Manipulation of Structures

```
int i;

stptr = &record;
inptr = fopen("stdata.dat", "r");

while((fscanf(inptr, "%%5c %%19c",
    stptr -> stid, stptr -> name))
    != EOF)
{
    for(i = 0; i < 4; i++)
        scanf(inptr, "%%d \ &stptr -> score[i]);

    stptr -> stid[5] = '\0';
stptr -> name[19] = '\0';

    stptr -> sum = 0;
    for(i = 0; i < 4; i++)
    {
        stptr -> sum += stptr -> score[i];
    }
    stptr->average = stptr -> sum / 4.0;

    if(stptr ->average >= 90.0)
        stptr ->ltgrd = 'A';
    else if(stptr -> average >= 80.0)
        stptr -> ltgrd = 'B';
    else if(stptr -> average >= 70.0)
        stptr -> ltgrd = 'C';
    else if(stptr -> average >= 60.0)
        stptr -> ltgrd = 'D';
    else
        stptr -> ltgrd = 'F';

    printf("%%s%%s %%s", stptr -> stid, stptr -> name);
    for(i = 0; i < 4; i++)
    {
        printf("%%d ", stptr -> score[i]);
    }
    printf("%%2f %%c\n\n",
        stptr -> average, stptr -> ltgrd);
}

return 0;
```
