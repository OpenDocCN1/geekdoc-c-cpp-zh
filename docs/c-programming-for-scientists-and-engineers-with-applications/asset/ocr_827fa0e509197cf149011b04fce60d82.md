```
#include <stdio.h>

int main()
{
    const float PI = 3.141593;
    float area, stress;
    int diam, p;

    printf("Input the Compression Load:\n");
    printf("and Diameter of the Column:\n");

    scanf("%d %d", &p, &diam);

    if(p > 0 && diam > 0) {
        printf("nCompression Load: %d pounds\n", p);
        printf("Diameter of the Steel Column: %2d inches\n", diam);

        area = PI * ( (diam * diam) / 4.0);
        stress = p / area;
        printf("Compression Stress: %7.2f pounds per square inch\n", stress);
    }

    return 0;
}
```
