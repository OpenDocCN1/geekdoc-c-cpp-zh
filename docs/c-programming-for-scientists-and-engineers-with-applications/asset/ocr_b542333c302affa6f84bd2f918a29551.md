```
int fininput(float array[], int n)
{
    FILE *inptr;
    int count = 0;
    float value;

    inptr = fopen("sortdata.dat", "r");
    while((fscanf(inptr, "%f", &value))!=Eof && count < n) {
        array[count] = value;
        count++;
    }
    fclose(inptr);
    return count;
}
```
