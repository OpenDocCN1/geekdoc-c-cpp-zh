/*
 * Coordinate Matrix and
 * the Transformation Matrix
 *
 *********************************************************************************/

 * Input parameters:
 * transmat1[IMAX] - transformation matrix
 * oldline1[IMAX] - the coordinates of old line
 *\n * Computed Variables:
 * newline[IMAX] - translated coordinates of new
 * line
 *\n * Output Variables:
 * newline[JMAX] - new coordinates of line
 *\n *********************************************************************************/

void funcmmultiply(float transmat1[ ] [IMAX], float oldline1[ ] [JMAX],
    float newline1[][][JMAX])
{
    int i, j, k;
    for(i = 0; i < IMAX; i++)
    {
        for(j = 0; j < JMAX; j++)
        {
            newline1[i][j] = 0.0;
            for(k = 0; k < KMAX; k++)
            {
                newline1[i][j] += transmat1[i][k] * oldline1[k][j];
            }
        }
    }
    return;
}
