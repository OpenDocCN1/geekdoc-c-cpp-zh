    Output Parameters:

```
void funcinput(char strname[ ][STRLNG]) {
    FILE *inptr;
    int i;
    inptr = fopen("sortin.dat","r");
    for(i = 0; i < NUMSTR; i++) {
        fgets(strname[i], STRLNG, inptr);
        printf("%s\n", strname[i]);
    }
    fclose(inptr);
    return;
}
```
