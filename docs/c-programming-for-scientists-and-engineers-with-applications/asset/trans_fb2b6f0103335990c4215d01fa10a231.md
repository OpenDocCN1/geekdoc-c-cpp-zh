void funccost(int numcars1 [][JMAX],
  float carunitcost1 [][JMAX],
  float totcostcars1 [][JMAX])
{
    int i, j;

    for(i = 0; i < IMAX; i++)
        for(j = 0; j < JMAX; j++)
            totcostcars1[i][j] = numcars1[i][j]*carunitcost1[i][j];
    return;
}
