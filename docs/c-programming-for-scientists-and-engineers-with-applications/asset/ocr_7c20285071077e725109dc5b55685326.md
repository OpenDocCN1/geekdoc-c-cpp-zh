/* Determine the Maximum and the Minimum                               */
/*********************************************************************
/* Input Parameters:                                                                    */
/*   number[count] - array of real numbers                             */
/*   count - size of array                                              */
/* Computed Variables:                                                                */
/*   amax1 - maximum value                                                           */
/*   amin1 - minimum value                                                           */
/* Output parameters:                                                                 */
/*   amax1 - maximum value                                                           */
/*   amin1 - minimum value                                                           */
/*********************************************************************
```

```
void maxmin(int count,float number1[],float *amax2,
             float *amin2)
{
  int i;
  *amax2 = number1[0];
  *amin2 = number1[0];
  for (i = 1; i < count; i++) {
    if(number1[i] > *amax2)
      *amax2 = number1[i];
    else if(number1[i] < *amin2)
      *amin2 = number1[i];
  }
}
