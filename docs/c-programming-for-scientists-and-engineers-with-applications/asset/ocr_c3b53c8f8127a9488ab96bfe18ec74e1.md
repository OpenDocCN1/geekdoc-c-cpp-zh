float length, period, constant;

```
fptrout = fopen("pend.dat", "w");

constant = 2.0 * PI;
length = 1.0;
period = constant * sqrt(length / GRAVITY);
fprintf(fptrout, "  OUTPUT\n");  "");
fprintf(fptrout, "  length");  "");
fprintf(fptrout, "  period\n");
fprintf(fptrout, "      ");  "");
fprintf(fptrout, "      \\n");

while(period <= 3.45)
{
    fprintf(fptrout, "  %2.0f,      %5.2f\n",
    length, period);
    length += 1.0;
    period = constant * sqrt(length / GRAVITY);
}

return 0;
```
