```c
int numsqrt(float num1, float *sqroot)
{
  int flag = 1;
  float sqroot0, error;
  if(num1 < 0)
  {
    flag = 0;
  } else if(num1 == 0)
    *sqroot = 0;
  } else
  {
    error = .0001;
    sqroot0 = num1;
    *sqroot = num1 / 2.0;
    do
    {
      sqroot0 = *sqroot;
      *sqroot = (sqroot0 + num1 / sqroot0) / 2.0;
    }while(abs(*sqroot - sqroot0) > error);
  }
  return flag;
}
```
