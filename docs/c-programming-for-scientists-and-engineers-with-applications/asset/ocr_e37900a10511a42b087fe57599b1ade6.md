/*
 * sqroot - square root of the number num
 * *
 * Output variables:
 * *
 * numsgroot - square root of the number num
 * *
 * Functions called:
 * *
 * numsqrt - function to compute the
 * square root
 * *
 *****************************************************************

#include <stdio.h>
#include <stdlib.h>

int numsqrt(float, float *);

int main()
{
    int flag, count;
    float num, sqroot;

    printf("How many numbers to calculate?\\n");
    scanf("%d", &count);
    printf("Input Data Number Square Root\\n");

    while(count != 0)
    {
        scanf("%f", &num);
        flag = numsqrt(num, &sqroot);
        if(flag == 0)
        {
            printf("%6.2f Negative number, \\n",
                num);
            printf("no real square root\\n");
        }
        else
        {
            printf("%6.2f %6.2f\\n",
                num, sqroot);
        }
    }
}
