    NSURL *theURL = [NSURL URLWithStr ing:theURLString];
    if ( [[NSWorkspace sharedWorkspace] openURL:theURL] )
        NSLog(@"Email Loaded");
}

- (IBAction)sendEmailManually:(id) sender
{
    NSStr ing *theEmailDest = [emailField str ingValue];
    NSStr ing *theSubject = @"Important Message";
    NSStr ing *theBody = @"Hello there!";
