```
#import "Account.h"

@implementation Account

-(float)balance {
 return balance;
}

-(void)setBalance:(float)aBalance {
 balance = aBalance;
}

-(void)deposit:(float)depositAmount {
 balance += depositAmount;
}

-(void)withdraw:(float)withdrawAmount {
 balance -= balance withdrawAmount;
}

- (id) init
{
 //initialize the superclass and assign it to self
 if (self = [super init]) {
 //now that you know self has been inited,
 //you can work with its instance variables
 balance = 100.0f;
 }
 return self;
}

@end
```

This code implements the two accessor methods and the `deposit` and `withdraw` methods, as well as initializes the balance by giving the account $100 to start.
