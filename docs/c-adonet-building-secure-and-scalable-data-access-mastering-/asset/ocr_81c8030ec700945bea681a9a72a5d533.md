Connection pooling is a technique that reuses database connections instead of creating a new connection for each request. By using connection pooling, you can improve the performance of your LINQ to SQL applications, as the overhead of creating and tearing down connections is eliminated.

To use connection pooling, create a static method that returns a LINQ to SQL data context, and use the data context in your LINQ to SQL queries. Here's an example:

```
```csharp
using System.Data.Linq;
using System.Data.Linq.Mapping;

namespace MyNamespace
{
    public class MyDataContext : DataContext
    {
        public MyDataContext(string connectionString) : base(connectionString) {}

        [Function(Name = "dbo.GetCustomers")]
        public IQueryable<Customer> GetCustomers()
        {
            return this.ExecuteQuery<Customer>("dbo.GetCustomers");
        }

        public static MyDataContext GetMyDataContext()
        {
            return new MyDataContext(Properties.Settings.Default.MyConnectionString);
