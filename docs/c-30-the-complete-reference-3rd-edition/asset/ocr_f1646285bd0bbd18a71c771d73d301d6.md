// Create a 4-bit type called Nibble.
using System;

// A 4-bit type.
class Nibble {
    int val; // underlying storage

    public Nibble() { val = 0; }

    public Nibble(int i) {
        val = i;
        val = val & 0xF; // retain lower 4 bits
    }

    // Overload binary + for Nibble + Nibble.
    public static Nibble operator +(Nibble op1, Nibble op2)
    {
        Nibble result = new Nibble();

        result.val = op1.val + op2.val;

        result.val = result.val & 0xF; // retain lower 4 bits

        return result;
    }

    // Overload binary + for Nibble + int.
    public static Nibble operator +(Nibble op1, int op2)
    {
        Nibble result = new Nibble();

        result.val = op1.val + op2;

        result.val = result.val & 0xF; // retain lower 4 bits

        return result;
    }

    // Overload binary + for int + Nibble.
    public static Nibble operator +(int op1, Nibble op2)
    {
        Nibble result = new Nibble();

        result.val = op1 + op2.val;

        result.val = result.val & 0xF; // retain lower 4 bits
