// Construct array given its size.
public RangeArray(int low, int high) {
    high++;
    if(high <= low) {
        throw new RangeArrayException("Low index not less than high.");
    }
    a = new int[high - low];
    Length = high - low;
    lowerBound = low;
    upperBound = --high;
}

private bool ok(int index) {
    if(index >= lowerBound & index <= upperBound) return true;
    return false;
