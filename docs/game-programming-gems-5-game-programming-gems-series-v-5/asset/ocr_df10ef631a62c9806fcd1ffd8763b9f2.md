The menuControl class contains information that is common to most types of user controls, such as control location, control type, and the current state. MenuControl also offers a render() function, again allowing the system to treat all controls similarly. Additionally, the menuControl class contains methods for altering the visibility, giving you complete control over whether a certain menu object will be displayed. Listing 1.17.2 contains the description of the menuControl class.

# Listing 1.17.2 Description of the menuControl Class

```
```cpp
class menuControl
{
public:
    static enum controlType {
        NONE = -1,
        BUTTON = 0,
        STATIC,
        SLIDER,
        LIST
    };

    static enum controlState { ACTIVE=0, DISABLED };
    static enum controlView { VISIBLE=0, HIDDEN };

    menuControl(void);
    virtual ~menuControl(void);

    virtual void render(void) = 0;

    void setControlXY(int X, int Y) { locX = X; locY = Y; }
    int getType(void) { return type; }

    void activateControl(void) { state = controlState::ACTIVE; }
    void deactivateControl(void) { state = controlState::DISABLED; }
    bool getControlState(void) { return state; }

    void showControl(void) { view = controlState::VISIBLE; }
    void hideControl(void) { view = controlState::HIDDEN; }
    bool getControlView(void) { return view; }

protected:
    // properties of a control
    int type; // the type of control
    int locX; // the X location
    int locY; // the Y location

    bool state; // whether the control is active
    bool view; // whether the control is visible
};
