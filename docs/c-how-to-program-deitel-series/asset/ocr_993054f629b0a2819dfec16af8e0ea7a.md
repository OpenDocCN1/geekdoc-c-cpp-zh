   while ( ( choice = enterChoice() ) != END )

            createTextFile( inOutCredit );
            break;

         case UPDATE: // update record
            updateRecord( inOutCredit );
            break;

         case NEW: // create record
            newRecord( inOutCredit );
            break;

            deleteRecord( inOutCredit );
