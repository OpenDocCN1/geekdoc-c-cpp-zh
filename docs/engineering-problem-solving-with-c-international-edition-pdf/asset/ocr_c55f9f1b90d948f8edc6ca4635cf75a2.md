```
1. pixel pixel::operator/(pixel p1)
{
    pixel temp;
    temp.red = red/p1.red;
    temp.green = green/p1.green;
    temp.blue = blue/p1.blue;
    return temp;
}
```

```
2. bool pixel::operator==(pixel p1)
{
    if (red==p1.red &&
        green==p1.green &&
        blue==p1.blue )
        return true;
    return false;
}
```
