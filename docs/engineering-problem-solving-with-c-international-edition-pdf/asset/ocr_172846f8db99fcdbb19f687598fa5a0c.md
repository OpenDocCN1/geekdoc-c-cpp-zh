void readPointFile(istream& in,

{
    int npts;
    // Read number of data points.
    in >> npts;
    v.resize(npts);
    Point p;
    // Read Points and atore in vector.
    // Points are formatted as x,y
    double x,y;
    char comma;
    for (int i=0; i<npts; ++i)
    {
        in >> x >> comma >> y;
        p.setX(x);
        p.setY(y);
        v[i] = p;
