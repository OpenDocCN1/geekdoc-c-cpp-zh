#include <stdio.h>
int main(void)
{
    int c;
    while ((c = getc(stdin)) != EOF) {
        if (c == '\r')          /* If \r, then ... */
            if ((c = getc(stdin)) != '\n')  /* ... check next character */
                ungetc(c, stdin);            /* Put it back if not \n */
            else
                putc('\n', stdout);
        else
            putc(c, stdout);
    }
    return 0;
}
