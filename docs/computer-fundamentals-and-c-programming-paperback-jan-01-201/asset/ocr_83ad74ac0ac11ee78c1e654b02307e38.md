int time_diff(short h1, short m1, short h2, short m2)   /* Function1 */
{
    short net_mins;
    net_mins = h1 * 60 + m1 - (h2 * 60 + m2);
    return absolute(net_mins);                         /* Calls another function */
}

int absolute(short a_mins)
{
    return (a_mins > 0 ? a_mins : a_mins * -1);
}
