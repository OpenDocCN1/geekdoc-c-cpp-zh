> /* baseDigit() finds a char representing a digit in another base.
> Receive: value, an int.
> Precondition: 0 <= value && value <= 35.
> Return: the character representation of value.
> ---------------------------------------------------------------*/

```
char baseDigit(int value)
{
    const int NUM_DIGITS = 36;    // number of digits in other base
    const string digits = "0123456789abcdefghijklmnopqrstuvwxyz";
    if (0 <= value && value < NUM_DIGITS)
        return digits[value];
    else
    {
        cerr << "\n** baseDigit(value): " << value
             << " outside of range 0.." << NUM_DIGITS - 1 << endl;
        return '*';
    }
}
```
