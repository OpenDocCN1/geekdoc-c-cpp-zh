18. void f(unsigned n)
{
	if (n == 0)
		return 0;
	else
		return n * f(n - 1);
}

20. double f(double x, unsigned n)

}

21. unsigned f(unsigned n)
{
	if (n < 2)
		return 0;
	else
		return 1 + f(n / 2);
}

22. unsigned f(unsigned n)
