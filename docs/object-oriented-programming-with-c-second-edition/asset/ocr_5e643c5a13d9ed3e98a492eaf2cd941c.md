{
  String temp;
  fin.read((char*)&temp.len, sizeof(int));
  temp.cStr= new char[temp.len+1];
  int i;
  for(i=0;i<temp.len;i++)
    fin.get(temp.cStr[i]);
  temp.cStr[i]='\0';
  *this = temp;
}
