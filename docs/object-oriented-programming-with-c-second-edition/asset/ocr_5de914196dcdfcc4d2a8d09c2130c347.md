# 274 Object-Oriented Programming with C++

```
;
class B : public A {
public:
    void f2()
    {
        cout<<"B::f2() called\n";
    }
};

```

```
class C : public A {
public:
    void f3()
    {
        cout<<"C::f3() called\n";
    }
};
```

```
void main() {
    B B0j;
    C B0j;
    A &ARef1=B0j; // ARef1 is a reference to an object of
//class B
```

```
try {
    B &BRef1=dynamic_cast<B &>(ARef1);
    //ARef1 is actually of type B and
    //type is also B&. Hence, cast
    //returns reference to B0j.

    BRef1.f2();
} catch(bad_cast) {
    cout<<"Invalid cast\n";
}
```
