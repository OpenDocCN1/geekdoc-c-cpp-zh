```
/*Beginning of Distance.cpp*/
#include "Distance.h"
Distance& larger(Distance& dd1, Distance& dd2)
{
    float i,j;
    d1=d1.getFeet()*12+dd1.getInches();
    d2=d2.getFeet()*12+dd2.getInches();
    if(i>j)
        return dd1;
    else
        return dd2;
}
/*definitions of the rest of the functions of class Distance*/
```
