```
/* Function to delete the first node.*/
void SingleLinkedList::delBeg() {
  /* Delete only if the list is empty.*/
  if(head != NULL) {
    /* Make a temporary pointer point at the first node.*/
    SingleLinkedList * temp = head;
    /* Make the head pointer point at the second node.*/
    head = head -> getNext(); //head becomes NULL if only //one node in the list.
    /* Delete the first node.*/
    delete temp;
  }}
```

```
/* Function to delete the last node.*/
void SingleLinkedList::delEnd() {
  /* Delete only if the list is empty.*/
  if(head != NULL) {
    /* If there is only one node ...*/
    if(head -> getNext() == NULL) {
      /* ... delete it and ...*/
      delete head;
    }}
```
