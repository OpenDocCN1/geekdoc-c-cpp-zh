```
CThreadPool::~CThreadPool()
{
    SetEvent(hEvent);
    if (WaitSingleObject(hProblemSolver, MAX_WAIT_TIME) != WAIT_OBJECT_0)
    {
        TerminateThread(hProblemSolver, (DWORD)-1);
    }
    CloseHandle(hProblemSolver);
    CloseHandle(hEvent);
}
```
