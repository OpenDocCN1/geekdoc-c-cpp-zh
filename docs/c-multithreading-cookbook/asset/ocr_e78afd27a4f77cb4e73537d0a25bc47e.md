System::Void SetState(bool bState)
{
    Lock* lock = gcnew Lock(this->lockObject);
    bContinue = bState;
    delete lock;
}
```

```
bool QueryState(void)
{
    Lock* lock = gcnew Lock(this->lockObject);
    bContinue = bContinue;
    delete lock;

    return bState;
}
