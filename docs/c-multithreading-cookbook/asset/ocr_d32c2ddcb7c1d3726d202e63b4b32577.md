#include "Clist.h"

template<class T>
class Cqueue : List<T>
{
public:
  Cqueue() : List<T>() {}
  Cqueue(T* tElement) : List<T>(tElement) {}
  virtual ~Cqueue() {}
  virtual void Enqueue(T* tElement) {
    Append(Head(), tElement);
  }
  virtual T* Dequeue() {
    T* tElement = GetFirst();
    Remove(tElement);
    return tElement;
  }
}
