DWORDAPI StartAddress (LPVOID lParameter)
{
    // We'll solve the linear system using Cramer's rule:
    CQueue<QueueElement>* pQueue = (CQueue<QueueElement>*)lpParameter;

    HANDLE hMutex = OpenMutex(MUTEX_ALL_ACCESS, FALSE, szMutex);
    HANDLE hSemaphore = OpenSemaphore(SEMAPHORE_ALL_ACCESS, FALSE, szSemaphore);
    QueueElement* pElement = NULL;

    while (true) {
        WaitForSingleObject(hSemaphore, INFINITE);
        ReleaseSemaphore(hSemaphore, 1, NULL);

        WaitForSingleObject(hMutex, INFINITE);
        pElement = pQueue->Dequeue();
        ReleaseMutex(hMutex);
        if (pElement == NULL) {
            break;
        }

        char szBuffer[1024];
        double dDeterminant = (pElement->iA1 * pElement->iB2) -
                              (pElement->iB1 * pElement->iA2);

        if (dDeterminant != 0) {
            double DX = (pElement->iC1 * pElement->iB2) -
                        (pElement->iB1 * pElement->iC2) / dDeterminant;
            double DY = (pElement->iA1 * pElement->iC2) -
                        (pElement->iC1 * pElement->iA2) / dDeterminant;
            sprintf_s(szBuffer, " %8.4f, y = %8.4f\n", DX, DY);
        } else {
            sprintf_s(szBuffer, " Determinant is zero.\n");
        }

        strcat_s(szResult, 4096, szBuffer);
        SetWindowTextA(pElement->hWndResult, szResult);

        SendMessage(pElement->hWndProgress, PBM_STEPIT, 0, 0);

        delete pElement;
        sleep(1000);
    }

    CloseHandle(hSemaphore);
    CloseHandle(hMutex);
    return 0L;
}

# How it works

This is a slightly different approach than the one used in our previous example. Instead of creating as many threads as there are processors, we instantiate a semaphore object where the initial count will not exceed the number of CPUs. It is similar; the only difference is the technique used and approach selected inside the StartAddress routine. After successful operation, we are releasing the semaphore by decrementing it:

```cpp
WaitForSingleObject(hSemaphore, INFINITE);
ReleaseSemaphore(hSemaphore, 1, NULL);
```

# Using event

The event object works as a signal for alerting the thread in a certain manner. It can be used to start or stop execution, when user input is performed, and so on. We will use the same example as event to signal threads that the user wants to abort and end the program. In this way, we will make our application even more user-friendly and responsive to user input.
