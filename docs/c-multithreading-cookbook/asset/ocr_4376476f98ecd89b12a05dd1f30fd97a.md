OPENFILENAMEA ofn;

ZeroMemory(&ofn, sizeof(ofn));
ofn.ofnStructSize = sizeof(ofn);
ofn.hwndOwner = hWnd;
ofn.lpstrFile = szFile;
ofn.lpstrFile[0] = '\0';
ofn.nMaxFile = sizeof(szFile);
ofn.lpstrFilter = "All\0*.*\0\TXT\0.*\0";
ofn.nFilterTitle = 1;
ofn.lpstrFilterTitle = NULL;
ofn.nMaxFileTitle = 0;
ofn.lpstrInitialDir = NULL;
ofn.Flags = OFN_PATHMUSTEXIST | OFN_FILEMUSTEXIST;
if (GetOpenFileNameA(&ofn) == TRUE) {
strncpy_s(szFile, MAX_PATH - 1, szFile);
return TRUE;
}
