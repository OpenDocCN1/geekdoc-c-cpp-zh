# Thread Synchronization and Concurrent Operations

```
if (!szFilePath) {
    MessageBox(hWnd,
    T("You must choose valid file path first!"),
    T("Error!"), MB_OK | MB_TOPMOST | MB_ICONERROR);
    break;
}

fstream infile(szFilePath);

if (infile.is_open()) {
    string line;
    while (std::getline(infile, line)) {
        QueueElement* pQueueElement = new QueueElement();
        istringstream iss(line);

        if ((iss >> pQueueElement->iA1 >> pQueueElement->iB1 >> pQueueElement->iC1 >> pQueueElement->iA2 >> pQueueElement->iB2 >> pQueueElement->iC2)) {
            break;
        }

        pQueueElement->WndProgress = GetDlgItem(hWnd, CONTROL_PROGRESS);
        pQueueElement->WndResult = GetDlgItem(hWnd, CONTROL_RESULT);

        queue.Enqueue(pQueueElement);
    }

infile.close();
    SendMessage(GetDlgItem(hWnd, CONTROL_PROGRESS), PBM_SETRANGE, 0, MAKELPARAM(0, queue.Count()));
}

int iCount = queue.Count();
for (int iIndex = 0; iIndex < iCount; iIndex++) {
    StartProcess(GetDlgItem(hWnd, CONTROL_TEXT));
    Sleep(100);
}
```
