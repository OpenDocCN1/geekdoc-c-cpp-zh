                break;
            }
        case WM_COMMAND:

                            if (!FileDialog(hWnd, szFilePath)) {
                                MessageBox(hWnd,

                                T("Error!"), MB_OK | MB_TOPMOST | MB_ICONERROR);
                            }
                        } else {
                            char szBuffer[MAX_PATH];

                            strchr(szFilePath, '\'\') + 1;
                            SetWindowTextA(GetDlgItem(hWnd), CONTROL_TEXT, szBuffer);
                        }
