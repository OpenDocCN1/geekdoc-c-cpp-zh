3. Open Solution Explorer and right-click on Header files. Under Visual C++, select UI.
Add a new Windows Form named MyForm. Open MyForm.h.

4. Add the following code:

```
#pragma once

namespace CLRAplication10
{
    using namespace System;
    using namespace System::Threading;
    using namespace System::Threading::Tasks;
    using namespace System::Windows::Forms;
    using namespace System::Drawing;

    ref class Lock
    {
        Object* lockObject;
        public:
            Lock(Object* lock) : lockObject(lock)
            {
                Monitor::Enter(lockObject);
            }
        protected:
            -Lock()
            {
                Monitor::Exit(lockObject);
            }
    };

    ref class MyForm : public MyForm
    {
        public:
            MyForm(void) : syncHandle(gcnew Object())
            {
                InitializeComponent();
                this->Load += gcnew EventHandler(this, &MyForm::Form_Load);
            }
        protected:
            -MyForm()
            {
                if (components)
                {
                    delete components;
                }
            }
    };
```
