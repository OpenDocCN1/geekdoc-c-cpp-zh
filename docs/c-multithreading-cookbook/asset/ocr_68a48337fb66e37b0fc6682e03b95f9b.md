hWND label = GetDlgItem(hWnd, LABEL_TEXT);
TCHAR szBuffer[BUFFER_SIZE];
sprintf(szBuffer,
#ifdef _UNICODE
	_("%u\n\t%s\tu\n%s\td\n\t%s\td")
#else
	_("%u\n\t%s\tu\n%s\td\n\t%s\td")
#endif
	_("Thread ID: %d", (DWORD)pData->dwThreadId,
	_("Current value: %d", (int)pData->value,
	_("Message ID: %d", pData->msgID);
SetWindowText(hLabel, szBuffer);

delete pData;

PostMessage(hWnd, WM_ENDTASK, 0, 0);

return 0;

# How it works...

First, we need to define certain message values. A message is represented as a unique integer, larger than the predefined WM_USER value. All messages larger than the WM_USER (1024) value are considered application specific (user defined). The first 1024 messages are used (reserved) by Windows. We decided to define the two messages as follows:

+   * WM_PLOTDATA for the calculation thread to inform the main thread that calculation has been completed
* WM_ENDTASK to inform the main thread that all the drawing has been completed
