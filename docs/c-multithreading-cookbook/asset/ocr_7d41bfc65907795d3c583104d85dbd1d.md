#include <stdfax.h>

double CSwefel::Evaluate(void* lpParameters)
{
    GetSystemTime(&stdStart);

    for (unsigned i = 0; i < uSize; i++)
    {
        dSum += pdAllele[i];

        if (dMin > pdAllele[i])
        {
            dMin = pdAllele[i];
        }

        double dTmp = dInnerSum * dInnerSum;
        dScore += dTmp;

        if (i != 0 && dBest > dTmp)
        {
            dBest = dTmp;
        }
    }
    GetSystemTime(&stdEnd);
    return dScore;
}

void CSwefel::Initialize(void)
{
    srand(Seed());

    dSum = 0;
    dMin = DBL_MAX;
    dScore = 0;
