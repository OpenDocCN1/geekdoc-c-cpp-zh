template <class T>
T* CList<T>::GetNext(T* tElement)
{
  if (head == NULL)
{
    return NULL;
  }
  if (tElement == NULL)
{
    return GetFirst();
  }
  if (head->Element() == tElement)
{
    return head->Next() != NULL ? head->Next()->Element() : NULL;
  }
  ListNode* lst = head->Next();
  while (lst != NULL)
{
    if (lst->Element() == tElement)
    {
      return lst->Next() != NULL ? lst->Next()->Element() : NULL;
    }
    lst = lst->Next();
  }
  return NULL;
}
