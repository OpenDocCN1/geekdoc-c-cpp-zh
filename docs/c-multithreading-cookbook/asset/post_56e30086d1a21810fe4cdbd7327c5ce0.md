LRESULT CALLBACK WindowProcedure(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam)
{
    switch (uMsg)
    {
    case WM_DESTROY:
        {
            PostQuitMessage(0);
            break;
        }
    case WM_CLOSE:
        {
            SetEvent(hEvent);
            DestroyWindow(hWnd);
            break;
        }
    case WM_COMMAND:
        {
            switch (LOWORD(wParam)) {
            case CONTROL_BROWSE:
                {
                    if (!FileDialog(hWnd, szFilePath)) {
                        MessageBox(hWnd, T("You must choose valid file!"),
                            T("Error!"), MB_OK | MB_ICONERROR);
                    }
                }
                break;
            case CONTROL_START:
                {
                    if (!szFilePath) {
                        MessageBox(hWnd, T("Cannot open file!"),
                            T("Error!"), MB_OK | MB_ICONERROR);
                    } else {
                        ifstream infile(szFilePath);
                        if (infile.is_open()) {
                            string line;
                            while (std::getline(infile, line)) {
                                QueueElement* pElement = new QueueElement();
                                istringstream iss(line);

                                if (!(iss >> pElement->A1 >> pElement->B1 >> pElement->C1 >>
                                    pElement->A2 >> pElement->B2 >> pElement->C2)) {
                                    delete pElement;
                                    break;
                                }

                                pElement->hWndProgress = GetDlgItem(hWnd, CONTROL_PROGRESS);
                                pElement->hWndResult = GetDlgItem(hWnd, CONTROL_RESULT);

                                queue.Enqueue(pElement);
                            }
                            infile.close();
                            SendMessage(GetDlgItem(hWnd, CONTROL_PROGRESS), PBM_SERANGE, 0, MAKELPARAM(0, queue.Count()));
                        }

                        int iCount = queue.Count();
                        for (int iIndex = 0; iIndex < iCount; iIndex++) {
                            StartProcess(GetDlgItem(hWnd, CONTROL_TEXT));
                            Sleep(100);
                        }
                    }
                }
                break;
            }
        }
        break;
    default:
    {
        return DefWindowProc(hWnd, uMsg, wParam, lParam);
    }
}
return 0;
}

DWINAPI ListenerRoutine(LPVOID lpParameter)
{
    void** lpParameters = (void**)lpParameter;
    HWND hWndProgress = (HWND)lpParameters[0];
    HWND hWndResult = (HWND)lpParameters[1];

    HANDLE hEvent = OpenEvent(EVENT_ALL_ACCESS, FALSE, EVENT_NAME);

    while (TRUE)
    {
        DWORD dwStatus = WaitSingleObject(hEvent, INFINITE);
        if (dwStatus == WAIT_OBJECT_0)
        {
            break;
        }
    }

    HANDLE hPipe = CreateNamedPipe(PIPE_NAME, PIPE_ACCESS_DUPLEX,
        PIPE_WAIT | PIPE_READMODE_MESSAGE |
        PIPE_TYPE_MESSAGE, PIPE_UNLIMITED_INSTANCES, BUFFER_SIZE,
        BUFFER_SIZE, 0, NULL);
    if (hPipe == INVALID_HANDLE_VALUE)
    {
        char szBuffer[MAX_PATH];
        sprintf(szBuffer, "Error: [%u]\n", GetLastError());
        SetWindowTextA(hWndResult, szBuffer);
    }
}
