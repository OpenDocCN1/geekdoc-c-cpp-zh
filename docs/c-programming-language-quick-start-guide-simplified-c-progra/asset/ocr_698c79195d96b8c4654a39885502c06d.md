void main (void)
{
int Two_D[10][10];
for (int i = 0; i < 10; i++)
{
for (int j = 0; j<10; j++)
{
Two_D[i][j] = i+j;
}
}
for(int i = 0; i < 10; i++)
{
for(int j = 0; j<10; j++)
{
printf("%d, ", Two_D[i][j]);
}
}

}
