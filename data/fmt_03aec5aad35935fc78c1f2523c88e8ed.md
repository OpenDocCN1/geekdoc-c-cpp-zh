```sas
*18-5;
title "描述性统计"

class Gender;
var GPA;

  Gender all;
keylabel n = 'Number'
  all = 'Total'
  mean = 'Average'
  min = 'Minimum'
  max = 'Maximum'
run;
```

```sas
*18-7;
title "更多描述性统计"
proc tabulate data=learn.college format=7.1 noseps;
class Gender SchoolSize;
var GPA ClassRank;

GPA*(median min max)(median=f=7. min*f=7. max*f=7.)/ rts=15;
keylabel all = 'Total'
  median = 'Median'
  min = 'Minimum'
  max = 'Maximum'
label ClassRank = 'Class Rank'
  SchoolSize = 'School Size'
run;
```
