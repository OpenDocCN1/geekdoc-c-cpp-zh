## 程序 11.3：枚举与函数演示

```c
#include <stdio.h>
/* 引入一个枚举数据类型 */
typedef enum {monday, tuesday, wednesday, thursday, friday, saturday, sunday} days;
/* 两个实用函数 */
days yesterday(days today);
days tomorrow(days today);
char *thdays[] = {"monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"};

int main(void) {
    days today;
    printf("today \tyesterday \ttomorrow\n");
    printf("========================================\n");
    for (today=monday; today<=sunday; today++) {
        printf("%s = %d %s = %d %s = %d %s = %d\n",
            thdays[today], today, thdays[yesterday(today)],
            yesterday(today), thdays[tomorrow(today)],
            tomorrow(today));
    }
    return 0;
}
days yesterday(days today) {
    return (today+6)%7;
}
days tomorrow(days today) {
    return (today+1)%7;
}
```

**输出结果：**

| today | yesterday | tomorrow |
|-------|-----------|----------|
| monday = 0 | sunday = 6 | tuesday = 1 |
| tuesday = 1 | monday = 0 | wednesday = 2 |
| wednesday = 2 | tuesday = 1 | thursday = 3 |
| thursday = 3 | wednesday = 2 | friday = 4 |
| friday = 4 | thursday = 3 | saturday = 5 |
| saturday = 5 | friday = 4 | sunday = 6 |
| sunday = 6 | saturday = 5 | monday = 0 |

## 程序 11.4：结构体的简单用法

```c
#include <stdio.h>

struct student {
    int mid;
    int final;
    int hmws;
};

void main(void)
{
    struct student sam = {85, 90, 88};
    struct student tom = {93, 88, 91};
    struct student *he = &tom; /* 我们可以通过指针访问结构体及其字段 */
}
```

我们不能像处理整数、实数等那样直接读取或写入结构体。此外，我们无法直接比较两个结构体（即不能说 `sam == tom`）。但我们可以将一个结构体赋值给另一个结构体。

例如：`tom = sam;`

然后我们可以通过打印它的各个字段来输出 `tom` 的值：

```c
printf("tom = %d, %d, %d\n", tom.mid, tom.final, tom.hmws);
```

注意用于访问 `tom` 字段的点号符号（`.`）。

输出将是：`85, 90, 88`

```c
printf("*he = %d, %d, %d\n", he->mid, he->final, he->hmws);
```

注意用于从 `he` 访问字段的箭头符号（`->`）。

输出将是：`*he = 85, 90, 88`

如果我们以不同的方式访问字段，将得到相同的输出：

```c
printf("*he = %d, %d, %d\n", (*he).mid, (*he).final, (*he).hmws);
```

## 程序 11.5：哈希表实现

```c
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define SIZE 50
#define MAX 10

typedef struct node {
    char symbol[MAX];
    int value;
    struct node *next;
} entry;

typedef entry *entry_ptr;

int hash_value(char *name) {
    int sum = 0;
    while (*name != '\0') {
        sum += *name;
        name++;
    }
    return (sum % SIZE);
}

void initialize(entry_ptr table[]) {
    for (int i = 0; i < SIZE; i++)
        table[i] = NULL;
}

void insert(entry_ptr table[], char *name, int val) {
    int h = hash_value(name);
    entry_ptr temp = table[h];
    int flag = 1;

    while (temp != NULL && flag) {
        if (strcmp(temp->symbol, name) == 0) {
            printf("符号 %s 已经存在于表中\n", name);
            flag = 0;
        }
        temp = temp->next;
    }

    if (flag) {
        temp = (entry_ptr) malloc(sizeof(entry));
        if (temp == NULL) {
            printf("ERRRR ......\n");
            exit(0);
        }
        strcpy(temp->symbol, name);
        temp->value = val;
        temp->next = table[h];
        table[h] = temp;
    }
}

void retrieve(entry_ptr table[], char *name) {
    int h = hash_value(name);
    entry_ptr temp = table[h];
    int flag = 1;

    while (temp != NULL && flag) {
        if (strcmp(temp->symbol, name) == 0) {
            printf("符号 %s 存在于表中且值为 %d", name, temp->value);
            flag = 0;
        }
        temp = temp->next;
    }

    if (flag == 1)
        printf("符号 %s 不在表中\n", name);
}

void main() {
    entry_ptr table[SIZE];
    char name[MAX];
    int value, n;

    initialize(table);

    do {
        printf("输入符号和值对\n");
        scanf("%s %d", &name, &value);
        insert(table, name, value);
        printf("输入 1 继续\n");
        scanf("%d", &n);
    } while (n == 1);

    do {
        printf("输入要查询的符号\n");
        scanf("%s", name);
        retrieve(table, name);
        printf("输入 1 继续\n");
        scanf("%d", &n);
    } while (n == 1);
}
```

## 程序 11.6：数组实现的线性表

```c
#include <stdio.h>
#include <conio.h>

#define MAX 20 // 列表最大元素

struct List {
    int list[MAX];         // 要插入的元素
    int element;
    int pos;               // 插入或删除的位置
    int length;            // 元素总数
} l;

enum boolean { false, true };

/* 函数原型 */
int menu(void);           /* 显示操作列表 */
void create(void);        /* 创建初始元素 */
void insert(int, int);    /* 在指定位置插入元素 */
void delete(int);         /* 删除指定位置的元素 */
void find(int);           /* 查找元素的位置 */
void display(void);       /* 显示列表中的元素 */
boolean isfull(void);     /* 检查列表是否满 */
boolean isEmpty(void);    /* 检查列表是否为空 */

void main() {
    int ch, element, pos;
    l.length = 0;

    while (1) {
        ch = menu();
        switch (ch) {
            case 1:
                l.length = 0;
                create();
                break;
            case 2:
                if (isfull() != true) {
                    printf("\t新元素 : ");
                    scanf("%d", &element);
                    printf("\t位置 : ");
                    scanf("%d", &pos);
                    insert(element, pos);
                } else {
                    printf("列表已满。无法插入...");
                }
                printf("\n按任意键继续...\n");
                getch();
                break;
            case 3:
                if (isEmpty() != true) {
                    printf("\t要删除的元素位置 : ");
                    scanf("%d", &pos);
                    delete(pos);
                } else {
                    printf("列表为空.");
                }
                printf("\n按任意键继续...\n");
                getch();
                break;
            case 4:
                printf("列表中的元素个数为 %d", l.length);
                printf("\n按任意键继续...\n");
                getch();
                break;
            case 5:
                printf("输入要查找的元素 : ");
                scanf("%d", &element);
                find(element);
                break;
            case 6:
                display();
                break;
            case 7:
                exit(0);
            default:
                printf("无效选择");
                printf("\n按任意键继续...\n");
                getch();
        }
    }
}

/* 显示操作列表的函数 */
int menu() {
    int ch;
    clrscr();
    printf("\n\t***********************\n");
    printf("\t\t数组实现的列表******\n");
    printf("\t***********************\n\n");
    printf("\t1. 创建\n\t2. 插入\n\t3. 删除\n\t4. 计数\n\t5. 查找\n\t6. 显示\n\t7. 退出\n");
    printf("\n输入你的选择 : ");
    scanf("%d", &ch);
    printf("\n");
    return ch;
}

/* 创建初始元素的函数 */
void create() {
    int element, flag = 1;
    while (flag == 1) {
        printf("输入一个元素 : ");
        scanf("%d", &element);
        l.list[l.length] = element;
        l.length++;
        printf("要插入另一个元素按 '1' : ");
        scanf("%d", &flag);
    }
}

/* 在指定位置插入给定元素的函数 */
void insert(int element, int pos) {
    int i;
    if (pos == 0) {
        for (i = l.length; i > 0; i--) {
            l.list[i] = l.list[i-1];
        }
        l.list[0] = element;
        l.length++;
    } else {
        for (i = l.length; i > pos; i--) {
            l.list[i] = l.list[i-1];
        }
        l.list[pos] = element;
        l.length++;
    }
}

/* 删除指定位置元素的函数 */
void delete(int pos) {
    int i;
    if (pos < l.length) {
        for (i = pos; i < l.length - 1; i++) {
            l.list[i] = l.list[i+1];
        }
        l.length--;
    }
}

/* 查找给定元素的位置函数 */
void find(int element) {
    for (int i = 0; i < l.length; i++) {
        if (l.list[i] == element) {
            printf("元素 %d 于位置 %d\n", element, i+1);
            return;
        }
    }
    printf("未找到元素\n");
}

/* 显示列表中的元素函数 */
void display() {
    for (int i = 0; i < l.length; i++) {
        printf("元素 %d : %d\n", i+1, l.list[i]);
    }
}

/* 检查列表是否满的函数 */
boolean isfull() {
    return (l.length == MAX) ? true : false;
}

/* 检查列表是否为空的函数 */
boolean isEmpty() {
    return (l.length == 0) ? true : false;
}
```
