# 104. 给定一个整数数组，找到给定长度且平均值最小的子数组。

```cpp
#include <iostream>
#include <vector>
int main() {
    int n, k;
    std::cout << "Enter the number of elements: ";
    std::cin >> n;
    std::vector<int> nums(n);
    std::cout << "Enter the elements of the array:\n";
    for (int i = 0; i < n; i++) {
        std::cin >> nums[i];
    }
    std::cout << "Enter the length of the subarray: ";
    std::cin >> k;
    if (k > n) {
        std::cout << "子数组长度大于数组大小!" << std::endl;
        return 0;
    }
    // 计算前 'k' 个元素的和
    int sum = 0;
    for (int i = 0; i < k; i++) {
        sum += nums[i];
    }
    int min_sum = sum;
    int start_index = 0; // 最小和子数组的起始索引
    // 滑动窗口法
    for (int i = k; i < n; i++) {
        sum += nums[i] - nums[i - k];
        if (sum < min_sum) {
            min_sum = sum;
            start_index = i - k + 1;
        }
    }
    std::cout << "平均值最小的长度为 " << k << "的子数组是:\n";
    for (int i = start_index; i < start_index + k; i++) {
        std::cout << nums[i] << " ";
    }
    std::cout << std::endl;
    return 0;
}
```

# 105. 给定两个排序数组，将它们合并为一个单一排序数组。

```cpp
#include <iostream>
#include <vector>
int main() {
    int n1, n2;
    // 读取第一个数组
    std::cout << "Enter the number of elements in the first array: ";
    std::cin >> n1;
    std::vector<int> arr1(n1);
    std::cout << "Enter the sorted elements of the first array:\n";
    for (int i = 0; i < n1; i++) {
        std::cin >> arr1[i];
    }
    // 读取第二个数组
    std::cout << "Enter the number of elements in the second array: ";
    std::cin >> n2;
    std::vector<int> arr2(n2);
    std::cout << "Enter the sorted elements of the second array:\n";
    for (int i = 0; i < n2; i++) {
        std::cin >> arr2[i];
    }
    // 合并数组
    int i = 0, j = 0;
    std::vector<int> merged(n1 + n2);
    int k = 0; // 合并数组的索引
    while (i < n1 && j < n2) {
        if (arr1[i] <= arr2[j]) {
            merged[k++] = arr1[i++];
        } else {
            merged[k++] = arr2[j++];
        }
    }
    // 如果 arr1 中还有剩余元素
    while (i < n1) {
        merged[k++] = arr1[i++];
    }
    // 如果 arr2 中还有剩余元素
    while (j < n2) {
        merged[k++] = arr2[j++];
    }
    // 打印合并后的数组
    std::cout << "Merged sorted array is:\n";
    for (int m = 0; m < k; m++) {
        std::cout << merged[m] << " ";
    }
    std::cout << std::endl;
    return 0;
}
```

## 112. 编写一个程序读取两个单词并检查第二个单词是否是第一个单词的变位词（anagram）。

```cpp
#include <iostream>
#include <string>
#include <algorithm>
int main() {
  std::string word1, word2;
  // 获取单词输入
  std::cout << "Enter the first word: ";
  std::cin >> word1;
  std::cout << "Enter the second word: ";
  std::cin >> word2;
  // 如果长度不同，则不可能是变位词
  if (word1.length() != word2.length()) {
    std::cout << "The words are not anagrams." << std::endl;
    return 0;
  }
  // 对两个单词排序
  std::sort(word1.begin(), word1.end());
  std::sort(word2.begin(), word2.end());
  // 检查排序后的单词是否相同
  if (word1 == word2) {
    std::cout << "The words are anagrams." << std::endl;
  } else {
    std::cout << "The words are not anagrams." << std::endl;
  }
  return 0;
}
```

## 113. 编写一个程序获取全名并仅显示名。

```cpp
    std::string fullName;
    // 获取全名输入
    std::cout << "Enter the full name: ";
    std::getline(std::cin, fullName);
    // 找到名字中的第一个空格
    size_t spacePosition = fullName.find(' ');
    std::string firstName;
    // 如果找到了空格，提取名
    if (spacePosition != std::string::npos) {
        firstName = fullName.substr(0, spacePosition);
    } else {
        // 如果没有找到空格，假设整个输入为名
        firstName = fullName;
    }
    std::cout << "First name: " << firstName << std::endl;
    return 0;
}
```
