10.16 (值是多少？) 下面程序使用位运算判断一个整数是否为 X 的倍数。

```c
#include <stdio.h>

int multiple(int num);

int main() {
    int y;
    printf("Enter integer: ");
    scanf("%d", &y);
    if (multiple(y)) {
        printf("%d is a multiple\n", y);
    } else {
        printf("%d is not a multiple\n", y);
    }
    return 0;
}

int multiple(int num) {
    int mask = 1;
    int mult = 1;
    for (int i = 0; i < 31; i++) {
        if (num & mask) {
            mult = 0;
            break;
        }
        mask <<= 1;
    }
    return mult;
}
```

10.17 (这个函数是做什么的?)

```c
#include <stdio.h>

int mystery(unsigned int bits);

int main() {
    unsigned int x;
    printf("Enter integer: ");
    scanf("%u", &x);
    printf("The result is %u\n", mystery(x));
    return 0;
}

int mystery(unsigned int bits) {
    unsigned int mask = 1 << 31;
    unsigned int total = 0;
    for (int i = 0; i < 32; i++) {
        if (bits & mask) {
            total++;
        }
        mask >>= 1;
    }
    return !(total % 2) ? 1 : 0;
}
```
