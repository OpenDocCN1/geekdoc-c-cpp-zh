```c
            for (j = 0; j <= 3; j++) {

                *(p + k * 5 + j) = *(p + k * 5 + j + 1);
                *(p + k * 5 + j + 1) = t;
            }
}
```

### (l) 矩阵行列式

通过键盘输入一个 6x6 矩阵。编写一个程序获取该矩阵的行列式值。

```c
/* nx矩阵的行列式 */
/* 此处的逻辑是评估一个 nx 行列式，我们首先将行列式的第一行元素与对应的 n-1 x n 行列式相乘，通过将 n-1 x n-1 行列式复制到预分配的内存中。我们继续这个过程直到达到 1x1 行列式，然后回溯步骤。 */
#include <stdlib.h>
#include <malloc.h>

void detmat(int *arr, int order);

int main() {
    int *arr, sum, n, i, j, pos, num;

    printf("输入 (nx) 矩阵的 n 值:\n");
    scanf("%d", &n);
    /* 分配内存以容纳行列式 */
    arr = calloc(n * n, 2);

    printf("输入数字:\n");
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            scanf("%d", &num);
            pos = i * n + j;
            arr[pos] = num;
        }
    }
    /* 打印输入的矩阵 */
    for (i = 0; i < n; i++) {
        for (j = i; j < n * n; j += n) {
            printf("%d\t", arr[j]);
        }
        printf("\n");
    }
    sum = detmat(arr, n);
    free(arr); /* 释放 arr 使用的内存 */

    printf("\n%d", sum);
    return 0;
}

/* 计算行列式的函数 */
void detmat(int *arr, int order) {
    int sign = 1, sum = 0, i, j, k, count, arr2;
    int newsize, newpos, pos;

    if (order == 1)
        return (arr[0]);

    for (i = 0; i < order; i++, sign *= -1) {
      /* 将 n-1 by n-1 拷贝到另一个数组 */
      newsize = (order - 1) * (order - 1);
      arr2 = calloc(newsize, 2); /* 为新数组分配内存 */
      for (j = 1; j < order; j++) {
        for (k = 0; k < order; k++) {
          if (k == i)
            continue;

          pos = j * order + k;
          newpos = (j - 1) * (order - 1) + count;

          arr2[newpos] = arr[pos];
          count++;
        }
        /* n-1 by n-1 数组的行列式值加到 sum */
        sum = sum + arr[i] * sign * detmat(arr2, order - 1);
        free(arr2); /* 释放数组 */
    }
    return (sum);
}
```
