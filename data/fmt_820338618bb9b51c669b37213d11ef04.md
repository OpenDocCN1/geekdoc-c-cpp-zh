
# 操作与维护队列

```c
// 操作与维护队列

#include <stdio.h>

#include <stdlib.h>

// 自引用结构

struct queueNode {

    char data; // 将数据定义为字符

    struct queueNode *nextPtr; // queueNode 指针

};

typedef struct queueNode QueueNode;

typedef QueueNode *QueueNodePtr;

// 函数原型

void printQueue(QueueNodePtr currentPtr);

int isEmpty(QueueNode *headPtr);

void dequeue(QueueNodePtr *headPtr, QueueNodePtr *tailPtr);

void enqueue(QueueNodePtr *headPtr, QueueNodePtr *tailPtr, char *value);

void instructions(void);

// 主函数 main，开始程序执行

int main(void)

{

    QueueNodePtr headPtr = NULL; // 初始化 headPtr

    QueueNodePtr tailPtr = NULL; // 初始化 tailPtr

    char item; // 用户输入的字符

    instructions(); // 显示菜单

    printf("%s", "2");

    unsigned int choice; // 用户菜单选择

    scanf("%u", &choice);

    // 当用户不输入 3 时

    while (choice != 3) {

        switch(choice) {

            // 入队值

            case 1:

                printf("%s", "Enter a character: ");

                scanf("%c\n", &item);

                enqueue(&headPtr, &tailPtr, item);

                printQueue(headPtr);

                break;

            // 出队值

            case 2:

                // 如果队列为空

                if (!isEmpty(headPtr)) {

                    item = dequeue(&headPtr, &tailPtr);

                    printf("%c has been dequeued.\n", item);

                }

        }

    }

}
```

```c
// 在队列末尾插入节点

void enqueue(QueueNode **headPtr, QueueNode **tailPtr, char value)

{

    QueueNode *newPtr = malloc(sizeof(QueueNode));

    if (newPtr != NULL) { // 是否有可用空间

        newPtr->data = value;

        newPtr->nextPtr = NULL;

        // 如果空，在头部插入节点

        if (isEmpty(*headPtr)) {

            *headPtr = newPtr;

        } else

            (*tailPtr)->nextPtr = newPtr;

        *tailPtr = newPtr;

    } else

        printf("%c not inserted. No memory available.\n", value);

}
```

```c
// 从队列中移除节点

char dequeue(QueueNode **headPtr, QueueNode **tailPtr)

{

    char value = (*headPtr)->data;

    QueueNodePtr tempPtr = *headPtr;

    *headPtr = (*headPtr)->nextPtr;

    // 如果队列为空

    if (*headPtr == NULL) {

        *tailPtr = NULL;

    }

    free(tempPtr);

    return value;

}
```

```c
// 在树中插入 0 到 14 之间的随机值
for (unsigned int i = 1; i <= 10; ++i) {
    int item = rand() % 15;
    printf("%3d", item);
    insertNode(&rootPtr, item);

puts("\n\n前序遍历结果为:\n\n");
preorder(rootPtr);

// 中序遍历树
puts("\n\n中序遍历结果为:\n\n");

// 后序遍历树
puts("\n\n后序遍历结果为:\n\n");
postOrder(rootPtr);

// 将节点插入树中
void insertNode(TreeNode *treePtr, int value) {
    // 如果树为空
    if (*treePtr == NULL) {
        *treePtr = malloc(sizeof(TreeNode));
        // 如果分配了内存，则分配数据
        if (*treePtr != NULL) {
            (*treePtr)->data = value;
            (*treePtr)->leftPtr = NULL;
            (*treePtr)->rightPtr = NULL;
        }
    } else {
        printf("%d 未插入。没有可用内存。\n", value);
    }
    // 待插入的数据小于当前节点的数据
    if (value < (*treePtr)->data) {
        insertNode(&(treePtr)->leftPtr, value);
    }
    // 待插入的数据大于当前节点的数据
    else if (value > (*treePtr)->rightPtr) {
        insertNode(&(treePtr)->rightPtr, value);
    }
    else { // 忽略重复数据值
        printf("%s", "dup");
    }
}
```
