
# 计算匹配样本中每个协变量的标准化差异

```sas
/*
计算匹配样本中每个协变量的标准化差异。 */
****************************************************************************************/
proc sort data=long;
by des;
run;
****************************************************************************************/
/* 计算连续变量标准化差异的宏。 */
****************************************************************************************/
%macro cont(var=label=);
proc means mean stddev data=long noprint
var &var;
by des;
out=outmean (keep = mean stddev) mean = mean stddev=stddev;
run;

data des0;
set outcome;
if des = 0;
mean_0 = mean;
s_0 = stddev;
keep mean_0 s_0;
run;

des1;
set outcome;
if des = 1;
mean_1 = mean;
s_1 = stddev;
keep mean_1 s_1;
run;

data newdata;
length label $ 25;
merge des0 des1;
d = (mean_1 - mean_0)/ sqrt(s_1*s_1 + s_0*s_0)/2);
d = round(abs(d),0.0001);
label = &label;
keep d label;
run;

proc append data=newata base=standiff force;
%mend cont;
****************************************************************************************************/
/* 计算分类变量标准化差异的宏。 */
************************************************************************************************
%macro binary(var,label)=;
proc means data=long noprint;
var &var;
by des;
output out=outmean (keep = des mean) mean = mean;
run;

data des0;
set outmean;
if des = 0;
mean_0 = mean;
run;

data des;
set outmean;
if des = 1;
mean_1 = mean;
keep mean_1;
run;

data newata;
length label $ 25;
merge des0;
d = (mean_1 - mean_0)/ sqrt(mean_1*(1-mean_1) + mean_0*(1-mean_0))/2;
d = round(abs(d),0.0001);
label = &label;
keep d label;
run;

proc append data=newata base=standiff force;
%mend binary;
```
