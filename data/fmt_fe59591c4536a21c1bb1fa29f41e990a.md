int count_x(const char* p, char x)
  // 计算 x 在 p 中出现的次数
  // 假设 p 指向一个零终止数组（或为空）
{
  if (p==nullptr)
    return 0;
  int count = 0;
  while (*p) {
    if (*p==x)
      ++count;
    ++p;
  }
  return count;
}
