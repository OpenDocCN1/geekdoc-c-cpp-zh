### (1) 编写一个使用位位运算符实现两个整数相乘的程序。

```c
#include <stdio.h>

int main()
{
    int a, b, result;
    printf("\nEnter the numbers to be multiplied: ");
    scanf("%d%d", &a, &b);
    result = 0;
    while (b != 0)
    {
        if (b & 1)
            result = add (result, a);
        a <<= 1;
        b >>= 1;
    }
    printf("\nResult:%d", result);
}

int add (int x, int y)
{
    while (y != 0)
    {
        int carry = x & y;
        x = x ^ y;
        y = carry << 1;
    }
    return x;
}
```

### (2) 编写一个程序统计给定文本文件中的单词数。

```c
#include<stdio.h>
int main()
{
    char ch; FILE *fp;
    char fname[13];
    int count = 0;
    printf("Enter File name");
    gets ( fname );
    fp = fopen(fname, "r");
    while ( ( ch = getc ( fp ) )!= EOF) {
        if ( ch==' ')
            count++;
    }
```

### (3) 编写一个程序，接收一组数字作为命令行参数并打印它们的平均值。

```c
#include <stdio.h>
#include <stdlib.h>
int main(int argc, char* argv[])
{
    int sum, i, avg;
    sum = 0;
    for (i = 1; i < argc; i++)
        sum = sum + atoi (argv[i]);
    avg = sum / argc ;
    printf("Average=%%%f", avg );
    return 0;
}
```

### (4) 编写一个程序，通过逐字节的比较来检查两个文件的内容是否相同。

```c
#include <stdio.h>
#include <stdlib.h>
int main()
{
    char ch1,ch2;
    FILE *fp,*fq;
    char fname1[ 13 ], fname2[ 13 ];
    printf("Enter File 1 name ");
    gets(fname1);
    printf("Enter File 2 name ");
    gets(fname2);
    fp = fopen(fname1, "rb");
    fq = fopen(fname2, "rb");
    while ( 1 )
    {
        ch1 = getc (fp);
        ch2 = getc (fq);
        if (ch1 != ch2 || (ch1 == EOF && ch2 != EOF) || (ch1 != EOF && ch2 == EOF))
        {
            printf("File contents do not match ");
            exit ( 1 );
        }
        if (ch1 == EOF && ch2 == EOF)
            break;
    }
    printf ( "File contents match" );
    fclose ( fp );
    fclose ( fq );
}
```
