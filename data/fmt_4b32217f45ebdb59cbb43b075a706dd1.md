
```c
double dmax;
int i;
int ix;
int value;
value = 0;
if ( n < 1 || incx <= 0 )
{
  return value;
}
value = 1;
if ( n == 1 )
{
  return value;
}
if ( incx == 1 )
{
  dmax = fabs(dx[0]);
  for ( i = 1; i < n; i++ )
  {
    if ( dmax < fabs(dx[i]) )
    {
      value = i + 1;
      dmax = fabs(dx[i]);
    }
  }
}
else {
  ix = 0;
  dmax = fabs(dx[0]);
  ix = ix + incx;
  for ( i = 1; i < n; i++ )
  {
    if ( dmax < fabs(dx[ix]) )
    {
      value = i + 1;
      dmax = fabs(dx[ix]);
    }
    ix = ix + incx;
  }
}
return value;
```

********************
`double r8_random ( int iseed[4] )`
********************

目的：

`R8_RANDOM` 返回 0 到 1 之间的均匀分布随机数。

讨论：

此程序使用模数为 `$2^{48}$` 的乘法同余方法，乘因子为 3395283406453。

48 位整数存储在每个 12 位的元素中。因此，该程序可以在具有 32 位整数的机器上移植。

参数：

输入/输出，integer `ISEED(4)`。进入时，随机数生成器的种子；数组元素必须在 0 到 4095 之间，且 `ISEED(4)` 必须为奇数。退出时，种子种子。输出，double `R8_RANDOM`，下一个伪随机数。

```c
int ipw2 = 4096;
int it1;
int it2;
int it3;
int it4;
int m1 = 494;
int m2 = 322;
int m3 = 2508;
int m4 = 2549;
double r = 1.0 / 4096.0;
double value;
/* 将种子乘以模 2**48 的乘因子 */
it4 = iseed[3] * m4;
it3 = it4 / ipw2;
it4 = it4 - ipw2 * it3;
it3_ = it3 + iseed[2] * m4 + iseed[3] * m3;
it2_ = it3 / ipw2;
it3_ = it3 - ipw2 * it2;
it2_ = it2 + iseed[1] * m4 + iseed[2] * m3 + iseed[3] * m2;
it1_ = it2 / ipw2;
it2_ = it2 - ipw2;
it1_ = it1 + iseed[0] * m4 + iseed[1] * m3 + iseed[3] * m1;
it1_ = ( it1 % ipw2 );
/* 返回更新的种子
seed[0] = it1;
seed[1] = it2;
seed[2] = it3;
seed[3] = it4;
/* 48 位整数转换为区间 (0,1) 内的实数 */
value = r * ( double )( it1 )
+ r * ( double )( it2 )
+ r * ( double )( it3 )
+ r * ( double )( it4 )
return value;
}
```
