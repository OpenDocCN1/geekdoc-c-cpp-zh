#include <stdio.h>

{
    int arr[3][3];
    i, j, count = 0;

    printf("输入矩阵元素:\n");
    for (i = 0; i <= 2; i++)
    {
        for (j = 0; j <= 2; j++)
        {
            scanf("%d", &arr[i][j]);
        }
    }

    printf("形成的矩阵为....");
    for (i = 0; i <= 2; i++)
    {
        for (j = 0; j <= 2; j++)
            printf("%d ", arr[i][j]);
        printf("\n");
    }

    /* 检查对称性 */
    for (i = 0; i <= 2; i++)
    {
        for (j = i; j <= 2; j++)
        {
            if (arr[i][j] == arr[j][i])
                count++;
        }
    }
