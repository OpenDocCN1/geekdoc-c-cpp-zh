#include <stdio.h>
#include <math.h>

#define PI 3.14159

void calculate_circle() {
    float radius, area;
    printf("输入半径: ");
    scanf("%f", &radius);
    area = PI * radius * radius;
    printf("圆形面积: %.2f\n", area);
}

void calculate_quad() {
    float length, width, area;
    printf("输入矩形长度: ");
    scanf("%f", &length);
    printf("输入宽度: ");
    scanf("%f", &width);
    area = length * width;
    printf("矩形面积: %.2f\n", area);
}

void calculate_triangle() {
    float a, b, c, s, area;
    printf("输入边A: ");
    scanf("%f", &a);
    printf("输入边B: ");
    scanf("%f", &b);
    printf("输入边C: ");
    scanf("%f", &c);
    s = (a + b + c) / 2;
    area = sqrt(s * (s - a) * (s - b) * (s - c));
    printf("三角形面积: %.2f\n", area);
}

int main() {
    int choice;
    while (1) {
        printf("\n1. 圆形\n2. 矩形\n3. 三角形\n4. 退出\n请选择: ");
        scanf("%d", &choice);
        switch (choice) {
            case 1: calculate_circle(); break;
            case 2: calculate_quad(); break;
            case 3: calculate_triangle(); break;
            case 4: return 0;
            default: printf("无效选择\n");
        }
    }
    return 0;
}
