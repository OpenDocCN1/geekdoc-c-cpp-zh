
### (a) 阶段1 顺序/读取

```c
void stage1(tid, arg);

version = 0;

while (TRUE) {

  if (ver.begin(tid)) {

    ver.recovery(tid); continue;

    ver.read(tid,node->next,node->next);

    node = node->next;

  }

  ver.terminate(tid);

  ver.recovery(tid); continue;

}

dest = 1 + (version % np);

ver.write(tid, dest, &node);

ver.commit(unit, node, node);

ver.commit(tid, unit, &node);

if (ver.end(tid)) {

  ver.recovery(tid); continue;

  version++;

}
```

### (b) 阶段2 并行/工作

```c
void stage2(tid, arg);

version = 0;

while(TRUE) {

  if (ver.begin(tid)) {

    ver.recovery(tid); continue;

    if (node == np - 1) {

      ver.read(tid,node, arg);

      res = work(node);

      ver.write(all, &res, res);

    }

  }

  if (ver.end(tid))

    ver.recovery(tid); continue;

  version++;

}
```

### (c) 阶段3 顺序/写入

```c
void stage3(tid, arg);

version = 0;

while(TRUE) {

  if (ver.begin(tid)) {

    ver.recovery(tid); continue;

    write(res);

  }

  if (ver.end(tid))

  ver.recovery(tid); continue;

version++;

}
```
