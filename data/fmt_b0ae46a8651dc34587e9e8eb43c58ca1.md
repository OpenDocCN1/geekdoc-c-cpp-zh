```c
struct node fun(struct node);

{
    struct node s1 , s2;

    s1.key2 = 20;
    s1.next = NULL;
    s2 = fun(s1);
    printf("%d %d",s1.key1,s1.key2);
    printf("%d %d",s2.key1,s2.key2);
    return 0;

    s3.key1 = s3.key1^s3.key2;

    s3.key1 = s3.key1^s3.key2;
    return s3;
}
```

- (d) fun 函数始终交换 s3 的 key1 和 key2。
