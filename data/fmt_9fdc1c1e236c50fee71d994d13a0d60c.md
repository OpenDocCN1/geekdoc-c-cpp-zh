```csharp
class clsErrorLog
{
    //==================== 实例成员 ==================
    private string fileName;
    private string path;
    private string errorMessage;
    private int errorFlag;

    StreamWriter sw = null;
    StreamReader sr = null;

    //==================== 构造函数 ==================
    public clsErrorLog(string msg)
    {
        errorMessage = msg;
        errorFlag = 0;
        fileName = "ErrorLog.txt";
    }

    //==================== 属性方法 ==================
    public string fileName
    {
        get
        {
            return fileName;
        }
        set
        {
            if (value.Length > 0)
                fileName = value;
        }
    }
    public string errorMessage
    {
        get
        {
            return errorMessage;
        }
        set
        {
            if (value.Length > 0)
                errorMessage = value;
        }
    }
    public string PathName
    {
        get
        {
            return path;
        }
        set
        {
            if (value.Length > 0)
                path = value;
        }
    }

    //========================= 通用方法 ===================
    * 用途：读取错误日志文件。
    * 参数列表:
    * n/a
    * 返回值
    * string 错误日志消息的内容
    public string ReadLog()
    {
        string buff;
        try {
            string pfn = Path.Combine(pathName, fileName);
            if (File.Exists(pfn)) {
                sr = new StreamReader(pfn);
                buff = sr.ReadToEnd();
                sr.Close();
            }
            return buff;
        } catch {
            return "";
        }
    }

    * 用途：将错误日志消息写入错误日志文件。
    消息包含日期、时间、错误类型以及错误的堆栈跟踪。
    * 参数列表:
    * n/a
    * 返回值
    * int 0 = 无错误，1 = 其他
    public int WriteErrorLog()
    {
        errorFlag = 0;
        DateTime currentDT = DateTime.Now();
        try{
            // 我们有这些字符串吗？
            if (errorMessage.Length != 0 && pathName.Length != 0 &&
                fileName.Length != 0)
            {
                sw = new StreamWriter(Path.Combine(pathName, fileName), true);
                sw.WriteLine(currentDT.ToShortDateString() + " * " +
                    currentDT.ToShortTimeString() + " * = " + errorMessage);
                sw.WriteLine("--------------------");
            }
            sw.Close();
        }
        else{
            errorFlag = 1; // 发生了错误
        }
    } catch (Exception ex)
    {
        errorMessage = ex.Message;
        errorFlag = 1; // 发生了错误
    }
    return errorFlag;
    }

    * 用途：将错误消息写入错误日志文件。
    * 参数列表:
    * string msg 要写入的错误消息
    * 返回值
    * int 0 = 无错误，1 = 其他
    public int WriteErrorLog(string msg)
    {
        errorMessage = msg; // 复制消息
        errorFlag = WriteErrorLog(); // 现在调用原始的那个
        return errorFlag;
    }
}
```
