```cpp
* // selectionSort 函数按升序对字符串向量进行排序。
*****************************************************************
*/
void selectionSort(vector<string> &v)
{
    int minIndex;
    string minValue;

    for (int start = 0; start < (v.size() - 1); start++)
    {
        minIndex = start;
        minValue = v[start];
        for (int index = start + 1; index < v.size(); index++)
        {
            if (v[index] < minValue)
            {
                minValue = v[index];
                minIndex = index;
            }
        }
        swap(v[minIndex], v[start]);
    }
}
```

```cpp
void swap(string &a, string &b)
{
    string temp = a;
    a = b;
    b = temp;
}
```

# 8.5 向量的排序与搜索

```cpp
int binarySearch(const vector<string> &v, string str)
{
    int first = 0,             // 向量第一个元素
        last = v.size() - 1,   // 向量最后一个元素
        middle,                // 搜索中间点
        position = -1;         // 搜索值的位置
    bool found = false;        // 标记位

    while (!found && first <= last)
    {
        middle = (first + last) / 2; // 计算中间点
        if (v[middle] == str)        // 如果在中间找到值
        {
            found = true;
            position = middle;
        }
        else if (v[middle] > str)    // 如果值在下半部分
            last = middle - 1;
        else
            first = middle + 1;      // 如果值在上半部分
    }
    return position;
}
```

## 9.1 获取变量的地址

+ 概念：变量的地址

每个变量都存储在内存的位置。每个位置都有唯一的地址。使用地址运算符 `&` 获取变量的地址。

```cpp
#include <iostream>
using namespace std;

int main()
{
    int x = 25;
    cout << "x的地址是 " << &x << endl;
    cout << "x的大小是 " << sizeof(x) << " 字节" << endl;
    cout << "x的值是 " << x << endl;
    return 0;
}
```

程序输出
x的地址是 0x7ffee15bc
x的大小是 4 字节
x的值是 25

## 9.2 指针变量

+ 概念：指针变量

通常称为指针的变量用于存储内存地址。通过指针变量，你可以间接地操作数据。就像 `int` 变量用于存储整数，`double` 变量用于存储浮点数一样，指针变量用于存储内存地址。

在数据类型和变量名之间使用星号 `*` 声明指针变量：

```cpp
int *ptr;
double *ptr;
char *ptr;
```

使用地址运算符 `&` 为指针初始化：

```cpp
int x = 25;
int *ptr = &x;
```

要访问指针所指向地址的值，使用解引用运算符 `*`：

```cpp
#include <iostream>
using namespace std;

int main()
{
    int x = 25;
    int *ptr = &x;
    cout << "x的地址是 " << &x << endl;
    cout << "ptr中存储的值是 " << ptr << endl;
    cout << "x的值是 " << *ptr << endl;
    return 0;
}
```

程序输出
x的地址是 0x7ffee15bc
ptr中存储的值是 0x7ffee15bc
x的值是 25

## 9.3 数组与指针之间的关系

+ 概念：数组与指针之间的关系

在 C++ 中，数组名是指向数组第一个元素的指针。这意味着数组和指针密切相关。

```cpp
int list1[5] = {10, 20, 30, 40, 50};
int *ptr = list1;
```

由于数组名是第一个元素的指针，你可以使用指针运算来访问数组元素：

```cpp
cout << *ptr;       // 输出10
cout << *(ptr + 1); // 输出20
```

以下程序演示了数组与指针的关系：

```cpp
#include <iostream>
using namespace std;

int main()
{
    int list1[5] = {10, 20, 30, 40, 50};
    int *ptr = list1;
    for (int i = 0; i < 5; i++)
    {
        cout << "元素 " << i << " 是 " << *(ptr + i) << endl;
    }
    return 0;
}
```
