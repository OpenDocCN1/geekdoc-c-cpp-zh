# 用于可重用的继承

```csharp
public class Stack : DataStructure

    public Stack(int size) : base(size)

    }

    {

        top++;

        SetElement(top, value);

    }

    public int Pop()

    {

        int value = GetElement(top);

        top--;

        return value;

    }

}

```
